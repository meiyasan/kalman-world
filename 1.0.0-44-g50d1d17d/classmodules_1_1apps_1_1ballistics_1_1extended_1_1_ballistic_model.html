281 def _acceleration(self, p, v, u=None):
282 """Gravity + drag + control at position p and velocity v (the forces f() integrates)."""
283 # Near-surface mode (g provided): constant downward acceleration in Z.
284 # Orbital mode (g=None): full geocentric G*M/r³ formula — requires
285 # positions in geocentric metres (origin = Earth's centre).
286 if self["g"] is not None:
287 a_grav = np.array([0.0, 0.0, -self["g"]], dtype=self.dtype)
288 else:
289 r = np.linalg.norm(p)
290 a_grav = -self["G"] * self["M"] * p / r**3 if r > 0 else np.zeros(3, dtype=self.dtype)
291
292 # Drag (depends on velocity, without control)
293 vmag = np.linalg.norm(v)
294 a_drag = -(self["c_drag"] / self["m"]) * vmag * v if vmag > 0 else np.zeros(3, dtype=self.dtype)
295
296 # Control (optional controlled thrust force)
297 a_ctrl = np.zeros(3, dtype=self.dtype)
298 if u is not None:
299 u_arr = np.asarray(u, dtype=self.dtype).ravel()
300 if u_arr.size > 0:
301 a_ctrl = u_arr.reshape(3) / self["m"]
302
