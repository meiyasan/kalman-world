1"""
2An instrument: a set of sensors, and what they share.
3
4The Python face of ``Kalman::Instrument`` (``include/Kalman/Instrument.h``), with
5the same fields. A sensor is one reading of the state; an instrument is the
6piece of hardware that carries one or more of them: a weather station, a
7receiver, a camera. Its sensors share
8
9- a **name**, under which the model groups them (``model.instruments``);
10- a **rate**, for the sensors that state none of their own;
11- a **calibration**: every reading goes through one gain and one offset,
12
13 z = gain * g(x, t) + offset
14
15 which the model that takes the instrument declares as two parameters,
16 ``"<name>.gain"`` and ``"<name>.offset"``. Being parameters, they can be
17 estimated with the state: ``model.joint({"station.offset": (sigma, drift)})``
18 calibrates the instrument while it measures.
19
20An instrument is made by composing sensors::
21
22 station = compose(thermometer, co2, name="station", rate=0.5)
23 station = thermometer | co2 # named after both
24 room = model.add_instrument(station) # a new model, read by both
25
26Two more things an instrument can say:
27
28- where its data comes from, ``provider=``: a provider (or any function) that
29 returns the instrument's own readings, one column per channel.
30 ``model.measurements()`` then fetches them and lays them on the model's steps;
31- ``sequential=True``: its sensors are read one after the other, one per step,
32 and not stacked into one measurement. This is a detector: its layers, each at
33 its ``station``. A model that is a detector reports itself this way
34 (``chamber.instruments``); such an instrument is not added to another model.
35
36A given piece of hardware is an **adapter**: a subclass of this class made of
37generic sensors (``kalman.framework.sensors``), with what its data sheet and
38its mounting say, and its provider. See ``kalman.adapters``.
39
40The arithmetic is the C++ model's (``Kalman::Models::Fusion``): nothing is
41computed here.
42"""

48class KalmanInstrument:
49 """A set of sensors.
50
51 Parameters
52 ----------
53 name : what the instrument is called.
54 sensors : its :class:`~.sensor.KalmanSensor` objects, in the order of their channels.
55 rate : how often it reports, for its sensors that state no rate of their
56 own; ``None`` for every step of the model.
57 gain, offset : give either and its readings go through
58 ``z = gain * g(x, t) + offset`` (see :meth:`calibrate`).
59 provider : where its readings come from: a provider (``fetch()`` returning
60 ``{"zk": ...}``) or a function returning the array, ``(N, channels)`` at
61 the instrument's own rate. See :meth:`read`.
62 sequential : ``True`` when its sensors are read one after the other, one per
63 step (a detector's layers), and not stacked into one measurement.
64 """

66 def __init__(self, name, sensors, *, rate=None, gain=None, offset=None, provider=None, sequential=False):
67 if not isinstance(name, str) or not name:
68 raise ValueError("KalmanInstrument: an instrument needs a name.")
69 sensors = list(sensors)
70 if not sensors:
71 raise ValueError(f"KalmanInstrument: {name!r} has no sensor.")
72 for sensor in sensors:
73 if not isinstance(sensor, KalmanSensor):
74 raise TypeError(f"KalmanInstrument: {name!r} is made of KalmanSensor objects, "
75 f"got {type(sensor).__name__}.")
76 names = [sensor.name for sensor in sensors]
77 if len(set(names)) != len(names):
78 raise ValueError(f"KalmanInstrument: {name!r} has two sensors of one name ({names}).")
79 if rate is not None and not float(rate) > 0.0:
80 raise ValueError(f"KalmanInstrument: the rate of {name!r} must be positive.")
81 self.name = name
82 self.sensors = sensors
83 self.rate = None if rate is None else float(rate)
84 self.provider = provider
85 self.sequential = bool(sequential)
86 self.calibrated, self.gain, self.offset = False, 1.0, 0.0
87 if gain is not None or offset is not None:
88 self.calibrate(1.0 if gain is None else gain, 0.0 if offset is None else offset)

90 def calibrate(self, gain=1.0, offset=0.0):
91 """Give the instrument a calibration to carry, ``z = gain * g + offset``: the
92 parameters ``"<name>.gain"`` and ``"<name>.offset"`` of the model that takes
93 it. Returns the instrument."""
94 self.calibrated, self.gain, self.offset = True, float(gain), float(offset)
95 return self

97 @property
98 def channels(self):
99 """The number of channels of its measurement, or ``None`` while a sensor
100 cannot tell: the sum over its sensors, or one sensor's when they are read
101 one after the other."""
102 counts = [sensor.channels for sensor in self.sensors]
103 if any(k is None for k in counts):
104 return None
105 return counts[0] if self.sequential else sum(counts)

107 def read(self, **fetch):
108 """Its own readings, from its provider: ``{sensor name: (N, k) array}``, one
109 entry per sensor, as :meth:`KalmanModel.measurements` takes them. ``fetch``
110 goes to the provider."""
111 import numpy as np
112 if self.provider is None:
113 raise ValueError(f"KalmanInstrument {self.name!r} has no provider: give provider= to read it.")
114 if self.sequential:
115 raise ValueError(f"KalmanInstrument {self.name!r} is a detector: its readings are hits per layer, "
116 "handed to track finding, not rows of a measurement.")
117 source = self.provider
118 data = source.fetch(**fetch) if hasattr(source, "fetch") else source(**fetch)
119 zk = np.asarray(data["zk"] if isinstance(data, dict) else data, dtype=np.float64)
120 widths = [sensor.channels for sensor in self.sensors]
121 if any(k is None for k in widths):
122 raise ValueError(f"KalmanInstrument {self.name!r}: a sensor does not say how many channels it has.")
123 zk = zk.reshape(len(zk), -1) if zk.size else zk.reshape(0, sum(widths))
124 if zk.shape[1] != sum(widths):
125 raise ValueError(f"KalmanInstrument {self.name!r}: its provider returned {zk.shape[1]} columns, its "
126 f"sensors have {sum(widths)} channels ({dict(zip([s.name for s in self.sensors], widths))}).")
127 out, first = {}, 0
128 for sensor, k in zip(self.sensors, widths):
129 out[sensor.name] = zk[:, first:first + k]
130 first += k
131 return out
