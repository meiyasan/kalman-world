25 Router::Router(std::vector<std::shared_ptr<Model>> modes, const Matrix &Pi, const Vector &mu0,
26 std::vector<std::string> names)
27 : fModes(std::move(modes)), fModeNames(std::move(names)), fPi(Pi), fDecision(std::make_shared<Routers::Bayesian>())
28 {
29 const std::size_t M = fModes.size();
30 if (M < 2) throw std::invalid_argument("Models::Router: at least two models are required.");
31 for (const auto &mode : fModes)
32 if (!mode) throw std::invalid_argument("Models::Router: a model is null.");
33 if (Pi.Rows() != M || Pi.Cols() != M)
34 throw std::invalid_argument("Models::Router: Pi has shape (" + std::to_string(Pi.Rows()) + ", "
35 + std::to_string(Pi.Cols()) + "), expected (" + std::to_string(M) + ", "
36 + std::to_string(M) + ").");
37 for (std::size_t i = 0; i < M; ++i)
38 {
39 double row = 0.;
40 for (std::size_t j = 0; j < M; ++j)
41 {
42 if (Pi(i, j) < 0.) throw std::invalid_argument("Models::Router: Pi has a negative entry.");
43 row += Pi(i, j);
44 }
45 if (std::abs(row - 1.) > 1e-9)
46 throw std::invalid_argument("Models::Router: row " + std::to_string(i) + " of Pi sums to "
47 + std::to_string(row) + ", expected 1.");
48 }
49
50 // Mixing averages states across modes, so it only means something when
51 // every mode means the same thing by "the state".
52 const Model &first = *fModes.front();
53 for (const auto &mode : fModes)
54 {
55 if (mode->NState() != first.NState() || mode->NMeasure() != first.NMeasure()
56 || mode->NControl() != first.NControl())
57 throw std::invalid_argument("Models::Router: model '" + mode->Name() + "' has (n, m, l) = ("
58 + std::to_string(mode->NState()) + ", " + std::to_string(mode->NMeasure())
59 + ", " + std::to_string(mode->NControl()) + ") but '" + first.Name()
60 + "' has (" + std::to_string(first.NState()) + ", "
61 + std::to_string(first.NMeasure()) + ", " + std::to_string(first.NControl())
62 + "). Every mode must share them.");
63 if (mode->SampleRate() != first.SampleRate())
64 throw std::invalid_argument("Models::Router: the modes must share one sample rate.");
65 }
66
67 if (fModeNames.empty())
68 {
69 std::map<std::string, int> seen;
70 for (const auto &mode : fModes)
71 {
72 const std::string base = mode->Name().empty() ? "mode" : mode->Name();
73 const int k = ++seen[base];
74 fModeNames.push_back(k == 1 ? base : base + std::to_string(k));
75 }
76 }
77 if (fModeNames.size() != M) throw std::invalid_argument("Models::Router: one name per model.");
78 {
79 std::set<std::string> seen;
80 for (const std::string &name : fModeNames)
81 if (name.empty() || !seen.insert(name).second)
82 throw std::invalid_argument("Models::Router: every mode needs a name of its own ('" + name + "').");
83 }
84
85 SetName(first.Name());
86 Allocate(first.NState(), first.NMeasure(), first.NControl());
87 SetSampleRate(first.SampleRate());
88 SetT0(first.T0());
89 SetTimeUnit(first.TimeUnit());
90 SetDefaultEngine(EngineKind::Interacting);
91 SetStructure(StructureKind::Maps);
92 SetBuiltInSensorName(first.BuiltInSensorName());
93 if (first.HasAnalyticJacobian()) SetAnalyticJacobian(true);
94 else SetDifferentiator(first.Differentiator(), first.Epsilon());
95 DeclareState(first.StateComponents());
96 DeclareMeasurement(first.MeasurementComponents());
97
98 SetInitialProbabilities(mu0.Size() ? mu0 : Vector(M, 1. / static_cast<double>(M)));
99 }

101 void Router::SetInitialProbabilities(const Vector &mu0)
102 {
103 if (mu0.Size() != fModes.size())
104 throw std::invalid_argument("Models::Router: one initial probability per model.");
105 fMu0 = Routers::Router::Normalise(mu0);

130 void Router::AdoptInitialState()
131 {
132 const std::size_t n = fNState;
133 Vector x(n, 0.);
134 for (std::size_t j = 0; j < fModes.size(); ++j)
135 if (fModes[j]->X0().Size() == n) x += fMu0(j) * fModes[j]->X0();
136 Matrix P(n, n, 0.);
137 for (std::size_t j = 0; j < fModes.size(); ++j)
138 {
139 const Matrix &Pj = fModes[j]->P0();
140 for (std::size_t a = 0; a < n; ++a)
141 for (std::size_t b = 0; b < n; ++b)
142 {
143 const double da = fModes[j]->X0().Size() == n ? fModes[j]->X0()(a) - x(a) : 0.;
144 const double db = fModes[j]->X0().Size() == n ? fModes[j]->X0()(b) - x(b) : 0.;
145 const double p = (Pj.Rows() == n && Pj.Cols() == n) ? Pj(a, b) : (a == b ? 1. : 0.);
146 P(a, b) += fMu0(j) * (p + da * db);
147 }
148 }
149 Utils::Symmetrize(P);
150 SetInitialState(x);
151 SetInitialCovariance(P);
152 }

167 std::pair<std::size_t, std::string> Router::Resolve(const std::string &name) const
168 {
169 std::size_t best = fModes.size(), length = 0;
170 for (std::size_t j = 0; j < fModes.size(); ++j)
171 {
172 const std::string &mode = fModeNames[j];
173 if (name.size() > mode.size() + 1 && name.compare(0, mode.size(), mode) == 0 && name[mode.size()] == '.'
174 && mode.size() >= length && fModes[j]->HasParameter(name.substr(mode.size() + 1)))
175 {
176 best = j;
177 length = mode.size();
178 }
179 }
180 if (best == fModes.size())
181 {
182 std::string known;
183 for (const std::string &n : ParameterNames()) known += (known.empty() ? "" : ", ") + n;
184 throw std::out_of_range("Models::Router: no parameter '" + name + "'; it has: " + (known.empty() ? "none" : known) + ".");
185 }
186 return {best, name.substr(length + 1)};
187 }
