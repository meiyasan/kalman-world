37 def test_three_ways(self):

44 def test_stacked(self):
45 a, b = cart("a", 1.0), cart("b", -2.0)
46 model = compose(a, b)
47 self.assertIs(model.structure, KalmanStructureKind.MATRICES)
48 self.assertIs(model.DEFAULT_ENGINE, KalmanEngineKind.LINEAR)
49 np.testing.assert_array_equal(model.x0.ravel(), [0, 1, 0, -2])
50 x = np.array([[1.0], [2.0], [10.0], [-4.0]])
51 np.testing.assert_allclose(model.f(x).ravel(), [1.2, 2.0, 9.6, -4.0])
52 np.testing.assert_array_equal(model.h(x).ravel(), [1.0, 10.0])
53 F, H = model.F(), model.H()
54 self.assertEqual((F[0, 1], F[2, 3], F[0, 2], H[0, 0], H[1, 2], H[0, 2]), (0.1, 0.1, 0.0, 1.0, 1.0, 0.0))
55 self.assertEqual(model.R()[1, 1], 0.25)
56 self.assertEqual([s.name for s in model.sensors], ["a", "b"]) # each part brings its sensor
57 self.assertEqual(float(model.sensors[1].measure(x)[0, 0]), 10.0)
58 self.assertEqual(model.part("b"), slice(2, 4))
59 self.assertEqual(model.part(b), slice(2, 4))
60 with self.assertRaises(KeyError):
61 model.part("c")

72 def test_the_first_part_is_filtered_as_it_would_be_alone(self):
73 a = cart("a")
74 model = compose(a, cart("b", -2.0))
75 rng = np.random.default_rng(3)
76 zk = rng.standard_normal((200, 2)).cumsum(axis=0) * 0.1
77 with warnings.catch_warnings():
78 warnings.simplefilter("ignore")
79 both = np.asarray(model.filter(zk, smooth=False, reset=True).xk).reshape(200, -1)
80 alone = np.asarray(cart("a").filter(zk[:, :1], smooth=False, reset=True).xk).reshape(200, -1)
81 np.testing.assert_allclose(both[:, :2], alone, atol=1e-10)

93 def test_parameters_of_a_part(self):
94 thermo = KalmanInstrument("thermo", [KalmanSensor("probe", G=[[1.0, 0.0]], R=0.04)], offset=0.5)
95 left = cart("left").add_instrument(thermo)

101 warnings.simplefilter("ignore")

103 self.assertEqual(joint.n, 5)
