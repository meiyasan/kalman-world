MPCKalmanController forwards to Kalman::Controllers::MPC; it has no Python
implementation, so it is pinned against LQR and against its own limits.

133 def rollout(controller, n_steps=200):
134 """A noise-free run of the car: the commands asked for, and the cost."""
135 model = controller.model
136 Qcost, Rcost, x_ref = model.control_cost()
137 x, asked, cost = model.x0.copy(), [], 0.0
138 for k in range(n_steps):
139 u = np.asarray(controller(x, 1e-4 * np.eye(3), k / model.fs), dtype=float).reshape(-1, 1)
140 applied = model.limit_control(u)
141 asked.append(float(u[0, 0]))
142 cost += ((x - x_ref).T @ Qcost @ (x - x_ref) + applied.T @ Rcost @ applied).item() # 1 x 1: numpy 2.5 refuses float() of it
143 x = np.asarray(model.decode_step(x, u=u, t=k / model.fs), dtype=float).reshape(-1, 1)
144 return np.array(asked), cost, x
