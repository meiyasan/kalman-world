48def _files():
49 for path in ROOT.rglob("*"):
50 rel = path.relative_to(ROOT)
51 if not path.is_file() or any(part in SKIP for part in rel.parts) or rel.as_posix() in ALLOWED:
52 continue
53 if path.suffix in TEXT or path.suffix == ".ipynb":
54 yield path, rel.as_posix()

57def _text(path):
58 """What a person wrote in the file: for a notebook the cell sources, not the stored outputs."""
59 raw = path.read_text(encoding="utf-8", errors="replace")
60 if path.suffix != ".ipynb":
61 return raw
62 try:
63 cells = json.loads(raw).get("cells", [])
64 except json.JSONDecodeError:
65 return raw
66 return "\n".join("".join(cell.get("source", [])) for cell in cells)
