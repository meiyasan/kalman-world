9class UnscentedKalmanEngine(KalmanEngine):
10 """
11 Sigma-point engine. Uses ``model.f`` / ``model.h`` directly (no Jacobians).
12
13 Noise handling is automatic per call:
14 - non-additive (``model.f(...,w)`` / ``model.h(...,v)``) → augmented-state sigma points;
15 - otherwise additive with ``L Q Lᵀ`` / ``M R Mᵀ`` coupling.
16
17 Returns an identity transition for the generic RTS step (``_smooth`` inherited).
18 The measurement update is missing-measurement aware (NaN channels dropped).
19 """

32 # Utilities

34 @staticmethod
35 def _to_vec(a):
36 a = np.asarray(a, dtype=np.float64)
37 return a.reshape(-1) if a.ndim > 1 else a
38
39 @staticmethod
40 def _sym_psd(S):
41 S = 0.5 * (S + S.T)
42 return S + 1e-12 * np.eye(S.shape[0], dtype=S.dtype)
43
44 @staticmethod
45 def _chol(S):
46 S = UnscentedKalmanEngine._sym_psd(S)
47 try:
48 return np.linalg.cholesky(S)
49 except np.linalg.LinAlgError:
50 d = S.shape[0]
51 for k in range(6):
52 try:
53 return np.linalg.cholesky(S + (10.0 ** k) * 1e-12 * np.eye(d))
54 except np.linalg.LinAlgError:
55 pass
56 raise

58 def _ut_weights(self, n_eff):
59 lam = self.alpha ** 2 * (n_eff + self.kappa) - n_eff
60 gamma = np.sqrt(n_eff + lam)
61 Wm = np.full(2 * n_eff + 1, 0.5 / (n_eff + lam), dtype=np.float64)
62 Wc = Wm.copy()
63 Wm[0] = lam / (n_eff + lam)
64 Wc[0] = lam / (n_eff + lam) + (1.0 - self.alpha ** 2 + self.beta)
65 return gamma, Wm, Wc
66
67 def _sigmas(self, mean, cov, gamma):
68 mean = self._to_vec(mean)
69 S = self._chol(cov)
70 n = mean.shape[0]
71 sig = np.zeros((2 * n + 1, n), dtype=np.float64)
72 sig[0] = mean
73 for i in range(n):
74 col = gamma * S[:, i]
75 sig[i + 1] = mean + col
76 sig[n + i + 1] = mean - col
77 return sig

129 @staticmethod
130 def _weighted_outer_sum(Wc, Y, mean):
131 """``sum(Wc[i] * outer(Y[i]-mean, Y[i]-mean))`` over sigma points."""
132 n_sigma, n = Y.shape
133 from ..native import HAS_C_ACCEL
134 if HAS_C_ACCEL:
135 from ..native import lib, as_contiguous, dptr
136 P_out = np.empty((n, n), dtype=np.float64)
137 lib.ukf_weighted_outer_sum(
138 n_sigma, n,
139 dptr(as_contiguous(Wc)), dptr(as_contiguous(Y)), dptr(as_contiguous(mean)),
140 dptr(P_out),
141 )
142 return P_out
143 return sum(Wc[i] * np.outer(Y[i] - mean, Y[i] - mean) for i in range(n_sigma))

152 from ..native import lib, as_contiguous, dptr

165 def _predict(self, model, x, P, u=None, t=None, **kw):
166 if u is not None:
167 u = np.asarray(u).reshape(-1, 1)
168 u = u if model.l > 0 else None
169
170 n = model.n
171 x_vec = self._to_vec(x)
172 P = np.asarray(P, dtype=np.float64)
173
174 if self._noise_through_f(model):
175 Qw = self._Q(model, t)
176 n_a = n + Qw.shape[0]
177 gamma, Wm, Wc = self._ut_weights(n_a)
178 mean_a = np.zeros(n_a); mean_a[:n] = x_vec
179 P_a = np.zeros((n_a, n_a)); P_a[:n, :n] = P; P_a[n:, n:] = Qw
180 sig_a = self._sigmas(mean_a, P_a, gamma)
181 Y = np.array([self._f_call(model, sp[:n], u=u, t=t, w=sp[n:]) for sp in sig_a])
182 x_pred = np.einsum("i,ij->j", Wm, Y)
183 P_pred = self._weighted_outer_sum(Wc, Y, x_pred)
184 P_pred = self._sym_psd(P_pred)
185 else:
186 gamma, Wm, Wc = self._ut_weights(n)
187 sig = self._sigmas(x_vec, P, gamma)
188 Y = np.array([self._f_call(model, sp, u=u, t=t) for sp in sig])
189 x_pred = np.einsum("i,ij->j", Wm, Y)
190 P_pred = self._weighted_outer_sum(Wc, Y, x_pred)
191 L = self._L(model, x_vec.reshape(-1, 1), u, t)
192 P_pred = self._sym_psd(P_pred + self._sym_psd(L @ self._Q(model, t) @ L.T))
193
194 return x_pred.reshape(-1, 1), P_pred, np.eye(n)

196 def _update(self, model, x, P, z, t=None, **kw):
197 n = model.n
198 x_pred = self._to_vec(x)
199 P_pred = np.asarray(P, dtype=np.float64)
200 z = self._to_vec(z)
201 keep = ~np.isnan(z)
202 if not keep.any():
203 return np.asarray(x).reshape(-1, 1), P_pred
204
205 if self._noise_through_h(model):
206 Rv = self._R(model, t)
207 n_a = n + Rv.shape[0]
208 gamma, Wm, Wc = self._ut_weights(n_a)
209 mean_a = np.zeros(n_a); mean_a[:n] = x_pred
210 P_a = np.zeros((n_a, n_a)); P_a[:n, :n] = P_pred; P_a[n:, n:] = Rv
211 sig_a = self._sigmas(mean_a, P_a, gamma)
212 Z = np.array([self._h_call(model, sp[:n], t=t, v=sp[n:]) for sp in sig_a])
213 z_pred = np.einsum("i,ij->j", Wm, Z)
214 S = self._weighted_outer_sum(Wc, Z, z_pred)
215 Pxz = self._weighted_cross_sum(Wc, sig_a[:, :n] - x_pred, Z - z_pred)
216 S = self._sym_psd(S)
217 else:
218 gamma, Wm, Wc = self._ut_weights(n)
219 sig = self._sigmas(x_pred, P_pred, gamma)
220 Z = np.array([self._h_call(model, sp, t=t) for sp in sig])
221 z_pred = np.einsum("i,ij->j", Wm, Z)
222 S = self._weighted_outer_sum(Wc, Z, z_pred)
223 Pxz = self._weighted_cross_sum(Wc, sig - x_pred, Z - z_pred)
224 M = self._M(model, x_pred.reshape(-1, 1), t)
225 S = self._sym_psd(S + self._sym_psd(M @ self._R(model, t) @ M.T))
226
227 # Multi-rate: restrict the update to the available measurement channels.
228 if not keep.all():
229 idx = np.where(keep)[0]
230 S = S[np.ix_(idx, idx)]
231 Pxz = Pxz[:, idx]
232 z = z[idx]
233 z_pred = z_pred[idx]
234
235 K = Pxz @ np.linalg.pinv(S)
236 x_new = (x_pred + K @ (z - z_pred)).reshape(-1, 1)
237 P_new = self._sym_psd(P_pred - K @ S @ K.T)
238 return x_new, P_new
