50class CalorimeterSampler:
51 """
52 Monte Carlo shower sampler for a CalorimeterGeometry.
53
54 Generates synthetic calorimeter hits without any external framework.
55 Transverse energy is distributed over cells by integrating a 2D Gaussian
56 profile using the error function — giving a physically realistic, dense
57 hit cluster per layer rather than a handful of randomly placed points.
58 """
59
60 def __init__(self, geometry: CalorimeterGeometry):
61 self._geo = geometry
62
63 def sample(self, E_GeV: float, particle: str = "gamma",
64 x0: float = 0., y0: float = 0.,
65 tx: float = 0., ty: float = 0.,
66 E_threshold_GeV: float = 1e-4,
67 seed=None) -> List[dict]:
68 """
69 Simulate one particle shower and return hit list.
70
71 Each active layer receives a fraction of the total energy (from the
72 longitudinal shower profile). That energy is then split across cells
73 by integrating a 2D Gaussian over each cell area (erf-based) — so the
74 hit pattern is physically correct and dense, not randomly scattered.
75
76 Parameters
77 ----------
78 E_GeV : particle kinetic energy [GeV]
79 particle : "gamma" (EM) or "neutral" / "neutron" (hadronic)
80 x0, y0 : entry transverse position [mm]
81 tx, ty : track slopes [mm/mm]
82 E_threshold_GeV : minimum cell energy to record (noise floor)
83 seed : random seed (int or None)
84
85 Returns
86 -------
87 list of {x, y, z, E} — multiple hits per active layer
88 """
89 from scipy.special import erf as _erf
90 _sqrt2 = np.sqrt(2.)
91
92 rng = np.random.default_rng(seed)
93 geo = self._geo
94 if geo.n_active == 0:
95 raise ValueError("Geometry has no active layers.")
96
97 if particle == "gamma":
98 fracs = _em_fracs_from_depth(geo.t_X0_active, geo.dt_X0_active, E_GeV)
99 # 2D Gaussian σ so that 90% falls within Molière radius R_M:
100 # P(r < R_M) = 1 - exp(-R_M²/2σ²) = 0.9 → σ = R_M / sqrt(2 ln 10)
101 sigma_t = geo.sigma_t_em_mm
102 sigma_s = geo.sigma_sampling_em
103 elif particle in ("neutral", "neutron", "hadron"):
104 fracs = _had_fracs_from_depth(geo.t_lam_active, geo.dt_lam_active, E_GeV)
105 sigma_t = geo.sigma_t_had_mm # 90% lat. containment ≈ 0.5 λ_int
106 sigma_s = geo.sigma_sampling_had
107 else:
108 raise ValueError(f"particle must be 'gamma' or 'neutral', got {particle!r}")
109
110 hits = []
111 for i, layer in enumerate(geo._active_layers):
112 E_dep = float(E_GeV * fracs[i])
113 if E_dep < 1e-9:
114 continue
115
116 # Sampling fluctuation at layer level: a gamma draw with mean E_dep and
117 # variance σ_s²·E_dep, so the layers add up to σ_E/E = σ_s/√E. It is never
118 # negative, where a Gaussian would have to be clipped at zero and a small
119 # deposit (σ_s·√E_dep > E_dep) would come out too large on average.
120 if sigma_s > 0 and E_dep > 1e-9:
121 E_dep = float(rng.gamma(E_dep / sigma_s ** 2, sigma_s ** 2))
122 if E_dep < E_threshold_GeV * 0.01:
123 continue
124
125 # Track position at this layer (slope propagated from calorimeter entrance)
126 dz = geo.z_active[i] - geo._z_start
127 x_c = x0 + tx * dz
128 y_c = y0 + ty * dz
129 cs = layer.cell_size_mm
130 z_c = float(geo.z_active[i])
131
132 # Grid of cell centres covering shower core (±3σ, at least ±1.5 cell)
133 r_cut = max(3. * sigma_t, 1.5 * cs)
134 x_lo = np.floor((x_c - r_cut) / cs) * cs + cs / 2
135 x_hi = x_c + r_cut + cs
136 y_lo = np.floor((y_c - r_cut) / cs) * cs + cs / 2
137 y_hi = y_c + r_cut + cs
138
139 xs_c = np.arange(x_lo, x_hi, cs)
140 ys_c = np.arange(y_lo, y_hi, cs)
141
142 for xi in xs_c:
143 # fraction of 1D Gaussian in x-strip [xi-cs/2, xi+cs/2]
144 fx = 0.5 * (_erf((xi + cs/2 - x_c) / (sigma_t * _sqrt2)) -
145 _erf((xi - cs/2 - x_c) / (sigma_t * _sqrt2)))
146 if fx < 1e-5:
147 continue
148 for yi in ys_c:
149 fy = 0.5 * (_erf((yi + cs/2 - y_c) / (sigma_t * _sqrt2)) -
150 _erf((yi - cs/2 - y_c) / (sigma_t * _sqrt2)))
151 frac_cell = float(fx * fy)
152 if frac_cell < 1e-5:
153 continue
154 E_cell = E_dep * frac_cell
155 if E_cell >= E_threshold_GeV:
156 hits.append({"x": float(xi), "y": float(yi),
157 "z": z_c, "E": float(E_cell)})
158 return hits

162 def show_shower_diagnostics(self, hits: List[dict], E_true: float,
163 particle: str = "gamma"):
164 """
165 Three-panel diagnostic to verify shower description:
166 1. Longitudinal profile dE/dz vs theory (Gamma distribution)
167 2. Transverse profile dE/dr at shower maximum vs Gaussian theory
168 3. Energy containment fraction vs depth (cumulative)
169
170 Parameters
171 ----------
172 hits : hit list from sample()
173 E_true : true particle energy [GeV]
174 particle : "gamma" or "neutral"
175 """
176 import matplotlib.pyplot as plt
177 import matplotlib.gridspec as gridspec
178 from scipy.special import erf as _erf
179
180 geo = self._geo
181 if not hits:
182 raise ValueError("Empty hit list.")
183
184 zs = np.array([h["z"] for h in hits])
185 es = np.array([h["E"] for h in hits])
186 xs = np.array([h["x"] for h in hits])
187 ys = np.array([h["y"] for h in hits])
188
189 # --- per-layer sums ---
190 z_unique = np.array(sorted(set(float(z) for z in zs)))
191 E_by_z = np.array([es[zs == z].sum() for z in z_unique])
192 E_total = E_by_z.sum()
193
194 # --- theory profile ---
195 # z_fine in absolute z; depth_fine = distance from calorimeter entrance.
196 # em/hadronic_shower_profile expect depth in mm, NOT absolute z.
197 z_fine = np.linspace(geo._z_start, z_unique[-1] + 10, 500)
198 dz_fine_v = float(z_fine[1] - z_fine[0])
199 depth_fine = z_fine - geo._z_start # [0, total_length_mm+10]
200 dz_arr = np.full_like(depth_fine, dz_fine_v)
201
202 if particle == "gamma":
203 fracs_th = em_shower_profile(depth_fine, dz_arr, E_true,
204 X0_mm=geo.X0_eff_mm)
205 sigma_t = geo.sigma_t_em_mm
206 depth_label, depth_arr, dt_arr = "X₀", geo.t_X0_active, geo.dt_X0_active
207 else:
208 fracs_th = hadronic_shower_profile(depth_fine, dz_arr, E_true,
209 lambda_mm=geo.lambda_eff_mm)
210 sigma_t = geo.sigma_t_had_mm
211 depth_label, depth_arr, dt_arr = "λ", geo.t_lam_active, geo.dt_lam_active
212
213 dz_scale = (z_unique[-1] - z_unique[0]) / len(z_unique) if len(z_unique) > 1 else 1.
214 # dE/dz theory [GeV/mm]: fraction per fine bin × E_true / bin_width
215 E_theory = fracs_th * E_true / dz_fine_v
216
217 # --- shower max layer ---
218 i_max = int(np.argmax(E_by_z))
219 z_max = z_unique[i_max]
220 mask_max = zs == z_max
221 xs_max = xs[mask_max]; ys_max = ys[mask_max]; es_max = es[mask_max]
222
223 # Transverse radial profile at shower max
224 x_c = float(np.average(xs_max, weights=es_max))
225 y_c = float(np.average(ys_max, weights=es_max))
226 rs = np.sqrt((xs_max - x_c) ** 2 + (ys_max - y_c) ** 2)
227 r_max = max(rs.max() if len(rs) > 0 else sigma_t * 3, sigma_t * 3)
228 r_fine = np.linspace(0, r_max, 200)
229 E_layer_max = float(es_max.sum())
230 # Ring-integrated 2D Gaussian: dE/dr = E * r/σ² * exp(-r²/2σ²)
231 E_ring_theory = (E_layer_max * r_fine / sigma_t**2
232 * np.exp(-r_fine**2 / (2 * sigma_t**2)))
233 # Cumulative containment
234 cum_theory = 1. - np.exp(-r_fine**2 / (2 * sigma_t**2))
235
236 # --- depth cumulative from active layers ---
237 depth_cum_data = np.cumsum(E_by_z) / max(E_total, 1e-9)
238 depth_cum_theory = np.cumsum(fracs_th)
239
240 fig = plt.figure(figsize=(15, 4.5))
241 gs = gridspec.GridSpec(1, 3, figure=fig, wspace=0.38)
242
243 # ── 1. Longitudinal profile ──────────────────────────────────────────
244 ax1 = fig.add_subplot(gs[0])
245 bar_w = dz_scale * 0.65
246 ax1.bar(z_unique, E_by_z / dz_scale, width=bar_w,
247 color="#1E88E5" if particle == "gamma" else "#43A047",
248 edgecolor="k", lw=0.5, alpha=0.85, label="Simulated hits")
249 ax1.plot(z_fine, E_theory, "r-", lw=2,
250 label="Theory (Gamma profile)")
251 ax1.axvline(z_max, color="orange", lw=1.2, ls="--", label=f"Shower max z={z_max:.0f}")
252 ax1.set_xlabel("z [mm]"); ax1.set_ylabel("dE/dz [GeV/mm]")
253 ax1.set_title("Longitudinal Profile")
254 ax1.legend(fontsize=8); ax1.grid(alpha=0.25)
255
256 # ── 2. Transverse profile at shower max ──────────────────────────────
257 ax2 = fig.add_subplot(gs[1])
258 # bin hits by radius
259 r_edges = np.linspace(0, r_max * 1.05, 12)
260 r_mids = 0.5 * (r_edges[:-1] + r_edges[1:])
261 dr = np.diff(r_edges)
262 E_ring = np.array([es_max[(rs >= r_edges[k]) & (rs < r_edges[k+1])].sum()
263 for k in range(len(r_edges)-1)])
264 ax2.bar(r_mids, E_ring / dr, width=dr * 0.8,
265 color="#1E88E5" if particle == "gamma" else "#43A047",
266 edgecolor="k", lw=0.5, alpha=0.85, label="Simulated (shower max)")
267 ax2.plot(r_fine, E_ring_theory, "r-", lw=2, label=f"Gaussian σ={sigma_t:.1f} mm")
268 ax2.axvline(sigma_t, color="purple", lw=1, ls=":", label=f"σ_t = {sigma_t:.1f} mm")
269 ax2.axvline(geo.moliere_mm if particle == "gamma" else geo.lambda_eff_mm * 0.5,
270 color="green", lw=1, ls=":", label="R₉₀%")
271 ax2.set_xlabel("r [mm]"); ax2.set_ylabel("dE/dr [GeV/mm]")
272 ax2.set_title(f"Transverse Profile (z={z_max:.0f} mm)")
273 ax2.legend(fontsize=8); ax2.grid(alpha=0.25)
274
275 # ── 3. Energy containment ────────────────────────────────────────────
276 ax3 = fig.add_subplot(gs[2])
277 rs_all = np.sqrt((xs - x_c) ** 2 + (ys - y_c) ** 2)
278 r_sort = np.sort(rs_all)
279 e_sort = np.array([es[np.argsort(rs_all)][:k+1].sum() for k in range(len(rs_all))])
280 if E_total > 0:
281 ax3.plot(r_sort, e_sort / E_total, "bo-", ms=3, lw=1.5, label="Simulated")
282 ax3.plot(r_fine, cum_theory, "r-", lw=2, label="2D Gaussian theory")
283 ax3.axhline(0.9, color="green", lw=1, ls="--", label="90% (Molière)")
284 ax3.axvline(geo.moliere_mm if particle == "gamma" else geo.lambda_eff_mm * 0.5,
285 color="green", lw=1, ls=":")
286 ax3.set_xlabel("r [mm]"); ax3.set_ylabel("Fraction contained")
287 ax3.set_title("Lateral Energy Containment")
288 ax3.set_ylim(0, 1.05); ax3.legend(fontsize=8); ax3.grid(alpha=0.25)
289
290 ptag = "gamma" if particle == "gamma" else "neutral"
291 fig.suptitle(f"Shower Diagnostics — {E_true:.0f} GeV {ptag} "
292 f"(E_reco = {E_total:.2f} GeV)", fontweight="bold")

298def _gamma_profile(t_centers, dt, a, b):
299 """Gamma-distribution shower fractions in depth units."""
300 t_lo = np.maximum(0., t_centers - 0.5 * dt)
301 t_hi = t_centers + 0.5 * dt
302 fracs = np.clip(sps.gammainc(a, b * t_hi) - sps.gammainc(a, b * t_lo), 0., 1.)
303 total = fracs.sum()
304 return fracs / total if total > 1e-9 else np.ones(len(t_centers)) / len(t_centers)

307def _em_fracs_from_depth(t_X0_centers, dt_X0, E_GeV, Ec_GeV=0.010,
308 a_EM=0.50, b_EM=0.50):
309 """EM shower fractions from cumulative X₀ depth arrays."""
310 a = max(0.1, a_EM * np.log(max(float(E_GeV), 1e-6) / max(Ec_GeV, 1e-9)))
311 return _gamma_profile(np.asarray(t_X0_centers), np.asarray(dt_X0), a, b_EM)

339def build_measurements(hits: List[dict]):
340 """
341 Collapse hit list into per-layer (x̄, ȳ, E_dep) measurements.
342
343 Returns
344 -------
345 z_layers : ndarray (N,)
346 R_xy : list of (2,2) ndarrays — per-layer transverse covariance
347 meas : ndarray (N, 3) — columns: [x̄, ȳ, E_dep]
348 """
349 if not hits:
350 raise ValueError("Empty hit list passed to build_measurements().")
351
352 layers = {}
353 for h in hits:
354 layers.setdefault(float(h["z"]), []).append(h)
355
356 z_sorted, meas_list, R_list = sorted(layers), [], []
357 for z in z_sorted:
358 hs = layers[z]
359 es = np.array([float(h["E"]) for h in hs])
360 xs = np.array([float(h["x"]) for h in hs])
361 ys = np.array([float(h["y"]) for h in hs])
362 E_w = es.sum() if es.sum() > 0. else 1.
363 xbar = np.dot(es, xs) / E_w
364 ybar = np.dot(es, ys) / E_w
365 sxx = np.dot(es, (xs - xbar) ** 2) / E_w
366 syy = np.dot(es, (ys - ybar) ** 2) / E_w
367 sxy = np.dot(es, (xs - xbar) * (ys - ybar)) / E_w
368 if len(hs) == 1:
369 sxx = syy = sxy = 0.
370 e2 = float(np.dot(es, es))
371 neff = max(1.0, (E_w ** 2 / e2) if e2 > 0. else 1.0)
372 meas_list.append([xbar, ybar, es.sum()])
373 R_list.append(np.array([[sxx / neff, sxy / neff],
374 [sxy / neff, syy / neff]]))
375
376 return np.asarray(z_sorted), R_list, np.asarray(meas_list)
