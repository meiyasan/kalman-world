27 // What a power-of-two transform of length n needs and that does not depend
28 // on the data: the bit-reversal permutation and the twiddle factors
29 // e^{-2 pi i k / n}, k < n/2. Built once per length and per thread.
30 struct Plan
31 {
32 std::vector<std::size_t> reversed;
33 std::vector<double> cosine, sine;
34 };
35
36 const Plan &PlanFor(std::size_t n)
37 {
38 thread_local std::unordered_map<std::size_t, Plan> plans;
39 const auto found = plans.find(n);
40 if (found != plans.end()) return found->second;
41
42 Plan plan;
43 plan.reversed.resize(n);
44 for (std::size_t i = 1, j = 0; i < n; ++i)
45 {
46 std::size_t bit = n >> 1;
47 for (; j & bit; bit >>= 1) j ^= bit;
48 j ^= bit;
49 plan.reversed[i] = j;
50 }
51 plan.cosine.resize(n / 2);
52 plan.sine.resize(n / 2);
53 for (std::size_t k = 0; k < n / 2; ++k)
54 {
55 const double angle = -2. * kPi * static_cast<double>(k) / static_cast<double>(n);
56 plan.cosine[k] = std::cos(angle);
57 plan.sine[k] = std::sin(angle);
58 }
59 return plans.emplace(n, std::move(plan)).first->second;
60 }

62 // Iterative Cooley-Tukey, n a power of two. The butterflies are written on
63 // the real and imaginary parts: std::complex's product checks for
64 // infinities at every call, which costs more than the product.
65 void Radix2(cd *a, std::size_t n, bool inverse)
66 {
67 const Plan &plan = PlanFor(n);
68 for (std::size_t i = 1; i < n; ++i)
69 if (i < plan.reversed[i]) std::swap(a[i], a[plan.reversed[i]]);
70
71 double *x = reinterpret_cast<double *>(a); // (re, im) pairs
72 const double sign = inverse ? -1. : 1.;
73 for (std::size_t len = 2; len <= n; len <<= 1)
74 {
75 const std::size_t half = len / 2, stride = n / len;
76 for (std::size_t i = 0; i < n; i += len)
77 for (std::size_t j = 0; j < half; ++j)
78 {
79 const double wr = plan.cosine[j * stride], wi = sign * plan.sine[j * stride];
80 double *p = x + 2 * (i + j), *q = x + 2 * (i + j + half);
81 const double vr = q[0] * wr - q[1] * wi, vi = q[0] * wi + q[1] * wr;
82 q[0] = p[0] - vr;
83 q[1] = p[1] - vi;
84 p[0] += vr;
85 p[1] += vi;
86 }
87 }
88 }

90 // Lengths with factors 3 and 5 (48, 96, 100, 120 ...): split by the odd
91 // factor into shorter transforms and recombine, down to a power of two.
92 // The n-th roots of unity are kept per length and per thread.
93 const std::vector<cd> &RootsFor(std::size_t n)
94 {
95 thread_local std::unordered_map<std::size_t, std::vector<cd>> roots;
96 const auto found = roots.find(n);
97 if (found != roots.end()) return found->second;
98 std::vector<cd> table(n);
99 for (std::size_t k = 0; k < n; ++k)
100 {
101 const double angle = -2. * kPi * static_cast<double>(k) / static_cast<double>(n);
102 table[k] = cd(std::cos(angle), std::sin(angle));
103 }
104 return roots.emplace(n, std::move(table)).first->second;
105 }

109 bool IsSmooth(std::size_t n)
110 {
111 for (std::size_t p : {std::size_t(2), std::size_t(3), std::size_t(5)})
112 while (n % p == 0) n /= p;
113 return n == 1;
114 }

116 // out[0..n) = transform of in[0], in[stride], ..., in[(n-1) stride].
117 void MixedRadix(const cd *in, std::size_t stride, cd *out, std::size_t n, bool inverse)
118 {
119 if (IsPowerOfTwo(n))
120 {
121 for (std::size_t k = 0; k < n; ++k) out[k] = in[k * stride];
122 if (n > 1) Radix2(out, n, inverse);
123 return;
124 }
125 const std::size_t p = OddFactor(n), m = n / p;
126 std::vector<cd> parts(n);
127 for (std::size_t r = 0; r < p; ++r) MixedRadix(in + r * stride, stride * p, parts.data() + r * m, m, inverse);
128
129 const std::vector<cd> &roots = RootsFor(n);
130 const double *w = reinterpret_cast<const double *>(roots.data());
131 const double *y = reinterpret_cast<const double *>(parts.data());
132 const double sign = inverse ? -1. : 1.;
133 for (std::size_t q = 0; q < p; ++q)
134 for (std::size_t k = 0; k < m; ++k)
135 {
136 const std::size_t index = k + q * m;
137 double re = y[2 * k], im = y[2 * k + 1]; // r = 0
138 for (std::size_t r = 1; r < p; ++r)
139 {
140 const std::size_t t = (r * index) % n;
141 const double wr = w[2 * t], wi = sign * w[2 * t + 1];
142 const double yr = y[2 * (r * m + k)], yi = y[2 * (r * m + k) + 1];
143 re += yr * wr - yi * wi;
144 im += yr * wi + yi * wr;
145 }
146 out[index] = cd(re, im);
147 }
148 }

150 // Bluestein, for any other length: x_k e^{-i pi k^2 / n} convolved with the
151 // chirp, through power-of-two transforms of length m >= 2n - 1. The chirp
152 // and the transform of its mirror are kept per length and direction.
153 struct Chirp
154 {
155 std::size_t m = 0;
156 std::vector<cd> chirp; // (n)
157 std::vector<cd> kernel; // (m) transform of the mirrored conjugate chirp
158 };
159
160 const Chirp &ChirpFor(std::size_t n, bool inverse)
161 {
162 thread_local std::unordered_map<std::size_t, Chirp> chirps[2];
163 auto &table = chirps[inverse ? 1 : 0];
164 const auto found = table.find(n);
165 if (found != table.end()) return found->second;
166
167 Chirp c;
168 c.m = 1;
169 while (c.m < 2 * n - 1) c.m <<= 1;
170 const double sign = inverse ? 1. : -1.;
171 c.chirp.resize(n);
172 for (std::size_t k = 0; k < n; ++k)
173 {
174 // k^2 mod 2n keeps the angle small for long transforms.
175 const double phase = sign * kPi * static_cast<double>((k * k) % (2 * n)) / static_cast<double>(n);
176 c.chirp[k] = cd(std::cos(phase), std::sin(phase));
177 }
178 c.kernel.assign(c.m, cd(0., 0.));
179 c.kernel[0] = std::conj(c.chirp[0]);
180 for (std::size_t k = 1; k < n; ++k) c.kernel[k] = c.kernel[c.m - k] = std::conj(c.chirp[k]);
181 Radix2(c.kernel.data(), c.m, false);
182 return table.emplace(n, std::move(c)).first->second;
183 }

185 void Bluestein(std::vector<cd> &a, bool inverse)
186 {
187 const std::size_t n = a.size();
188 const Chirp &c = ChirpFor(n, inverse);
189 const std::size_t m = c.m;
190
191 thread_local std::vector<cd> A;
192 A.assign(m, cd(0., 0.));
193 for (std::size_t k = 0; k < n; ++k) A[k] = a[k] * c.chirp[k];
194 Radix2(A.data(), m, false);
195 double *x = reinterpret_cast<double *>(A.data());
196 const double *y = reinterpret_cast<const double *>(c.kernel.data());
197 for (std::size_t k = 0; k < m; ++k)
198 {
199 const double re = x[2 * k] * y[2 * k] - x[2 * k + 1] * y[2 * k + 1];
200 const double im = x[2 * k] * y[2 * k + 1] + x[2 * k + 1] * y[2 * k];
201 x[2 * k] = re;
202 x[2 * k + 1] = im;
203 }
204 Radix2(A.data(), m, true);
205 const double norm = 1. / static_cast<double>(m);
206 for (std::size_t k = 0; k < n; ++k) a[k] = A[k] * norm * c.chirp[k];
207 }
208 }

210 void Transform(std::vector<cd> &x, bool inverse)
211 {
212 const std::size_t n = x.size();
213 if (n <= 1) return;
214 if (IsPowerOfTwo(n)) Radix2(x.data(), n, inverse);
215 else if (IsSmooth(n))
216 {
217 thread_local std::vector<cd> scratch;
218 scratch.resize(n);
219 MixedRadix(x.data(), 1, scratch.data(), n, inverse);
220 std::copy(scratch.begin(), scratch.end(), x.begin());
221 }
222 else Bluestein(x, inverse);
223 if (inverse)
224 {
225 const double norm = 1. / static_cast<double>(n);
226 for (cd &v : x) v *= norm;
227 }
228 }
