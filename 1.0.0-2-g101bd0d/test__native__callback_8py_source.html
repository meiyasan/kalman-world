24class Pendulum(ExtendedKalmanModel):
25 """Damped pendulum written in Python only; the angle is observed."""
26
27 def __init__(self, **kw):
28 self.nnnn, self.mmmm, self.lll = 2, 1, 0
29 self.dtypedtype = np.float64
30 super().__init__("pendulum", Q=np.diag([1e-6, 1e-4]), R=np.array([[1e-2]]), L=np.eye(2),
31 fs=50.0, x0=np.array([[0.5], [0.0]]), P0=np.eye(2), **kw)
32
33 def f(self, x, u=None, w=None, t=None):
34 th, om = np.asarray(x, float).reshape(-1)
35 dt = 1.0 / self.fsfsfs
36 return np.array([[th + dt * om], [om + dt * (-9.81 * np.sin(th) - 0.1 * om)]])
37
38 def h(self, x, v=None, t=None):
39 return np.asarray(x, float).reshape(-1)[:1].reshape(1, 1)

211class _NoisyPendulum(Pendulum):
212 """The pendulum with its noise inside the maps: a random torque, and a
213 sensor error that grows with the angle."""
214
215 def f(self, x, u=None, w=None, t=None):
216 out = Pendulum.f(self, x)
217 if w is not None:
218 w = np.asarray(w, float).reshape(-1)
219 out[1, 0] += w[1] * np.cos(float(np.asarray(x).reshape(-1)[0]))
220 out[0, 0] += w[0]
221 return out
222
223 def h(self, x, v=None, t=None):
224 out = Pendulum.h(self, x)
225 if v is not None:
226 out = out + float(np.asarray(v).reshape(-1)[0]) * (1.0 + 0.5 * out)
227 return out

231 """A pendulum trajectory with process noise, and noisy readings of its angle."""
232 rng = np.random.default_rng(seed)
233 model = Pendulum()

235 truth, z = [], []

237 x = model.f(x) + rng.multivariate_normal(np.zeros(2), model._Q).reshape(2, 1)

410 @staticmethod
411 def _measurements(n=200, seed=4):
412 rng = np.random.default_rng(seed)
413 t = np.arange(n) / 50.0
414 z = 0.4 * np.cos(3.0 * t) * np.exp(-0.3 * t) + 0.05 * rng.standard_normal(n)
415 z[60:70] = np.nan # a gap in the sensor
416 return z.reshape(-1, 1)
417
418 @staticmethod
419 def _policy(calls):
420 def policy(x, P, t, nu=None):
421 calls.append((np.shape(x), np.shape(P), np.shape(nu)))
422 th, om = np.asarray(x).reshape(-1)
423 feel = 0.0 if nu is None or not np.all(np.isfinite(nu)) else float(np.asarray(nu).reshape(-1)[0])
424 return np.array([[-4.0 * th - 1.5 * om + 0.2 * feel]])
425 return policy

427 def _both(self, engine_kind):
428 from modules.framework.filter import KalmanFilter
429 zk = self._measurements()
430 out = []
431 for reference in (True, False):
432 calls = []
433 f = KalmanFilter(DrivenPendulum(), engine=engine_kind())
434 if reference:
435 with native.python_reference():
436 f.forward(zk, self._policy(calls), record=False)
437 else:
438 f.forward(zk, self._policy(calls), record=False)
439 out.append((f, calls))
440 return out
