36 bool IsIdentity(const Matrix &M)
37 {
38 if (M.Rows() != M.Cols()) return false;
39 for (std::size_t i = 0; i < M.Rows(); ++i)
40 for (std::size_t j = 0; j < M.Cols(); ++j)
41 if (M(i, j) != (i == j ? 1. : 0.)) return false;
42 return true;
43 }

45 bool Same(const Matrix &A, const Matrix &B)
46 {
47 if (A.Rows() != B.Rows() || A.Cols() != B.Cols()) return false;
48 for (std::size_t i = 0; i < A.Rows(); ++i)
49 for (std::size_t j = 0; j < A.Cols(); ++j)
50 if (A(i, j) != B(i, j)) return false;
51 return true;
52 }

88 std::string Pack(const std::vector<Matrix> &table)
89 {
90 std::string bytes;
91 for (const auto &M : table)
92 for (std::size_t i = 0; i < M.Rows(); ++i)
93 for (std::size_t j = 0; j < M.Cols(); ++j)
94 {
95 const double v = M(i, j);
96 bytes.append(reinterpret_cast<const char *>(&v), sizeof v);
97 }
98 return bytes;
99 }

101 std::vector<Matrix> Unpack(const std::string &bytes, std::size_t steps, std::size_t rows, std::size_t cols,
102 const std::string &what)
103 {
104 if (bytes.size() != steps * rows * cols * sizeof(double))
105 Bad(what + " holds " + std::to_string(bytes.size()) + " bytes, its shape asks for "
106 + std::to_string(steps * rows * cols * sizeof(double)) + ".");
107 std::vector<Matrix> table(steps, Matrix(rows, cols, 0.));
108 const char *at = bytes.data();
109 for (auto &M : table)
110 for (std::size_t i = 0; i < rows; ++i)
111 for (std::size_t j = 0; j < cols; ++j)
112 {
113 std::memcpy(&M(i, j), at, sizeof(double));
114 at += sizeof(double);
115 }
116 return table;
117 }

150 ModelFile ModelFile::Export(const Model &model, const Options &options)
151 {
152 const StructureKind structure = model.Structure();
153 const double t0 = model.T0();
154 const std::size_t n = model.NState(), m = model.NMeasure(), l = model.NControl();
155
156 std::string kind = "complete";
157 if (options.linearize) kind = "linearization";
158 else if (structure == StructureKind::Maps) kind = "reference";
159
160 if (kind == "reference" && options.registered.empty())
161 Bad("'" + model.Name() + "' is given by maps (code), which a file cannot hold. It can be written as a "
162 "reference to its registered name (give Options::registered), or as a linearization "
163 "(Options::linearize).");
164 if (kind == "complete" && structure == StructureKind::TimeVarying && options.steps == 0)
165 Bad("'" + model.Name() + "' has matrices that change with time: say over how many samples to "
166 "tabulate them (Options::steps), or ask for the matrices at the first sample only "
167 "(Options::linearize).");
168
169 Json document = Json::Object();
170 document.Set("format", kFormat).Set("format_version", kVersion).Set("kind", kind)
171 .Set("generator", std::string("libKalman ") + KALMAN_VERSION);
172
173 Json origin = Json::Object();
174 if (!options.registered.empty()) origin.Set("model", options.registered);
175 if (!options.plugin.empty()) origin.Set("plugin", options.plugin);
176 origin.Set("title", model.Title()).Set("description", model.Description()).Set("version", model.Version());
177 document.Set("origin", options.origin.IsObject() ? options.origin : origin);
178
179 document.Set("structure", kind == "linearization" ? ToString(StructureKind::Matrices) : ToString(structure));
180 document.Set("dynamics", ToString(model.Dynamics()));
181 document.Set("state", Components(model.StateComponents()));
182 document.Set("measurement", Components(model.MeasurementComponents()));
183
184 // ---- the spec, in the form the JSON API takes ----------------------
185 Json spec = Json::Object(), dynamics = Json::Object(), noise = Json::Object(), initial = Json::Object();
186 ModelFile file;
187
188 if (kind == "reference")
189 {
190 spec.Set("base", options.registered);
191 Json parameters = Json::Object();
192 for (const auto &name : model.ParameterNames()) parameters.Set(name, model.Parameter(name));
193 for (const auto &name : model.ArrayNames()) parameters.Set(name, Numbers(model.Array(name)));
194 dynamics.Set("params", parameters);
195 }
196 else
197 {
198 const bool continuous = kind == "complete" && !model.Generator().Empty() && !model.NoiseIntensity().Empty();
199 spec.Set("base", continuous ? "continuous" : "matrices");
200 spec.Set("name", options.registered.empty() ? model.Name() : options.registered);
201
202 if (kind == "linearization")
203 {
204 const Vector x = options.at.Size() ? options.at : model.X0();
205 if (x.Size() != n) Bad("the linearization state has " + std::to_string(x.Size()) + " components, the "
206 "model has " + std::to_string(n) + ".");
207 dynamics.Set("F", Json::From(model.Jacobian(x, Vector(l, 0.), t0)));
208 dynamics.Set("H", Json::From(model.JacobianH(x, t0)));
209 document.Set("linearized_at", Json::Object().Set("x", Json::From(x)).Set("t", t0));
210 }
211 else if (continuous)
212 {
213 dynamics.Set("A", Json::From(model.Generator()));
214 dynamics.Set("H", Json::From(model.H(t0)));
215 dynamics.Set("Qc", Json::From(model.NoiseIntensity()));
216 if (!model.NoiseCoupling().Empty()) dynamics.Set("Lc", Json::From(model.NoiseCoupling()));
217 }
218 else
219 {
220 dynamics.Set("F", Json::From(model.F(t0)));
221 dynamics.Set("H", Json::From(model.H(t0)));
222 }
223 if (l > 0)
224 if (const Matrix B = model.B(t0); !B.Empty()) dynamics.Set("B", Json::From(B));
225 if (const Matrix L = model.L(t0); !L.Empty() && !IsIdentity(L)) dynamics.Set("L", Json::From(L));
226 }
227
228 const bool continuousNoise = spec["base"].AsString() == "continuous";
229 if (!continuousNoise) noise.Set("Q", Json::From(model.Q(t0)));
230 noise.Set("R", Json::From(model.R(t0)));
231 initial.Set("x0", Json::From(model.X0())).Set("P0", Json::From(model.P0()));
232
233 spec.Set("fs", model.SampleRate()).Set("dynamics", dynamics).Set("noise", noise).Set("initial", initial);
234 spec.Set("labels", Json::Object().Set("states", Names(model.StateComponents()))
235 .Set("observables", Names(model.MeasurementComponents())));
236 spec.Set("bindings", Json::Object().Set("engine", kind == "reference" ? Lower(ToString(model.DefaultEngine()))
237 : std::string("linear")));
238 document.Set("spec", spec);
239
240 // ---- the tables: the matrices that change with time ------------------
241 if (kind == "complete" && structure == StructureKind::TimeVarying)
242 {
243 const double dt = model.DeltaT();
244 Json entries = Json::Object();
245 const auto tabulate = [&](const char *which, const std::function<Matrix(double)> &at)
246 {
247 std::vector<Matrix> table;
248 bool varies = false;
249 for (std::size_t k = 0; k < options.steps; ++k)
250 {
251 table.push_back(at(t0 + static_cast<double>(k) * dt));
252 varies = varies || !Same(table.back(), table.front());
253 }
254 if (!varies) return; // constant: the spec already holds it
255 Json shape = Json::Array();
256 shape.Add(options.steps).Add(table.front().Rows()).Add(table.front().Cols());
257 entries.Set(which, Json::Object().Set("file", TablePath(which)).Set("shape", shape));
258 file.fTables.Set(TablePath(which), Pack(table));
259 };
260 tabulate("F", [&model](double t) { return model.F(t); });
261 tabulate("H", [&model](double t) { return model.H(t); });
262 tabulate("Q", [&model](double t) { return model.Q(t); });
263 tabulate("R", [&model](double t) { return model.R(t); });
264 document.Set("tables", Json::Object().Set("steps", options.steps).Set("t0", t0)
265 .Set("dtype", "float64, little-endian, row-major")
266 .Set("entries", entries));
267 }
268 (void) m;
269
270 file.fDocument = std::move(document);
271 return file;
272 }

280 std::unique_ptr<Model> ModelFile::Import() const
281 {
282 // A bare spec (no envelope) is accepted: it is what the JSON API takes.
283 const bool envelope = fDocument.Has("spec");
284 if (envelope)
285 {
286 if (!fDocument.Has("format") || fDocument["format"].AsString("format") != kFormat)
287 Bad(std::string("not a model file: \"format\" is not \"") + kFormat + "\".");
288 const double version = fDocument["format_version"].AsNumber("format_version");
289 if (version > kVersion)
290 Bad("the file is of format version " + std::to_string(static_cast<int>(version)) + ", this library "
291 "reads up to " + std::to_string(kVersion) + ".");
292 }
293 const Json &spec = envelope ? fDocument["spec"] : fDocument;
294 if (!spec.IsObject()) Bad("the document has no spec.");
295
296 const std::string base = spec.Has("base") ? spec["base"].AsString("spec.base") : std::string("matrices");
297 const Json &dynamics = spec["dynamics"], &noise = spec["noise"], &initial = spec["initial"];
298 std::unique_ptr<Model> model;
299
300 if (base == "matrices" || base == "continuous")
301 {
302 const bool continuous = dynamics.Has("A");
303 if (continuous && dynamics.Has("F")) Bad("the spec gives both dynamics.F and dynamics.A.");
304 if (!continuous && !dynamics.Has("F")) Bad("a matrices spec needs dynamics.F (or dynamics.A).");
305 if (!dynamics.Has("H")) Bad("a matrices spec needs dynamics.H.");
306 const Matrix first = (continuous ? dynamics["A"] : dynamics["F"]).AsMatrix(continuous ? "dynamics.A" : "dynamics.F");
307 const Matrix H = dynamics["H"].AsMatrix("dynamics.H");
308 const Matrix B = dynamics.Has("B") ? dynamics["B"].AsMatrix("dynamics.B") : Matrix();
309 const std::size_t n = first.Rows();
310
311 auto tabulated = std::make_unique<Models::Tabulated>(
312 spec.Has("name") ? spec["name"].AsString("spec.name") : std::string("model"), n, H.Rows(), B.Cols());
313 if (spec.Has("fs")) tabulated->SetSampleRate(spec["fs"].AsNumber("spec.fs"));
314 tabulated->SetH(H);
315 if (!B.Empty()) tabulated->SetB(B);
316 if (dynamics.Has("L")) tabulated->SetL(dynamics["L"].AsMatrix("dynamics.L"));
317 if (continuous)
318 {
319 tabulated->SetGenerator(first);
320 if (dynamics.Has("Qc"))
321 tabulated->SetNoiseIntensity(dynamics["Qc"].AsMatrix("dynamics.Qc"),
322 dynamics.Has("Lc") ? dynamics["Lc"].AsMatrix("dynamics.Lc") : Matrix());
323 }
324 else tabulated->SetF(first);
325
326 if (envelope)
327 tabulated->SetComponents(ReadComponents(fDocument["state"], "state"),
328 ReadComponents(fDocument["measurement"], "measurement"));
329
330 if (envelope && fDocument.Has("tables"))
331 {
332 const Json &tables = fDocument["tables"];
333 const std::size_t steps = static_cast<std::size_t>(tables["steps"].AsNumber("tables.steps"));
334 if (tables.Has("t0")) tabulated->SetT0(tables["t0"].AsNumber("tables.t0"));
335 for (const auto &entry : tables["entries"].Members("tables.entries"))
336 {
337 const std::string where = entry.second["file"].AsString("tables.entries." + entry.first + ".file");
338 const auto &shape = entry.second["shape"].Items("tables.entries." + entry.first + ".shape");
339 if (shape.size() != 3 || static_cast<std::size_t>(shape[0].AsNumber("shape")) != steps)
340 Bad("the table of " + entry.first + " does not have the shape [steps, rows, columns].");
341 if (!fTables.Has(where))
342 Bad("the document refers to '" + where + "', which is not in the file. A model with tables "
343 "is one archive (" + kExtension + "); its JSON alone is not the whole model.");
344 tabulated->SetTable(entry.first, Unpack(fTables.Get(where), steps,
345 static_cast<std::size_t>(shape[1].AsNumber("shape")),
346 static_cast<std::size_t>(shape[2].AsNumber("shape")), where));
347 }
348 }
349 model = std::move(tabulated);
350 }
351 else
352 {
353 auto &factory = Factory::Instance();
354 if (!factory.Has(base))
355 {
356 std::string hint;
357 if (envelope && fDocument["origin"].Has("python"))
358 hint = " It refers to the Python class " + fDocument["origin"]["python"].AsString("origin.python")
359 + ": open it from Python (kalman.native.open_model).";
360 else if (envelope && fDocument["origin"].Has("plugin"))
361 hint = " It was written from the plugin '" + fDocument["origin"]["plugin"].AsString("origin.plugin")
362 + "': load it first.";
363 Bad("the file refers to the model '" + base + "', which is not registered here." + hint
364 + " A reference holds a model's settings, not its code.");
365 }
366 model = factory.Create(base);
367 if (dynamics.Has("params"))
368 {
369 // Arrays first: a model may size itself from them.
370 for (const auto &p : dynamics["params"].Members("dynamics.params"))
371 if (p.second.IsArray()) model->SetArray(p.first, Doubles(p.second, "dynamics.params." + p.first));
372 for (const auto &p : dynamics["params"].Members("dynamics.params"))
373 if (!p.second.IsArray())
374 model->SetParameter(p.first, p.second.AsNumber("dynamics.params." + p.first));
375 }
376 if (spec.Has("fs"))
377 if (const double fs = spec["fs"].AsNumber("spec.fs"); fs != model->SampleRate()) model->SetSampleRate(fs);
378 }
379
380 if (noise.Has("Q")) model->SetQ(noise["Q"].AsMatrix("noise.Q"));
381 if (noise.Has("R")) model->SetR(noise["R"].AsMatrix("noise.R"));
382 if (initial.Has("x0")) model->SetInitialState(initial["x0"].AsVector("initial.x0"));
383 if (initial.Has("P0")) model->SetInitialCovariance(initial["P0"].AsMatrix("initial.P0"));
384 model->CheckShapes();
385 return model;
386 }

419 std::vector<std::string> ModelFile::Apply(Model &model, const Json &settings, const std::string &identity)
420 {
421 if (!settings.IsObject() || !settings.Has("format") || settings["format"].AsString("format") != kSettingsFormat)
422 Bad(std::string("not a settings file: \"format\" is not \"") + kSettingsFormat + "\". A settings file "
423 "holds what changes from a model file (fs, parameters, noise, initial), not a model.");
424 if (settings.Has("format_version") && settings["format_version"].AsNumber("format_version") > kVersion)
425 Bad("the settings are of a later format version than this library reads.");
426 if (settings.Has("for") && settings["for"].AsString("for") != identity)
427 Bad("these settings were written for the model '" + settings["for"].AsString("for") + "', not for '"
428 + identity + "'.");
429 const auto allowed = [](const Json &object, std::initializer_list<const char *> keys, const std::string &where)
430 {
431 for (const auto &member : object.Members(where))
432 {
433 bool known = false;
434 for (const char *k : keys) known = known || member.first == k;
435 if (!known)
436 {
437 std::string list;
438 for (const char *k : keys) list += (list.empty() ? "" : ", ") + std::string(k);
439 Bad("'" + member.first + "' is not a setting" + (where.empty() ? "" : " of " + where)
440 + " (settings are: " + list + "). A settings file cannot change what the model is.");
441 }
442 }
443 };
444 allowed(settings, {"format", "format_version", "for", "fs", "parameters", "noise", "initial"}, "");
445
446 std::vector<std::string> changed;
447 const auto differs = [](const Matrix &A, const Matrix &B) { return !Same(A, B); };
448 if (settings.Has("parameters"))
449 {
450 for (const auto &p : settings["parameters"].Members("parameters"))
451 {
452 if (!p.second.IsArray()) continue;
453 const std::vector<double> values = Doubles(p.second, "parameters." + p.first);
454 if (model.Array(p.first) == values) continue; // throws for an unknown name
455 model.SetArray(p.first, values);
456 changed.push_back(p.first);
457 }
458 for (const auto &p : settings["parameters"].Members("parameters"))
459 {
460 if (p.second.IsArray()) continue;
461 const double value = p.second.AsNumber("parameters." + p.first);
462 if (model.Parameter(p.first) == value) continue; // throws for an unknown name
463 model.SetParameter(p.first, value);
464 changed.push_back(p.first + " = " + Plain(value));
465 }
466 }
467 if (settings.Has("fs"))
468 if (const double fs = settings["fs"].AsNumber("fs"); fs != model.SampleRate())
469 {
470 // Matrices written for one step length are that step's: another rate needs
471 // the model stated in continuous time (a generator), or the code itself.
472 if (model.Structure() != StructureKind::Maps && model.Generator().Empty())
473 Bad("fs cannot change here: the matrices of this model were written for "
474 + Plain(model.SampleRate()) + " Hz. Export it again at the rate you want.");
475 model.SetSampleRate(fs);
476 changed.push_back("fs = " + Plain(fs));
477 }
478 if (settings.Has("noise"))
479 {
480 const Json &noise = settings["noise"];
481 allowed(noise, {"Q", "R"}, "noise");
482 if (noise.Has("Q"))
483 if (const Matrix Q = noise["Q"].AsMatrix("noise.Q"); differs(Q, model.Q(model.T0())))
484 {
485 model.SetQ(Q);
486 changed.push_back("Q");
487 }
488 if (noise.Has("R"))
489 if (const Matrix R = noise["R"].AsMatrix("noise.R"); differs(R, model.R(model.T0())))
490 {
491 model.SetR(R);
492 changed.push_back("R");
493 }
494 }
495 if (settings.Has("initial"))
496 {
497 const Json &initial = settings["initial"];
498 allowed(initial, {"x0", "P0"}, "initial");
499 if (initial.Has("x0"))
500 {
501 const Vector x0 = initial["x0"].AsVector("initial.x0");
502 bool same = x0.Size() == model.X0().Size();
503 for (std::size_t i = 0; same && i < x0.Size(); ++i) same = x0(i) == model.X0()(i);
504 if (!same)
505 {
506 model.SetInitialState(x0);
507 changed.push_back("x0");
508 }
509 }
510 if (initial.Has("P0"))
511 if (const Matrix P0 = initial["P0"].AsMatrix("initial.P0"); differs(P0, model.P0()))
512 {
513 model.SetInitialCovariance(P0);
514 changed.push_back("P0");
515 }
516 }
517 model.CheckShapes();
518 return changed;
519 }

521 ModelFile::Opened ModelFile::Open(const std::string &path, bool settings)
522 {
523 const ModelFile file = Load(path);
524 if (file.fDocument.Has("format") && file.fDocument["format"].AsString("format") == kSettingsFormat)
525 Bad("'" + path + "' is a settings file, not a model: open the model file it sits beside.");
526 Opened opened;
527 opened.model = file.Import();
528 opened.kind = file.Kind();
529 opened.tables = file.HasTables();
530 const std::string beside = SettingsPath(path);
531 if (settings)
532 if (std::ifstream in(beside, std::ios::binary); in)
533 {
534 std::ostringstream buffer;
535 buffer << in.rdbuf();
536 try { opened.changed = Apply(*opened.model, Json::Parse(buffer.str()), file.Identity()); }
537 catch (const std::exception &e) { Bad("'" + beside + "': " + e.what()); }
538 // A tabulated matrix is read from its table: a constant would be ignored without a word.
539 if (const auto *tabulated = dynamic_cast<const Models::Tabulated *>(opened.model.get()))
540 for (const char *which : {"Q", "R"})
541 if (!tabulated->Table(which).empty()
542 && std::find(opened.changed.begin(), opened.changed.end(), which) != opened.changed.end())
543 Bad("'" + beside + "': " + which + " changes with time in '" + path + "' (it is a table), "
544 "so one constant cannot replace it.");
545 opened.settings = beside;
546 }
547 return opened;
548 }

561 void ModelFile::Save(const std::string &path) const
562 {
563 const std::string json = ".json";
564 const bool plain = path.size() >= json.size() && path.compare(path.size() - json.size(), json.size(), json) == 0;
565 if (plain && HasTables())
566 Bad("this model has tables, so its JSON is not the whole model: write it as one archive ('"
567 + std::string(kExtension) + "'), not as '" + path + "'.");
568 const std::string reserved = std::string(kExtension) + kSettingsSuffix;
