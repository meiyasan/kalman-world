42 Field FiniteDifference::Discretize(const Field &w) const
43 {
44 CheckShape(w);
45 Field p(fNy + 2, fNx + 2, fBoundaryValue);
46 for (std::size_t i = 0; i < fNy; ++i)
47 for (std::size_t j = 0; j < fNx; ++j) p(i + 1, j + 1) = w(i, j);
48
49 auto wrap = [](std::size_t i, std::size_t n) { return (i + n) % n; };
50 for (std::size_t i = 0; i < fNy + 2; ++i)
51 for (std::size_t j = 0; j < fNx + 2; ++j)
52 {
53 const bool edge = i == 0 || j == 0 || i == fNy + 1 || j == fNx + 1;
54 if (!edge) continue;
55 if (fBoundary == Boundary::Periodic)
56 p(i, j) = w(wrap(i + fNy - 1, fNy), wrap(j + fNx - 1, fNx));
57 else if (fBoundary == Boundary::Neumann)
58 p(i, j) = w(std::min(std::max<std::size_t>(i, 1), fNy) - 1,
59 std::min(std::max<std::size_t>(j, 1), fNx) - 1);
60 // Dirichlet: already fBoundaryValue
61 }
62 return p;
63 }

65 void FiniteDifference::Gradient(const Field &w, Field &wx, Field &wy) const
66 {
67 const Field p = Discretize(w);
68 wx = Field(fNy, fNx);
69 wy = Field(fNy, fNx);
70 for (std::size_t i = 0; i < fNy; ++i)
71 for (std::size_t j = 0; j < fNx; ++j)
72 {
73 wx(i, j) = (p(i + 1, j + 2) - p(i + 1, j)) / (2. * fDx);
74 wy(i, j) = (p(i + 2, j + 1) - p(i, j + 1)) / (2. * fDy);
75 }
76 }

78 Field FiniteDifference::Laplacian(const Field &w) const
79 {
80 const Field p = Discretize(w);
81 Field out(fNy, fNx);
82 for (std::size_t i = 0; i < fNy; ++i)
83 for (std::size_t j = 0; j < fNx; ++j)
84 {
85 const double c = p(i + 1, j + 1);
86 out(i, j) = (p(i + 1, j + 2) - 2. * c + p(i + 1, j)) / (fDx * fDx)
87 + (p(i + 2, j + 1) - 2. * c + p(i, j + 1)) / (fDy * fDy);
88 }
89 return out;
90 }

123 Field FiniteDifference::RandomIC(double amp) const
124 {
125 std::normal_distribution<double> gauss(0., 1.);
126 Field f(fNy, fNx);
127 for (std::size_t i = 0; i < f.Size(); ++i) f.Data()[i] = gauss(fRandom);
128
129 const double h2 = std::pow(std::min(fDx, fDy), 2);
130 for (int sweep = 0; sweep < 8; ++sweep) // smooth by diffusion sweeps
131 {
132 Field lap = Laplacian(f);
133 lap *= 0.2 * h2;
134 f += lap;
135 }
136
137 double mean = 0., var = 0.;
138 for (std::size_t i = 0; i < f.Size(); ++i) mean += f.Data()[i];
139 mean /= static_cast<double>(f.Size());
140 for (std::size_t i = 0; i < f.Size(); ++i) var += std::pow(f.Data()[i] - mean, 2);
141 f *= amp / (std::sqrt(var / static_cast<double>(f.Size())) + 1e-12);
142 return f;
143 }
