1"""
2Scaffold a reviewable app module from a ``KalmanSpec``.
3
4:func:`scaffold` returns **Python source** for a thin ``modules/apps/<name>.py``
5— the path from an LLM-authored spec to a real app, with a *human* as the commit
6gate. It only *emits* source (never ``exec``s it); :func:`write_scaffold` writes

19def scaffold(spec: dict, class_name: str | None = None, *, credential: str | None = None) -> str:
20 """Return Python source for a ``KalmanModel``/``LinearKalmanModel`` subclass
21 built from ``spec`` (matrices or continuous-linear bases)."""
22 from .auth import authorize
23 authorize(credential, "write_models")
24
25 spec = dict(spec or {})
26 base = spec.get("base", "matrices")
27 if base not in ("matrices", "continuous"):
28 raise ValueError("scaffold supports the 'matrices'/'continuous' declarative bases; "
29 "parameterized catalog models are already importable.")
30 cls = class_name or spec.get("name") or "AuthoredModel"
31 cls = "".join(w[:1].upper() + w[1:] for w in str(cls).replace("-", "_").split("_") if w)
32 dyn = spec.get("dynamics", {}) or {}
33 noise = spec.get("noise", {}) or {}
34 init = spec.get("initial", {}) or {}
35 fs = spec.get("fs", 1.0)
36 sensor = spec.get("sensor_id", "S")
37
38 kwargs = []
39 if base == "matrices":
40 for k in ("F", "H", "B", "L", "M"):
41 if dyn.get(k) is not None:
42 kwargs.append(f"{k}=np.array({_fmt(dyn[k])})")
43 else:
44 for k in ("A", "Qc", "Lc"):
45 if dyn.get(k) is not None:
46 kwargs.append(f"{k}=np.array({_fmt(dyn[k])})")
47 for k in ("Q", "R"):
48 if noise.get(k) is not None:
49 kwargs.append(f"{k}=np.array({_fmt(noise[k])})")
50 for k in ("x0", "P0"):
51 if init.get(k) is not None:
52 kwargs.append(f"{k}=np.array({_fmt(init[k])})")
53 kwargs.append(f"fs={fs}")
54 args = ",\n ".join(kwargs)
55 parent = "LinearKalmanModel" if base == "matrices" else "KalmanModel"
56
57 return f'''"""Scaffolded from a KalmanSpec by kalman.api.scaffold — review before committing."""
58import numpy as np
59from kalman.framework import {parent}

77def write_scaffold(spec: dict, path: str, class_name: str | None = None,
78 *, credential: str | None = None) -> str:
79 """Write :func:`scaffold` source to ``path`` (a human-chosen staging location,
