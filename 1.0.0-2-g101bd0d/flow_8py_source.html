1"""2D turbulence data assimilation — Kolmogorov flow + ensemble OSSE harness.
2
3The geophysical-DA testbed: a pseudo-spectral 2D incompressible Navier–Stokes
4solver (vorticity form, Kolmogorov forcing) wrapped as a :class:`KalmanModel`
5whose state is the full vorticity *field* (n = grid²), observed at ``p`` sparse
6point probes. The natural filter is the :class:`EnsembleKalmanFilter` — the
7covariance lives in the ensemble, never densely — with Gaspari–Cohn
8localization on the periodic torus supplied by the model itself.
9
10:class:`TwinExperiment` is the OSSE (observing-system simulation experiment)
11scaffold: truth from the real solver, synthetic observations at known probe
12locations, cycled analysis, RMSE/spread diagnostics against a free-running
13control. It is the standard validation harness for any learned component
14(surrogate forecast, learned observation operator, …) before it is trusted in
15a filter; see ``kalman.apps.weather.surrogate`` for the CNN forecast surrogate.
16
17**RNG discipline** (hard-won): the harness draws the *observation* noise from
18one shared stream so every method sees literally the same measurement
19realization, while each method's engine consumes its **own** stream for the
20perturbed-obs draws — so adding or removing a method can never silently change
21another method's results.
22"""

37class NS2D(SpectralSolver):
38 """Pseudo-spectral 2D incompressible Navier–Stokes (vorticity form).
39
40 .. code-block:: text
41
42 ∂ω/∂t = −u·∇ω + ν ∇²ω + F, F = χ cos(k_f y) (Kolmogorov forcing)
43
44 Domain ``[0, 2π]²`` periodic, 2/3-rule dealiasing, RK4 time stepping.
45 State vector = vorticity field flattened, shape ``(n²,)``.
46
47 A concrete :class:`~kalman.framework.solvers.SpectralSolver` — the spectral
48 grid, ``gradient``/``velocity`` operators, RK4 ``advance`` and numpy/torch
49 backends live in the base; this subclass only fixes the **Kolmogorov
50 physics** (``rhs_hat``: advection − viscosity + forcing).
51 """
52
53 NAME = "NS2D (Kolmogorov)"
54 DESCRIPTION = "Pseudo-spectral 2D Navier–Stokes, vorticity form, Kolmogorov forcing."
55
56 def __init__(self, n=64, nu=1/70.0, kf=4, chi=1.0, dt=0.01, seed=0,
57 *, backend=None, device=None, integrator=None):
58 super().__init__(n=n, dt=dt, backend=backend, device=device, seed=seed,
59 integrator=integrator)
60 self.nu, self.kf, self.chi = nu, kf, chi
61 ny, nx = self.shape
62 y = self.Ly * np.arange(ny) / ny # forcing varies with y
63 self.F = chi * np.cos(kf * y)[:, None] * np.ones((1, nx))
64 self.Fh = np.fft.fft2(self.F)
65
66 def _vorticity_terms(self):
67 return dict(nu=self.nu, forcing_hat=self.Fh)
68
69 def rhs_hat(self, wh, t=None):
70 u, v = self._vel_spectral(wh)
71 wx, wy = self._grad_spectral(wh)
72 adv = np.fft.fft2(u*wx + v*wy) * self.dealias
73 return -adv - self.nu*self.k2*wh + self.Fh

83class KolmogorovFlowModel(NativeTwin, KalmanModel):
84 """
85 Kolmogorov-flow vorticity field as a Kalman state (n = grid², chaotic).
86
87 - ``f`` — advance the field one assimilation cycle (``cycle`` RK4 steps of
88 the pseudo-spectral solver); no dense Jacobian exists, so the model
89 allows only the ensemble and particle engines.
90 - ``h`` — sample the field at ``n_obs`` fixed probe points (+ iid noise
91 ``sig_o``); the probes are drawn once from ``seed``.
92 - ``localization()`` — Gaspari–Cohn tapers from periodic torus distances
93 between grid cells and probes (radius ``loc_radius`` grid units).
94 - ``initial_ensemble(N)`` — random climatology snapshots (large, honest
95 initial error), from a cached spin-up run.
96 - ``forecast(X)`` — batched member propagation, **the AI plug-in point**:
97 assign ``model.surrogate = fn`` (columns → columns) to swap the solver
98 for a learned forecast without touching the filter.
99
100 Validated lessons carried over from the prototype OSSE (see notebook 13):
101 a 16-member localized+inflated EnKF converges to well below the free-run
102 error; a *linear* latent space (POD/DMD) hits a representation floor and
103 is not offered here; an under-trained one-step surrogate diverges in
104 closed loop — check its open-loop rollout error first
105 (``kalman.apps.weather.surrogate.rollout_rmse``) and budget separate inflation for it.
106 """

258 def f(self, x, u=None, w=None, t=None):
259 """One assimilation cycle of the pseudo-spectral solver."""
260 x = np.asarray(x, dtype=float)
261 out = self.solver.advance(x.reshape(-1), self.cycle)
262 return out.reshape(x.shape) if x.ndim > 1 else out

264 def forecast(self, X, t=None):
265 """Batched member propagation (d, N) → (d, N); surrogate if assigned."""
266 X = np.asarray(X, dtype=float)
267 if self.surrogate is not None:
268 return np.asarray(self.surrogate(X), dtype=float)
269 out = np.empty_like(X)
270 for i in range(X.shape[1]):
271 out[:, i] = self.solver.advance(X[:, i], self.cycle)
272 return out

274 def h(self, x, v=None, t=None):
275 x = np.asarray(x, dtype=float)
276 if self.obs_type == "gradient":
277 gx, gy = self.vorticity_gradient(x.reshape(-1))
278 z = np.concatenate([gx.ravel()[self.obs_idx],
279 gy.ravel()[self.obs_idx]]).reshape(-1, 1)
280 else:
281 z = x.reshape(-1)[self.obs_idx].reshape(-1, 1)
282 return z if v is None else z + np.asarray(v, dtype=float).reshape(-1, 1)

284 def observe(self, X, t=None):
285 """Batched observation map (d, N) → (m, N). For ``obs_type='gradient'``
286 the first ``p`` rows are ∂ω/∂x at the probes, the next ``p`` are ∂ω/∂y."""
287 X = np.asarray(X, dtype=float)
288 if self.obs_type == "gradient":
289 gx, gy = self.vorticity_gradient(X) # (d, N) each
290 return np.vstack([gx[self.obs_idx], gy[self.obs_idx]])
291 return X[self.obs_idx, :]

293 def H(self, x=None, t=None):
294 """Linear observation operator (m × d). Point-selection for vorticity
295 obs; a spectral-gradient sampling operator for gradient obs (built once
296 and cached — only the ensemble/particle engines run here, which use
297 ``observe`` instead, so this serves shape-checking / inspection)."""
298 if self.obs_type == "gradient":
299 if getattr(self, "_H_grad", None) is None:
300 gx = np.zeros((self.n_probes, self.nnnnn))
301 gy = np.zeros((self.n_probes, self.nnnnn))
302 e = np.zeros(self.nnnnn)
303 for j, idx in enumerate(self.obs_idx): # columns of the operator
304 e[:] = 0.0; e[idx] = 1.0
305 ggx, ggy = self.vorticity_gradient(e)
306 gx[j] = ggx.ravel(); gy[j] = ggy.ravel()
307 self._H_grad = np.vstack([gx, gy])
308 return self._H_grad
309 H = np.zeros((self.mmmm, self.nnnnn))
310 H[np.arange(self.mmmm), self.obs_idx] = 1.0
311 return H

322 def torus_distance(self, a_idx, b_idx):
323 """Pairwise grid distance between flat indices on the periodic torus."""
324 n = self.grid
325 ay, ax = np.divmod(np.asarray(a_idx), n)
326 by, bx = np.divmod(np.asarray(b_idx), n)
327 def wrap(p, q):
328 dd = np.abs(p[:, None] - q[None, :])
329 return np.minimum(dd, n - dd)
330 return np.sqrt(wrap(ax, bx)**2 + wrap(ay, by)**2)

332 def localization(self):
333 """(rho_xy, rho_yy) Gaspari–Cohn tapers for the ensemble engine.
334
335 For ``obs_type='gradient'`` the two channels of each probe sit at the
336 same location, so the tapers are block-tiled: ``rho_xy`` duplicated
337 across the ∂x/∂y channels and ``rho_yy`` tiled 2×2."""
338 all_idx = np.arange(self.nnnnn)
339 rho_xy = gaspari_cohn(self.torus_distance(all_idx, self.obs_idx), self.loc_radius)
340 rho_yy = gaspari_cohn(self.torus_distance(self.obs_idx, self.obs_idx), self.loc_radius)
341 if self.obs_type == "gradient":
342 rho_xy = np.hstack([rho_xy, rho_xy]) # (d, 2p)
343 rho_yy = np.block([[rho_yy, rho_yy], [rho_yy, rho_yy]]) # (2p, 2p)
344 return rho_xy, rho_yy

349 def vorticity_gradient(self, field):
350 """Spectral vorticity gradient ∇ω = (∂ω/∂x, ∂ω/∂y).
351
352 The spectral ``ik`` operator itself lives on the solver (``NS2D``, the
353 single source of truth — it is the same one the dynamics use); this
354 wraps it for the DA layer: a *linear* operator on the field, so it drops
355 straight into the Kalman machinery as an observation operator
356 (:meth:`observe`) or an analysis diagnostic (:meth:`gradient_diagnostic`).
357 Accepts a flat ``(d,)`` field, a ``(g, g)`` field, or a batched ensemble
358 ``(d, N)``; returns ``(∂ω/∂x, ∂ω/∂y)`` in the matching layout."""
359 F = np.asarray(field, dtype=float)
360 if F.ndim == 2 and F.shape[0] == self.nnnnn: # batched (d, N)
361 g = self.grid
362 w = F.T.reshape(-1, g, g)
363 wh = np.fft.fft2(w, axes=(-2, -1))
364 gx = np.real(np.fft.ifft2(1j * self.solver.kx[None] * wh, axes=(-2, -1)))
365 gy = np.real(np.fft.ifft2(1j * self.solver.ky[None] * wh, axes=(-2, -1)))
366 return gx.reshape(-1, self.nnnnn).T, gy.reshape(-1, self.nnnnn).T
367 return self.solver.gradient(F) # single field

376 def velocity_sampler(self, field, region, *, speed=40.0):
377 """Return a callable ``(lon, lat) → (v_e, v_n)`` km/h that reads this
378 flow field's velocity at the geographic point.
379
380 Maps a geographic ``region = (lon_min, lon_max, lat_min, lat_max)`` onto
381 the flow's periodic box (bilinear interpolation, torus-wrapped) and
382 scales the non-dimensional velocity to ``speed`` km/h at unit magnitude.
383 This is the bridge that lets the field EnKF **supply the deep-layer
384 steering flow** to :func:`kalman.apps.weather.storms.field_steering` — the
385 environment estimate (with its ensemble uncertainty) driving the
386 typhoon tracker.
387 """
388 u, v = self.velocity(field)
389 g = self.grid
390 lo0, lo1, la0, la1 = region
391 norm = (np.abs(u) + np.abs(v)).mean() + 1e-9
392
393 def sample(lon, lat):
394 fx = ((np.asarray(lon) - lo0) / (lo1 - lo0)) % 1.0 * g
395 fy = ((np.asarray(lat) - la0) / (la1 - la0)) % 1.0 * g
396 i0 = np.floor(fx).astype(int) % g; j0 = np.floor(fy).astype(int) % g
397 i1 = (i0 + 1) % g; j1 = (j0 + 1) % g
398 ax = fx - np.floor(fx); ay = fy - np.floor(fy)
399 def bilinear(A):
400 return (A[j0, i0]*(1-ax)*(1-ay) + A[j0, i1]*ax*(1-ay)
401 + A[j1, i0]*(1-ax)*ay + A[j1, i1]*ax*ay)
402 ue = bilinear(u) / norm * speed
403 vn = bilinear(v) / norm * speed
404 return float(ue), float(vn)
405 return sample

407 def gradient_diagnostic(self, members, k_sigma=1.0):
408 """Ensemble estimate of ∇ω **with its uncertainty**.
409
410 ``members`` is the ``(d, N)`` ensemble (e.g. ``engine.members`` after a
411 run). Each member's gradient is a linear image of the member, so the
412 ensemble of gradients gives a calibrated mean field and a ±σ band —
413 the field-scale analogue of the storm tracker's covariance ellipse.
414 Returns a dict of ``(g, g)`` maps: ``gx_mean``/``gy_mean`` (components),
415 ``mag_mean`` (|∇ω| mean), ``mag_sigma`` (its ``k_sigma``-σ uncertainty)."""
416 gx, gy = self.vorticity_gradient(np.asarray(members, dtype=float)) # (d, N)
417 mag = np.sqrt(gx ** 2 + gy ** 2)
418 g = self.grid
419 return {"gx_mean": gx.mean(1).reshape(g, g),
420 "gy_mean": gy.mean(1).reshape(g, g),
421 "mag_mean": mag.mean(1).reshape(g, g),
422 "mag_sigma": (k_sigma * mag.std(1)).reshape(g, g)}

424 def climatology(self, n_snapshots=150, spinup=2500, stride=10):
425 """Spin up to the attractor and collect snapshots — cached ``(d, K)``.
426
427 The snapshots seed initial ensembles and set the climatological error
428 scale that any assimilation run must beat. The first call fixes the
429 cache — prime it explicitly (e.g. with a shorter spin-up in tests)
430 before anything that consumes it.
431 """
432 if self._clim is None:
433 w = self.solver.advance(self.solver.random_ic(), spinup)
434 snaps = []
435 for _ in range(n_snapshots):
436 w = self.solver.advance(w, stride)
437 snaps.append(w.ravel().copy())
438 self._clim = np.array(snaps).T
439 return self._clim

483# ======================================================================
484# OSSE twin-experiment harness
485# ======================================================================
486class TwinExperiment:
487 """
488 Cycled identical-twin experiment: truth from the model's real solver,
489 shared synthetic observations, one ensemble per registered method::
490
491 osse = TwinExperiment(model, n_cycles=60, n_members=16)
492 osse.add_method("enkf", inflation=1.06)
493 osse.add_method("surrogate", inflation=1.15, forecast=fn) # AI forecast
494 osse.add_free_run()
495 hist = osse.run()
496
497 Every method sees the **same** truth, probes, observation-noise
498 realization (one shared ``rng_obs`` stream) and initial ensemble; each
499 method's engine draws its perturbed-obs noise from its **own** stream
500 (``seed_eps + k``), so the comparison is invariant to which methods are
501 registered — the RNG discipline the prototype had to learn the hard way.
502
503 ``run()`` returns ``{name: {"rmse_f", "rmse_a", "spread", "time"}}``
504 (forecast/analysis RMSE vs truth, mean ensemble spread, wall time).
505
506 **Asynchronous stations** — real observation networks are neither
507 synchronized nor equal-rate. ``obs_periods=(lo, hi)`` gives every probe
508 its own reporting period (uniform in ``[lo, hi]`` cycles) and random
509 phase; silent stations contribute NaN, which the engines' masked update
510 skips natively. The schedule is deterministic given ``seed_setup`` and
511 drawn from its own stream, so enabling it never changes any other draw.
512
513 **Truth replay** — by default the truth evolves with the model's own solver
514 (a *perfect-model* twin). Pass ``truth_trajectory=`` (one field per cycle,
515 ``(T, n)`` or ``(T, grid, grid)``) to replay an **external** truth instead:
516 a run of the solver with different physics (imperfect-model experiment) or
517 a real reanalysis sequence (see ``kalman.apps.weather.geo`` for the ERA5
518 loader chain). Observations are then synthesized from the replayed fields
519 at the model's probes, and every RMSE is measured against them — so the
520 same harness, ratios (:meth:`skill`) and RNG discipline cover real data.
521 """
522
523 def __init__(self, model: KolmogorovFlowModel, *, n_cycles=60, n_members=16,
524 seed_obs=999, seed_setup=123, seed_eps=200, truth0=None,
525 truth_trajectory=None, obs_periods=None):
526 self.model = model
527 self.n_cycles = int(n_cycles)
528 self.n_members = int(n_members)
529 self.rng_obs = np.random.default_rng(seed_obs)
530 self._seed_eps = int(seed_eps)
531 self._truth0 = truth0
532 self._truth_traj = None
533 if truth_trajectory is not None:
534 traj = np.asarray(truth_trajectory, dtype=float)
535 if traj.ndim == 3:
536 traj = traj.reshape(traj.shape[0], -1)
537 if traj.ndim != 2 or traj.shape[1] != model.n:
538 raise ValueError(f"truth_trajectory must be (T, {model.n}) or "
539 f"(T, {model.grid}, {model.grid}); got {traj.shape}")
540 self._truth_traj = traj
541 self.n_cycles = min(self.n_cycles, traj.shape[0])
542 self._ens0 = model.initial_ensemble(self.n_members,
543 rng=np.random.default_rng(seed_setup))
544 # --- asynchronous observation network (optional) -------------------
545 # Real stations are neither synchronized nor equal-rate: each probe
546 # gets its own reporting period and phase, and cycles where it stays
547 # silent contribute NaN — which the engines' masked update skips
548 # natively. obs_periods=(lo, hi) draws per-station periods uniformly;
549 # an array gives them explicitly; None (default) = everyone, every
550 # cycle. The schedule comes from its own seed_setup-derived stream so
551 # enabling it never perturbs the ensemble/observation draws.
552 self.obs_mask = self.obs_periods = None
553 if obs_periods is not None:
554 rng_sched = np.random.default_rng(seed_setup + 1)
555 p = model.m
556 if isinstance(obs_periods, tuple) and len(obs_periods) == 2:
557 periods = rng_sched.integers(obs_periods[0], obs_periods[1] + 1, size=p)
558 else:
559 periods = np.broadcast_to(np.asarray(obs_periods, dtype=int), (p,)).copy()
560 phases = rng_sched.integers(0, periods)
561 cyc = np.arange(self.n_cycles)[:, None]
562 self.obs_mask = ((cyc + phases[None, :]) % periods[None, :]) == 0
563 self.obs_periods = periods
564 self.methods = {}

566 class _MethodView:
567 """Per-method model view: optional surrogate forecast and/or additive
568 model-error covariance override (``Q = q_std² I``), everything else
569 proxied to the shared model."""
570 def __init__(self, model, fn=None, q_std=None):
571 object.__setattr__(self, "_model", model)
572 object.__setattr__(self, "_fn", fn)
573 object.__setattr__(self, "_q_std", q_std)
574 def forecast(self, X, t=None):
575 return self._fn(X) if self._fn is not None else self._model.forecast(X, t=t)
576 def Q(self, t=None):
577 if self._q_std is None:
578 return self._model.Q(t=t)
579 return np.diag(np.full(self._model.n, float(self._q_std) ** 2))
580 def __getattr__(self, name):
581 return getattr(self._model, name)

583 def add_method(self, name, engine=None, *, inflation=1.06, forecast=None,
584 q_std=None, assimilate=True, seed=None):
585 """Register a method. ``forecast`` (columns → columns) swaps the solver
586 for a surrogate; ``q_std`` gives this branch its own additive
587 model-error noise (``Q = q_std² I`` drawn each forecast — the standard
588 answer when the forecast model is known to be wrong);
589 ``assimilate=False`` makes it a free-running control."""
590 if engine is None:
591 engine = EnsembleKalmanEngine(
592 n_members=self.n_members, inflation=inflation,
593 seed=self._seed_eps + len(self.methods) if seed is None else seed)
594 view = (self.model if forecast is None and q_std is None
595 else self._MethodView_MethodView(self.model, fn=forecast, q_std=q_std))
596 self.methods[name] = dict(engine=engine, view=view, assimilate=assimilate)
597 return engine

599 def add_free_run(self, name="free"):
600 """No-assimilation control — the baseline any filter must beat."""
601 return self.add_method(name, assimilate=False)

603 def _native_twin(self, method, callback):
604 """The C++ twin a method's whole run can use, or ``None`` (Python loop): a
605 surrogate forecast, a callback watching every cycle, another engine, or a
606 model with no twin."""
607 if callback is not None or type(method["engine"]) is not EnsembleKalmanEngine:
608 return None
609 view = method["view"]
610 if view is not self.model and (not isinstance(view, self._MethodView_MethodView) or view._fn is not None):
611 return None
612 return method["engine"]._native_twin(self.model, Q=view.Q(t=None))

614 def run(self, verbose=True, every=15, callback=None, record_fields=False):
615 """Run every method over the cycles; see the class docstring.
616
617 A method whose model has a C++ twin runs its whole run in C++
618 (``Kalman::Filters::EnKF::Cycle``: the forecasts of its members in
619 parallel, the observations, the analyses), on the truth and the
620 measurements made here first; the others run the Python loop, cycle by
621 cycle, on the same truth and measurements."""
622 model = self.model
623 if self._truth_traj is not None:
624 truth = self._truth_traj[0]
625 else:
626 truth = model.equilibrium_state() if self._truth0 is None else np.asarray(self._truth0).ravel().copy()
627 sig = np.sqrt(np.diag(model.R()))
628
629 hist = {}
630 for name, m in self.methods.items():
631 m["engine"].reset(m["view"])
632 m["engine"]._members = self._ens0.copy() # shared start: fair comparison
633 hist[name] = {"rmse_f": [], "rmse_a": [], "spread": [], "time": 0.0}
634 self.fields = ({"truth": [], **{n: {"forecast": [], "analysis": []} for n in self.methods}}
635 if record_fields else None)
636
637 def observe(c, truth):
638 truth = model.f(truth) if self._truth_traj is None else self._truth_traj[c]
639 y = np.asarray(model.h(truth)).reshape(-1) + self.rng_obs.normal(0.0, sig)
640 if self.obs_mask is not None:
641 y = np.where(self.obs_mask[c], y, np.nan) # silent stations this cycle
642 return truth, y
643
644 twins = {name: self._native_twin(m, callback) for name, m in self.methods.items()}
645 twins = {name: handle for name, handle in twins.items() if handle is not None}
646 if self.n_cycles == 0:
647 twins = {}
648 cycles = None
649 if twins:
650 # The truth and the measurements first (the same draws of rng_obs, in
651 # the same order), then each method's whole run in one C++ call.
652 cycles = []
653 for c in range(self.n_cycles):
654 truth, y = observe(c, truth)
655 cycles.append((truth.copy(), y))
656 truths = [tr for tr, _ in cycles]
657 ys = np.asarray([y for _, y in cycles])
658 for name, handle in twins.items():
659 m, h = self.methods[name], hist[name]
660 t0 = time.time()
661 out = m["engine"]._cycle_native(m["view"], handle, ys, analyse=m["assimilate"], Q=m["view"].Q(t=None))
662 h["time"] += time.time() - t0
663 h["rmse_f"] = [float(np.sqrt(np.mean((mu - tr) ** 2))) for mu, tr in zip(out["forecast"], truths)]
664 h["rmse_a"] = [float(np.sqrt(np.mean((mu - tr) ** 2))) for mu, tr in zip(out["analysis"], truths)]
665 h["spread"] = [float(v) for v in out["spread"]]
666 if record_fields:
667 self.fields[name]["forecast"] = [mu.copy() for mu in out["forecast"]]
668 self.fields[name]["analysis"] = [mu.copy() for mu in out["analysis"]]
669
670 for c in range(self.n_cycles):
671 truth, y = cycles[c] if cycles is not None else observe(c, truth)
672 if record_fields:
673 self.fields["truth"].append(truth.copy())
674 for name, m in self.methods.items():
675 if name in twins:
676 continue
677 eng, h = m["engine"], hist[name]
678 t0 = time.time()
679 mu_f = eng.forecast(m["view"]).ravel()
680 h["rmse_f"].append(float(np.sqrt(np.mean((mu_f - truth) ** 2))))
681 if m["assimilate"]:
682 eng.analyze(m["view"], y)
683 h["time"] += time.time() - t0
684 mu_a = eng.mean.ravel()
685 h["rmse_a"].append(float(np.sqrt(np.mean((mu_a - truth) ** 2))))
686 h["spread"].append(eng.spread())
687 if record_fields:
688 self.fields[name]["forecast"].append(mu_f.copy())
689 self.fields[name]["analysis"].append(mu_a.copy())
690 if callback is not None:
691 callback(c, truth, hist)
692 if verbose and (c + 1) % every == 0:
693 line = " ".join(f"{k}={v['rmse_a'][c]:.3f}" for k, v in hist.items())
694 print(f"cycle {c+1:3d}: RMSE_a {line}")
695
696 self.truth = truth
697 self.hist = hist
698 return hist

717 for name, h in self.hist.items()
718 if not (isinstance(reference, str) and name == reference)}
