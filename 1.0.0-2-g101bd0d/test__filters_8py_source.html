47 def test_forward_backward_shapes(self):

67 def test_smoother_not_worse_than_forward(self):
68 m = _cv_model()
69 truth = _make_track(m, seed=7)
70 kf = LinearKalmanFilter(m)
71 fwd = kf.forward(truth.zk, record=True)
72 sm = kf.backward()
73 rmse_f = np.sqrt(np.mean((fwd.xk[:, 0] - truth.xk[:, 0]) ** 2))
74 rmse_s = np.sqrt(np.mean((sm.xk[:, 0] - truth.xk[:, 0]) ** 2))
75 self.assertLessEqual(rmse_s, rmse_f * 1.05) # smoothing should help (allow tiny slack)

114class _Projectile(ExtendedKalmanModel):
115 """2-D projectile with quadratic drag; position observed."""
116
117 def __init__(self):
118 self.n, self.m, self.l = 4, 2, 0
119 super().__init__("proj", Q=np.eye(4) * 1e-4,
120 R=np.eye(2) * 0.25, fs=20)
121
122 def f(self, x, u=None, w=None, t=None):
123 px, py, vx, vy = x.ravel()
124 dt, k, g = 0.05, 0.02, 9.81
125 speed = np.hypot(vx, vy)
126 return np.array([[px + vx * dt],
127 [py + vy * dt],
128 [vx - k * speed * vx * dt],
129 [vy - (g + k * speed * vy) * dt]])
130
131 def h(self, x, v=None, t=None):
132 # additive measurement noise (sampler passes v drawn from R)
133 return x[:2] if v is None else x[:2] + v

146 tail = slice(15, None) # skip the initial-covariance transient
147 rmse_meas = np.sqrt(np.mean((truth.zk[tail] - truth.xk[tail, :2]) ** 2))

152 rmse = np.sqrt(np.mean((rec.xk[tail, :2] - truth.xk[tail, :2]) ** 2))
153 self.assertLess(rmse, rmse_meas, f"{Filt.__name__} should beat raw position")
