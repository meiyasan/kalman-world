1"""The toy model written in Python: the same physics as examples/cxx/toy_plugin.cc.
2
3A cup of coffee cools toward the temperature of the room (Newton's law of
4cooling), and nobody knows how fast: the cooling rate is part of the state.
5
6 state x = [T, k] temperature [degC], cooling rate [1/s]
7 dynamics T' = -k (T - T_room), k' = 0
8 measurement z = T a thermometer
9
10This file and ``toy_plugin.cc`` are the two ways to write a model for the
11framework. They state the same thing line for line:
12
13 ===================== ========================== ===========================
14 what C++ (toy_plugin.cc) Python (this file)
15 ===================== ========================== ===========================
16 the class Kalman::Models::Extended ExtendedKalmanModel
17 one step Vector f(x, u, t) def f(self, x, u, w, t)
18 the measurement Vector h(x, t) def h(self, x, v, t)
19 a setting DeclareParameter("room") a constructor argument
20 noise, initial state SetQ, SetR, SetInitial... Q=, R=, x0=, P0=
21 found by name KALMAN_REGISTER_MODEL the class itself
22 ===================== ========================== ===========================
23
24A model written in C++ runs wholly in C++. A model written in Python is run by
25the same C++ filter, which calls back into ``f`` and ``h`` below at each step:
26slower, and nothing to compile. ``toy_sensor.py`` runs both on the same
27readings and prints how far apart they are.
28"""

74 def __init__(self, sensor_id="thermometer", room=21.0, fs=1.0, **kwargs):
75 self.room = float(room) # degC
76 self.n, self.m, self.l = 2, 1, 0
77 # The model is nearly exact: a little slack on T, and a very slow
78 # random walk on k (the cup's cooling rate barely changes).
79 kwargs.setdefault("Q", np.diag([1e-4, 1e-12]))
80 kwargs.setdefault("R", np.array([[0.5 ** 2]])) # thermometer noise: 0.5 degC
81 # What the filter believes before the first reading: hot coffee, and a
82 # cooling rate that is only a guess.
83 kwargs.setdefault("x0", np.array([[80.0], [1e-3]]))
84 kwargs.setdefault("P0", np.diag([25.0, 1e-6]))
85 super().__init__(sensor_id, fs=fs, L=np.eye(2), **kwargs)

87 def f(self, x, u=None, w=None, t=None):
88 """The exact solution over one step, so no integrator is needed."""
89 T, k = np.asarray(x, dtype=float).reshape(-1)
90 out = np.array([[self.room + (T - self.room) * np.exp(-k / self.fs)], [k]])
91 return out if w is None else out + np.asarray(w, dtype=float).reshape(2, 1)
92
93 def h(self, x, v=None, t=None):
94 z = np.asarray(x, dtype=float).reshape(-1)[:1].reshape(1, 1)
95 return z if v is None else z + np.asarray(v, dtype=float).reshape(1, 1)
