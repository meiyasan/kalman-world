115 Record Filter::Run(const std::vector<Vector> &zk, const std::vector<Vector> *uk,
116 Controller *controller, bool reset)
117 {
118 if (zk.empty()) throw std::runtime_error("Kalman::Filter::Forward: no measurements given.");
119 fModel->CheckShapes();
120
121 const std::size_t m = fModel->NMeasure();
122 const std::size_t l = fModel->NControl();
123
124 if (uk && l > 0 && uk->size() != zk.size())
125 throw std::runtime_error("Kalman::Filter::Forward: uk and zk must have the same length.");
126
127 if (reset) Reset();
128
129 const std::size_t N = zk.size();
130 const double dt = fModel->DeltaT();
131 double t = fModel->T0();
132
133 fPriorX.reserve(fPriorX.size() + N);
134 fPostX.reserve(fPostX.size() + N);
135
136 // The closed loop hands the controller the previous innovation; on the first
137 // step there is none yet.
138 Vector command;
139 const Vector *lastNu = nullptr;
140
141 for (std::size_t k = 0; k < N; ++k)
142 {
143 const Vector &z = zk[k];
144 if (z.Size() != m)
145 {
146 std::ostringstream os;
147 os << "Kalman::Filter::Forward: measurement " << k << " has " << z.Size()
148 << " channels, expected " << m << ".";
149 throw std::runtime_error(os.str());
150 }
151
152 const Vector *u = (uk && l > 0) ? &(*uk)[k] : nullptr;
153 if (controller)
154 {
155 command = controller->Command(fX, fP, t, lastNu);
156 u = l > 0 ? &command : nullptr;
157 }
158
159 // --- predict --------------------------------------------------
160 Vector xPrior;
161 Matrix PPrior, F;
162 fEngine->Predict(*fModel, fX, fP, u, t, xPrior, PPrior, F);
163
164 // --- innovation (kept for diagnostics, before the correction) --
165 Vector nu;
166 Matrix S;
167 fEngine->Innovation(*fModel, xPrior, PPrior, z, t, nu, S);
168
169 // --- update ---------------------------------------------------
170 Vector xPost;
171 Matrix PPost;
172 fEngine->Update(*fModel, xPrior, PPrior, z, t, xPost, PPost);
173
174 fPriorX.push_back(xPrior);
175 fPriorP.push_back(PPrior);
176 fPostX.push_back(xPost);
177 fPostP.push_back(PPost);
178 fTransition.push_back(F);
179 fMeasure.push_back(z);
180 fControl.push_back(u ? *u : Vector());
181 fInnovation.push_back(nu);
182 fTime.push_back(t);
183
184 fX = xPost;
185 fP = PPost;
186 lastNu = &fInnovation.back();
187
188 t += dt;
189 }
190
191 return MakeRecord(fPostX, fPostP, StateFlavor::Posterior);
192 }

194 Record Filter::Backward()
195 {
196 const std::size_t N = NSteps();
197 if (N == 0) throw std::runtime_error("Kalman::Filter::Backward: run Forward() first.");
198 if (!fEngine->HasSmoother())
199 throw std::runtime_error(std::string("Kalman::Filter::Backward: the ") + fEngine->Title()
200 + " has no RTS smoother; use Forward() only.");
201
202 fSmoothX = fPostX;
203 fSmoothP = fPostP;
204
205 // The last posterior is already the smoothed estimate; walk backwards from it.
206 for (std::size_t k = N - 1; k-- > 0;)
207 {
208 Vector xs;
209 Matrix Ps;
210 fEngine->Smooth(fPostX[k], fPostP[k],
211 fPriorX[k + 1], fPriorP[k + 1],
212 fSmoothX[k + 1], fSmoothP[k + 1],
213 fTransition[k + 1],
214 xs, Ps);
215 fSmoothX[k] = xs;
216 fSmoothP[k] = Ps;
217 }
218
219 return MakeRecord(fSmoothX, fSmoothP, StateFlavor::Smoothed);
220 }
