361def test_every_python_catalogue_model_goes_through_a_file():
362 """Each Python model that can be built with its defaults is written, read
363 back and filters the same. A reference carries the constructor's arguments."""
364 import inspect
365 import warnings
366 import numpy as np
367 import kalman.native as native
368 from kalman import api
369 from kalman.framework.native import core
370
371 if not core.HAS_CXX_CORE:
372 pytest.skip("libKalman is not built")
373 done = []
374 for name, cls in sorted(api._model_catalog().items()):
375 if hasattr(cls, "NATIVE_NAME"):
376 continue
377 first = list(inspect.signature(cls.__init__).parameters)[1:2]
378 try:
379 with warnings.catch_warnings():
380 warnings.simplefilter("ignore")
381 model = cls("s") if first in (["sensor_id"], ["name"]) else cls()
382 structure = model.structure
383 except Exception:
384 continue # needs arguments: not this test's business
385 with warnings.catch_warnings():
386 warnings.simplefilter("ignore")
387 if structure.name == "TIME_VARYING":
388 continue # tables need a file: covered by test_python_models_export_and_open
389 document = model.export()
390 assert document["kind"] == ("complete" if structure.name == "MATRICES" else "reference"), name
391 back = api.make_model(document["spec"])
392 if document["kind"] == "reference":
393 assert type(back) is cls, name
394 written = document["spec"]["dynamics"]["params"]
395 left_out = set(document["origin"].get("parameters_not_written", []))
396 for key, value in written.items():
397 again = back._init_arguments[key]
398 again = getattr(again, "_name_", again)
399 assert np.all(np.asarray(again) == np.asarray(value)), (name, key)
400 assert not left_out, (name, left_out)
401 assert np.allclose(np.asarray(back.x0, float), np.asarray(model.x0, float), equal_nan=True), name
402 assert np.allclose(back.Q(), model.Q(), equal_nan=True) and np.allclose(back.R(), model.R()), name
403 done.append(name)
404 assert len(done) >= 15, done
405 assert {"CarLaneModel", "RocketAscentModel", "InterplanetaryModel", "GPSModel"} <= set(done)
