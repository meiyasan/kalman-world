1"""
2GPS positioning with general (and special) relativity.
3
4A GPS receiver solves for its position **and clock** from pseudoranges to several
5satellites. The satellites carry atomic clocks that, relative to a ground clock, run
6fast by a net **+38.6 µs/day** — the sum of special-relativistic time dilation
7(−7.2 µs/day, they move fast) and the general-relativistic blueshift (+45.8 µs/day, they
8sit higher in Earth's potential well). Left uncorrected that is **≈ 11.6 km/day** of
9range error. On top of the constant rate, each satellite's eccentric orbit adds a
10*periodic* relativistic term
11
12 Δt_rel(t) = F · e · √a · sin E(t), F = −2√μ / c² ≈ −4.443×10⁻¹⁰ s/√m,
13
14which is the correction a receiver actually applies per satellite (up to ~14 m).
15
16Idealisations: the satellite clocks are *not* detuned before launch (real ones are, which
17removes the secular rate), the receiver clock is perfect, the Earth does not rotate (no
18Sagnac term) and there is no atmosphere.
19
20`GPSModel` is an EKF on the state ``x = [x, y, z, b, ḃ]`` (ECEF position, receiver clock
21bias and clock drift, all in metres / m·s⁻¹). The measurement is the per-satellite
22pseudorange ``ρ_i = |s_i(t) − p| + b`` (nonlinear → analytic line-of-sight Jacobian). The
23relativistic terms are baked into the simulated pseudoranges; toggling
24``model.apply_relativistic`` decides whether the receiver removes them in
25:meth:`GPSModel.preprocess`, giving a clean **before/after** comparison.
26
27The famous secular rate is *common-mode* and is absorbed by the estimated clock drift
28``ḃ`` (which is exactly why a 4th satellite is needed — to solve for the clock); the part
29that genuinely corrupts **position** is the per-satellite eccentricity term.
30"""

49 c, s = np.cos(a), np.sin(a)

58class GPSConstellation:
59 """A small Keplerian GPS constellation (vectorised positions + relativistic offsets)."""
60
61 def __init__(self, a, e, inc, raan, argp, M0):
62 self.a = np.asarray(a, float)
63 self.e = np.asarray(e, float)
64 self.inc = np.asarray(inc, float)
65 self.raan = np.asarray(raan, float)
66 self.argp = np.asarray(argp, float)
67 self.M0 = np.asarray(M0, float)
68 self.n_mean = np.sqrt(MU_EARTH / self.a**3) # mean motion [rad/s]
69 # Per-satellite perifocal→ECI rotation R = Rz(Ω) Rx(i) Rz(ω).
70 self.R = np.stack([_Rz(O) @ _Rx(i) @ _Rz(w)
71 for O, i, w in zip(self.raan, self.inc, self.argp)])
72
73 @property
74 def size(self):
75 return self.a.shape[0]
76
77 def _eccentric_anomaly(self, M):
78 """Solve Kepler's equation E − e sinE = M (Newton), vectorised over any shape."""
79 M = np.mod(M + np.pi, 2.0 * np.pi) - np.pi
80 E = M.copy()
81 for _ in range(12):
82 E = E - (E - self.e * np.sin(E) - M) / (1.0 - self.e * np.cos(E))
83 return E
84
85 def _anomalies(self, t):
86 t1 = np.atleast_1d(np.asarray(t, float))
87 M = self.M0[None, :] + self.n_mean[None, :] * t1[:, None] # (N, S)
88 return t1, self._eccentric_anomaly(M)
89
90 def positions(self, t):
91 """Satellite ECEF positions: ``(S,3)`` for scalar ``t`` or ``(N,S,3)`` for array ``t``."""
92 scalar = np.ndim(t) == 0
93 _, E = self._anomalies(t)
94 nu = 2.0 * np.arctan2(np.sqrt(1 + self.e) * np.sin(E / 2),
95 np.sqrt(1 - self.e) * np.cos(E / 2))
96 r = self.a * (1.0 - self.e * np.cos(E)) # (N, S)
97 pf = np.stack([r * np.cos(nu), r * np.sin(nu), np.zeros_like(r)], axis=-1) # (N,S,3)
98 pos = np.einsum("sij,nsj->nsi", self.R, pf) # rotate to ECI/ECEF
99 return pos[0] if scalar else pos
100
101 def relativistic_seconds(self, t):
102 """Per-satellite eccentricity relativistic clock offset Δt_rel [s]."""
103 scalar = np.ndim(t) == 0
104 _, E = self._anomalies(t)
105 off = F_REL * self.e * np.sqrt(self.a) * np.sin(E) # (N, S)
106 return off[0] if scalar else off

160 def __init__(self, sensor_id: str | int,
161 receiver_lat: float = 45.0,
162 receiver_lon: float = 10.0,
163 n_sat: int = 8,
164 fs: float = 1.0,
165 pseudorange_uncertainty: float = 1.0,
166 process_uncertainty: float = 0.2,
167 clock_drift_uncertainty: float = 1e-3,
168 apply_relativistic: bool = True,
169 secular: bool = True,
170 elevation_mask: float | None = None,
171 **kwargs):
172
173 self.dtypedtype = kwargs.pop("dtype", np.float64)
174 self.fsfsfsfsfs = float(fs)
175 self.apply_relativistic = bool(apply_relativistic)
176 self._sigma_rho = float(pseudorange_uncertainty)
177 self._sigma_process = float(process_uncertainty)
178 self._sigma_drift = float(clock_drift_uncertainty)
179 self._secular_on = bool(secular)
180 # Elevation-cutoff [deg] below which a satellite is "out of sight". When set, the model
181 # tracks the FULL constellation and masks each satellite epoch-by-epoch as it rises/sets
182 # (the EKF skips the masked pseudoranges); when None, a fixed set of satellites is used.
183 self.elevation_mask = None if elevation_mask is None else float(elevation_mask)
184
185 # Receiver truth (ECEF, spherical) + local ENU basis.
186 lat, lon = np.radians(receiver_lat), np.radians(receiver_lon)
187 self.receiver_ecef = R_EARTH * np.array(
188 [np.cos(lat) * np.cos(lon), np.cos(lat) * np.sin(lon), np.sin(lat)], dtype=self.dtypedtype)
189 self._enu = np.array([
190 [-np.sin(lon), np.cos(lon), 0.0],
191 [-np.sin(lat) * np.cos(lon), -np.sin(lat) * np.sin(lon), np.cos(lat)],
192 [ np.cos(lat) * np.cos(lon), np.cos(lat) * np.sin(lon), np.sin(lat)],
193 ], dtype=self.dtypedtype)
194
195 # Build a 24-satellite pool. With an elevation mask we keep the WHOLE pool and decide
196 # visibility dynamically (satellites set and rise); otherwise we keep the n_sat highest
197 # above the horizon at t0.
198 pool = self._build_pool()
199 self.constellation = pool if self.elevation_mask is not None \
200 else self._select_visible(pool, n_sat)
201 M = self.constellation.size
202
203 # State [x,y,z, b, b_dot]; continuous generator: static receiver
204 # (ẋ=ẏ=ż=0), clock bias driven by drift (ḃ=b_dot), drift itself
205 # constant (ḃ_dot=0) — only entry is the b<-b_dot coupling. Passed as
206 # A= (not F=) below: the base KalmanModel discretizes exactly via
207 # expm(A/fs) into self._F, which f() below reads directly.
208 A = np.zeros((5, 5), dtype=self.dtypedtype)
209 A[3, 4] = 1.0
210
211 # Continuous noise: direct random-walk disturbance on x,y,z (receiver
212 # isn't perfectly static) plus a white-noise disturbance driving b_dot's
213 # own rate (clock-drift wander) — same Lc/Qc-per-column pattern as the
214 # v̇ disturbance elsewhere; passed as Qc=/Lc= for exact Van Loan discretization.
215 Lc = np.zeros((5, 4), dtype=self.dtypedtype)
216 Lc[0, 0] = Lc[1, 1] = Lc[2, 2] = 1.0
217 Lc[4, 3] = 1.0
218 qp = process_uncertainty ** 2
219 qd = clock_drift_uncertainty ** 2
220 Qc = np.diag([qp, qp, qp, qd]).astype(self.dtypedtype)
221 R = (self._sigma_rho ** 2) * np.eye(M, dtype=self.dtypedtype)
222
223 self.nnnnn = 5
224 self.mmmm = M
225 if kwargs.get("x0") is None:
226 kwargs["x0"] = np.concatenate([self.receiver_ecef, [0.0, 0.0]]).reshape(-1, 1)
227 if kwargs.get("P0") is None:
228 kwargs["P0"] = np.diag([100.0 ** 2, 100.0 ** 2, 100.0 ** 2,
229 100.0 ** 2, 1.0 ** 2]).astype(self.dtypedtype)
230
231 super().__init__(sensor_id, A=A, Qc=Qc, Lc=Lc, R=R, fs=fs, **kwargs)

255 def _build_pool(self, n_planes=6, per_plane=4):
256 rng = np.random.default_rng(2024)
257 a, e, inc, raan, argp, M0 = [], [], [], [], [], []
258 for p in range(n_planes):
259 for s in range(per_plane):
260 a.append(GPS_A)
261 e.append(rng.uniform(0.002, 0.020)) # eccentricity, as in the real constellation
262 inc.append(np.radians(55.0))
263 raan.append(2.0 * np.pi * p / n_planes)
264 argp.append(rng.uniform(0.0, 2.0 * np.pi))
265 M0.append(2.0 * np.pi * s / per_plane + 0.5 * p)
266 return GPSConstellation(a, e, inc, raan, argp, M0)

268 def _select_visible(self, pool, n_sat):
269 pos0 = pool.positions(0.0) # (S,3) at t0
270 up = self.receiver_ecef / np.linalg.norm(self.receiver_ecef)
271 los = pos0 - self.receiver_ecef
272 elev = np.arcsin(np.clip((los @ up) / np.linalg.norm(los, axis=1), -1, 1))
273 order = np.argsort(elev)[::-1] # highest elevation first
274 keep = order[:n_sat]
275 return GPSConstellation(pool.a[keep], pool.e[keep], pool.inc[keep],
276 pool.raan[keep], pool.argp[keep], pool.M0[keep])

288 def h(self, x, v=None, t=None):
289 x = np.asarray(x, dtype=self.dtypedtype).reshape(-1, 1)
290 sat = self.constellation.positions(float(t) if t is not None else 0.0) # (M,3)
291 rng = np.linalg.norm(sat - x[:3, 0], axis=1) # geometric range
292 z = (rng + x[3, 0]).reshape(-1, 1) # + receiver clock bias
293 if v is not None:

297 def H(self, x=None, t=None):
298 x = (self.x0x0x0 if x is None else np.asarray(x, dtype=self.dtypedtype)).reshape(-1, 1)
299 sat = self.constellation.positions(float(t) if t is not None else 0.0) # (M,3)
300 diff = x[:3, 0] - sat # ∂range/∂p = (p − s)/|p − s|
301 rng = np.linalg.norm(diff, axis=1, keepdims=True)
302 H = np.zeros((self.mmmm, 5), dtype=self.dtypedtype)
303 H[:, :3] = diff / rng
304 H[:, 3] = 1.0 # ∂ρ/∂(clock bias)
305 return H

310 def _correction_meters(self, N):
311 """Total relativistic range correction c·(Δt_rel + secular), shape (N, M)."""
312 t = self.t0t0 + np.arange(N) / self.fsfsfsfsfs
313 corr = self.constellation.relativistic_seconds(t) # (N, M) eccentricity
314 if self._secular_on:
315 corr = corr + (SECULAR_RATE * t)[:, None] # common-mode secular
316 return C_LIGHT * corr

330 def simulate(self, duration: float, seed=None) -> np.ndarray:
331 """Generate ``(N, n_sat)`` pseudoranges from the true receiver position, with the
332 relativistic terms (eccentricity + secular) and Gaussian noise baked in."""
333 if seed is not None:
334 np.random.seed(seed)
335 N = int(round(duration * self.fsfsfsfsfs))
336 t = self.t0t0 + np.arange(N) / self.fsfsfsfsfs
337 pos = self.constellation.positions(t) # (N, M, 3)
338 rng = np.linalg.norm(pos - self.receiver_ecef[None, None, :], axis=2) # (N, M)
339 pr = rng + self._correction_meters(N) # cb_true = 0
340 pr = pr + self._sigma_rho * np.random.randn(*pr.shape)
341 if self.elevation_mask is not None: # blank out-of-sight satellites
342 pr = np.where(self.visible_mask(t), pr, np.nan) # NaN → EKF skips them
343 return pr

395 def dop(self, t):
396 """Geometric/position/horizontal/vertical **dilution of precision** over time from the
397 *visible* satellites' geometry → dict of ``(N,)`` arrays (NaN when < 4 are in view).
398
399 DOP multiplies the per-satellite ranging error into position error: e.g. the expected
400 horizontal error is ``HDOP · σ_pseudorange``. It is the single number that says how good
401 the current sky geometry is for a GPS user — and it spikes whenever satellites set."""
402 t = np.atleast_1d(np.asarray(t, float))
403 pos = self.constellation.positions(t) # (N, M, 3)
404 los = pos - self.receiver_ecef[None, None, :]
405 e = los / np.linalg.norm(los, axis=2, keepdims=True) # unit line-of-sight (N, M, 3)
406 vis = self.visible_mask(t)
407 out = {k: np.full(len(t), np.nan) for k in ("gdop", "pdop", "hdop", "vdop")}
408 for k in range(len(t)):
409 m = vis[k]
410 if m.sum() < 4:
411 continue
412 G = np.column_stack([e[k, m], np.ones(int(m.sum()))]) # rows [ex,ey,ez,1]
413 try:
414 Cov = np.linalg.inv(G.T @ G) # in local ENU? -> rotate diag
415 except np.linalg.LinAlgError:
416 continue
417 # rotate the position block into East/North/Up to split H/V DOP
418 Cenu = self._enu @ Cov[:3, :3] @ self._enu.T
419 out["gdop"][k] = np.sqrt(np.trace(Cov))
420 out["pdop"][k] = np.sqrt(np.trace(Cov[:3, :3]))
421 out["hdop"][k] = np.sqrt(Cenu[0, 0] + Cenu[1, 1])
422 out["vdop"][k] = np.sqrt(Cenu[2, 2])
423 return out
