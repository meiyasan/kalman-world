23 Vector Router::Normalise(Vector mu)
24 {
25 double sum = 0.;
26 for (std::size_t i = 0; i < mu.Size(); ++i)
27 {
28 // A mode that reaches exactly zero can never come back: every later
29 // update multiplies it by something finite. The floor is what keeps a
30 // regime recoverable after the filter has stopped believing in it.
31 if (!(mu(i) > 1e-12)) mu(i) = 1e-12;
32 sum += mu(i);
33 }
34 if (!(sum > 0.)) // every entry was NaN
35 return Vector(mu.Size(), 1. / static_cast<double>(mu.Size()));
36
37 mu *= 1. / sum;
38 return mu;
39 }

41 Matrix Router::NisWindow(const Matrix &nis, std::size_t window)
42 {
43 const std::size_t M = nis.Cols();
44 const std::size_t have = std::min(window, nis.Rows());
45 const std::size_t pad = window - have;
46
47 Matrix out(window, M, 1.0); // 1.0 = the chi2(1) mean
48 for (std::size_t r = 0; r < have; ++r)
49 for (std::size_t j = 0; j < M; ++j)
50 out(pad + r, j) = nis(nis.Rows() - have + r, j);
51 return out;
52 }

56 Vector Bayesian::Route(const Matrix &, const Vector &c, const Vector &likelihoods) const
57 {
58 if (c.Size() != likelihoods.Size())
59 throw std::runtime_error("Kalman::Routers::Bayesian: c and likelihoods differ in length.");
60
61 Vector mu(c.Size());
62 for (std::size_t j = 0; j < c.Size(); ++j) mu(j) = c(j) * likelihoods(j);
63 return Normalise(mu);
64 }

68 Vector Threshold::Route(const Matrix &nis, const Vector &c, const Vector &) const
69 {
70 const std::size_t M = nis.Cols() ? nis.Cols() : c.Size();
71 if (nis.Rows() == 0)
72 return Vector(M, 1. / static_cast<double>(M)); // nothing observed yet
73
74 // The mean of the last `window` rows, or of all of them while there are
75 // fewer: a short history is averaged as it is, not padded (padding with the
76 // chi2(1) mean would pull every model toward 1 whatever m is). The same
77 // rule as routers/threshold.py, which forwards here; a window of 0 is the
78 // whole history, as `nis[-0:]` is in numpy.
79 const std::size_t rows = fWindow == 0 ? nis.Rows() : std::min(fWindow, nis.Rows());
80 Vector dist(M);
81 for (std::size_t j = 0; j < M; ++j)
82 {
83 double mean = 0.;
84 for (std::size_t r = nis.Rows() - rows; r < nis.Rows(); ++r) mean += nis(r, j);
85 mean /= static_cast<double>(rows);
86 dist(j) = std::abs(mean - static_cast<double>(fM));
87 }
88
89 if (fSoft)
90 {
91 Vector mu(M);
92 for (std::size_t j = 0; j < M; ++j) mu(j) = 1. / (dist(j) + 1e-6);
93 return Normalise(mu);
94 }
95
96 std::size_t best = 0;
97 for (std::size_t j = 1; j < M; ++j)
98 if (dist(j) < dist(best)) best = j;
99
100 Vector mu(M, 0.);
101 mu(best) = 1.;
102 return mu; // hard: deliberately not floored
103 }

121 {
122 const Vector a = Normalise(fA->Route(nis, c, likelihoods));
123 const Vector b = Normalise(fB->Route(nis, c, likelihoods));
124
125 Vector mu(a.Size());
