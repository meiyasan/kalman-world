47 bool SymmetricEigen(const Matrix &S, Vector &values, Matrix &vectors)
48 {
49 const std::size_t n = S.Rows();
50 Matrix A = SymmetricPart(S);
51 vectors = Matrix::Identity(n);
52
53 // Cyclic Jacobi: repeatedly zero the largest off-diagonal entry by a plane
54 // rotation. Converges quadratically and needs no pivoting or workspace.
55 for (int sweep = 0; sweep < 100; ++sweep)
56 {
57 double off = 0.;
58 for (std::size_t i = 0; i < n; ++i)
59 for (std::size_t j = i + 1; j < n; ++j) off += A(i, j) * A(i, j);
60 if (off <= 1e-30)
61 {
62 values.Resize(n);
63 for (std::size_t i = 0; i < n; ++i) values(i) = A(i, i);
64 return true;
65 }
66
67 for (std::size_t p = 0; p < n; ++p)
68 for (std::size_t q = p + 1; q < n; ++q)
69 {
70 if (std::abs(A(p, q)) < 1e-300) continue;
71
72 const double theta = (A(q, q) - A(p, p)) / (2. * A(p, q));
73 const double sign = theta >= 0. ? 1. : -1.;
74 const double t = sign / (std::abs(theta) + std::sqrt(theta * theta + 1.));
75 const double c = 1. / std::sqrt(t * t + 1.);
76 const double s = t * c;
77
78 for (std::size_t k = 0; k < n; ++k)
79 {
80 const double akp = A(k, p), akq = A(k, q);
81 A(k, p) = c * akp - s * akq;
82 A(k, q) = s * akp + c * akq;
83 }
84 for (std::size_t k = 0; k < n; ++k)
85 {
86 const double apk = A(p, k), aqk = A(q, k);
87 A(p, k) = c * apk - s * aqk;
88 A(q, k) = s * apk + c * aqk;
89 }
90 for (std::size_t k = 0; k < n; ++k)
91 {
92 const double vkp = vectors(k, p), vkq = vectors(k, q);
93 vectors(k, p) = c * vkp - s * vkq;
94 vectors(k, q) = s * vkp + c * vkq;
95 }
96 }
97 }
98
99 values.Resize(n);
100 for (std::size_t i = 0; i < n; ++i) values(i) = A(i, i);
101 return false;
102 }

104 double ConditionNumber(const Matrix &M)
105 {
106 Vector values;
107 Matrix vectors;
108 SymmetricEigen(M, values, vectors);
109

111 for (std::size_t i = 0; i < values.Size(); ++i)
