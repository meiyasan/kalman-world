27def barycentric_interpolation(*colors, space="rgb"):
28 """
29 Compute the barycentric (average) color of a collection of colors.
30
31 Parameters
32 ----------
33 *colors : str | tuple | list | dict
34 Color specifications (names, hex codes, RGB tuples, or nested containers).
35 Dicts are flattened using their values.
36 space : {"rgb","lab"}, default="rgb"
37 Color space in which to average:
38 - "rgb": arithmetic mean in RGB (numerical average).
39 - "lab": perceptual mean in Lab (requires colorspacious).
40
41 Returns
42 -------
43 tuple of floats
44 (R, G, B) each in [0,1].
45 """
46 def _flatten(items):
47 for i in items:
48 if isinstance(i, str):
49 yield i
50 elif isinstance(i, Mapping):
51 yield from _flatten(i.values())
52 elif isinstance(i, Iterable):
53 yield from _flatten(i)
54 else:
55 yield i
56
57 flat_colors = list(_flatten(colors))
58 if not flat_colors:
59 raise ValueError("No colors provided")
60
61 # Convert to RGB triples
62 rgbs = np.array([mcolors.to_rgb(c) for c in flat_colors])
63
64 if space == "lab":
65 if not _HAS_COLORSPACIOUS:
66 raise RuntimeError("Lab averaging requires `colorspacious` installed.")
67 labs = cspace_convert(rgbs, "sRGB1", "CIELab")
68 mean_lab = labs.mean(axis=0)
69 mean_rgb = cspace_convert(mean_lab[np.newaxis, :], "CIELab", "sRGB1")[0]
70 else:
71 mean_rgb = rgbs.mean(axis=0)
72
73 return tuple(np.clip(mean_rgb, 0, 1))

211def animate(fig, draw, frames, *, interval=90, jshtml=None, blit=False):
212 """Build a matplotlib ``FuncAnimation`` and return it ready for the current environment.
213
214 One animation primitive serves both worlds, which is why every ``_plot_interactive`` /
215 replay routine funnels through ``FuncAnimation``:
216
217 - inside a **Jupyter** kernel it returns a self-contained ``IPython.display.HTML``
218 play/scrub player (via ``anim.to_jshtml()``) that survives notebook serialization
219 with no live kernel;
220 - in a plain **Python** session it returns the raw ``FuncAnimation`` so a GUI
221 backend's ``plt.show()`` plays it live.
222
223 Parameters
224 ----------
225 fig : matplotlib Figure the figure to animate (closed here so the static frame
226 isn't also rendered alongside the player).
227 draw : callable per-frame update ``draw(frame)`` (the ``_draw`` closure).
228 frames : iterable | int frame indices forwarded to ``FuncAnimation``.
229 interval, blit standard ``FuncAnimation`` knobs.
230 jshtml : bool | None force the HTML player (True) or the raw anim (False);
231 ``None`` auto-detects whether an IPython kernel is active.
232 """
233 from matplotlib.animation import FuncAnimation
234 anim = FuncAnimation(fig, draw, frames=frames, interval=interval, blit=blit)
235 plt.close(fig)
236 if jshtml is None:
237 try:
238 from IPython import get_ipython
239 jshtml = getattr(get_ipython(), "kernel", None) is not None
240 except Exception:
241 jshtml = False
242 if jshtml:
243 try:
244 from IPython.display import HTML
245 import matplotlib as mpl
246 # to_jshtml() silently DROPS frames once the embedded size passes
247 # animation.embed_limit (20 MB default) — the usual "animation doesn't finish"
248 # cause. Lift it for the render so long replays play to the end, then restore.
249 prev = mpl.rcParams.get("animation.embed_limit", 20.0)
250 mpl.rcParams["animation.embed_limit"] = max(prev, 100.0) # MB
251 try:
252 html = anim.to_jshtml()
253 finally:
254 mpl.rcParams["animation.embed_limit"] = prev
255 return HTML(html)
256 except Exception:
257 return anim
258 return anim

292def numerical_jacobian(func, x: np.ndarray, *args, eps: float = None, scheme: str = "forward") -> np.ndarray:
293 """``d func / dx`` at ``x`` by finite differences.
294
295 ``scheme``: ``"forward"`` (one-sided, first order; step ``1e-6``) or
296 ``"central"`` (two-sided, second order; step ``1e-5``). The step is relative:
297 ``eps * max(1, |x_j|)``. Computed in C++ (Kalman::Differentiators) when
298 libKalman is loaded and the values are real double precision.
299 """
300 if scheme not in ("forward", "central"):
301 raise ValueError(f"numerical_jacobian: unknown scheme {scheme!r} (forward or central).")
302 if eps is None:
303 eps = 1e-6 if scheme == "forward" else 1e-5
304 x = as_col(x)
305 y0 = as_col(func(x, *args))
306 core = _native_core()
307 if core is not None and x.dtype == np.float64 and y0.dtype == np.float64 and y0.size and x.size:
308 return core.jacobian(scheme, lambda point: func(point, *args), x, y0, step=eps)
309 m, n = y0.shape[0], x.shape[0] # reference, and other dtypes
310 J = np.zeros((m, n), dtype=float)
311 for j in range(n):
312 dx = np.zeros_like(x)
313 step = eps * max(1.0, abs(x[j, 0]))
314 dx[j, 0] = step
315 if scheme == "forward":
316 J[:, j] = ((as_col(func(x + dx, *args)) - y0) / step).ravel()
317 else:
318 J[:, j] = ((as_col(func(x + dx, *args)) - as_col(func(x - dx, *args))) / (2.0 * step)).ravel()
319 return J

321def van_loan_discretize(A: np.ndarray, L: np.ndarray, Qc: np.ndarray, dt: float):
322 """Exact continuous->discrete conversion (Van Loan, 1978) for a linear system
323 ``ẋ = Ax + Lw`` driven by continuous white noise of spectral density ``Qc``.
324
325 Returns ``(F, Q)`` for the equivalent discrete model ``x_{k+1} = F x_k + q_k``,
326 ``Cov(q_k) = Q``, exact over the step ``dt`` — including the off-diagonal
327 position/velocity-type coupling that a naive ``Q ≈ L Qc Lᵀ · dt`` misses, and
328 correct even when ``A`` is singular/nilpotent (as every kinematic chain here is),
329 where the closed-form ``Q = A⁻¹(...)`` alternative doesn't apply.
330
331 Built from the block matrix exponential
332 ``expm([[-A, L·Qc·Lᵀ], [0, Aᵀ]]·dt) = [[M11, M12], [0, M22]]``,
333 giving ``F = M22ᵀ`` and ``Q = F·M12``.
334 """
335 A = np.asarray(A, dtype=float)
336 G = np.asarray(L, dtype=float) @ np.asarray(Qc, dtype=float) @ np.asarray(L, dtype=float).T
337 core = _native_core()
338 if core is not None: # the implementation: Kalman::Utils::VanLoan
339 return core.van_loan(A, G, dt)
340 n = A.shape[0] # reference, when libKalman is not there
341 Z = np.zeros((2 * n, 2 * n), dtype=float)
342 Z[:n, :n] = -A
343 Z[:n, n:] = G
344 Z[n:, n:] = A.T
345 M = expm(Z * dt)
346 F = M[n:, n:].T
347 Q = F @ M[:n, n:]
348 return F, 0.5 * (Q + Q.T)

368def symbolic_discretize(A: np.ndarray, t: str = "dt"):
369 """Closed-form ``F(t) = exp(A·t)`` as a symbolic expression, for reading a
370 model's discretization as an equation (the way it's designed) rather than a
371 numeric matrix you have to run code to see. Lazy ``sympy`` import — this is a
372 read/inspect tool (call it from a notebook on a model's ``A``), not something
373 any model needs at runtime, so it isn't a hard dependency of the framework.
374
375 Display with ``print(...)`` (flat ``dt**2/2`` form), not ``sympy.pprint``
376 (its 2D box/fraction layout is ambiguous in a plain terminal); in a Jupyter
377 cell the bare returned Matrix already renders as typeset math.
378 """
379 import sympy as sp
380 t_sym = sp.symbols(t, positive=True)
381 A_sym = sp.nsimplify(sp.Matrix(A), rational=True)
382 return sp.simplify((A_sym * t_sym).exp())

384def zoh_discretize(A: np.ndarray, B: np.ndarray, dt: float):
385 """Exact zero-order-hold discretization of ``ẋ = Ax + Bu`` into ``F, B_d`` such
386 that ``x_{k+1} = F x_k + B_d u_k``. Uses the augmented matrix exponential
387 ``expm([[A, B], [0, 0]]·dt) = [[F, B_d], [0, I]]`` rather than the closed form
388 ``B_d = A⁻¹(F − I)B``, so it stays valid when ``A`` is singular/nilpotent.
389 """
390 n, l = A.shape[0], B.shape[1]
391 Z = np.zeros((n + l, n + l), dtype=float)
392 Z[:n, :n] = A
393 Z[:n, n:] = B
394 M = expm(Z * dt)
395 return M[:n, :n], M[:n, n:]

422def resample_poly_float(y, fs_old, fs_new, max_denominator=1e12):
423 """
424 Resample a 1-D signal y from sample rate fs_old to fs_new
425 using polyphase filtering with integer up/down factors.
426
427 Parameters
428 ----------
429 y : array_like
430 Input signal.
431 fs_old : float
432 Original sample rate.
433 fs_new : float
434 Desired sample rate.
435 max_denominator : int
436 Maximum denominator when rationally approximating fs_new/fs_old.
437
438 Returns
439 -------
440 y_resampled : ndarray
441 The resampled signal.
442 """
443 # 1) Find a simple rational approximation to the rate change
444 frac = Fraction(fs_new, fs_old).limit_denominator(max_denominator)
445 up, down = frac.numerator, frac.denominator
446
447 # 2) Do the polyphase resampling
448 y_resampled = resample_poly(y, up, down)
449
450 return y_resampled

452def make_multirate_input(sensor_data, base_fs):
453 """
454 Interleave multiple sensors running at different rates into a single
455 NaN-padded measurement matrix at `base_fs`.
456
457 Parameters
458 ----------
459 sensor_data : list of (zk, fs_sensor) tuples
460 Each tuple is an (N_i, m_i) measurement array and its sample rate.
461 The duration is inferred from each array's length and its fs_sensor.
462 base_fs : float
463 The base filter sample rate.
464
465 Returns
466 -------
467 zk_out : ndarray, shape (N_base, m_total)
468 NaN where the sensor did not fire at that base-rate step.
469 """
470 if not sensor_data:
471 raise ValueError("sensor_data must contain at least one (zk, fs) pair.")
472
473 # Determine total duration from the longest sensor
474 durations = [len(zk) / fs for zk, fs in sensor_data]
475 duration = max(durations)
476 N_base = int(round(duration * base_fs))
477
478 # Total observable dimension
479 m_total = sum(zk.shape[1] if zk.ndim == 2 else 1 for zk, _ in sensor_data)
480
481 zk_out = np.full((N_base, m_total), np.nan)
482
483 col = 0
484 for zk, fs_sensor in sensor_data:
485 zk = np.atleast_2d(zk) if zk.ndim == 1 else zk
486 m_i = zk.shape[1]
487 n_sensor = len(zk)
488 # Map each sensor sample index to the nearest base-rate index
489 for k in range(n_sensor):
490 base_idx = int(round(k * base_fs / fs_sensor))
491 if base_idx < N_base:
492 zk_out[base_idx, col:col + m_i] = zk[k]
493 col += m_i
494
495 return zk_out

564def cached_fit(path, fit, *, load, save, force=None, verbose=True):
565 """Run ``fit()`` unless ``path`` already holds its result, in which case ``load(path)``.
566
567 After a fit the result is stored with ``save(path)``. A training step that takes
568 minutes then costs a file read on every later run of the notebook -- and the
569 plots downstream see the very same weights, not a retrained near-copy.
570
571 ``force=True`` (or ``KALMAN_CACHE_REFRESH=1`` in the environment) retrains and
572 overwrites, which is what to do after changing the training parameters: the
573 cache is keyed by path alone, on purpose, so that a notebook stays a plain
574 notebook and a stale entry is deleted rather than reasoned about.
575
576 Returns whatever ``fit()`` returned, or ``load(path)``'s result on a cache hit.
577 """
578 import os
579 from pathlib import Path
580 path = Path(path)
581 if force is None:
582 force = os.environ.get("KALMAN_CACHE_REFRESH", "").strip() not in ("", "0", "false", "no")
583 if path.exists() and not force:
584 if verbose:
585 print(f"[cache] loading {path.name} (set KALMAN_CACHE_REFRESH=1 to retrain)")
586 return load(path)
587 result = fit()
588 path.parent.mkdir(parents=True, exist_ok=True)
589 save(path)
590 if verbose:
591 print(f"[cache] saved {path.name}")
592 return result
