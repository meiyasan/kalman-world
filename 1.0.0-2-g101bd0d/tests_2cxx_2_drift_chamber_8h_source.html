57 DriftChamber(std::size_t nLayers = 8, double Bz = 1.5, double rInner = 20., double rStep = 12.,
58 double q = 1., double sigmaHit = 0.013, double pSeed = 1.,
59 double qPos = 1e-6, double qMom = 1e-4, double step = 0.5)
60 : Kalman::Models::Extended("CDC", 4, 2), fBz(Bz), fQ(q), fSigmaHit(sigmaHit),
61 fPSeed(pSeed), fStep(step)
62 {
63 for (std::size_t l = 0; l < nLayers; ++l) fRadii.push_back(rInner + rStep * static_cast<double>(l));
64 SetDefaultEngine(Kalman::EngineKind::Combinatorial);
65 SetSampleRate(1.);
66
67 Kalman::Matrix Q(4, 4, 0.);
68 Q(0, 0) = qPos; Q(1, 1) = qPos; Q(2, 2) = qMom; Q(3, 3) = qMom;
69 SetQ(Q);
70 Kalman::Matrix R = Kalman::Matrix::Identity(2);
71 R *= sigmaHit * sigmaHit;
72 SetR(R);
73 Kalman::Matrix H(2, 4, 0.);
74 H(0, 0) = 1.; H(1, 1) = 1.;
75 SetH(H);
76 }

84 void SeedState(const Kalman::Vector &hit, Kalman::Vector &x0, Kalman::Matrix &P0) const override
85 {
86 double r = std::hypot(hit(0), hit(1));
87 if (r == 0.) r = 1.;
88 x0 = Kalman::Vector(4);
89 x0(0) = hit(0); x0(1) = hit(1);
90 x0(2) = fPSeed * hit(0) / r; x0(3) = fPSeed * hit(1) / r;
91 P0 = Kalman::Matrix(4, 4, 0.);
92 P0(0, 0) = P0(1, 1) = fSigmaHit * fSigmaHit;
93 P0(2, 2) = P0(3, 3) = fPSeed * fPSeed;

96 // --- dynamics: a circular arc, propagated by radius ------------------------
97 Kalman::Vector Rhs(const Kalman::Vector &s) const
98 {
99 const double p = std::hypot(s(2), s(3));
100 Kalman::Vector d(4, 0.);
101 if (p == 0.) return d;
102 const double c = kKappa * fQ * fBz / p;
103 d(0) = s(2) / p; d(1) = s(3) / p; d(2) = c * s(3); d(3) = -c * s(2);
104 return d;
105 }
106
107 Kalman::Vector RK4(const Kalman::Vector &s, double dl) const
108 {
109 const Kalman::Vector k1 = Rhs(s);
110 Kalman::Vector s2 = s; s2 += 0.5 * dl * k1;
111 const Kalman::Vector k2 = Rhs(s2);
112 Kalman::Vector s3 = s; s3 += 0.5 * dl * k2;
113 const Kalman::Vector k3 = Rhs(s3);
114 Kalman::Vector s4 = s; s4 += dl * k3;
115 const Kalman::Vector k4 = Rhs(s4);
116 Kalman::Vector out = s;
117 out += (dl / 6.) * (k1 + 2. * k2 + 2. * k3 + k4);
118 return out;
119 }

122 bool PropagateToRadius(Kalman::Vector s, double rTarget, Kalman::Vector &out) const
123 {
124 const double r0 = std::hypot(s(0), s(1));
125 if (rTarget <= r0 + 1e-12) { out = s; return true; }
126 const double budget = 4. * (rTarget - r0) + 20.;
127 double travelled = 0.;
128 while (travelled < budget)
129 {
130 const Kalman::Vector next = RK4(s, fStep);
131 if (std::hypot(next(0), next(1)) >= rTarget)
132 {
133 double lo = 0., hi = fStep; // bisect the last sub-step onto rTarget
134 for (int i = 0; i < 40; ++i)
135 {
136 const double mid = 0.5 * (lo + hi);
137 const Kalman::Vector sm = RK4(s, mid);
138 if (std::hypot(sm(0), sm(1)) >= rTarget) hi = mid; else lo = mid;
139 }
140 out = RK4(s, hi);
141 return true;
142 }
143 s = next;
144 travelled += fStep;
145 }
146 return false;
147 }

149 Kalman::Vector f(const Kalman::Vector &x, const Kalman::Vector &, double t = 0.) const override
150 {
151 Kalman::Vector out;
152 if (PropagateToRadius(x, t, out)) return out;

154 }

163 // --- synthetic events, with this model's own propagator -------------------
164 TrackingEvent GenerateEvent(std::size_t nTracks = 3, std::size_t nBackground = 4,
165 double efficiency = 1., std::uint64_t seed = 0,
166 double ptMin = 0.7, double ptMax = 1.5,
167 double phiMin = 0.2, double phiMax = 1.3,
168 double originSigma = 0.05) const
169 {
170 std::mt19937_64 rng(seed);
171 std::uniform_real_distribution<double> uniform(0., 1.);
172 std::normal_distribution<double> gauss(0., 1.);
173
174 const std::size_t L = fRadii.size();
175 TrackingEvent ev;
176 ev.hits.assign(L, {});
177 ev.truth.assign(L, {});
178 ev.radii = fRadii;
179
180 const double spread = (phiMax - phiMin) / static_cast<double>(std::max<std::size_t>(1, nTracks));
181 for (std::size_t tid = 0; tid < nTracks; ++tid)
182 {
183 const double base = nTracks > 1
184 ? phiMin + (phiMax - phiMin) * static_cast<double>(tid) / static_cast<double>(nTracks - 1)
185 : 0.5 * (phiMin + phiMax);
186 const double phi = base + (uniform(rng) * 0.3 - 0.15) * spread;
187 const double pt = ptMin + (ptMax - ptMin) * uniform(rng);
188 Kalman::Vector s(4);
189 s(0) = originSigma * gauss(rng); s(1) = originSigma * gauss(rng);
190 s(2) = pt * std::cos(phi); s(3) = pt * std::sin(phi);
191
192 std::vector<Kalman::Vector> states;
193 for (std::size_t l = 0; l < L; ++l)
194 {
195 Kalman::Vector next;
196 if (!PropagateToRadius(s, fRadii[l], next)) break;
197 s = next;
198 states.push_back(s);
199 if (uniform(rng) <= efficiency)
200 {
201 Kalman::Vector pos(2);
202 pos(0) = s(0) + fSigmaHit * gauss(rng);
203 pos(1) = s(1) + fSigmaHit * gauss(rng);
204 ev.hits[l].push_back(pos);
205 ev.truth[l].push_back(static_cast<long>(tid));
206 }
207 }
208 if (!states.empty()) ev.tracks.push_back(states);
209 }
210
211 for (std::size_t l = 0; l < L; ++l)
212 for (std::size_t b = 0; b < nBackground; ++b)
213 {
214 const double ang = (phiMin - 0.3) + (phiMax - phiMin + 0.6) * uniform(rng);
215 Kalman::Vector pos(2);
216 pos(0) = fRadii[l] * std::cos(ang) + fSigmaHit * gauss(rng);
217 pos(1) = fRadii[l] * std::sin(ang) + fSigmaHit * gauss(rng);
218 ev.hits[l].push_back(pos);
219 ev.truth[l].push_back(-1L);
220 }
221 return ev;
222 }
223 };
