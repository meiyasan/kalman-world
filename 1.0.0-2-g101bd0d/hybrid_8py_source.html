7class HybridRouter(KalmanRouter):
8 """Geometric blend (log-linear opinion pool) of two routers.
9
10 μ = router_a(...)^(1−α) · router_b(...)^α / Z
11
12 At α = 0 → pure router_a. At α = 1 → pure router_b.
13
14 Common uses::
15
16 # Bayesian anchor + learned refinement
17 HybridRouter(BayesianRouter(), NeuralRouter(MLP()), alpha=0.3)
18
19 # Learned primary + rule-based guard
20 HybridRouter(NeuralRouter(MLP()), ThresholdRouter(soft=True), alpha=0.5)
21
22 ``fit()`` is forwarded to both children.
23
24 Parameters
25 ----------
26 router_a, router_b : KalmanRouter
27 alpha : float in [0, 1]
28 Blend weight toward router_b.
29
30 Blended by ``Kalman::Routers::Hybrid`` when libKalman is installed. A child
31 with a C++ counterpart (Bayesian, Threshold, Hybrid) is evaluated there; any
32 other child (a NeuralRouter, a plain callable) is called back from C++.
33 :meth:`_reference_route` is the Python reference.
34 """

50 def _reference_route(self, nis_arr, c, likelihoods):
51 mu_a = np.maximum(np.asarray(self.router_a(nis_arr, c, likelihoods), float), 1e-12)
52 mu_b = np.maximum(np.asarray(self.router_b(nis_arr, c, likelihoods), float), 1e-12)
53 return self._normalise(mu_a ** (1 - self.alpha) * mu_b ** self.alpha)
