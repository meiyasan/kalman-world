35 void Sampler::Generate(std::size_t nsteps, const Vector &x0)
36 {
37 if (nsteps == 0) throw std::runtime_error("Kalman::Sampler::Generate: nsteps must be positive.");
38
39 fTruth.clear();
40 fMeasure.clear();
41 fTime.clear();
42 fTruth.reserve(nsteps);
43 fMeasure.reserve(nsteps);
44 fTime.reserve(nsteps);
45
46 Vector x = (x0.Size() == fModel->NState()) ? x0 : fModel->X0();
47 const Vector u(fModel->NControl(), 0.);
48
49 const double dt = fModel->DeltaT();
50 double t = fModel->T0();
51
52 // Q and R are evaluated per step so a time-varying model is sampled
53 // correctly; the factorisation is redone only when the matrix changed.
54 Matrix Qfac, Rfac;
55 bool haveQ = false, haveR = false;
56 double factoredAt = 0.;
57
58 for (std::size_t k = 0; k < nsteps; ++k)
59 {
60 Matrix Q = fModel->Q(t);
61 const Matrix L = fModel->L(t);
62 if (!L.Empty()) Q = Utils::Similarity(L, Q);
63
64 if (!haveQ || t != factoredAt) { Qfac = Utils::CholeskyFactor(Q); haveQ = true; }
65 if (!haveR || t != factoredAt) { Rfac = Utils::CholeskyFactor(fModel->R(t)); haveR = true; }
66 factoredAt = t;
67
68 x = fModel->f(x, u, t);
69 const Vector qw = Draw(Qfac);
70 if (!qw.Empty()) x += qw;
71
72 Vector z = fModel->h(x, t);
73 const Vector rw = Draw(Rfac);
74 if (!rw.Empty()) z += rw;
75
76 fTruth.push_back(x);
77 fMeasure.push_back(z);
78 fTime.push_back(t);
79
80 t += dt;
81 }
82 }
