46class Ballistics : public Kalman::Models::Extended
47{
48 private:
49 double fGravity = 9.81; // g [m/s^2]
50 double fDragOverMass = 0.; // c_drag / m [1/m]
51
52 public:
53 Ballistics(double fs, double dragOverMass)
54 : Kalman::Models::Extended("ballistics", 6, 3), fDragOverMass(dragOverMass)
55 {
56 // Models::Extended already declared the extended engine and switched
57 // the Jacobian to central differences, which is what a nonlinear f()
58 // needs -- F(t) is not its Jacobian.
59 SetSampleRate(fs);
60
61 Kalman::Matrix H(3, 6, 0.);
62 for (std::size_t i = 0; i < 3; ++i) H(i, i) = 1.; // observe position
63 SetH(H);
64
65 const double dt = DeltaT();
66 const double q = 0.5 * 0.5; // sigma_a^2 [m/s^2]^2
67
68 // Discrete white-noise acceleration per axis: an unmodelled acceleration
69 // of standard deviation sigma_a, held constant over each step. It moves
70 // position and velocity together, hence the off-diagonal terms.
71 Kalman::Matrix Q(6, 6, 0.);
72 for (std::size_t i = 0; i < 3; ++i)
73 {
74 Q(i, i) = q * dt * dt * dt * dt / 4.;
75 Q(i, i + 3) = q * dt * dt * dt / 2.;
76 Q(i + 3, i) = Q(i, i + 3);
77 Q(i + 3, i + 3) = q * dt * dt;
78 }
79 SetQ(Q);
80
81 Kalman::Matrix R = Kalman::Matrix::Identity(3);
82 R *= 4.0; // sigma_z = 2 m per axis
83 SetR(R);
84
85 Kalman::Matrix P0 = Kalman::Matrix::Identity(6);
86 P0 *= 100.;
87 SetInitialCovariance(P0);
88 }
89
91 Kalman::Vector f(const Kalman::Vector &x, const Kalman::Vector &,
92 double = 0.) const override
93 {
94 const double dt = DeltaT();
95
96 const double vx = x(3), vy = x(4), vz = x(5);
97 const double speed = std::sqrt(vx * vx + vy * vy + vz * vz);
98
99 // a = -g z_hat - (c/m) |v| v
100 const double ax = -fDragOverMass * speed * vx;
101 const double ay = -fDragOverMass * speed * vy;
102 const double az = -fGravity - fDragOverMass * speed * vz;
103
104 Kalman::Vector out(6);
105 // Position and velocity advance with the acceleration held constant over
106 // the step. That is exact for a constant acceleration; drag changes the
107 // acceleration during the step, so this is an explicit first-order step
108 // whose error shrinks with dt.
109 out(0) = x(0) + dt * vx + 0.5 * ax * dt * dt;
110 out(1) = x(1) + dt * vy + 0.5 * ay * dt * dt;
111 out(2) = x(2) + dt * vz + 0.5 * az * dt * dt;
112 out(3) = vx + dt * ax;
113 out(4) = vy + dt * ay;
114 out(5) = vz + dt * az;
115 return out;
116 }
117
118 Kalman::Vector h(const Kalman::Vector &x, double = 0.) const override
119 {
120 Kalman::Vector z(3);
121 z(0) = x(0); z(1) = x(1); z(2) = x(2);
122 return z;
123 }
124};

126int main()
127{
128 // --- the system ------------------------------------------------------
129 // c_drag/m = 2e-3 1/m is a light ball (terminal speed sqrt(g m/c) = 70 m/s),
130 // not an artillery shell.
131 Ballistics model(/* fs */ 50., /* c_drag/m */ 2e-3);
132
133 Kalman::Vector x0(6);
134 x0(0) = 0.; x0(1) = 0.; x0(2) = 0.; // launch at the origin
135 x0(3) = 60.; x0(4) = 15.; x0(5) = 90.; // m/s
136 model.SetInitialState(x0);
137
138 Printf("%s", model.Describe().c_str());
139 Printf("engine kind declared by the model : %s", Kalman::ToString(model.DefaultEngine()));
140
141 // --- the data (truth + noisy position fixes) --------------------------
142 Kalman::Sampler sampler(&model, 20260804);
143 sampler.Generate(600, x0);
144
145 // The altitude coordinate (channel 2) is missing from 15% of the fixes. NaN
146 // marks the gap: the filter updates with the two coordinates it has and
147 // predicts the altitude, with no special case anywhere.
148 sampler.DropChannel(2, 0.15);
149
150 // --- the estimate -----------------------------------------------------
151 Kalman::Filters::EKF filter(&model);
152 const Kalman::Record forward = filter.Forward(sampler.Measurements());
153 const Kalman::Record smoothed = filter.Backward();
154
155 Printf("mean NIS = %.3f (consistent near m = %zu)", filter.MeanNIS(), model.NMeasure());
156 Printf("mean NEES = %.3f (consistent near n = %zu)", filter.MeanNEES(sampler.Truth()),
157 model.NState());
158 Printf("altitude RMSE : filtered = %.3f m, smoothed = %.3f m",
159 forward.RMSE(2, sampler.Truth()), smoothed.RMSE(2, sampler.Truth()));
160
161 // --- ROOT takes it from here -----------------------------------------
162 gStyle->SetOptStat(0);
163 TCanvas *c = new TCanvas("ballistics", "Ballistic tracking (EKF)", 1100, 620);
164 c->SetGrid();
165
166 const std::vector<double> t = forward.Times();
167 const std::vector<double> zEst = forward.Series(2); // altitude estimate
168 const std::vector<double> zSig = forward.Sigma(2); // its 1-sigma
169 const std::vector<double> zSm = smoothed.Series(2);
170 const std::vector<double> zMeas = forward.MeasurementSeries(2);
171 std::vector<double> ex(forward.NSteps(), 0.);
172 std::vector<double> zTruth(forward.NSteps());
173 for (std::size_t k = 0; k < forward.NSteps(); ++k) zTruth[k] = sampler.Truth()[k](2);
174
175 TMultiGraph *mg = new TMultiGraph();
176
177 TGraphErrors *band = new TGraphErrors(forward.NSteps(), t.data(), zEst.data(),
178 ex.data(), zSig.data());
179 band->SetFillColorAlpha(kAzure - 9, 0.6);
180 band->SetLineColor(kAzure + 2);
181 band->SetLineWidth(2);
182 mg->Add(band, "L3");
183
184 // The dropped fixes are NaN; skip them rather than drawing a spike to zero.
185 TGraph *gz = new TGraph();
186 for (std::size_t k = 0; k < forward.NSteps(); ++k)
187 if (!std::isnan(zMeas[k])) gz->SetPoint(gz->GetN(), t[k], zMeas[k]);
188 gz->SetMarkerStyle(kFullCircle);
189 gz->SetMarkerSize(0.4);
190 gz->SetMarkerColor(kGray + 2);
191 mg->Add(gz, "P");
192
193 TGraph *gs = new TGraph(forward.NSteps(), t.data(), zSm.data());
194 gs->SetLineColor(kRed + 1);
195 gs->SetLineWidth(2);
196 mg->Add(gs, "L");
197
198 TGraph *gt = new TGraph(forward.NSteps(), t.data(), zTruth.data());
199 gt->SetLineColor(kBlack);
200 gt->SetLineStyle(kDashed);
201 mg->Add(gt, "L");
202
203 mg->SetTitle("Ballistic trajectory: 15% of altitude fixes dropped;time [s];altitude [m]");
204 mg->Draw("A");
205
206 // Bottom-right: the only corner the trajectory does not pass through. A
207 // legend defaults to an opaque fill, so parking it over the curve silently
208 // erases a stretch of the data and reads exactly like a gap in the input.
209 TLegend *leg = new TLegend(0.62, 0.15, 0.88, 0.39);
210 leg->SetFillStyle(0);
211 leg->SetBorderSize(1);
212 leg->AddEntry(gt, "truth", "l");
213 leg->AddEntry(gz, "radar fix", "p");
214 leg->AddEntry(band, "EKF #pm 1#sigma", "lf");
215 leg->AddEntry(gs, "RTS smoother", "l");
216 leg->Draw();
217
218 c->SaveAs("ballistics_root.pdf");
219 c->SaveAs("ballistics_root.png"); // handy for a README or a log
220
221 TFile out("ballistics_root.root", "RECREATE");
222 mg->Write("trajectory");
223 out.Close();
224
225 Printf("wrote ballistics_root.pdf, .png and .root");
226 return 0;
227}
