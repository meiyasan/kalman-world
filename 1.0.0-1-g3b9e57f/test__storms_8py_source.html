174 def test_learned_intensity_beats_persistence(self):
175 """The ML elaboration: a learned SST/shear intensity model cuts the
176 intensity-forecast error of the physics-persistence baseline."""
177 def steer(lon, lat, t=None): return analytic_steering(lon, lat, hemisphere="N")
178 rng = np.random.default_rng(0)
179 train = [synthetic_steered_track(n_steps=48, seed=s, I0=rng.uniform(25, 55),
180 lat0=rng.uniform(7, 13), lon0=rng.uniform(155, 172))[0]

185 def build(ifn, tr):
186 m = StormModel("TC", steering=steer, intensity_fn=ifn, q_pos=0.25,
187 x0=np.array([[tr[0, 0]], [tr[0, 1]], [-20.], [3.], [35.]]))
188 m.initial_state(m.x0, np.diag([0.5, 0.5, 30, 30, 100.]))
189 return m
190 t_a, lead = 6, 14
191 gains = []
192 for s in range(200, 220):
193 tr, fx = synthetic_steered_track(n_steps=48, seed=s)
194 r = ExtendedKalmanFilter(build(None, tr)).forward(fx[:t_a])
195 x0, P0 = np.asarray(r.xk)[t_a-1].reshape(-1, 1), np.asarray(r.Pk)[t_a-1]
196 cp = build(None, tr).forecast_cone(x0, P0, steps=lead)
197 cm = build(lim.as_intensity_fn(), tr).forecast_cone(x0, P0, steps=lead)
198 tt = tr[t_a:t_a+lead, 4]
199 ep = np.sqrt(np.mean((cp["I"] - tt) ** 2))
200 em = np.sqrt(np.mean((cm["I"] - tt) ** 2))
201 gains.append(1 - em / ep)
202 self.assertGreater(np.mean(gains), 0.10) # ML consistently helps intensity
