1"""Pseudo-spectral 2D field solver (FFT discretization + composed time integrator).
2
3The `SPECTRAL` :class:`~kalman.framework.solver.KalmanSolver`: it owns the spatial
4part (FFT, 2/3-rule dealiasing, spectral ``ik`` operators) and delegates the time
5march to a composed :class:`~kalman.framework.integrator.KalmanIntegrator` (RK4)
6applied to the **spectral** field. Subclasses supply the physics via
7``rhs_hat(wh) → dω̂/dt``; this class is domain-agnostic (the Kolmogorov forcing
8lives in ``apps.weather.flow.NS2D``, which subclasses this).
9
10**Backend.** ``backend="numpy"`` (default) or ``"torch"`` with ``device`` in
11``cpu/cuda/mps``. The torch path accepts a batched ``(N, g, g)`` / ``(N, g²)``
12ensemble as one advance — the EnKF forecast's dominant cost — and returns numpy,
13so callers are unchanged. It runs in C++ on libTorch
14(``Kalman::Learn::Solvers::Vorticity`` in ``libKalmanLearn``) for a solver that
15states its physics in ``_vorticity_terms``. Python's torch package is not involved.
16"""

32 def __init__(self, n=64, dt=0.01, *, length=2*np.pi, integrator=None,
33 backend=None, device=None, seed=0):
34 """``n`` is the grid: an ``int`` for a square ``n×n`` mesh, or
35 ``(ny, nx)`` for a **rectangular** one. ``length`` is the physical
36 domain size — a scalar or ``(Ly, Lx)`` — so wavenumbers are scaled
37 ``2π/L`` (default ``2π`` reproduces integer wavenumbers exactly).
38
39 Spectral discretization requires a **regular periodic** mesh; bounded /
40 irregular grids are the job of the finite-difference/volume/element
41 solver kinds."""
42 super().__init__(integrator=integrator, backend=backend, device=device)
43 ny, nx = (int(n), int(n)) if np.isscalar(n) else (int(n[0]), int(n[1]))
44 Ly, Lx = (float(length), float(length)) if np.isscalar(length) else (float(length[0]), float(length[1]))
45 self.ny, self.nx = ny, nx
46 self.shape = (ny, nx)
47 self.Ly, self.Lx = Ly, Lx
48 self.n = nx if nx == ny else None # square-grid back-compat (None if rectangular)
49 self.dt = float(dt)
50 # Integer mode indices (used for the 2/3 dealias rule) and the physical
51 # wavenumbers k = 2π/L · index (identical to the index for L = 2π).
52 ix = np.fft.fftfreq(nx, 1.0 / nx)
53 iy = np.fft.fftfreq(ny, 1.0 / ny)
54 self._ix = ix[None, :] * np.ones((ny, 1))
55 self._iy = iy[:, None] * np.ones((1, nx))
56 self.kx = (2 * np.pi / Lx) * self._ix
57 self.ky = (2 * np.pi / Ly) * self._iy
58 self.k2 = self.kx**2 + self.ky**2
59 self.k2inv = np.where(self.k2 > 0, 1.0 / np.maximum(self.k2, 1e-12), 0.0)
60 self.dealias = (np.abs(self._ix) <= nx // 3) & (np.abs(self._iy) <= ny // 3)
61 self.rng = np.random.default_rng(seed)
62 self._native = None # the libTorch solver, made on first advance
63
64 # ------------------------------------------------------------------
65 def _as_grid(self, w):
66 """Validate/reshape an input field to ``(ny, nx)``; raise a clear error
67 when the given 2-D gridding does not match this solver's mesh."""
68 arr = np.asarray(w, dtype=float)
69 ny, nx = self.shape
70 if arr.ndim == 1:
71 if arr.size != ny * nx:
72 raise ValueError(f"field of size {arr.size} does not fit this solver's "
73 f"{ny}x{nx} grid ({ny*nx} cells).")
74 return arr.reshape(ny, nx)
75 if arr.shape[-2:] != (ny, nx):
76 raise ValueError(f"field grid {arr.shape[-2:]} != solver grid {(ny, nx)}; "
77 f"construct the solver with n=({arr.shape[-2]}, {arr.shape[-1]}).")
78 return arr

90 # ------------------------------------------------------------------
91 # Spectral differential operators (numpy; single source of truth)
92 # ------------------------------------------------------------------
93 def _grad_spectral(self, wh):
94 wx = np.real(np.fft.ifft2(1j * self.kx * wh))
95 wy = np.real(np.fft.ifft2(1j * self.ky * wh))
96 return wx, wy
97
98 def _vel_spectral(self, wh):
99 psih = wh * self.k2inv
100 u = np.real(np.fft.ifft2(1j * self.ky * psih))
101 v = np.real(np.fft.ifft2(-1j * self.kx * psih))
102 return u, v

104 def gradient(self, w):
105 """∇ω = (∂ω/∂x, ∂ω/∂y) of a real field ``(ny,nx)``/flat."""
106 return self._grad_spectral(np.fft.fft2(self._as_grid(w)))
107
108 def velocity(self, w):
109 """Incompressible velocity (u, v) of a real field ``(ny,nx)``/flat."""
110 return self._vel_spectral(np.fft.fft2(self._as_grid(w)))

112 # ------------------------------------------------------------------
113 # Time stepping — RK4 (the composed integrator) on the spectral field
114 # ------------------------------------------------------------------
115 @staticmethod
116 def _rk_step(rhs, x, dt, tableau):
117 """One explicit Runge–Kutta step from a Butcher ``tableau`` (c, A, b),
118 so *any*
119 explicit integrator drives the spectral field uniformly."""
120 c, A, b = tableau
121 ks = []
122 for i in range(len(c)):
123 if i == 0:
124 xi = x
125 else:
126 acc = A[i][0] * ks[0]
127 for j in range(1, i):
128 acc = acc + A[i][j] * ks[j]
129 xi = x + dt * acc
130 ks.append(rhs(xi))
131 out = b[0] * ks[0]
132 for i in range(1, len(b)):
133 out = out + b[i] * ks[i]
134 return x + dt * out
135
136 def _tableau(self):
137 tab = self.integrator.tableau
138 if tab is None:
139 raise TypeError(f"{self.integrator.name()} has no explicit tableau; the spectral "
140 f"solver needs an explicit RK integrator (Euler/RK4/Dormand-Prince).")
141 return tab

143 def step(self, wh):
144 """One step of the spectral field with the composed integrator (numpy)."""
145 return self._rk_step(self.rhs_hatrhs_hat, wh, self.dt, self._tableau())

147 def advance(self, w, nsteps):
148 """Advance a real field ``w`` by ``nsteps``. Accepts ``(ny,nx)``, flat
149 ``(ny·nx,)``, or a batched ensemble ``(N, ny, nx)`` / ``(N, ny·nx)``;
150 returns the same layout. Uses the torch backend when configured."""
151 if self.backendbackend == "torch":
152 return self._advance_batched(w, int(nsteps))
153 arr = np.asarray(w, dtype=float)
154 flat = arr.ndim == 1
155 scheme = self._native_scheme()
156 if scheme is not None: # C++: Kalman::Solvers::Vorticity
157 from ..native import core
158 terms = self._vorticity_terms()
159 grid = self._as_grid(arr)
160 out = core.solve_vorticity(self.shape, self.dt, (self.Ly, self.Lx), terms["nu"], terms.get("beta", 0.0),
161 terms.get("forcing_hat"), scheme, grid, int(nsteps)).reshape(grid.shape)
162 return out.ravel() if flat else out
163 wh = np.fft.fft2(self._as_grid(arr))
164 for _ in range(int(nsteps)):
165 wh = self.step(wh)
166 out = np.real(np.fft.ifft2(wh))
167 return out.ravel() if flat else out

169 def solve(self, field, steps=1, u=None, t=None):
170 return self.advance(field, int(steps))

172 def random_ic(self, amp=1.0):
173 """Random large-scale initial field, unit std (numpy RNG)."""
174 wh = (self.rng.standard_normal(self.shape) +
175 1j * self.rng.standard_normal(self.shape))
176 kmag = np.sqrt(self._ix**2 + self._iy**2) # in mode-index units
177 wh *= np.exp(-0.5 * (kmag / 6.0) ** 2) * self.dealias
178 w = np.real(np.fft.ifft2(wh))
179 return amp * w / (w.std() + 1e-12)

182 # torch backend (CPU/CUDA/MPS) — batched spectral steps on libTorch
183 #
184 # The steps run in C++: Kalman::Learn::Solvers::Vorticity in libKalmanLearn.
185 # Python's torch package is not used. A solver says what its physics is in
186 # ``_vorticity_terms``; the grid and the integrator's tableau come from here.
187 # ------------------------------------------------------------------
188 def _vorticity_terms(self):
189 """The physics as the terms of ``dw/dt = -u.grad(w) - beta v + nu lap(w) + F``:
190 a dict with ``nu`` and optionally ``beta`` and ``forcing_hat`` (the transform
191 of ``F``). ``None`` (the default) when the equation is not of that form."""
192 return None
193
194 def _states_its_physics(self) -> bool:
195 """Whether ``_vorticity_terms`` describes what this object computes: the
196 class that gives the terms is also the one whose ``rhs_hat`` runs. A
197 subclass that changes ``rhs_hat`` without its terms is other physics."""
198 if self._vorticity_terms() is None:
199 return False
200 cls = type(self)
201 owner = next(k for k in cls.__mro__ if "_vorticity_terms" in k.__dict__)
202 return cls.rhs_hatrhs_hat is owner.rhs_hat and "rhs_hat" not in self.__dict__
203
204 def _native_scheme(self):
205 """The C++ integrator's name when this solver's numpy path can run in C++
206 (Kalman::Solvers::Vorticity): its physics is stated, its integrator is a
207 built-in scheme and its ``step`` is not overridden. Else ``None``."""
208 from ..integrator import _BUILTIN_STEPS
209 from ..models import native
210 if not native.native_loop_enabled() or type(self).step is not SpectralSolver.step:
211 return None
212 integrator = self.integrator
213 name = getattr(integrator, "NATIVE", None)
214 if name is None or type(integrator)._step is not _BUILTIN_STEPS.get(name):
215 return None
216 return name if self._states_its_physics() else None
217
218 def _libtorch(self):
219 """The libTorch solver for this grid and physics, made on first use."""
220 if self._native is None:
221 terms = self._vorticity_terms() if self._states_its_physics() else None
222 if terms is None:
223 raise NotImplementedError(
224 f"{type(self).__name__}: backend='torch' runs the C++ libTorch solver, which steps "
225 "dw/dt = -u.grad(w) - beta v + nu lap(w) + F. Return those terms from "
226 "_vorticity_terms(), or use backend='numpy' with rhs_hat().")
227 from ..native import learn
228 self._native = learn.Vorticity(self.kx, self.ky, self.k2, self.k2inv, self.dealias,
229 device=self.devicedevice or "cpu", **terms)
230 self.devicedevice = self._native.device
231 return self._native

233 def _advance_batched(self, w, nsteps):
234 arr = np.asarray(w, dtype=float)
235 ny, nx = self.shape
236 squeeze = False
237 if arr.ndim == 1: # (ny·nx,) single
238 x = self._as_grid(arr)[None]; flatten = "flat"
239 elif arr.ndim == 2 and arr.shape == (ny, nx): # (ny,nx) single
240 x = arr.reshape(1, ny, nx); flatten = "grid"; squeeze = True
241 elif arr.ndim == 2: # (N, ny·nx) batch
242 if arr.shape[1] != ny * nx:
243 raise ValueError(f"batched rows of size {arr.shape[1]} do not fit the "
244 f"{ny}x{nx} grid.")
245 x = arr.reshape(-1, ny, nx); flatten = "flat"
246 else: # (N, ny, nx) batch
247 x = self._as_grid(arr); flatten = "grid"
248 tab = self._tableau() # same integrator as numpy path
249 out = self._libtorch().advance(x, nsteps, self.dt, tab)
250 if squeeze:
251 return out[0]
252 if arr.ndim == 1:
253 return out[0].ravel()
254 if flatten == "flat":
255 return out.reshape(out.shape[0], ny * nx)
256 return out
