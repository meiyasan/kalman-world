1"""
2Sensor fusion — one latent target, several asynchronous, differently-noisy sensors.
3
4The essence of Kalman sensor fusion (cf. satya15july, *"Sensor Fusion with Kalman
5Filter"*) is that **stacking measurements** in a single update optimally weights each
6sensor by its inverse covariance, so the fused estimate beats *any* single sensor — and
7sensors running at different rates fall out for free: a sensor that does not fire on a
8given step is simply a ``NaN`` row, which the engine skips (predict-only for that
9channel).
10
11A :class:`SensorFusionModel` tracks a 1-D constant-velocity target ``x = [position,
12velocity]`` observed by a **configurable list of** :class:`SensorSpec` (each declares what
13it observes — ``"position"`` or ``"velocity"`` — its noise, and its rate). This single,
14declarative model covers both demonstrated regimes:
15
16 * **redundant** — several position sensors (the stereo-camera + lidar analog);
17 * **complementary** — a slow noisy position sensor + a fast velocity sensor (the
18 GPS + wheel-odometry analog).
19
20Multi-rate measurements are built with :func:`kalman.helpers.make_multirate_input`.
21"""

69class SensorFusionModel(LinearKalmanModel):
70 """Constant-velocity 1-D target fused from a list of :class:`SensorSpec`.
71
72 Parameters
73 ----------
74 sensor_id : str
75 Channel / group name.
76 sensors : list[SensorSpec]
77 The sensors to fuse (stacked as rows of ``H``; ``R = diag(σ_i²)``).
78 fs : float
79 Base filter sample rate [Hz].
80 process_uncertainty : float
81 White-noise-acceleration std [m/s²/√Hz] driving the constant-velocity ``Q`` block.
82 """
83
84 NAME = "Sensor Fusion Model"
85 DESCRIPTION = ("Constant-velocity 1-D target fused from a configurable list of sensors "
86 "(position/velocity, per-sensor noise and rate).")
87 VERSION = "1.0"

97 def __init__(self, sensor_id: str | int,
98 sensors: list[SensorSpec],
99 fs: float = 50.0,
100 process_uncertainty: float = 1.0,
101 **kwargs):
102
103 self.dtypedtype = kwargs.pop("dtype", np.float64)
104 if not sensors:
105 raise ValueError("SensorFusionModel needs at least one SensorSpec.")
106 self.sensors = list(sensors)
107
108 # Continuous-time generator: ṗ=v, v̇=0 (constant velocity). Passed as
109 # A= (not F=): the base KalmanModel discretizes exactly via expm(A/fs).
110 A = np.array([[0.0, 1.0],
111 [0.0, 0.0]], dtype=self.dtypedtype)
112
113 # Stack one observation row per sensor.
114 rows, rvar = [], []
115 for s in self.sensors:
116 rows.append([1.0, 0.0] if s.observes == "position" else [0.0, 1.0])
117 rvar.append(self.dtypedtype(s.uncertainty) ** 2)
118 H = np.array(rows, dtype=self.dtypedtype)
119 R = np.diag(rvar).astype(self.dtypedtype)
120
121 # Continuous white-noise-acceleration disturbance driving v̇ (Lc picks
122 # out that row); passed as Qc=/Lc= so the base model Van Loan-discretizes
123 # it exactly alongside A=.
124 Lc = np.array([[0.0], [1.0]], dtype=self.dtypedtype)
125 Qc = np.array([[self.dtypedtype(process_uncertainty) ** 2]], dtype=self.dtypedtype)
126
127 self.mmmm = len(self.sensors)
128 if kwargs.get("P0") is None:
129 kwargs["P0"] = np.diag([10.0, 10.0]).astype(self.dtypedtype)
130
131 super().__init__(sensor_id, A=A, H=H, Qc=Qc, Lc=Lc, R=R, fs=fs, **kwargs)

138 # Multi-rate synthetic measurements from a sampled truth trajectory
139 # ------------------------------------------------------------------
140 def make_measurements(self, sample, seed=None) -> np.ndarray:
141 """Build the ``NaN``-padded ``(N, n_sensors)`` measurement matrix from a truth record.
142
143 Each sensor samples its channel of ``sample.xk`` at its own rate, adds its own
144 noise, and the streams are interleaved at the base rate via
145 :func:`make_multirate_input` (missing steps → ``NaN`` → predict-only).
146 """
147 if seed is not None:
148 set_random_seed(seed)
149 truth = np.asarray(sample.xk).reshape(len(sample), -1)
150 N = truth.shape[0]
151 base_fs = float(self.fsfsfs)
152 duration = N / base_fs
153
154 sensor_data = []
155 for s in self.sensors:
156 idx = 0 if s.observes == "position" else 1
157 rate = float(s.rate or base_fs)
158 # Sample the truth at this sensor's own firing times (robust to rates that
159 # do not evenly divide base_fs); make_multirate_input re-derives the same
160 # base-rate slots, so the streams interleave consistently.
161 n_i = int(round(duration * rate))
162 base_idx = np.round(np.arange(n_i) * base_fs / rate).astype(int)
163 base_idx = base_idx[base_idx < N]
164 clean = truth[base_idx, idx]
165 noisy = clean + s.uncertainty * np.random.randn(base_idx.size)
166 sensor_data.append((noisy.reshape(-1, 1), rate))
167
168 zk = make_multirate_input(sensor_data, base_fs)
169 # Pin the length to the truth length (guards off-by-one from fractional rates).
170 if zk.shape[0] != N:
171 fixed = np.full((N, zk.shape[1]), np.nan)
172 n = min(N, zk.shape[0])
173 fixed[:n] = zk[:n]
174 zk = fixed
175 return zk

177 def single_sensor_record(self, zk, index: int):
178 """Filter with only sensor ``index`` active (all other channels masked to ``NaN``).
179
180 Yields the baseline each single sensor would achieve on its own — the reference the
181 fused estimate is meant to beat.
182 """
183 zk = np.asarray(zk, dtype=float)
184 masked = np.full_like(zk, np.nan)
185 masked[:, index] = zk[:, index]
186 return self.filter(masked, reset=True)
