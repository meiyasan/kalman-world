1"""Barotropic vorticity on a mid-latitude beta-plane — a model of the atmosphere.
2
3Why this exists
4---------------
5:class:`~.flow.KolmogorovFlowModel` is a superb data-assimilation testbed and a poor model of
6the atmosphere. Assimilating measured ERA5 500 hPa vorticity into it is *worse than
7persistence* — worse, in fact, than the window's own time mean — and no inflation setting
8repairs that, because the failure is not tuning. Kolmogorov flow is forced 2D turbulence on a
9doubly-periodic torus with no rotation; mid-latitude flow is rotation-dominated. Measured at
1050°N, relative vorticity is ~0.40 of the planetary value, so the term the Kolmogorov model
11omits is the same size as the ones it keeps.
12
13The barotropic vorticity equation is the smallest correction that is still *about the right
14system*:
15
16.. code-block:: text
17
18 ∂ζ/∂t = −u·∇ζ − β v + ν ∇²ζ
19
20The one new term, ``−β v``, is advection of planetary vorticity: move north and f increases, so
21ζ must fall. That term alone is what makes the flow support **Rossby waves**, the westward-
22propagating structures that organise mid-latitude weather, and it is the physics the
23Kolmogorov model cannot express at any parameter setting.
24
25This is still not a sphere. It is a beta-plane: f linearised about a reference latitude on a
26doubly-periodic Cartesian box. For a 20°×20° window in the mid-latitudes — which is exactly
27what the ERA5 slice is — that is the approximation the beta-plane was invented for, and it
28buys the dominant missing physics without spherical harmonics. Anything global needs real
29spherical geometry and this class should not be stretched to pretend otherwise.
30
31Units
32-----
33Deliberately **physical** (SI), unlike the non-dimensional Kolmogorov testbed: β is a real
34number about the real planet and the model is worthless if the field it acts on has had its
35amplitude rescaled to something else. Feed it ``reanalysis_truth(..., rescale=False)``.
36"""

51def beta_at(latitude_deg: float) -> float:
52 """Rossby parameter β = df/dy = 2Ω cos(φ) / a, in 1/(m·s).
53
54 About 1.47e-11 in the mid-latitudes. It is small in absolute terms and dominant in
55 practice, because it multiplies a velocity rather than a vorticity.
56 """
57 return 2.0 * OMEGA * np.cos(np.radians(latitude_deg)) / EARTH_RADIUS

60def coriolis_at(latitude_deg: float) -> float:
61 """Planetary vorticity f = 2Ω sin(φ), 1/s — the scale ζ is measured against."""
62 return 2.0 * OMEGA * np.sin(np.radians(latitude_deg))

65def box_size(lat_span_deg: float, lon_span_deg: float, latitude_deg: float):
66 """Physical ``(Ly, Lx)`` in metres for a lat/lon window at a reference latitude.
67
68 Longitude degrees shorten with cos(φ), which is why a 20°×20° window at 50°N is not
69 square: about 2,224 km north-south against 1,430 km east-west.
70 """
71 ly = np.radians(lat_span_deg) * EARTH_RADIUS
72 lx = np.radians(lon_span_deg) * EARTH_RADIUS * np.cos(np.radians(latitude_deg))
73 return float(ly), float(lx)

76class BetaPlaneNS2D(SpectralSolver):
77 """Barotropic vorticity on a beta-plane, pseudo-spectral, in SI units.
78
79 A sibling of :class:`~.flow.NS2D` rather than a subclass: that one hard-codes a
80 non-dimensional ``[0, 2π]²`` box and its Kolmogorov forcing, and this needs a physical
81 rectangular domain and no forcing at all. Real mid-latitude flow is not stirred at one
82 wavenumber — it is driven baroclinically from below, which a barotropic model does not
83 carry — so leaving it unforced is the honest omission. It decays.
84 """
85
86 NAME = "Barotropic beta-plane"
87 DESCRIPTION = "2D barotropic vorticity with the beta effect — mid-latitude flow, SI units."
88
89 def __init__(self, n=64, nu=2.0e4, beta=None, latitude_deg=50.0, dt=60.0,
90 length=(2.224e6, 1.430e6), seed=0, **kw):
91 super().__init__(n=n, dt=dt, length=length, seed=seed, **kw)
92 self.nu = float(nu)
93 self.beta = beta_at(latitude_deg) if beta is None else float(beta)
94 self.latitude_deg = float(latitude_deg)
95
96 def rhs_hat(self, wh, t=None):
97 u, v = self._vel_spectral(wh)
98 wx, wy = self._grad_spectral(wh)
99 # −u·∇ζ − βv, dealiased together since both are products evaluated on the grid.
100 adv = np.fft.fft2(u * wx + v * wy + self.beta * v) * self.dealias
101 return -adv - self.nu * self.k2 * wh
102
103 def _vorticity_terms(self):
104 return dict(nu=self.nu, beta=self.beta)

139 def climatology_from(self, fields):
140 """Use MEASURED fields as the climatology, instead of a synthetic spin-up.
141
142 An unforced barotropic model has no attractor worth sampling — it decays — so its own
143 spin-up produces states that are dynamically self-consistent and meteorologically
144 unlike anything the atmosphere is doing this week. An ensemble drawn from those starts
145 far from the truth in a way no number of assimilation cycles can repair on a short
146 window. Drawing from real fields instead is standard practice (a climatological
147 ensemble) and is the honest fix.
148
149 **The caller must pass a period DISJOINT from the one being scored.** Seeding from the
150 scoring window would put the answer in the initial ensemble and every score after it
151 would be circular — the same failure as tuning process noise on truth. This method
152 cannot check that for you; it is the caller's contract, and the OSSE script splits the
153 window explicitly.
154
155 ``fields`` is ``(T, n)`` or ``(T, grid, grid)``; columns become ensemble candidates.
156 """
157 arr = np.asarray(fields, dtype=float)
158 arr = arr.reshape(arr.shape[0], -1)
159 if arr.shape[1] != self.nnnnn:
160 raise ValueError(f"climatology fields have {arr.shape[1]} states, model has {self.n}")
161 if not np.all(np.isfinite(arr)):
162 raise ValueError("climatology fields contain non-finite values")
163 self._clim_clim = arr.T.copy()
164 return self._clim_clim

166 def climatology(self, n_snapshots=150, spinup=600, stride=10):
167 """Snapshots at a **physical** vorticity amplitude.
168
169 The inherited version spins up from ``solver.random_ic()``, whose amplitude is O(1) —
170 correct for the non-dimensional Kolmogorov box and catastrophic here, where vorticity
171 is O(1e-5) 1/s. An O(1) field on this grid violates CFL on the first step and the
172 climatology comes back all-NaN, which then propagates silently into the initial
173 ensemble and surfaces much later as ``SVD did not converge`` inside the analysis.
174
175 So the initial condition is rescaled to ``clim_vorticity_rms`` (4.5e-5 1/s by default,
176 which is what the ERA5 slice actually measures at 500 hPa) before the spin-up, and the
177 spin-up is shorter because this model is unforced: it decays on the viscous timescale
178 (~30 days at nu = 2e4), so a long spin-up would quietly deflate the very scale being
179 set here.
180 """
181 if self._clim_clim is None:
182 w0 = np.asarray(self.solversolver.random_ic(), dtype=float).ravel()
183 scale = np.sqrt(np.mean(w0 ** 2))
184 if scale > 0:
185 w0 = w0 * (self.clim_vorticity_rms / scale)
186 w = self.solversolver.advance(w0, spinup)
187 snaps = []
188 for _ in range(n_snapshots):
189 w = self.solversolver.advance(w, stride)
190 snaps.append(np.asarray(w).ravel().copy())
191 self._clim_clim = np.array(snaps).T
192 return self._clim_clim
