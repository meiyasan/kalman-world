21def _time(fn, n_reps=5):
22 fn() # warm-up
23 t0 = time.perf_counter()
24 for _ in range(n_reps):
25 fn()
26 return (time.perf_counter() - t0) / n_reps

92def bench_particle_update(n_particles=500, mv=2, n_reps=20):
93 from modules.framework.native import lib, as_contiguous, dptr
94
95 np.random.seed(2)
96 z = np.random.randn(mv)
97 zp = np.random.randn(n_particles, mv)
98 Rinv = np.eye(mv) * 2.0
99
100 def run_c():
101 logw = np.zeros(n_particles)
102 lib.pf_weight_update(n_particles, mv,
103 dptr(as_contiguous(z)), dptr(as_contiguous(zp)),
104 dptr(as_contiguous(Rinv)), dptr(logw))
105
106 def run_py():
107 logw = np.zeros(n_particles)
108 for i in range(n_particles):
109 dz = z - zp[i]
110 logw[i] += -0.5 * dz @ Rinv @ dz
111
112 t_c = _time(run_c, n_reps)
113 t_py = _time(run_py, n_reps)
114 speedup = t_py / t_c
115 print(f"PF weight update ({n_particles} particles, mv={mv}): "
116 f"C={t_c*1e6:.1f} µs Py={t_py*1e6:.1f} µs "
117 f"→ {speedup:.1f}×")
118 return speedup

121def bench_ukf_outer_sum(n_sigma=9, n=4, n_reps=50):
122 from modules.framework.native import lib, as_contiguous, dptr
123
124 np.random.seed(3)
125 Wc = np.random.dirichlet(np.ones(n_sigma))
126 Y = np.random.randn(n_sigma, n)
127 mean = np.einsum("i,ij->j", Wc, Y)
128
129 def run_c():
130 P_out = np.empty((n, n))
131 lib.ukf_weighted_outer_sum(n_sigma, n,
132 dptr(as_contiguous(Wc)),
133 dptr(as_contiguous(Y)),
134 dptr(as_contiguous(mean)),
135 dptr(P_out))
136
137 def run_py():
138 sum(Wc[i] * np.outer(Y[i] - mean, Y[i] - mean) for i in range(n_sigma))
139
140 t_c = _time(run_c, n_reps)
141 t_py = _time(run_py, n_reps)
142 speedup = t_py / t_c
143 print(f"UKF weighted outer sum (2n+1={n_sigma}, n={n}): "
144 f"C={t_c*1e6:.1f} µs Py={t_py*1e6:.1f} µs "
145 f"→ {speedup:.1f}×")
146 return speedup

149def bench_diag_nis(N=1000, m=2, n_reps=20):
150 from modules.framework.native import lib, as_contiguous, dptr
151
152 np.random.seed(4)
153 nu = np.random.randn(N, m)
154 S = np.array([np.eye(m) + 0.1 * k for k in range(N)])
155
156 def run_c():
157 nis = np.empty(N)
158 lib.diag_nis_batch(N, m, dptr(as_contiguous(nu)), dptr(as_contiguous(S)), dptr(nis))
159
160 def run_py():
161 np.array([float(nu[k] @ np.linalg.solve(S[k], nu[k])) for k in range(N)])
