62 void SetMeasurementNoise(double sigmaMeas) { fSigmaMeas = sigmaMeas; Build(); }

64 void Build()
65 {
66 const double dt = DeltaT();
67 const double q = fSigmaAcc * fSigmaAcc;
68
69 Kalman::Matrix F = Kalman::Matrix::Identity(NState());
70 Kalman::Matrix H(NMeasure(), NState(), 0.);
71 Kalman::Matrix Q(NState(), NState(), 0.);
72 Kalman::Matrix R(NMeasure(), NMeasure(), 0.);
73
74 for (std::size_t a = 0; a < fNAxis; ++a)
75 {
76 const std::size_t p = 2 * a; // position index of this axis
77 const std::size_t v = p + 1; // velocity index
78
79 F(p, v) = dt;
80 H(a, p) = 1.;
81
82 Q(p, p) = q * dt * dt * dt * dt / 4.;
83 Q(p, v) = q * dt * dt * dt / 2.;
84 Q(v, p) = Q(p, v);
85 Q(v, v) = q * dt * dt;
86
87 R(a, a) = fSigmaMeas * fSigmaMeas;
88 }
89
90 SetF(F); SetH(H); SetQ(Q); SetR(R);
91 }

149 class Pendulum : public Kalman::Models::Extended
150 {
151 private:
152 double fLength = 1.;
153 double fGravity = 9.81;
154 double fDamping = 0.;
155
156 public:
157 Pendulum(double length, double damping, double fs)
158 : Kalman::Models::Extended("pendulum", 2, 1), fLength(length), fDamping(damping)
159 {
160 // Models::Extended already declared the engine and switched the
161 // Jacobian to central differences -- f() is nonlinear, so F(t) is
162 // not its Jacobian.
163 if (length <= 0.) throw std::runtime_error("Pendulum: the length must be positive.");
164 SetSampleRate(fs);
165
166 Kalman::Matrix H(1, 2, 0.);
167 H(0, 0) = 1.;
168 SetH(H);
169
170 SetNoise(1e-3, 1e-2);
171 SetInitialCovariance(Kalman::Matrix::Identity(2));
172 }
173
174 double Length() const { return fLength; }
175 double Damping() const { return fDamping; }
176 double Gravity() const { return fGravity; }
177
178 Kalman::Vector Derivative(const Kalman::Vector &x) const
179 {
180 Kalman::Vector dx(2);
181 dx(0) = x(1);
182 dx(1) = -(fGravity / fLength) * std::sin(x(0)) - fDamping * x(1);
183 return dx;
184 }
185
186 Kalman::Vector f(const Kalman::Vector &x, const Kalman::Vector &,
187 double = 0.) const override
188 {
189 const double dt = DeltaT();
190
191 const Kalman::Vector k1 = Derivative(x);
192 Kalman::Vector x2(x); x2 += 0.5 * dt * k1;
193 const Kalman::Vector k2 = Derivative(x2);
194 Kalman::Vector x3(x); x3 += 0.5 * dt * k2;
195 const Kalman::Vector k3 = Derivative(x3);
196 Kalman::Vector x4(x); x4 += dt * k3;
197 const Kalman::Vector k4 = Derivative(x4);
198
199 Kalman::Vector out(x);
200 out += (dt / 6.) * (k1 + 2. * k2 + 2. * k3 + k4);
201 return out;
202 }
203
204 Kalman::Vector h(const Kalman::Vector &x, double = 0.) const override
205 {
206 Kalman::Vector z(1);
207 z(0) = x(0);
208 return z;
209 }
210
211 void SetNoise(double sigmaTorque, double sigmaAngle)
212 {
213 const double dt = DeltaT();
214 const double q = sigmaTorque * sigmaTorque;
215
216 Kalman::Matrix Q(2, 2, 0.);
217 Q(0, 0) = q * dt * dt * dt * dt / 4.;
218 Q(0, 1) = q * dt * dt * dt / 2.;
219 Q(1, 0) = Q(0, 1);
220 Q(1, 1) = q * dt * dt;
221 SetQ(Q);
222
223 Kalman::Matrix R(1, 1, 0.);
224 R(0, 0) = sigmaAngle * sigmaAngle;
225 SetR(R);
226 }
