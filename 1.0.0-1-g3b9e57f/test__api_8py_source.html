17@contextlib.contextmanager
18def _kalman_importable():
19 """Make ``import kalman...`` resolve while a scaffolded module is executed.
20
21 The scaffold imports the *installed* package name, ``kalman`` -- right for
22 the user who will commit it, but a bare checkout (CI runs ``make tests`` on
23 the source tree, where the package is only importable as ``modules``) has
24 no such name. Alias every loaded ``modules.*`` module under ``kalman.*`` for
25 the duration, so the scaffold binds to the very same classes the test
26 compares against, and remove the aliases afterwards. A real installation
27 is left alone.
28 """
29 # Not find_spec(): a stale editable-install finder or a stray namespace
30 # directory can *claim* the name and still fail to import it, which is
31 # exactly the case that has to be caught. Only a working import counts.
32 try:
33 importlib.import_module("kalman.framework")
34 yield
35 return
36 except ImportError:
37 pass
38 previous = {}
39 for name, mod in list(sys.modules.items()):
40 if name == "modules" or name.startswith("modules."):
41 alias = "kalman" + name[len("modules"):]
42 previous[alias] = sys.modules.get(alias) # None when absent
43 sys.modules[alias] = mod # override a broken claimant too
44 try:
45 yield
46 finally:
47 for alias, before in previous.items():
48 if before is None:
49 sys.modules.pop(alias, None)
50 else:
51 sys.modules[alias] = before

336class TestDiagnosticsUseTheMeasurements(unittest.TestCase):
337 """NIS is the filter's self-check: innovations must be the size the claimed
338 noise predicts (mean ~ m). It is only that check if it is fed the
339 measurements — ``record.zk`` on a filtered run holds the filter's OUTPUT, so
340 diagnosing it re-filtered estimates and reported NIS ~ 0.1, i.e. a
341 consistent filter flagged as wildly over-conservative. A contract that
342 fails in the direction of looking fine is worse than no contract.
343 """
344
345 SPEC = {"base": "matrices", "fs": 10,
346 "dynamics": {"F": [[1, 0.1], [0, 1]], "H": [[1, 0]]},
347 "noise": {"Q": [[1e-4, 0], [0, 1e-4]], "R": [[0.25]]},
348 "initial": {"x0": [[0], [7.669]]}}
349
350 def test_nis_is_consistent_for_a_correctly_specified_filter(self):
351 zk = np.asarray(api.make_model(self.SPEC).sample(30.0).zk)
352 out = api.run_filter(self.SPEC, zk.tolist())
353 nis = np.asarray(api.diagnostics(out["run_id"])["nis"])
354 # chi2(1): mean 1, and a filter whose noise model is honest lands near it.
355 self.assertGreater(nis.mean(), 0.4)
356 self.assertLess(nis.mean(), 2.5)
357
358 def test_diagnostics_sees_the_input_not_the_output(self):
359 zk = np.asarray(api.make_model(self.SPEC).sample(20.0).zk)
360 out = api.run_filter(self.SPEC, zk.tolist())
361 stored = api._RUNS[out["run_id"]]["measurements"]
362 np.testing.assert_allclose(np.asarray(stored).reshape(zk.shape), zk)
363 # The filtered record is NOT the measurement set — that is the trap.
364 filtered = np.asarray(api._RUNS[out["run_id"]]["record"].zk).reshape(zk.shape)
365 self.assertGreater(np.abs(filtered - zk).max(), 1e-6)
366
367 def test_innovation_covariance_is_reported_alongside(self):
368 zk = np.asarray(api.make_model(self.SPEC).sample(10.0).zk)
369 out = api.run_filter(self.SPEC, zk.tolist())
370 d = api.diagnostics(out["run_id"])
371 S = np.asarray(d["innovation_covariance"])
372 # S = H P- H' + R >= R, so the reported spread is never below the sensor's.
373 self.assertGreaterEqual(S.mean(), 0.25 - 1e-9)
