1"""Rocket ascent — powered gravity-turn climb to orbit."""
2from __future__ import annotations
3"""
4Rocket science — getting off the pad and back: **takeoff** and **landing**.
5
6This module is the *near-surface, powered-flight* half of the rocketry models (no third
7bodies). The space-travel half — flying to the Moon and interplanetary transfers — lives in
8:mod:`kalman.apps.space`.
9
10The "rocket equation" family (powered flight where what differs is the **phase**):
11
12* :class:`RocketAscentModel` — a planar **guided ascent**: gravity versus **thrust supplied
13 as the control input** ``u`` (constant mass). Demonstrates control channels (``l>0``).
14 *Scaled units.* Legacy alias: ``RocketLaunchModel``.
15* :class:`RocketLiftoffModel` — a 1-D **vertical liftoff** whose thrust is internal and whose
16 **mass burns down** (Tsiolkovsky): acceleration rises until **burnout**. *SI units.*
17 Legacy alias: ``RocketTakeoffModel``.
18* :class:`RocketLandingModel` — the same 1-D mass-burning dynamics run in reverse: a
19 SpaceX-style **hoverslam**, free-falling then firing a closed-loop landing burn to touch
20 down near zero velocity. *SI units.*
21
22All are :class:`ExtendedKalmanModel` subclasses whose ``f()`` advances the state by one
234th-order Runge–Kutta step; the EKF linearises ``f`` with a numerical Jacobian.
24
25The ascent model uses *scaled units* (``length = 1 Mm = 1000 km``, ``time = 1 ks = 1000 s``)
26to stay well-conditioned; the near-surface liftoff/landing models use plain **SI** (m, s, kg).
27Physical constants come from :mod:`kalman.runtime.astro` (astropy-sourced where available).
28"""

53class RocketAscentModel(NativeTwin, ExtendedKalmanModel):
54 """
55 Planar **guided** powered ascent from Earth's surface (the steering/guidance model).
56
57 State ``x = [x, y, vx, vy]`` in Mm and km/s, in Earth-centred axes; **control**
58 ``u = [ax, ay]`` is the thrust acceleration in Mm/ks², which is m/s², so this model has
59 ``l = 2`` control channels. Dynamics: central gravity +
60 thrust (+ optional exponential-atmosphere drag). A radar measures position only.
61
62 The distinguishing trait is the **2-D control input**: the thrust program is fed in,
63 constant mass. Contrast :class:`RocketLiftoffModel` (1-D vertical liftoff whose thrust
64 is internal and whose **mass burns down**). Importable as the legacy alias
65 ``RocketLaunchModel``.
66 """
67
68 NAME = "Rocket Ascent (guided, thrust as control)"; VERSION = "1.0"

72 @property

75 def __init__(self, dt_s=1.0, track_km=1.0, drag=False, ballistic_coeff=2.0e-3,
76 length=40.0, width=3.7, q_pos=1e-9, q_vel=1e-9, **kw):
77 self.dtypedtype = np.float64
78 self._dt = dt_s / TU
79 self.drag = bool(drag)
80 self.Cb = float(ballistic_coeff) # lumped 0.5*Cd*A/m (scaled)
81 self.length, self.width = float(length), float(width)
82 self.nnnnn, self.mmmm, self.lll = 4, 2, 2
83 self.fsfsfsfsfs = 1.0 / self._dt
84 Q = np.diag([q_pos, q_pos, q_vel, q_vel]).astype(float)
85 R = (track_km * KM) ** 2 * np.eye(2)
86 # L = I: Q enters additively on every state. f() does not route an
87 # additive `w` through, so without it the engine would difference f with
88 # respect to w, find no sensitivity, and multiply Q by a zero matrix at
89 # every prediction.
90 super().__init__("LAUNCH", Q=Q, R=R, L=np.eye(self.nnnnn), fs=self.fsfsfsfsfs, **kw)

92 # --- C++ twin (libKalmanApps: Kalman::Apps::Astro::RocketAscent) -------
93 # The thrust program is the control input u (l = 2), handed to the twin
94 # with the measurements; a controller function keeps the Python model.
95 _TWIN_CONTRACT = NativeTwin._TWIN_CONTRACT + ("acceleration",)

97 def native_name(self):
98 return "rocket-ascent" if native_integrator_kind(self) is not None else None
99
100 def native_parameters(self):
101 return {"GM": float(GM_EARTH), "R": float(R_EARTH), "H": float(H_SCALE),
102 "ballistic_coeff": self.Cb, "drag": float(self.drag),
103 "substeps": float(self.substeps()), "integrator": float(native_integrator_kind(self))}
104
105 def acceleration(self, r, v, u):
106 a = -GM_EARTH * r / np.linalg.norm(r) ** 3 # gravity
107 if u is not None:
108 a = a + np.asarray(u, float).reshape(2) # thrust (control)
109 if self.drag:
110 alt = np.linalg.norm(r) - R_EARTH
111 rho = np.exp(-max(alt, 0.0) / H_SCALE)
112 a = a - self.Cb * rho * np.linalg.norm(v) * v # quadratic drag
113 return a
114
115 DYNAMICS = KalmanDynamicsKind.CONTINUOUS # f is ds/dt; the framework integrates it (RK4)
116
117 def f(self, x, u=None, w=None, t=None):
118 s = np.asarray(x, float).reshape(-1)
119 return np.concatenate([s[2:], self.acceleration(s[:2], s[2:], u)]).reshape(np.shape(x))
120
121 def h(self, x, v=None, t=None):
122 z = np.asarray(x, float).reshape(-1)[:2].reshape(2, 1)
123 return z if v is None else z + np.asarray(v, float).reshape(2, 1)
