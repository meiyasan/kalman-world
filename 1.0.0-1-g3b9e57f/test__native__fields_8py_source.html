1"""A field model's ensemble pass runs in C++ (libKalmanApps: Weather::KolmogorovFlow).
2
3With the ensemble engine, ``KolmogorovFlowModel`` and ``BarotropicFlowModel``
4hand the whole pass to their twin: the forecast of every member, the
5observation of the ensemble, the localization tapers, the analysis and the
6loop over the cycles. The draws stay numpy's (the engine's generator, in the
7order the Python steps take them), so the C++ pass is compared here with the
8Python reference on the same seed, member for member.
9
10How close: the reference steps the field with numpy's FFT and the twin with
11the library's; one solver step agrees to ~1e-16 relative, and the chaotic flow
12grows that to ~1e-13 over the 1,200 steps of notebook 13 (measured, then held
13by the analysis). The tolerances below are 1e-9 relative to the largest entry.
14
15Skipped when libKalmanApps is not built.
16"""

36def rel(a, b):

49def measurements(model, steps=12, seed=21):
50 rng = np.random.default_rng(seed)
51 w = model.equilibrium_state(decorrelate=100)
52 zs = []
53 for _ in range(steps):
54 w = model.f(w)
55 zs.append(np.asarray(model.h(w)).ravel() + rng.normal(0, model.sig_o, model.m))
56 zs = np.asarray(zs)
57 zs[3, ::4] = np.nan # some probes silent
58 zs[5, :] = np.nan # and one cycle with none
59 return zs

62def filter_pass(make, engine_kw, reference, expect_twin=True):
63 def go():
64 model = make()
65 zs = measurements(model)
66 f = EnsembleKalmanFilter(model, **engine_kw)
67 if not reference:
68 assert (f.engine._native_twin(model) is not None) == expect_twin
69 rec = f.forward(zs)
70 return (np.asarray(rec.xk), np.asarray(rec.Pk), f.engine.members.copy(),
71 int(f.engine.rng.integers(0, 1 << 62)), [np.asarray(nu).ravel() for nu in f.innovation])
72 if reference:
73 with native.python_reference():
74 return go()
75 return go()

192 return go()

196 return go()

199 return go()
