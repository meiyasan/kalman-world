22 // Two modes of the same 1-D system, differing in how much unmodelled
23 // acceleration they expect: a quiet cruise and a manoeuvre.
24 //
25 // The contrast has to be large enough to *show up in the data*. With a
26 // measurement noise comparable to the process noise, both modes explain the
27 // track equally well (measured: innovation covariances 8% apart, both with
28 // mean NIS ~ 1.09) and a correct Bayesian router then returns ~0.5 — not
29 // because the IKF is broken but because there is genuinely nothing to
30 // choose between. Hence a sharp sensor and a violent manoeuvre.
31 const double sigmaZ = 0.05;
32
33 Fixture::ConstantVelocity quiet(1, 50.);
34 quiet.SetProcessNoise(0.05);
35 quiet.SetMeasurementNoise(sigmaZ);
36
37 Fixture::ConstantVelocity agile(1, 50.);
38 agile.SetProcessNoise(50.0);
39 agile.SetMeasurementNoise(sigmaZ);
40
41 Matrix Pi(2, 2, 0.);
42 Pi(0, 0) = 0.95; Pi(0, 1) = 0.05;
43 Pi(1, 0) = 0.05; Pi(1, 1) = 0.95;
44
45 // --- construction is validated ------------------------------------------
46 {
47 bool threw = false;
48 try { Matrix bad(3, 3, 1.); Filters::IKF ikf({&quiet, &agile}, bad); }
49 catch (const std::runtime_error &) { threw = true; }
50 CHECK(threw, "IKF: a Pi of the wrong shape throws");
51
52 threw = false;
53 try { Matrix bad(2, 2, 0.7); Filters::IKF ikf({&quiet, &agile}, bad); }
54 catch (const std::runtime_error &) { threw = true; }
55 CHECK(threw, "IKF: a Pi whose rows do not sum to 1 throws");

57 threw = false;
58 try { Filters::IKF ikf({&quiet}, Matrix::Identity(1)); }

72 // --- a quiet track: the IKF should settle on the quiet mode ---------------
73 {
74 Vector x0(2); x0(0) = 0.; x0(1) = 2.;
75 quiet.SetInitialState(x0);
76 agile.SetInitialState(x0);
77
78 Sampler sampler(&quiet, 4242);
79 sampler.Generate(400, x0);
80

82 Filters::IKF ikf({&quiet, &agile}, Pi, &router);
83 const Record rec = ikf.Forward(sampler.Measurements());
84
85 CHECK(rec.NSteps() == 400, "IKF: one record entry per measurement");
86 CHECK(rec.NState() == 2, "IKF: the combined estimate has the shared state size");
87
88 const Matrix &modes = ikf.ModeHistory();
89 CHECK(modes.Rows() == 400 && modes.Cols() == 2, "IKF: mode history is (T, M)");
90
91 bool normalised = true;
92 for (std::size_t k = 0; k < modes.Rows(); ++k)
93 if (std::abs(modes(k, 0) + modes(k, 1) - 1.) > 1e-9) normalised = false;
94 CHECK(normalised, "IKF: mode probabilities sum to 1 at every step");
95
96 // Average over the second half, once the filter has had time to decide.
97 double quietShare = 0.;
98 for (std::size_t k = 200; k < 400; ++k) quietShare += modes(k, 0);
99 quietShare /= 200.;
100 std::printf(" quiet-mode probability on a quiet track: %.3f\n", quietShare);
101 CHECK(quietShare > 0.6, "IKF: settles on the mode that generated the data");
102
103 // The estimate must still be a good estimate, not just a good classifier.
104 double eraw = 0.;
105 for (std::size_t k = 0; k < 400; ++k)
106 eraw += std::pow(sampler.Measurements()[k](0) - sampler.Truth()[k](0), 2);
107 eraw = std::sqrt(eraw / 400.);
108 CHECK(rec.RMSE(0, sampler.Truth()) < eraw, "IKF: the combined estimate beats the raw measurement");
109 }

111 // --- a switching track: the IKF should follow the switch ------------------
112 {
113 // Cruise for 200 steps, then manoeuvre hard for 200. Built by hand so the
114 // regime boundary is known exactly.
115 //
116 // One track is a coin toss: of its 100 manoeuvring steps the agile mode
117 // wins anywhere from ~45 to ~100 (300 seeds: mean 76, sd 11.5), so a
118 // threshold on one draw passes or fails with the random stream -- which
119 // libstdc++ and libc++ produce differently from the same seed. The checks
120 // are on the mean over 20 tracks, which moves by a few steps, not tens.
121 const std::size_t nTracks = 20;
122 const double dt = 1. / 50.;
123 double before = 0., after = 0., agileWins = 0.;
124 bool nisShape = true, finite = true;
125 for (std::uint64_t seed = 0; seed < nTracks; ++seed)
126 {
127 std::mt19937_64 rng(seed);
128 std::normal_distribution<double> gauss(0., 1.);
129
130 std::vector<Vector> zk;
131 double p = 0., v = 2.;
132 for (std::size_t k = 0; k < 400; ++k)
133 {
134 const double sigmaA = (k < 200) ? 0.05 : 50.0; // the regime change
135 const double a = sigmaA * gauss(rng);
136 p += dt * v + 0.5 * a * dt * dt;
137 v += dt * a;
138
139 Vector z(1);
140 z(0) = p + sigmaZ * gauss(rng);
141 zk.push_back(z);
142 }
143
144 Vector x0(2); x0(0) = 0.; x0(1) = 2.;
145 quiet.SetInitialState(x0);
146 agile.SetInitialState(x0);
147
148 Routers::Bayesian router;
149 Filters::IKF ikf({&quiet, &agile}, Pi, &router);
150 ikf.Forward(zk);
151
152 const Matrix &modes = ikf.ModeHistory();
153 for (std::size_t k = 100; k < 200; ++k) before += modes(k, 1) / 100.; // agile share, cruising
154 for (std::size_t k = 300; k < 400; ++k) after += modes(k, 1) / 100.; // agile share, manoeuvring
155
156 // A single step can go either way on noise; the question is which mode
157 // wins across the manoeuvre.
158 for (std::size_t k = 300; k < 400; ++k)
159 if (ikf.DominantMode(k) == 1) ++agileWins;
160
161 const Matrix &nis = ikf.NisHistory();
162 nisShape = nisShape && nis.Rows() == 400 && nis.Cols() == 2;
163 for (std::size_t k = 0; k < nis.Rows(); ++k)
164 for (std::size_t j = 0; j < nis.Cols(); ++j)
165 if (!std::isfinite(nis(k, j))) finite = false;
166 }
167 before /= nTracks; after /= nTracks; agileWins /= nTracks;
168
169 std::printf(" agile-mode probability: %.3f while cruising -> %.3f while manoeuvring"
170 " (mean of %zu tracks)\n", before, after, nTracks);
171 std::printf(" agile mode dominant on %.1f of 100 manoeuvring steps\n", agileWins);
172 CHECK(after > before + 0.2, "IKF: the agile mode gains sharply once the target manoeuvres");
173 CHECK(before < 0.45, "IKF: the quiet mode is preferred while cruising");
174 CHECK(after > 0.55, "IKF: the agile mode dominates during the manoeuvre");
175 CHECK(agileWins > 65, "IKF: DominantMode reports the manoeuvring mode most of the time");
176 CHECK(nisShape, "IKF: NIS history is (T, M)");
177 CHECK(finite, "IKF: every NIS entry is finite");
178 }

180 // --- the router really is pluggable, and dropouts are handled -------------
181 {
182 Vector x0(2); x0(0) = 0.; x0(1) = 2.;
183 quiet.SetInitialState(x0);
184 agile.SetInitialState(x0);
185
186 Sampler sampler(&agile, 99);
187 sampler.Generate(300, x0);
188 sampler.DropChannel(0, 0.25);
189
190 Routers::Bayesian bayes;
191 Routers::Threshold thresh(1, 5, true);
192 Routers::Hybrid hybrid(&bayes, &thresh, 0.5);
193
194 for (const Routers::Router *r :
195 {static_cast<const Routers::Router *>(&bayes),
196 static_cast<const Routers::Router *>(&thresh),
197 static_cast<const Routers::Router *>(&hybrid)})
198 {
199 Filters::IKF ikf({&quiet, &agile}, Pi, r);
200 const Record rec = ikf.Forward(sampler.Measurements());
201
202 bool ok = true;
203 for (std::size_t k = 0; k < rec.NSteps(); ++k)
204 if (!std::isfinite(rec.State(k)(0)) || !std::isfinite(rec.Covariance(k)(0, 0)))
205 ok = false;
206 CHECK(ok, (std::string("IKF + ") + r->Title() + ": finite through 25% dropouts").c_str());
207 }
208
209 // Swapping the decision layer must not require rebuilding the bank.
210 Filters::IKF ikf({&quiet, &agile}, Pi, &bayes);
211 CHECK(ikf.GetRouter() == &bayes, "IKF: holds the router it was given");
212 ikf.SetRouter(&thresh);
213 CHECK(ikf.GetRouter() == &thresh, "IKF: SetRouter swaps the decision layer");
214
215 // No router at all means the classical update, not a crash.
216 Filters::IKF plain({&quiet, &agile}, Pi);
217 CHECK(plain.GetRouter() != nullptr, "IKF: defaults to the Bayesian router");
218 CHECK(plain.Forward(sampler.Measurements()).NSteps() == 300, "IKF: the default runs");
219 }
