1"""Bounding-box tracking + SORT association — the TRACK variant."""
2from __future__ import annotations
3"""
4Computer-vision object tracking with linear Kalman filters — the input is always a
5per-frame object **detector**'s output; this module owns the *filtering*, so a notebook
6never has to assemble ``F``/``H`` or a tracking loop by hand. Three levels, sharing one
7idea (detections in → smoothed, occlusion-robust state out):
8
9* :class:`DetectionModel` — **single object, centroid** ``x = [x, y, vx, vy]`` observing
10 ``(x, y)``. A constant-velocity filter that smooths a detector's centroid jitter and
11 **coasts through occlusions** (a ``NaN`` measurement row → predict-only, covariance grows
12 until re-acquisition).
13* :class:`BoxTrackModel` — **single object, bounding box** in the SORT parameterization
14 ``x = [x, y, s, r, vx, vy, vs]`` observing ``[x, y, s, r]`` (centre, area ``s = w·h``,
15 aspect ``r = w/h``). Now the *box size* is filtered/predicted too, so it coasts at its
16 last size rather than being faked from the covariance.
17* :class:`SortTracker` — **multiple objects** (SORT): per-frame detector **boxes** are
18 matched to existing tracks by IoU + Hungarian assignment, with track **birth/death** and
19 ``max_age`` occlusion coasting. Each track is one :class:`KalmanTrack` over a shared
20 :class:`BoxTrackModel`. :func:`synthetic_multi` generates a reproducible crossing-objects
21 scene (with occlusion) to drive it.
22
23The detector is swappable (a Haar cascade, an Otsu blob, a neural net…); this module only
24needs centroids (``DetectionModel``) or boxes (``BoxTrackModel``/``SortTracker``). See the
25``10_ComputerVision`` notebook. Distinct from ``apps.tracking`` — the combinatorial
26drift-chamber finder (HEP) — which shares the *combinatorial* idea but a different domain.
27"""

66# ----- box <-> [x, y, s, r] conversions (SORT parameterization) -----
67def box_to_xysr(bbox) -> np.ndarray:
68 """``[x1, y1, x2, y2]`` → column ``[x, y, s, r]`` (centre, area s=w·h, aspect r=w/h)."""
69 b = np.asarray(bbox, dtype=float).ravel()
70 w, h = b[2] - b[0], b[3] - b[1]
71 return np.array([b[0] + w / 2.0, b[1] + h / 2.0, w * h, w / h if h else 0.0]).reshape(4, 1)

74def xysr_to_box(state) -> np.ndarray:
75 """``[x, y, s, r, ...]`` → ``[x1, y1, x2, y2]`` (inverse of :func:`box_to_xysr`)."""
76 b = np.asarray(state, dtype=float).ravel()
77 s, r = max(float(b[2]), 1e-6), max(float(b[3]), 1e-6)
78 w = np.sqrt(s * r)
79 h = s / w
80 return np.array([b[0] - w / 2.0, b[1] - h / 2.0, b[0] + w / 2.0, b[1] + h / 2.0])

83def iou_batch(det, trk) -> np.ndarray:
84 """Pairwise IoU between detection boxes ``det (N,4)`` and track boxes ``trk (M,4)`` → ``(N, M)``."""
85 det = np.asarray(det, dtype=float).reshape(-1, 1, 4)
86 trk = np.asarray(trk, dtype=float).reshape(1, -1, 4)
87 xx1 = np.maximum(det[..., 0], trk[..., 0]); yy1 = np.maximum(det[..., 1], trk[..., 1])
88 xx2 = np.minimum(det[..., 2], trk[..., 2]); yy2 = np.minimum(det[..., 3], trk[..., 3])
89 w = np.clip(xx2 - xx1, 0.0, None); h = np.clip(yy2 - yy1, 0.0, None)
90 inter = w * h
91 area_d = (det[..., 2] - det[..., 0]) * (det[..., 3] - det[..., 1])
92 area_t = (trk[..., 2] - trk[..., 0]) * (trk[..., 3] - trk[..., 1])
93 return inter / np.clip(area_d + area_t - inter, 1e-9, None)

96def _greedy_match(iou: np.ndarray):
97 """Fallback assignment (no SciPy): repeatedly take the highest remaining IoU pair."""
98 iou = iou.copy()
99 pairs = []
100 while iou.size and iou.max() > 0:
101 d, t = np.unravel_index(np.argmax(iou), iou.shape)
102 pairs.append((int(d), int(t)))
103 iou[d, :] = -1.0
104 iou[:, t] = -1.0
105 return pairs

108def associate_detections(det, trk, iou_threshold: float = 0.3):
109 """Match detections to tracks by IoU (Hungarian if SciPy is present, else greedy).
110
111 Returns ``(matches, unmatched_detections, unmatched_tracks)`` where ``matches`` is a
112 list of ``(det_index, track_index)`` pairs with IoU ≥ ``iou_threshold``.
113 """
114 det = np.asarray(det, dtype=float).reshape(-1, 4)
115 trk = np.asarray(trk, dtype=float).reshape(-1, 4)
116 if len(trk) == 0:
117 return [], list(range(len(det))), []
118 if len(det) == 0:
119 return [], [], list(range(len(trk)))
120
121 iou = iou_batch(det, trk)
122 try:
123 from scipy.optimize import linear_sum_assignment
124 di, ti = linear_sum_assignment(-iou)
125 pairs = list(zip(di.tolist(), ti.tolist()))
126 except Exception: # SciPy missing → greedy fallback
127 pairs = _greedy_match(iou)
128
129 matched_d = {d for d, _ in pairs}
130 matched_t = {t for _, t in pairs}
131 un_det = [d for d in range(len(det)) if d not in matched_d]
132 un_trk = [t for t in range(len(trk)) if t not in matched_t]
133 matches = []
134 for d, t in pairs:
135 if iou[d, t] < iou_threshold: # too weak → leave both unmatched
136 un_det.append(d); un_trk.append(t)
137 else:
138 matches.append((d, t))
139 return matches, un_det, un_trk

142class BoxTrackModel(LinearKalmanModel):
143 """Constant-velocity **bounding-box** tracker — SORT state ``[x, y, s, r, vx, vy, vs]``.
144
145 Observes ``[x, y, s, r]`` (centre, area, aspect); the aspect ratio ``r`` is modelled as
146 constant (no rate). ``Q``/``R``/``P0`` follow the canonical SORT recipe (relative
147 weights), scaled by ``process_uncertainty`` / ``measurement_uncertainty`` (default 1.0).
148
149 Single-object use: inject a sequence of detector boxes and filter::
150
151 model = BoxTrackModel("ball", fs=25)
152 record = model.filter_boxes(boxes) # boxes: [x1,y1,x2,y2] or None per frame
153 smoothed = model.record_to_boxes(record) # back to [x1,y1,x2,y2] (coasts on misses)
154 """
155
156 NAME = "SORT Box Tracker"
157 DESCRIPTION = ("Constant-velocity bounding-box tracker [x, y, s, r, vx, vy, vs]; observes "
158 "[x, y, s, r] and coasts (predict-only) through detector misses.")
159 VERSION = "1.0"
160
161 @property
162 def states(self):
163 return BoxTrackState
164
165 @property
166 def observables(self):
167 return BoxTrackObservable
168
169 def __init__(self, sensor_id: str | int,
170 measurement_uncertainty: float = 1.0,
171 process_uncertainty: float = 1.0,
172 fs: float = 30.0,
173 frame: tuple[int, int] = (640, 480),
174 **kwargs):
175 self.dtypedtype = kwargs.pop("dtype", np.float64)
176 self.frame = tuple(frame)
177 dt = self.dtypedtype(1.0 / fs)
178
179 F = np.eye(7, dtype=self.dtypedtype)
180 F[0, 4] = dt; F[1, 5] = dt; F[2, 6] = dt # x,y,s advance by their velocities
181 H = np.zeros((4, 7), dtype=self.dtypedtype)
182 H[0, 0] = H[1, 1] = H[2, 2] = H[3, 3] = 1.0 # observe [x, y, s, r]
183
184 # Canonical SORT covariances (relative), scaled by the *_uncertainty knobs.
185 R = np.diag([1.0, 1.0, 10.0, 10.0]).astype(self.dtypedtype) * float(measurement_uncertainty)
186 Q = np.eye(7, dtype=self.dtypedtype)
187 Q[-1, -1] *= 0.01 # low process noise on area-rate
188 Q[4:, 4:] *= 0.01 # and on the velocities
189 Q *= float(process_uncertainty)
190
191 if kwargs.get("P0") is None:
192 P0 = np.eye(7, dtype=self.dtypedtype) * 10.0
193 P0[4:, 4:] *= 1000.0 # large prior on unobserved velocities
194 kwargs["P0"] = P0
195
196 super().__init__(sensor_id, F=F, H=H, Q=Q, R=R, fs=fs, **kwargs)

198 # ---- explicit input injection: detector boxes -> measurement rows ----
199 @staticmethod
200 def boxes_to_measurements(boxes) -> np.ndarray:
201 """Sequence of boxes (``[x1,y1,x2,y2]`` or ``None``/``NaN`` for a miss) → ``(N, 4)``
202 measurement rows ``[x, y, s, r]`` with ``NaN`` rows where the detector missed."""
203 rows = []
204 for b in boxes:
205 a = None if b is None else np.asarray(b, dtype=float).ravel()
206 if a is None or a.size < 4 or np.any(np.isnan(a)):
207 rows.append([np.nan] * 4)
208 else:
209 rows.append(box_to_xysr(a).ravel())
210 return np.asarray(rows, dtype=float)
211
212 @staticmethod
213 def record_to_boxes(record) -> np.ndarray:
214 """Filtered record (state rows ``[x,y,s,r,...]``) → ``(N, 4)`` boxes ``[x1,y1,x2,y2]``."""
215 return np.array([xysr_to_box(x) for x in np.asarray(record.xk)])

217 def initial_from_box(self, box):
218 """Seed ``x0`` from a single detector box (velocities zero)."""
219 x0 = np.zeros((7, 1), dtype=self.dtypedtype)
220 x0[:4, 0] = box_to_xysr(box).ravel()
221 self.initial_state(x0=x0)
222 return self

224 def filter_boxes(self, boxes, **kwargs) -> KalmanRecord:
225 """Inject a per-frame box sequence and run the filter (seeds ``x0`` from the first
226 valid box if not already set). The explicit, named alternative to assembling the
227 measurement stream and calling :meth:`filter` by hand."""
228 z = self.boxes_to_measurements(boxes)
229 valid = np.where(~np.isnan(z[:, 0]))[0]
230 if len(valid):
231 x0 = np.zeros((7, 1), dtype=self.dtypedtype)
232 x0[:4, 0] = z[valid[0]]
233 self.initial_state(x0=x0)
234 return self.filter(z, **kwargs)

241class KalmanTrack:
242 """One object's online box filter — a thin wrapper holding its own ``(x, P)`` and
243 stepping the shared :class:`BoxTrackModel` through the linear engine."""
244
245 def __init__(self, model: BoxTrackModel, engine, box, track_id: int):
246 self.model = model
247 self.engine = engine
248 self.id = int(track_id)
249 self.x = np.zeros((7, 1), dtype=float)
250 self.x[:4, 0] = box_to_xysr(box).ravel()
251 self.P = np.array(model.P0, dtype=float).copy()
252 self.time_since_update = 0
253 self.hits = 1
254 self.age = 0
255
256 def predict(self, t=None) -> np.ndarray:
257 """Advance one frame (predict-only); returns the predicted box."""
258 if self.x[6, 0] + self.x[2, 0] <= 0: # keep area from going negative
259 self.x[6, 0] = 0.0
260 self.x, self.P, _ = self.engine._predict(self.model, self.x, self.P, t=t)
261 self.age += 1
262 self.time_since_update += 1
263 return xysr_to_box(self.x)
264
265 def update(self, box, t=None) -> None:
266 """Correct with a matched detection box."""
267 self.time_since_update = 0
268 self.hits += 1
269 self.x, self.P = self.engine._update(self.model, self.x, self.P, box_to_xysr(box), t=t)
270
271 @property
272 def box(self) -> np.ndarray:
273 return xysr_to_box(self.x)

276class SortTracker:
277 """Multi-object SORT tracker: per-frame detector **boxes** in, stable per-ID tracks out.
278
279 Each :meth:`step` predicts every track, matches detections by IoU + Hungarian
280 assignment (:func:`associate_detections`), updates matches, **spawns** a track per
281 unmatched detection, and **retires** tracks unseen for more than ``max_age`` frames. A
282 track is reported once it has ``min_hits`` hits; while a confirmed track is between
283 detections it is still reported with ``coasting=True`` (predict-only through occlusion).
284 """
285
286 NAME = "SortTracker" # single source of truth for the method name (eval, plots, tables)
287
288 def __init__(self, max_age: int = 10, min_hits: int = 3, iou_threshold: float = 0.3,
289 measurement_uncertainty: float = 1.0, process_uncertainty: float = 1.0,
290 fs: float = 30.0, frame: tuple[int, int] = (640, 480), **model_kwargs):
291 self.model = BoxTrackModel("sort", measurement_uncertainty=measurement_uncertainty,
292 process_uncertainty=process_uncertainty, fs=fs,
293 frame=frame, **model_kwargs)
294 self.engine = self.model.default_engine()
295 self.max_age = int(max_age)
296 self.min_hits = int(min_hits)
297 self.iou_threshold = float(iou_threshold)
298 self.tracks: list[KalmanTrack] = []
299 self.frame_count = 0
300 self._next_id = 0
301
302 def step(self, boxes) -> list[TrackedBox]:
303 """Process one frame's detections (``(M, 4)`` boxes, possibly empty)."""
304 self.frame_count += 1
305 det = (np.asarray(boxes, dtype=float).reshape(-1, 4)
306 if boxes is not None and len(boxes) else np.empty((0, 4)))
307
308 # 1) advance every existing track
309 trk_boxes = np.zeros((len(self.tracks), 4))
310 for i, trk in enumerate(self.tracks):
311 trk_boxes[i] = trk.predict()
312
313 # 2) IoU/Hungarian association
314 matches, un_det, un_trk = associate_detections(det, trk_boxes, self.iou_threshold)
315
316 # 3) update matched, 4) birth unmatched detections
317 for d, t in matches:
318 self.tracks[t].update(det[d])
319 for d in un_det:
320 self.tracks.append(KalmanTrack(self.model, self.engine, det[d], self._next_id))
321 self._next_id += 1
322
323 # 5) death: drop tracks unseen for too long
324 self.tracks = [t for t in self.tracks if t.time_since_update <= self.max_age]
325
326 # 6) report confirmed tracks (coasting flag for predict-only frames)
327 out = []
328 for t in self.tracks:
329 if t.hits >= self.min_hits or self.frame_count <= self.min_hits:
330 out.append(TrackedBox(id=t.id, box=t.box, coasting=t.time_since_update > 0,
331 hits=t.hits, age=t.age))
332 return out
333
334 def run(self, detections) -> list[list[TrackedBox]]:
335 """Run over a whole sequence — an iterable of per-frame ``(M, 4)`` detection arrays —
336 returning the per-frame list of :class:`TrackedBox`."""
337 return [self.step(d) for d in detections]

340def synthetic_multi(n_frames: int = 120, frame: tuple[int, int] = (640, 480),
341 n_objects: int = 3, box_wh: tuple[int, int] = (50, 100),
342 noise: float = 3.0, seed=None) -> list[np.ndarray]:
343 """Reproducible crossing-objects scene for :class:`SortTracker` testing.
344
345 Returns a list of ``n_frames`` per-frame ``(M, 4)`` ``[x1, y1, x2, y2]`` detection arrays.
346 Three objects move on crossing linear paths; object 1 is occluded (no detection) for
347 ~15 frames in the middle to exercise coasting. Object 2 drifts diagonally.
348 """
349 if seed is not None:
350 set_random_seed(seed)
351 W, H = frame
352 bw, bh = box_wh
353 t = np.linspace(0.0, 1.0, n_frames)
354
355 cx = np.column_stack([
356 0.05 * W + 0.90 * W * t, # 0: left → right
357 0.95 * W - 0.90 * W * t, # 1: right → left (crosses 0 near centre)
358 0.20 * W + 0.60 * W * t, # 2: diagonal
359 ])
360 cy = np.column_stack([
361 np.full(n_frames, 0.30 * H), # 0: horizontal band
362 np.full(n_frames, 0.30 * H), # 1: same band → crossing with 0
363 0.10 * H + 0.80 * H * t, # 2: top → bottom
364 ])
365
366 occ_start = int(0.38 * n_frames)
367 occ_end = int(0.52 * n_frames)
368
369 out = []
370 for k in range(n_frames):
371 row = []
372 for i in range(n_objects):
373 if i == 1 and occ_start <= k < occ_end:
374 continue
375 x1 = cx[k, i] - bw / 2.0 + noise * np.random.randn()
376 y1 = cy[k, i] - bh / 2.0 + noise * np.random.randn()
377 row.append([x1, y1, x1 + bw, y1 + bh])
378 out.append(np.asarray(row, float) if row else np.empty((0, 4)))
379 return out
