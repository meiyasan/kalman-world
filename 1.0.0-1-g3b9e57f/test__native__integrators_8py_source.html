12def pendulum(x, u, t):
13 th, om = np.asarray(x).reshape(-1)
14 torque = 0.0 if u is None else float(np.asarray(u).reshape(-1)[0])
15 return np.array([om, -9.81 * np.sin(th) - 0.1 * om + torque + 0.3 * np.cos(t)]).reshape(np.shape(x))

33 def test_step_and_integrate_match_the_python_schemes(self):
34 x = np.array([[0.7], [-0.2]])
35 for integrator, atol in ((EulerIntegrator(), 1e-14), (RK4Integrator(), 1e-14),
36 (DormandPrinceIntegrator(), 1e-12), (DormandPrinceIntegrator(rtol=1e-9, atol=1e-12), 1e-12)):
37 for u in (None, np.array([[0.4]])):
38 got, want, n_cxx, n_py = self.both(integrator, lambda i, f: i.step(f, x, u=u, t=1.5, dt=0.05))
39 self.assertEqual(got.shape, (2, 1))
40 np.testing.assert_allclose(got, want, rtol=0, atol=atol, err_msg=integrator.name())
41 self.assertEqual(n_cxx, n_py, "the same number of derivative evaluations")
42 got, want, _, _ = self.both(integrator, lambda i, f: i.integrate(f, x.ravel(), u=u, t=0.0, dt=0.02, n=25))
43 self.assertEqual(got.shape, (2,))
44 np.testing.assert_allclose(got, want, rtol=0, atol=100 * atol, err_msg=integrator.name())

77 class Midpoint(KalmanIntegrator):
