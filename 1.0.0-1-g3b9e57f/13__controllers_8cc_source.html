27 std::pair<double, double> Simulate(Fixture::ControlledVelocity &plant, Controller &ctrl,
28 std::size_t steps, std::uint64_t seed = 1)
29 {
30 std::mt19937_64 rng(seed);
31 std::normal_distribution<double> gauss(0., 1.);
32 Engines::Linear engine;
33 const Matrix Qf = Utils::CholeskyFactor(plant.Q()), Rf = Utils::CholeskyFactor(plant.R());
34
35 Vector x = plant.X0(), truth = plant.X0(), nu;
36 Matrix P = plant.P0();
37 const Vector *lastNu = nullptr;
38 double t = 0., late = 0.;
39 for (std::size_t k = 0; k < steps; ++k)
40 {
41 const Vector u = ctrl(x, P, t, lastNu);
42 truth = plant.f(truth, u, t);
43 Vector w(2); w(0) = gauss(rng); w(1) = gauss(rng);
44 truth += Qf * w;
45 Vector z = plant.h(truth, t);
46 Vector v(1); v(0) = gauss(rng);
47 z += Rf * v;
48
49 Vector xp; Matrix Pp, F, S;
50 engine.Predict(plant, x, P, &u, t, xp, Pp, F);
51 engine.Innovation(plant, xp, Pp, z, t, nu, S);
52 engine.Update(plant, xp, Pp, z, t, x, P);
53 lastNu = &nu;
54 t += plant.DeltaT();
55 if (k >= steps / 2) late += std::abs(truth(0));
56 }
57 return {late / static_cast<double>(steps - steps / 2), ctrl.TotalCost()};
58 }
59}

171 // The DARE fixed point is a fixed point: plugging P back changes nothing.
172 const Matrix A = plant.F(), B = plant.B();
173 const Matrix P = Controllers::LQR::SolveDARE(A, B, Matrix::Identity(2), Matrix::Identity(1));
174 Matrix S = B.Transpose() * P * B; S += Matrix::Identity(1);
175 Matrix rhs = A.Transpose() * P * A;
176 rhs -= (A.Transpose() * P * B) * (Utils::Inverse(S) * (B.Transpose() * P * A));
177 rhs += Matrix::Identity(2);

307 // --- the Riccati and Stein solvers (Kalman/Algebra/Riccati.h) ----------------------
308 {
309 // Scalar DARE: p = q + a^2 p - a^2 b^2 p^2 / (r + b^2 p), a quadratic in p.
310 const double a = 1.1, b = 0.5, q = 2., r = 3.;
311 const double A2 = b * b, B2 = r - a * a * r - q * b * b, C2 = -q * r; // A2 p^2 + B2 p + C2 = 0
312 const double p = (-B2 + std::sqrt(B2 * B2 - 4. * A2 * C2)) / (2. * A2);
313 const Matrix P = Algebra::SolveDARE(Matrix(1, 1, a), Matrix(1, 1, b), Matrix(1, 1, q), Matrix(1, 1, r));
314 CHECK_CLOSE(P(0, 0), p, 1e-12 * p, "Riccati: the scalar DARE's positive root");
315
316 // A pole at 0.999: the plain recursion stopped 6e-8 short; Newton reaches rounding.
317 Matrix As(2, 2, 0.); As(0, 0) = 0.999; As(0, 1) = 0.01; As(1, 1) = 0.999;
318 Matrix Bs(2, 1, 0.); Bs(1, 0) = 0.01;
319 const Matrix Rs(1, 1, 1e3);
320 const Matrix Ps = Algebra::SolveDARE(As, Bs, Matrix::Identity(2), Rs);
321 Matrix next = As.Transpose() * Ps * As;
322 next -= (As.Transpose() * Ps * Bs) * Algebra::RiccatiGain(As, Bs, Rs, Ps);
323 next += Matrix::Identity(2);
324 double resid = 0., size = 0.;
325 for (std::size_t i = 0; i < 4; ++i)
326 {
327 resid = std::max(resid, std::abs(next.Data()[i] - Ps.Data()[i]));
328 size = std::max(size, std::abs(Ps.Data()[i]));
329 }
330 CHECK(resid < 1e-14 * size, "Riccati: the residual is at rounding for a slow pole");
331
332 // Stein: X = A^T X A + W, checked by substitution; a divergent sum is refused.
333 Matrix W = Matrix::Identity(2);
334 const Matrix X = Algebra::SolveStein(As, W);
335 Matrix back = As.Transpose() * X * As;
336 back += W;
337 double e = 0.;
338 for (std::size_t i = 0; i < 4; ++i) e = std::max(e, std::abs(back.Data()[i] - X.Data()[i]));
339 CHECK(e < 1e-12 * X(0, 0), "Stein: the doubling sum solves the equation");
340 bool threw = false;
341 try { Algebra::SolveStein(Matrix(1, 1, 1.5), Matrix(1, 1, 1.)); } catch (const std::runtime_error &) { threw = true; }
342 CHECK(threw, "Stein: an unstable A is refused");

344 Matrix Au(2, 2, 0.); Au(0, 0) = 1.2; Au(1, 1) = 0.5;
