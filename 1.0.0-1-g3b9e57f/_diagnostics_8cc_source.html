43 std::vector<double> Diagnostics::NIS() const
44 {
45 const std::size_t N = fFilter.NSteps();
46 std::vector<double> out(N);
47
48 for (std::size_t k = 0; k < N; ++k)
49 {
50 const std::vector<std::size_t> keep = Utils::ValidChannels(fFilter.Innovations()[k]);
51 if (keep.empty())
52 {
53 // Nothing was measured at this step, so there is no innovation to
54 // normalise; NaN keeps it out of averages instead of biasing them to 0.
55 out[k] = std::numeric_limits<double>::quiet_NaN();
56 continue;
57 }
58
59 const Matrix S = InnovationCovarianceAt(fFilter, k, keep);
60 const Vector nu = Utils::Compress(fFilter.Innovations()[k], keep);
61 out[k] = Utils::Mahalanobis(nu, Utils::Inverse(S));
62 }
63 return out;
64 }

78 {
79 const std::size_t N = fFilter.NSteps();
80 if (truth.size() != N)
81 throw std::runtime_error("Kalman::Diagnostics::NEES: truth must have one state per step.");

85 {
86 Vector e(fFilter.PosteriorStates()[k]);
