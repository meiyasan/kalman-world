49 Vector MLP::operator()(const Vector &features) const
50 {
51 if (fWeights.empty())
52 throw std::runtime_error("Kalman::Routers::NN::MLP: no layers have been added.");
53 if (features.Size() != fWeights.front().Cols())
54 {
55 std::ostringstream os;
56 os << "Kalman::Routers::NN::MLP: expected " << fWeights.front().Cols()
57 << " features, got " << features.Size() << ".";
58 throw std::runtime_error(os.str());
59 }
60
61 Vector x = features;
62 for (std::size_t l = 0; l < fWeights.size(); ++l)
63 {
64 x = fWeights[l] * x;
65 x += fBiases[l];
66
67 // tanh between layers, nothing after the last: the head is the softmax
68 // below. Same arrangement as nn/mlp.py.
69 if (l + 1 < fWeights.size())
70 for (std::size_t i = 0; i < x.Size(); ++i) x(i) = std::tanh(x(i));
71 }
72
73 // Softmax, shifted by the maximum. The shift is not cosmetic: logits of a
74 // few hundred are ordinary for a confident policy, and exp() of those
75 // overflows to inf, which then divides to NaN.
76 double hi = x(0);
77 for (std::size_t i = 1; i < x.Size(); ++i) hi = std::max(hi, x(i));
78
79 double sum = 0.;
80 for (std::size_t i = 0; i < x.Size(); ++i)
81 {
82 x(i) = std::exp(x(i) - hi);
83 sum += x(i);
84 }
85 for (std::size_t i = 0; i < x.Size(); ++i) x(i) /= sum;
86 return x;
87 }

89 void MLP::Load(const std::string &path)
90 {
91 std::ifstream in(path);
92 if (!in)
93 throw std::runtime_error("Kalman::Routers::NN::MLP: cannot open '" + path + "'.");
94
95 Clear();
96
97 std::size_t nLayers = 0;
98 if (!(in >> nLayers))
99 throw std::runtime_error("Kalman::Routers::NN::MLP: '" + path +
100 "' does not start with a layer count.");
101
102 for (std::size_t l = 0; l < nLayers; ++l)
103 {
104 std::size_t rows = 0, cols = 0;
105 if (!(in >> rows >> cols))
106 throw std::runtime_error("Kalman::Routers::NN::MLP: truncated shape for layer " +
107 std::to_string(l) + " in '" + path + "'.");
108
109 Matrix W(rows, cols);
110 for (std::size_t r = 0; r < rows; ++r)
111 for (std::size_t c = 0; c < cols; ++c)
112 if (!(in >> W(r, c)))
113 throw std::runtime_error(
114 "Kalman::Routers::NN::MLP: truncated weights for layer " +
115 std::to_string(l) + " in '" + path + "'.");
116
117 Vector b(rows);
118 for (std::size_t r = 0; r < rows; ++r)
119 if (!(in >> b(r)))
120 throw std::runtime_error(
121 "Kalman::Routers::NN::MLP: truncated bias for layer " +
122 std::to_string(l) + " in '" + path + "'.");
123
124 AddLayer(W, b);
125 }
126 }
