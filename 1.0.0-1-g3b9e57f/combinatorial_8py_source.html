1"""
2Combinatorial Kalman engine (CKF) — track *finding* by track following.
3
4Where the other engines estimate **one** trajectory from a time series of single
5measurements, this one reconstructs **several tracks** from an event whose every layer
6holds **many hits** (signal + background). For each seed it follows the track outward
7layer by layer; at each layer it opens a **branch** per hit that passes a χ² gate (plus a
8**hole** branch for detector inefficiency), **prunes** to a bounded beam of best
9candidates, and finally **arbitrates** shared hits so the reported tracks are disjoint.
10This is the Belle II CKF in miniature.
11
12Unlike the Jacobian engines, a single predict/update step is not the whole algorithm here:
13the *search* is. So this engine declares ``OWNS_PASS`` and implements :meth:`forward` /
14:meth:`backward` itself, and :class:`~kalman.framework.filters.ckf.CombinatorialKalmanFilter`
15stays the same thin leaf as ``LinearKalmanFilter`` — it names the engine and passes the
16search hyper-parameters, and owns no algorithm of its own.
17
18The per-step math still comes from :class:`ExtendedKalmanEngine`: nonlinear propagation
19(``model.f`` / Jacobian ``model.F``), the measurement map (``model.h`` / ``model.H``) and
20the RTS smoother are inherited unchanged. The additions are the χ² :meth:`gate` and the
21search around it.
22"""

32def _default_chi2_cut(dof: int, p: float = 0.99) -> float:
33 """χ² gate threshold for ``dof`` measurement channels (falls back if SciPy absent)."""
34 try:
35 from scipy.stats import chi2
36 return float(chi2.ppf(p, df=dof))
37 except Exception:
38 return {1: 6.63, 2: 9.21, 3: 11.34}.get(int(dof), 9.21)

98class GateContext:
99 """Everything about a gate that depends on the **prediction only**, not on the hit.
100
101 At one layer a candidate is tested against every hit on it. The predicted
102 measurement ``h(x)``, its Jacobian ``H``, the innovation covariance ``S``, the gain
103 ``K`` — and, because the Joseph form ``(I-KH) P (I-KH)ᵀ + K R Kᵀ`` never mentions
104 ``z``, the *posterior covariance itself* — are identical for all of them. Computing
105 them once per (candidate, layer) instead of once per (candidate, hit) lifts the
106 pseudo-inverse and the n×n products out of the inner loop.
107
108 Build one with :meth:`CombinatorialKalmanEngine.gate_context`, then call
109 :meth:`score` per hit.
110 """
111
112 __slots__ = ("zpred", "H", "R", "Sinv", "K", "P")
113
114 def __init__(self, zpred, H, R, Sinv, K, P):
115 self.zpred = zpred # (m,1) predicted measurement h(x)
116 self.H = H # (m,n) observation Jacobian
117 self.R = R # (m,m) measurement noise
118 self.Sinv = Sinv # (m,m) inverse innovation covariance
119 self.K = K # (n,m) Kalman gain
120 self.P = P # (n,n) posterior covariance — shared by every hit
121
122 def score(self, x, z):
123 """Score one candidate measurement → ``(chi2, x_post)``.
124
125 ``z`` is a single hit ``(m,)`` or ``(m,1)`` with no missing channels. The
126 posterior covariance is :attr:`P` for every hit, so it is not returned.
127 """
128 y = np.asarray(z, dtype=float).reshape(-1, 1) - self.zpred
129 chi2 = float((y.T @ self.Sinv @ y).item())
130 return chi2, x + self.K @ y

133class CombinatorialKalmanEngine(ExtendedKalmanEngine):
134 """
135 Extended-Kalman step + a measurement **gate**, driving a branch/prune/arbitrate search.
136
137 Search hyper-parameters, the CKF's counterpart of the UKF's ``alpha/beta/kappa``::
138
139 chi2_cut per-hit χ² gate (default: 99% of a χ²_m distribution)
140 max_candidates beam width kept per seed after each layer (default 6)
141 max_holes missed layers allowed on a track (default 2)
142 min_hits minimum assigned hits to accept a track (default 3)
143 shared_hit_frac max fraction of a track's hits shared with an already-accepted
144 track during arbitration (default 0.3)
145 hole_penalty score added per hole (default: one gate's worth of χ²)
146
147 Two gate entry points, same math:
148
149 - :meth:`gate` — one hit at a time, tolerates missing (NaN) channels.
150 - :meth:`gate_context` — hoists the hit-independent work out of the loop when a
151 whole layer is scored against one prediction. This is what :meth:`forward` uses.
152 """
153
154 NAME = "Combinatorial Kalman Engine"
155 DESCRIPTION = "EKF track-following with a χ² gate, beam pruning and hit arbitration."
156 VERSION = "2.0"
157 KIND = KalmanEngineKind.COMBINATORIAL
158
159 # The search *is* the algorithm here, so this engine drives the whole pass
160 # rather than exposing a single step for the filter's loop to call.
161 OWNS_PASS = True

164 min_hits=3, shared_hit_frac=0.3, hole_penalty=None):

166 # chi2_cut/hole_penalty stay None until a model is in hand: the default
167 # threshold depends on the number of measurement channels.
168 self.chi2_cut = None if chi2_cut is None else float(chi2_cut)
169 self.hole_penalty = None if hole_penalty is None else float(hole_penalty)
170 self.max_candidates = int(max_candidates)
171 self.max_holes = int(max_holes)
172 self.min_hits = int(min_hits)
173 self.shared_hit_frac = float(shared_hit_frac)

190 # ------------------------------------------------------------------
191 # The gate
192 # ------------------------------------------------------------------
193 def gate_context(self, model, xk, Pk, t=None) -> GateContext:
194 """Prepare the hit-independent half of the gate for the prediction ``(xk, Pk)``.
195
196 Assumes every measurement channel is present; for hits with NaN channels use
197 :meth:`gate`, which compresses the system per hit.
198 """
199 zpred, H = self._measure_measure(model, xk, t)
200 return self._context(model, Pk,
201 np.asarray(zpred, dtype=float).reshape(-1, 1),
202 np.asarray(H, dtype=float),
203 np.asarray(model.R(t=t), dtype=float))
204
205 @staticmethod
206 def _context(model, Pk, zpred, H, R) -> GateContext:
207 """Shared gain/covariance algebra — see :class:`GateContext`."""
208 S = R + H @ Pk @ H.T
209 Sinv = np.linalg.pinv(S)
210 K = Pk @ H.T @ Sinv
211 IKH = np.eye(model.n) - K @ H
212 P = IKH @ Pk @ IKH.T + K @ R @ K.T # Joseph form (numerically stable)
213 return GateContext(zpred, H, R, Sinv, K, P)
214
215 def gate(self, model, xk, Pk, z, t=None):
216 """Score and apply one candidate measurement.
217
218 Parameters
219 ----------
220 model : KalmanModel the system (provides ``h``/``H``/``R``).
221 xk, Pk the *predicted* state mean (n×1) and covariance (n×n).
222 z one candidate measurement (m×1); NaN channels are skipped.
223 t station/time tag forwarded to ``h``/``H``/``R``.
224
225 Returns
226 -------
227 (chi2, xk, Pk) : the innovation χ² (float), and the Kalman-updated mean
228 and covariance. ``chi2`` is ``+inf`` if every channel is missing.
229 """
230 z = np.asarray(z, dtype=float).reshape(-1, 1)
231 keep = ~np.isnan(z).ravel()
232 if not keep.any():
233 return np.inf, xk, Pk
234
235 zpred, H = self._measure_measure(model, xk, t)
236 zpred = np.asarray(zpred, dtype=float).reshape(-1, 1)
237 H = np.asarray(H, dtype=float)
238 R = np.asarray(model.R(t=t), dtype=float)
239
240 ctx = self._context(model, Pk, zpred[keep], H[keep, :], R[np.ix_(keep, keep)])
241 chi2, xk = ctx.score(xk, z[keep])
242 return chi2, xk, ctx.P

244 # ------------------------------------------------------------------
245 # Forward pass: seed → grow (branch/prune) → arbitrate
246 # ------------------------------------------------------------------
247 def forward(self, model, hits_by_layer, *, layer_radii=None, seeds=None, **kwargs):
248 """Find tracks in one event.
249
250 Parameters
251 ----------
252 hits_by_layer : sequence of (n_hits_l, m) arrays, innermost layer first.
253 layer_radii : (L,) radii of those layers; defaults to ``model.radii``.
254 seeds : explicit :class:`TrackCandidate` seeds, or None to build one
255 per hit on the innermost populated layer.
256
257 Returns
258 -------
259 list[TrackCandidate] — the accepted, arbitrated tracks.
260 """
261 radii = np.asarray(layer_radii if layer_radii is not None else model.radii, dtype=float)
262 hits = [np.asarray(h, dtype=float).reshape(-1, model.m) for h in hits_by_layer]
263 L = len(hits)
264 I = np.eye(model.n)
265
266 cut = self.gate_threshold(model)
267 penalty = self.hole_cost(model)
268 seeds = self.seed(model, hits, radii) if seeds is None else list(seeds)
269
270 if self._native_search(model, seeds):
271 self.candidates = self._find_native(model, hits, radii, seeds)
272 return self.arbitrate(self.candidates)
273
274 seed_tracks = []
275 for seed in seeds:
276 cands = [seed]
277 # Follow outward from the seed's own layer, whichever that is: seed() picks
278 # the innermost *populated* layer (not necessarily layer 0), and callers may
279 # pass seeds of their own. Starting at a fixed layer 1 would re-scan the
280 # seeding layer, letting a track take a second hit at the same radius off a
281 # zero-length propagation.
282 for l in range(seed.seed_layer + 1, L):
283 r = float(radii[l]); layer = hits[l]
284 # NaN channels need the per-hit compression in gate(); without them the
285 # whole layer shares one prepared gate (see GateContext).
286 masked = layer.size and bool(np.isnan(layer).any())
287 grown = []
288 for c in cands:
289 x, P, F = self._predict_predict(model, c.x, c.P, t=r)
290 if not np.all(np.isfinite(x)):
291 continue # track left the detector
292 Faccum = F @ c.Faccum
293 ctx = None if masked else self.gate_context(model, x, P, t=r)
294 for j in range(len(layer)):
295 if ctx is None:
296 chi2, xk, Pk = self.gate(model, x, P, layer[j].reshape(model.m, 1), t=r)
297 else:
298 chi2, xk = ctx.score(x, layer[j]); Pk = ctx.P
299 if chi2 < cut:
300 nc = c.branch()
301 nc.x, nc.P = xk, Pk
302 nc.chi2 += chi2; nc.score += chi2
303 nc.hit_ids.append((l, j))
304 nc.radii.append(r)
305 nc.xk.append(xk); nc.Pk.append(Pk)
306 nc.zk.append(layer[j])
307 nc.priors.append((x, P)); nc.Fc.append(Faccum)
308 nc.Faccum = I
309 grown.append(nc)
310 # no-hit (hole) branch: the track survives a missed layer
311 if c.holes < self.max_holes:
312 nc = c.branch()
313 nc.x, nc.P = x, P
314 nc.holes += 1; nc.score += penalty
315 nc.Faccum = Faccum
316 grown.append(nc)
317 if not grown:
318 break
319 grown.sort(key=lambda c: c.score)
320 cands = grown[: self.max_candidates]
321
322 finished = [c for c in cands if c.n_hits >= self.min_hits]
323 if finished:
324 seed_tracks.append(min(finished, key=lambda c: c.quality))
325
326 # Kept for inspection: the best candidate of every seed, before arbitration
327 # decides which of them survive as tracks (CombinatorialKalmanFilter.candidates).
328 self.candidates = seed_tracks
329 return self.arbitrate(seed_tracks)

410 def backward(self, model, tracks, **kwargs):
411 """RTS-smooth every track from :meth:`forward` → new, smoothed candidates."""
412 smoothed = []
413 for c in tracks:
414 N = c.n_hits
415 xs = [None] * N; Ps = [None] * N
416 xs[-1], Ps[-1] = c.xk[-1], c.Pk[-1]
417 for k in range(N - 2, -1, -1):
418 xs[k], Ps[k] = self._smooth_smooth(
419 model, (c.xk[k], c.Pk[k]), c.priors[k + 1], (xs[k + 1], Ps[k + 1]), c.Fc[k + 1]

421 out = c.branch()
422 out.xk, out.Pk = xs, Ps
423 smoothed.append(out)
424 return smoothed

429 def seed(self, model, hits, radii):
430 """One seed per hit on the innermost non-empty layer."""
431 seeds = []
432 I = np.eye(model.n)
433 for l in range(len(hits)):
434 if len(hits[l]) == 0:
435 continue
436 r = float(radii[l])
437 for j in range(len(hits[l])):
438 hit = hits[l][j]
439 x0, P0 = model.seed_state(hit)
440 seeds.append(TrackCandidate(
441 x=x0, P=P0, hit_ids=[(l, j)], radii=[r],
442 xk=[x0], Pk=[P0], zk=[np.asarray(hit, float).reshape(-1)],

444 ))

448 def arbitrate(self, tracks):
449 """Greedy hit competition: accept best tracks, reject those overlapping too much."""
450 accepted, used = [], set()
451 for c in sorted(tracks, key=lambda c: c.quality):
452 ids = set(c.hit_ids)
453 if len(ids & used) <= self.shared_hit_frac * len(ids):
454 accepted.append(c)
455 used |= ids
456 return accepted
