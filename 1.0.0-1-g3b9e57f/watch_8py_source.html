1"""Tell connected MCP clients when the shared registry changes.
2
3A client that subscribed to one of our resources gets a
4``notifications/resources/updated`` for it when somebody — this server, the
5REST API, another agent, a Unity scene — registers a model, loads a plugin,
6sets up a source or pushes measurements. The client then reads the resource
7again; the notification itself carries no registry content, so it needs no
8scope check of its own.
9
10Both ways to subscribe are served, because both are in use:
11
12 - ``resources/subscribe`` (MCP up to 2025-11-25): the watcher keeps the
13 subscribed sessions per URI and notifies each one;
14 - ``subscriptions/listen`` (MCP 2026-07-28): the watcher publishes to the
15 SDK's subscription bus, which owns the streams.
16
17 ============================ =======================================
18 registry event resource reported as updated
19 ============================ =======================================
20 ``model.*``, ``plugin.*`` ``catalog://models``
21 ``source.*`` ``catalog://sources``
22 ``measurement.received`` ``catalog://sources``
23 every event ``registry://events``
24 ============================ =======================================
25
26The watcher polls for events — the registry file in-process, or the REST API
27— because the change may come from any process. In front of a REST API whose
28events it has no token to read, it falls back to ``GET /events/cursor`` and
29reports every resource on every change (see :meth:`RegistryWatcher._poll`). It
30is one task per server however many sessions are open: the SDK enters the
31lifespan once per session on stdio and SSE, and once per server on streamable
32HTTP.
33
34A client that does neither calls ``list_events`` instead.
35"""

48def uris_for(event_type: str) -> set:
49 """The resources a registry event of this type makes stale."""
50 family = event_type.split(".", 1)[0]
51 out = {EVENTS}
52 if family in ("model", "plugin"):
53 out.add(MODELS)
54 elif family in ("source", "measurement"):
55 out.add(SOURCES)
56 return out

68 # new session object for every request, all on the same connection, so a

73 def attach(self, server) -> bool:
74 """Serve ``resources/subscribe`` / ``unsubscribe`` on ``server``, for
75 clients older than ``subscriptions/listen``. Registering the handlers
76 is also what makes the server advertise ``resources.subscribe``."""
77 # The SDK's server class has no public hook for these two requests; its
78 # low-level server does (add_request_handler), reached through an
79 # attribute that is not part of the public API. If an SDK release moves
80 # it, say so at start-up -- and tests/test_registry.py fails -- instead
81 # of clients quietly never hearing anything.
82 lowlevel = getattr(server, "_lowlevel_server", None)
83 if lowlevel is None or not hasattr(lowlevel, "add_request_handler"):
84 print("kalman.mcp: this MCP SDK version gives no way to serve resources/subscribe; "
85 "clients before MCP 2026-07-28 get no change notifications and must call "
86 "list_events. subscriptions/listen is not affected.", file=sys.stderr)
87 return False
88 import mcp.types as types
89
90 def key(session):
91 return id(getattr(session, "_connection", session))
92
93 async def subscribe(ctx, params):
94 held = self._subscribers.setdefault(str(params.uri), {})
95 held[key(ctx.session)] = ctx.session
96 while len(held) > MAX_SUBSCRIBERS: # oldest first
97 held.pop(next(iter(held)))
98 return types.EmptyResult()
99
100 async def unsubscribe(ctx, params):
101 self._subscribers.get(str(params.uri), {}).pop(key(ctx.session), None)
102 return types.EmptyResult()
103
104 lowlevel.add_request_handler("resources/subscribe", types.SubscribeRequestParams, subscribe)
105 lowlevel.add_request_handler("resources/unsubscribe", types.UnsubscribeRequestParams,
106 unsubscribe)
107 return True

109 async def _notify(self, uri: str):
110 from mcp.server.subscriptions import ResourceUpdated
111 await self.bus.publish(ResourceUpdated(uri=uri)) # subscriptions/listen
112 for k, session in list(self._subscribers.get(uri, {}).items()): # resources/subscribe
113 try:
114 await session.send_resource_updated(uri)
115 except Exception: # the connection is gone
116 self._subscribers[uri].pop(k, None)

135 async def _poll(self, cursor: int | None):
136 """``(new cursor, resources to report)`` since ``cursor``; ``None``
137 means "start from now", which reports nothing.
138
139 Precise when the events can be read: our own registry file, a REST API
140 with auth off, or one that accepts ``KALMAN_MCP_TOKEN``. Otherwise
141 coarse: ``GET /events/cursor`` needs no token and says only *that*
142 something changed, so every resource is reported and the clients read
143 again what they care about — each read checked against their own token.
144 Notifications therefore never depend on this server holding a token."""
145 from ..api.auth import AuthError
146 if self._precise:
147 try:
148 if self.backend.kind == "local":
149 # Our own registry file. No scope check: nothing read here
150 # leaves the process, only "this resource changed".
151 read = lambda since: self.backend.api.events(since, 100) # noqa: E731
152 else:
153 read = lambda since: self.backend.events(since, 100, self.token) # noqa: E731
154 if cursor is None:
155 return (await asyncio.to_thread(read, -1))["last"], set()
156 res = await asyncio.to_thread(read, cursor)
157 stale = set()
158 for e in res["events"]:
159 stale |= uris_for(e["type"])
160 return res["last"], stale
161 except AuthError as e:
162 self._precise = False
163 print(f"kalman.mcp: no token that may read registry events ({e}) Clients are "
164 "still notified, of every resource on every change; set KALMAN_MCP_TOKEN "
165 "to a read token for per-resource notifications.", file=sys.stderr)
166 last = await asyncio.to_thread(self.backend.cursor)
167 if cursor is None or last == cursor:
168 return last, set()
169 return last, {MODELS, SOURCES, EVENTS}
170
171 async def _run(self):
172 cursor, delay = None, self.poll
173 while True:
174 try:
175 cursor, stale = await self._poll(cursor)
176 for uri in sorted(stale):
177 await self._notify(uri)
178 delay = self.poll
179 except asyncio.CancelledError:
180 raise
181 except Exception: # the API is down: wait, then look again
182 delay = min(delay * 2, 30.0)
183 await asyncio.sleep(delay)
