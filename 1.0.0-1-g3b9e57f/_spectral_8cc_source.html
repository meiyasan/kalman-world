27 Spectral::Spectral(std::size_t ny, std::size_t nx, double dt, double Ly, double Lx,
28 std::shared_ptr<Integrator> integrator, std::uint64_t seed)
29 : Solver(std::move(integrator)), fNy(ny), fNx(nx), fLy(Ly), fLx(Lx), fDt(dt), fRandom(seed)
30 {
31 if (ny < 2 || nx < 2) throw std::runtime_error("Solvers::Spectral: the grid needs at least 2 cells a side.");
32 if (!(Ly > 0.) || !(Lx > 0.)) throw std::runtime_error("Solvers::Spectral: domain lengths must be positive.");
33
34 const std::vector<double> ix = FFT::Frequencies(nx);
35 const std::vector<double> iy = FFT::Frequencies(ny);
36
37 fIx = Field(ny, nx); fIy = Field(ny, nx);
38 fKx = Field(ny, nx); fKy = Field(ny, nx);
39 fK2 = Field(ny, nx); fK2Inv = Field(ny, nx);
40 fDealias = Field(ny, nx);
41
42 const double sx = 2. * kPi / Lx, sy = 2. * kPi / Ly;
43 for (std::size_t i = 0; i < ny; ++i)
44 for (std::size_t j = 0; j < nx; ++j)
45 {
46 fIx(i, j) = ix[j];
47 fIy(i, j) = iy[i];
48 fKx(i, j) = sx * ix[j];
49 fKy(i, j) = sy * iy[i];
50 fK2(i, j) = fKx(i, j) * fKx(i, j) + fKy(i, j) * fKy(i, j);
51 fK2Inv(i, j) = fK2(i, j) > 0. ? 1. / std::max(fK2(i, j), 1e-12) : 0.;
52 fDealias(i, j) = (std::abs(ix[j]) <= static_cast<double>(nx / 3)
53 && std::abs(iy[i]) <= static_cast<double>(ny / 3)) ? 1. : 0.;
54 }
55 }

96 void Spectral::Velocity(const Field &w, Field &u, Field &v) const
97 {
98 CheckShape(w);
99 SpectralField psih = FFT::Forward(w);
100 for (std::size_t i = 0; i < psih.Size(); ++i) psih.Data()[i] *= fK2Inv.Data()[i];
101 u = Real(FFT::Inverse(DerivativeY(psih)));
102 SpectralField vh = DerivativeX(psih);
103 vh *= -1.;
104 v = Real(FFT::Inverse(vh));
105 }

131 Field Spectral::RandomIC(double amp) const
132 {
133 std::normal_distribution<double> gauss(0., 1.);
134 SpectralField wh(fNy, fNx);
135 for (std::size_t i = 0; i < fNy; ++i)
136 for (std::size_t j = 0; j < fNx; ++j)
137 {
138 const double kmag = std::sqrt(fIx(i, j) * fIx(i, j) + fIy(i, j) * fIy(i, j));
139 const double taper = std::exp(-0.5 * std::pow(kmag / 6., 2)) * fDealias(i, j);
140 wh(i, j) = cd(gauss(fRandom), gauss(fRandom)) * taper;
141 }
142 Field w = Real(FFT::Inverse(wh));
143
144 double mean = 0., var = 0.;
145 for (std::size_t i = 0; i < w.Size(); ++i) mean += w.Data()[i];
146 mean /= static_cast<double>(w.Size());
147 for (std::size_t i = 0; i < w.Size(); ++i) var += std::pow(w.Data()[i] - mean, 2);
148 const double sd = std::sqrt(var / static_cast<double>(w.Size()));
149 w *= amp / (sd + 1e-12);
150 return w;
151 }
