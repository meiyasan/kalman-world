43 def test_every_layout_and_integrator_matches_numpy(self):
44 from modules.apps.weather.flow import NS2D
45 for kind in (KalmanIntegratorKind.EULER, KalmanIntegratorKind.RK4,
46 KalmanIntegratorKind.DORMAND_PRINCE):
47 ref = NS2D(n=16, nu=1/40., kf=4, chi=2.0, dt=0.01, seed=3, integrator=kind)
48 s = NS2D(n=16, nu=1/40., kf=4, chi=2.0, dt=0.01, seed=3, integrator=kind,
49 backend="torch", device="cpu")
50 ens = np.stack([ref.random_ic() for _ in range(3)])
51 want = np.stack([ref.advance(e.copy(), 5) for e in ens])
52 np.testing.assert_allclose(s.advance(ens.copy(), 5), want, atol=1e-10) # (N, ny, nx)
53 np.testing.assert_allclose(s.advance(ens.reshape(3, -1).copy(), 5),
54 want.reshape(3, -1), atol=1e-10) # (N, ny*nx)
55 np.testing.assert_allclose(s.advance(ens[0].copy(), 5), want[0], atol=1e-10) # (ny, nx)
56 np.testing.assert_allclose(s.advance(ens[0].ravel().copy(), 5),
57 want[0].ravel(), atol=1e-10) # (ny*nx,)
