29 def test_recovers_two_unrelated_tones(self):
30 """No demodulation, no shared structure: two unrelated real tones recovered
31 jointly via the time-varying H(t), F=I lab-frame formulation."""
32 np.random.seed(0)
33 fs, T = 1024.0, 4.0
34 t = np.arange(int(T * fs)) / fs
35 f_a, f_b = 60.0, 173.2
36 A_a, A_b = 1.0, 0.5
37 sig = (A_a * np.cos(2 * np.pi * f_a * t + 0.4)
38 + A_b * np.cos(2 * np.pi * f_b * t - 1.1)
39 + 0.05 * np.random.randn(len(t)))
40 m = PhasorOscillatorModel("demo", [f_a, f_b], fs=fs,
41 measurement_uncertainty=0.05,
42 process_uncertainty=1e-4, p0_std=10.0)
43 rec = m.filter(sig, smooth=True)
44 amp = m.amplitudes(rec)[len(t) // 2:].mean(axis=0)
45 self.assertAlmostEqual(amp[0], A_a, delta=0.15)
46 self.assertAlmostEqual(amp[1], A_b, delta=0.15)

83 def test_odd_harmonic_dominance_even_settle_near_zero(self):
84 """For a half-wave-symmetric (ideal) grid, joint estimation should leave
85 the even harmonics (H2, H4) near zero while H1/H3/H5 converge to truth."""
86 np.random.seed(0)
87 f0, fs, T = 50.0, 1024.0, 4.0
88 t = np.arange(int(T * fs)) / fs
89 A1, A3, A5 = 230.0, 15.0, 5.0
90 v_clean = (A1 * np.cos(2 * np.pi * f0 * t)
91 + A3 * np.cos(2 * np.pi * 3 * f0 * t + np.pi / 4)
92 + A5 * np.cos(2 * np.pi * 5 * f0 * t + np.pi / 3))
93 noise_std = 10.0
94 v_noisy = v_clean + np.random.normal(0, noise_std, len(t))
95
96 grid = PowerGridStateModel("L1", f0=f0, harmonics=5, fs=fs,
97 measurement_uncertainty=noise_std,
98 process_uncertainty=0.01)
99 grid.initial_state([[A1, 0], [0, 0], [A3, 0], [0, 0], [A5, 0]])
100 rec = grid.filter(v_noisy, smooth=True, reset=True)
101 amp, _ = grid.harmonic_envelope(rec)
102 steady = amp[-len(t) // 10:].mean(axis=0)
103
104 self.assertAlmostEqual(steady[0], A1, delta=15.0) # H1
105 self.assertAlmostEqual(steady[2], A3, delta=5.0) # H3
106 self.assertAlmostEqual(steady[4], A5, delta=3.0) # H5
107 self.assertLess(steady[1], 10.0) # H2 ~ 0
108 self.assertLess(steady[3], 10.0) # H4 ~ 0

110 def test_h2_rises_under_half_wave_rectifier_fault(self):
