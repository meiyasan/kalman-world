2"""
3Kalman filter **consistency diagnostics**.
4
5Everything here is computed *post-hoc* from a completed forward pass — the stored
6``prior`` / ``posterior`` / ``measurement`` logs plus the model's ``H`` / ``R`` / ``h`` —
7so it needs no engine changes and works for any filter (linear, EKF, UKF, …) and any
8model (named state enum or generic matrix model).
9
10:class:`KalmanDiagnostics` wraps a :class:`~kalman.framework.filter.KalmanFilter` that has
11already run ``forward()`` / ``filter()`` and exposes:
12
13* :meth:`innovations` — the measurement residual ν and its covariance S,
14* :meth:`gains` — the Kalman gain over time,
15* :meth:`nis` — Normalized Innovation Squared (no ground truth needed),
16* :meth:`nees` — Normalized Estimation Error Squared (needs the truth),
17* :meth:`plot` — the model-agnostic consistency dashboard.
18
19``KalmanFilter`` delegates its ``innovations()/gains()/nis()/nees()/diagnostics()`` to this
20class, so ``kf.diagnostics(sample)`` simply *draws* the dashboard.
21"""
22from __future__ import annotations
23import numpy as np
24
25from .io.record import KalmanRecord
26
27
28class KalmanDiagnostics:
29 """Post-hoc consistency diagnostics for one Kalman forward pass.
30
31 Construct from a filter that has already run ``forward()``/``filter()``::
32
33 KalmanDiagnostics(kf).nis()
34 KalmanDiagnostics(kf).plot(sample=truth)
35 """
36
37 def __init__(self, filter):
38 self.filter = filter
39
40 # ------------------------------------------------------------------
41 # Stored forward-pass quantities → plain arrays
42 # ------------------------------------------------------------------
43 def _arrays(self):
44 """Stack the stored forward-pass quantities into plain arrays.
45
46 Returns (N, n, m, z, x_prior, P_prior, x_post, P_post)."""
47 f = self.filter
48 N = len(f.posterior)
49 if N == 0:
50 raise RuntimeError("No forward pass to diagnose — run forward()/filter() first.")
51 n, m = f.n, f.m
52 z = np.array([np.reshape(zz, (m, 1)) for zz in f.measurement[:N]])
53 xpr = np.array([np.reshape(p[0], (n, 1)) for p in f.prior])
54 Ppr = np.array([np.asarray(p[1]).reshape(n, n) for p in f.prior])
55 xpo = np.array([np.reshape(p[0], (n, 1)) for p in f.posterior])
56 Ppo = np.array([np.asarray(p[1]).reshape(n, n) for p in f.posterior])
57 return N, n, m, z, xpr, Ppr, xpo, Ppo
58
59 # ------------------------------------------------------------------
60 # Numeric diagnostics
61 # ------------------------------------------------------------------
62 def innovations(self):
63 """Per-step innovation (measurement residual) and its covariance.
64
65 ν_k = z_k − h(x⁻_k), S_k = H_k P⁻_k H_kᵀ + R (H_k = ∂h/∂x at the prior).
66 A well-tuned filter has ν zero-mean and white, ~95% inside ±2·√diag(S).
67
68 Returns (nu, S): shapes (N, m) and (N, m, m)."""
69 model = self.filter.model
70 N, n, m, z, xpr, Ppr, _, _ = self._arrays()
71 R = np.asarray(model.R()).reshape(m, m)
72 nu = np.zeros((N, m)); S = np.zeros((N, m, m))
73 for k in range(N):
74 Hk = np.asarray(model.H(xpr[k])).reshape(m, n)
75 nu[k] = (z[k] - np.asarray(model.h(xpr[k])).reshape(m, 1)).ravel()
76 S[k] = Hk @ Ppr[k] @ Hk.T + R
77 return nu, S
78
79 def gains(self):
80 """Per-step Kalman gain K_k = P⁻_k H_kᵀ S_k⁻¹. Shape (N, n, m).
81
82 K[k, i, j] = how strongly measurement channel j moves state i at step k."""
83 model = self.filter.model
84 N, n, m, z, xpr, Ppr, _, _ = self._arrays()
85 R = np.asarray(model.R()).reshape(m, m)
86 K = np.zeros((N, n, m))
87 for k in range(N):
88 Hk = np.asarray(model.H(xpr[k])).reshape(m, n)
89 S = Hk @ Ppr[k] @ Hk.T + R
90 K[k] = Ppr[k] @ Hk.T @ np.linalg.inv(S)
91 return K
92
93 def nis(self):
94 """Normalized Innovation Squared ε_k = ν_kᵀ S_k⁻¹ ν_k (≈ χ²(m)). Shape (N,).
95
96 Consistency check that needs NO ground truth — usable on real data."""
97 nu, S = self.innovations()
98 N, m = nu.shape
99 from .native import HAS_C_ACCEL
100 if HAS_C_ACCEL:
101 from .native import lib, as_contiguous, dptr
102 nis = np.empty(N, dtype=np.float64)
103 lib.diag_nis_batch(N, m, dptr(as_contiguous(nu)), dptr(as_contiguous(S)), dptr(nis))
104 return nis
105 return np.array([float(nu[k] @ np.linalg.solve(S[k], nu[k])) for k in range(N)])
106
107 def nees(self, truth):
108 """Normalized Estimation Error Squared (needs ground truth, e.g. ``sample.xk``):
109 (x_k − x̂_k)ᵀ P_k⁻¹ (x_k − x̂_k) (≈ χ²(n)). Shape (N,).
110
111 Checks whether P is honest: time-mean ≈ n means well-calibrated, ≫ n means
112 the filter is over-confident, ≪ n means over-conservative."""
113 N, n, m, z, xpr, Ppr, xpo, Ppo = self._arrays()
114 truth = np.asarray(truth)
115 truth = truth.reshape(truth.shape[0], -1)
116 e = truth[:N] - xpo.reshape(N, n)
117 from .native import HAS_C_ACCEL
118 if HAS_C_ACCEL:
119 from .native import lib, as_contiguous, dptr
120 nees = np.empty(N, dtype=np.float64)
121 lib.diag_nees_batch(N, n, dptr(as_contiguous(e)), dptr(as_contiguous(Ppo)), dptr(nees))
122 return nees
123 return np.array([float(e[k] @ np.linalg.pinv(Ppo[k]) @ e[k]) for k in range(N)])

125 # ------------------------------------------------------------------
126 # Dashboard
127 # ------------------------------------------------------------------
128 def plot(self, sample=None, *, sigma=3, figsize=(14, 9), **kwargs):
129 """Filter-consistency dashboard computed from the forward pass:
130
131 1. **Innovation** ν per measurement channel, shaded ±``sigma``·√S — should
132 be zero-mean and stay (mostly) inside the band.
133 2. **Kalman gain** K(t) for measurement channel 0 across states — shows how
134 strongly a measurement moves each state, converging to steady state.
135 3. **NIS** with a χ²(m) 95% band (no truth needed).
136 4. **NEES** with a χ²(n) 95% band — only if ``sample`` (ground truth) given.
137
138 ``sample`` may be a ``KalmanRecord`` (uses ``sample.xk``) or a truth array.
139 """
140 import matplotlib.pyplot as plt
141 f = self.filter
142 N = len(f.posterior)
143 n = f.n; m = f.m
144 fs = f.sample_rate() or 1.0
145 t0 = float(f.model.t0) if f.model.t0 is not None else 0.0
146 t = t0 + np.arange(N) / fs
147
148 nu, S = self.innovations()
149 K = self.gains()
150 nis = self.nis()
151 truth = sample.xk if isinstance(sample, KalmanRecord) else sample
152 has_nees = truth is not None
153 nees = self.nees(truth) if has_nees else None
154
155 try:
156 from scipy.stats import chi2
157 band = lambda df: (chi2.ppf(0.025, df), df, chi2.ppf(0.975, df))
158 except Exception:
159 band = lambda df: (None, df, None)
160
161 # state labels (use the model's state enum when it lines up with n)
162 try:
163 names = [str(s).lower() for s in list(f.model.states)]
164 labels = names if len(names) == n else [f"x{i}" for i in range(n)]
165 except Exception:
166 labels = [f"x{i}" for i in range(n)]
167
168 npan = 4 if has_nees else 3
169 fig, ax = plt.subplots(2, 2, figsize=figsize) if npan == 4 else plt.subplots(1, 3, figsize=(figsize[0], figsize[1] / 2.2))
170 ax = np.atleast_1d(ax).ravel()
171 fig.suptitle("Kalman filter consistency diagnostics", fontweight="bold")
172
173 # 1) Innovations + ±sigma·sqrt(S)
174 a0 = ax[0]
175 for j in range(m):
176 sd = np.sqrt(np.clip(S[:, j, j], 0, None))
177 line, = a0.plot(t, nu[:, j], lw=1.0, label=f"ν[{j}]")
178 a0.fill_between(t, -sigma * sd, sigma * sd, color=line.get_color(), alpha=0.15, lw=0,
179 label=(f"±{sigma:g}σ" if j == 0 else None))
180 a0.axhline(0, color="gray", lw=0.8, ls="--")
181 a0.set_title("Innovation (measurement residual) ν = z − H·x⁻"); a0.set_xlabel("Time [s]"); a0.set_ylabel("residual")
182 a0.legend(fontsize=8); a0.grid(True, ls=":", alpha=0.6)
183
184 # 2) Kalman gain for measurement channel 0 (log y-scale)
185 a1 = ax[1]
186 for i in range(n):
187 a1.plot(t, K[:, i, 0], lw=1.0, label=labels[i])
188 # symlog: log-like compression so gains spanning orders of magnitude are all
189 # visible, while still handling zero / negative (off-diagonal) gains.
190 kmax = float(np.max(np.abs(K[:, :, 0]))) if K.size else 1.0
191 linthresh = max(kmax * 1e-3, 1e-12)
192 a1.set_yscale("symlog", linthresh=linthresh)
193 a1.set_title("Kalman gain |K(t)| (meas. channel 0, log scale)"); a1.set_xlabel("Time [s]"); a1.set_ylabel("gain (symlog)")
194 a1.legend(fontsize=8); a1.grid(True, ls=":", alpha=0.6, which="both")
195
196 # 3) NIS + chi2(m) band
197 a2 = ax[2]
198 a2.plot(t, nis, lw=0.8, color="steelblue", label="NIS")
199 lo, mid, hi = band(m)
200 a2.axhline(mid, color="gray", lw=1.0, ls="--", label=f"E=m={m}")
201 if lo is not None:
202 a2.axhline(lo, color="crimson", lw=0.8, ls=":"); a2.axhline(hi, color="crimson", lw=0.8, ls=":", label="95% χ²")
203 a2.set_title(f"Normalized Innovation Squared\ntime-mean ⟨NIS⟩ = {nis.mean():.2f} ≈ m = {m} (meas. dim)"); a2.set_xlabel("Time [s]"); a2.set_ylabel("νᵀS⁻¹ν")
204 a2.legend(fontsize=8); a2.grid(True, ls=":", alpha=0.6)
205
206 # 4) NEES + chi2(n) band
207 if has_nees:
208 a3 = ax[3]
209 a3.plot(t, nees, lw=0.8, color="darkorange", label="NEES")
210 lo, mid, hi = band(n)
211 a3.axhline(mid, color="gray", lw=1.0, ls="--", label=f"E=n={n}")
212 if lo is not None:
213 a3.axhline(lo, color="crimson", lw=0.8, ls=":"); a3.axhline(hi, color="crimson", lw=0.8, ls=":", label="95% χ²")
214 a3.set_title(f"Normalized Estimation Error Squared\ntime-mean ⟨NEES⟩ = {nees.mean():.2f} ≈ n = {n} (state dim)"); a3.set_xlabel("Time [s]"); a3.set_ylabel("eᵀP⁻¹e")
215 a3.legend(fontsize=8); a3.grid(True, ls=":", alpha=0.6)
216
217 fig.tight_layout()
218 return fig
