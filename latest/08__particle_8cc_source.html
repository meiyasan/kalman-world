16 // RMSE of the raw measurement against the truth, over the steps where one arrived.
17 double RawRMSE(const Sampler &s, std::size_t channel = 0)
18 {
19 double e = 0.;
20 std::size_t count = 0;
21 for (std::size_t k = 0; k < s.NSteps(); ++k)
22 {
23 const double d = s.Measurements()[k](channel) - s.Truth()[k](channel);
24 if (!std::isfinite(d)) continue; // a dropped channel is no measurement
25 e += d * d;
26 ++count;
27 }
28 return std::sqrt(e / static_cast<double>(count));
29 }
30}
