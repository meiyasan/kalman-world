1"""Netatmo indoor modules: the CO2 of a room, five minutes at a time.
2
3A Netatmo weather station has a main indoor module and up to three additional
4indoor modules; each reads CO2 (with temperature, humidity and, for the main one,
5noise and pressure) and reports about every five minutes. A room watched by five
6modules takes two stations: each module says which station it belongs to.
7
8What the data sheet gives, and what this adapter takes from it:
9
10- CO2 from 0 to 5000 ppm, accuracy +/- 50 ppm or +/- 5 %. Most of that is an
11 offset that moves slowly (the sensor ages, and the module recalibrates
12 itself), so a module is an instrument with a calibration: its
13 ``"<name>.offset"`` is estimated with ``joint``, as a slow random walk (a
14 ``drift`` per step, no ``tau``: an offset does not return to where it started);
15- the noise from one reading to the next is much smaller than that accuracy:
16 ``sigma`` (20 ppm by default) is that noise, to be checked on a quiet night;
17- one reading every five minutes: the room model steps every five minutes
18 (``RoomCO2Model(..., fs=1/300)``).
19
20Where the readings come from:
21
22- :class:`NetatmoCloud`: the Netatmo web API (``getmeasure``), the only interface
23 Netatmo documents. ``scale="max"`` gives every reading; ``"30min"`` and longer
24 give averages. The credentials are outbound secrets, never in the
25 repository (see the class);
26- :class:`NetatmoCSV`: files on disk, the export of the Netatmo web page or a
27 local logger's ``time, co2`` columns. The station has no documented local
28 interface; a local reader (HomeKit, a logger) writes such a file, or is any
29 function returning ``(times, ppm)``, given as ``source=``.
30
31Either way the readings are laid on the model's steps from ``start``: a step
32with no reading is NaN, which the filter skips::
33
34 from kalman.apps import RoomCO2Model, empty_at_night
35 from kalman.adapters import NetatmoModule, NetatmoCloud
36
37 room = RoomCO2Model("unread", count_people=True, fs=1 / 300) # every reading comes from a module
38 cloud = NetatmoCloud() # credentials from the environment
39 modules = [NetatmoModule(room, "main", (2, 2), station=STATION_1, source=cloud),
40 NetatmoModule(room, "desk", (2, 9), station=STATION_1, module=MODULE_1, source=cloud),
41 ...]
42 watched = room
43 for module in modules:
44 watched = watched.add_instrument(module)
45 watched = watched.add_instrument(empty_at_night(room)) # nobody between midnight and 6 am
46 zk = watched.measurements(start=t0, end=t1) # fetched, on the 5-minute steps
47 run = watched.joint({f"{m.name}.offset": (50.0, 0.5) for m in modules[1:]}).filter(zk, uk=np.ones((len(zk), 1)))
48
49The outdoor level (``outdoor=``) is what the empty room settles to, so the
50offsets are measured against it. It is not known better than about 20 ppm, so
51one module is kept as the reference and the others' offsets are estimated.
52"""

74def on_steps(times, values, start, end, step=STEP):
75 """Readings at ``times`` [unix s] laid on the steps ``start, start + step, ...``
76 before ``end``: one row per step, the mean of the readings within half a step
77 of it, NaN where there is none."""
78 n = int(np.floor((float(end) - float(start)) / float(step)))
79 out = np.full(max(n, 0), np.nan)
80 if n <= 0:
81 return out.reshape(0, 1)
82 t = np.asarray(times, dtype=float).reshape(-1)
83 v = np.asarray(values, dtype=float).reshape(-1)
84 k = np.rint((t - float(start)) / float(step)).astype(np.int64)
85 keep = (k >= 0) & (k < n) & np.isfinite(v)
86 sums, counts = np.zeros(n), np.zeros(n)
87 np.add.at(sums, k[keep], v[keep])
88 np.add.at(counts, k[keep], 1.0)
89 read = counts > 0
90 out[read] = sums[read] / counts[read]
91 return out.reshape(n, 1)

94class NetatmoModule(KalmanInstrument):
95 """One Netatmo indoor module, reading the CO2 of one cell of a room.
96
97 Parameters
98 ----------
99 room : the room model (a :class:`~kalman.apps.RoomCO2Model`); the module reads its cell ``at``.
100 name : what the module is called; its offset is the parameter ``"<name>.offset"``.
101 at : ``(row, column)`` of the cell where it stands.
102 station : the station's id (the main module's MAC address).
103 module : the additional module's id; ``None`` for the main module.
104 source : where the readings come from: :class:`NetatmoCloud`, :class:`NetatmoCSV`,
105 or a function ``(station, module, start, end) -> (times, ppm)``.
106 sigma : the noise from one reading to the next [ppm].
107 offset : the offset the module is believed to have [ppm].
108 """
109
110 def __init__(self, room, name, at, *, station, module=None, source=None, sigma=20.0, offset=0.0):
111 self.station, self.module, self.source = station, module, source
112 provider = None if source is None else self._readings
113 super().__init__(name, [room.sensor(name, at, sigma=sigma)], rate=1.0 / STEP, offset=offset,
114 provider=provider)
115
116 def _readings(self, *, start, end, **query):
117 fetch = self.source.readings if hasattr(self.source, "readings") else self.source
118 times, ppm = fetch(self.station, self.module, start, end, **query)
119 return on_steps(times, ppm, start, end)

122class NetatmoCloud(KalmanProvider):
123 """The CO2 of Netatmo modules, from the Netatmo web API.
124
125 The credentials of a Netatmo app (dev.netatmo.com, scope ``read_station``)
126 are outbound secrets, resolved like every provider's
127 (:mod:`kalman.providers.credentials`): from the environment,
128 ``NETATMO_CLIENT_ID``, ``NETATMO_CLIENT_SECRET`` and ``NETATMO_REFRESH_TOKEN``,
129 or from ``~/.kalman/credentials.json`` (``netatmo_client_id``, ...). They are
130 never in the repository.
131
132 Netatmo replaces the refresh token each time it is used, and the old one stops
133 working: the new one is kept in ``~/.kalman/netatmo.json`` (``token_file=`` or
134 ``NETATMO_TOKEN_FILE`` for another place), readable by its owner only, and is
135 the one used next time.
136
137 ``scale``: ``"max"`` for every reading, ``"30min"``, ``"1hour"``, ... for averages.
138 """
139
140 NAME = "Netatmo Cloud"
141 DESCRIPTION = "CO2 readings of Netatmo indoor modules, from the Netatmo web API (getmeasure)."
142 VERSION = "1.0"
143 KIND = KalmanProviderKind.REMOTE
144 MEASUREMENTS = ({"name": "co2", "units": "ppm", "description": "CO2 of one indoor module"},)
145 REQUIRES = ("NETATMO_CLIENT_ID", "NETATMO_CLIENT_SECRET", "NETATMO_REFRESH_TOKEN")
146 PAGE = 1024 # readings per request, at most
147 TOKENS = "~/.kalman/netatmo.json" # where the replaced refresh token is kept
148
149 def __init__(self, *, scale="max", token_file=None, opener=None, cache=True):
150 super().__init__(cache=cache)
151 self.scale = scale
152 self.token_file = os.path.expanduser(token_file or os.environ.get("NETATMO_TOKEN_FILE") or self.TOKENS)
153 self._open = opener or (lambda request: urllib.request.urlopen(request, timeout=30))
154 self._access, self._expires = None, 0.0
155
156 def _call(self, path, params, *, post=False):
157 data = urllib.parse.urlencode(params)
158 if post:
159 request = urllib.request.Request(API + path, data=data.encode(), method="POST")
160 else:
161 request = urllib.request.Request(f"{API}{path}?{data}")
162 request.add_header("Authorization", f"Bearer {self._token()}")
163 with self._open(request) as response:
164 return json.loads(response.read().decode())
165
166 def _kept(self):
167 """The refresh token kept from the last run, or ``None``."""
168 try:
169 with open(self.token_file, encoding="utf-8") as f:
170 return json.load(f).get("refresh_token")
171 except (OSError, ValueError):
172 return None
173
174 def _keep(self, refresh_token):
175 """Write the new refresh token: owner-only from its creation, replaced in one move."""
176 os.makedirs(os.path.dirname(self.token_file), mode=0o700, exist_ok=True)
177 draft = self.token_file + ".new"
178 with os.fdopen(os.open(draft, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600), "w", encoding="utf-8") as f:
179 json.dump({"refresh_token": refresh_token}, f)
180 os.replace(draft, self.token_file)
181
182 def _token(self):
183 if self._access and _time.time() < self._expires - 60:
184 return self._access
185 secret = lambda key: get_credential(f"netatmo_{key}", env=f"NETATMO_{key.upper()}") # fail-closed
186 client_id, client_secret = secret("client_id"), secret("client_secret")
187 kept = self._kept()
188 given = get_credential("netatmo_refresh_token", env="NETATMO_REFRESH_TOKEN", required=kept is None)
189 answer = None
190 for refresh_token in dict.fromkeys(t for t in (kept, given) if t): # the kept one first
191 try:
192 answer = self._call("/oauth2/token", {"grant_type": "refresh_token", "refresh_token": refresh_token,
193 "client_id": client_id, "client_secret": client_secret},
194 post=True)
195 break
196 except urllib.error.HTTPError as refused:
197 last = refused
198 if answer is None:
199 raise PermissionError("NetatmoCloud: Netatmo refused the refresh token; make a new one on dev.netatmo.com "
200 f"and set NETATMO_REFRESH_TOKEN ({last}).") from last
201 self._access, self._expires = answer["access_token"], _time.time() + float(answer.get("expires_in", 10800))
202 if answer.get("refresh_token"):
203 self._keep(answer["refresh_token"])
204 return self._access

206 def readings(self, station, module, start, end):
207 """``(times, ppm)`` of one module between ``start`` and ``end`` [unix s]."""
208 def produce():
209 times, ppm, begin = [], [], int(start)
210 while begin < int(end):
211 params = {"device_id": station, "scale": self.scale, "type": "co2", "date_begin": begin,
212 "date_end": int(end), "limit": self.PAGEPAGE, "optimize": "false", "real_time": "true"}
213 if module:
214 params["module_id"] = module
215 body = self._call("/api/getmeasure", params).get("body") or {}
216 page = sorted((int(t), v[0]) for t, v in body.items() if v and v[0] is not None)
217 if not page:
218 break
219 times += [t for t, _ in page]
220 ppm += [v for _, v in page]
221 if len(page) < self.PAGEPAGE:
222 break
223 begin = page[-1][0] + 1
224 return np.asarray(times, dtype=float), np.asarray(ppm, dtype=float)
225 return self._cached((station, module, int(start), int(end), self.scale), produce)
226
227 def fetch(self, station, module=None, start=None, end=None):
228 times, ppm = self.readings(station, module, start, end)
229 return {"time": times, "zk": ppm.reshape(-1, 1), "meta": {"units": "ppm", "scale": self.scale}}

232class NetatmoCSV(KalmanProvider):
233 """The CO2 of Netatmo modules, from files: one per module.
234
235 ``files`` maps a module id (the station's id for the main module) to a file.
236 A file is the export of the Netatmo web page (``;``-separated, a few lines of
237 title, then a header with ``Timestamp`` and ``CO2``), or any table whose
238 header names a time column (``Timestamp``, ``time``: unix seconds) and a
239 ``CO2`` column.
240 """
241
242 NAME = "Netatmo CSV"
243 DESCRIPTION = "CO2 readings of Netatmo indoor modules, from exported or logged files."
244 VERSION = "1.0"
245 KIND = KalmanProviderKind.FILE
246 MEASUREMENTS = NetatmoCloud.MEASUREMENTS
247
248 def __init__(self, files, *, cache=True):
249 super().__init__(cache=cache)
250 self.files = dict(files)

252 @staticmethod
253 def parse(path):
254 """``(times, ppm)`` from one file."""
255 with open(path, encoding="utf-8-sig", newline="") as f:
256 lines = f.read().splitlines()
257 for i, line in enumerate(lines):
258 lower = line.lower()
259 if "co2" in lower and ("timestamp" in lower or "time" in lower):
260 break
261 else:
262 raise ValueError(f"NetatmoCSV: {path} has no header with a time and a CO2 column.")
263 delimiter = ";" if lines[i].count(";") >= lines[i].count(",") else ","
264 rows = list(csv.reader(lines[i:], delimiter=delimiter))
265 header = [h.strip().strip('"').lower() for h in rows[0]]
266 when = next((j for j, h in enumerate(header) if h in ("timestamp", "time", "unix", "t")), None)
267 what = next((j for j, h in enumerate(header) if h.startswith("co2")), None)
268 if when is None or what is None:
269 raise ValueError(f"NetatmoCSV: {path} needs a time column (Timestamp or time, unix seconds) and a CO2 "
270 f"column; its header is {rows[0]}.")
271 times, ppm = [], []
272 for row in rows[1:]:
273 try:
274 moment, value = float(row[when]), float(row[what])
275 except (ValueError, IndexError): # a missing reading: the row is skipped whole
276 continue
277 times.append(moment)
278 ppm.append(value)
279 return np.asarray(times), np.asarray(ppm)
280
281 def readings(self, station, module, start, end):
282 key = module or station
283 if key not in self.files:
284 raise KeyError(f"NetatmoCSV: no file for module {key!r}; it has {sorted(self.files)}.")
285 times, ppm = self._cached(key, lambda: self.parse(self.files[key]))
286 keep = (times >= float(start)) & (times < float(end))
287 return times[keep], ppm[keep]

289 def fetch(self, station, module=None, start=None, end=None):
290 times, ppm = self.readings(station, module, start, end)
291 return {"time": times, "zk": ppm.reshape(-1, 1), "meta": {"units": "ppm"}}
