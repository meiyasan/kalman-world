11 """Load a measurement set (and optional truth) from a file.
12
13 - ``.npz`` — keys ``zk`` (required) and optionally ``xk``/``time``;
14 - ``.npy`` — a single ``(T, m)`` measurement array;
15 - ``.json`` — ``{"zk": [...], "xk": [...]?, "time": [...]?}``;
16 - ``.csv`` — a numeric table taken as ``zk``.
17 """

41 def save(path, zk, xk=None, time=None):
42 """Write a measurement set (and optional truth) to ``.npz``/``.json``."""
43 zk = np.asarray(zk, dtype=float)
44 payload = {"zk": zk}
45 if xk is not None:
46 payload["xk"] = np.asarray(xk, dtype=float)
47 if time is not None:
48 payload["time"] = np.asarray(time, dtype=float)
49 if str(path).endswith(".json"):
50 with open(path, "w") as fh:
51 json.dump({k: v.tolist() for k, v in payload.items()}, fh)
52 else:
53 np.savez(path, **payload)
54 return path

61 def _load():
62 if p.endswith(".npz"):
63 z = np.load(p)
64 out = {k: np.asarray(z[k], dtype=float) for k in z.files}
65 elif p.endswith(".npy"):
66 out = {"zk": np.asarray(np.load(p), dtype=float)}
67 elif p.endswith(".json"):
68 with open(p) as fh:
69 d = json.load(fh)
70 out = {k: np.asarray(v, dtype=float) for k, v in d.items()}
71 elif p.endswith(".csv"):
72 out = {"zk": np.atleast_2d(np.loadtxt(p, delimiter=","))}
73 else:
74 raise ValueError(f"unsupported file type: {p}")
75 out.setdefault("truth", out.get("xk"))
76 out["meta"] = {"provider": "file", "path": p}
