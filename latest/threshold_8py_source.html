7class ThresholdRouter(KalmanRouter):
8 """Route based on how close each model's smoothed NIS is to the χ²(m) mean.
9
10 Under the correct model NIS_j ~ χ²(m) with E[NIS_j] = m.
11 This router assigns credit to the model whose recent NIS is closest to m.
12
13 Hard mode (``soft=False``, default):
14 Full weight on the single best-matching model.
15
16 Soft mode (``soft=True``):
17 Inverse-distance weighting over all models.
18
19 No training required — ``fit()`` is a no-op.
20
21 Parameters
22 ----------
23 m : int
24 Measurement dimension (expected NIS under the correct model).
25 window : int
26 Number of past NIS values to average before comparing to m.
27 soft : bool
28 Hard argmin (False) vs inverse-distance softmax (True).
29
30 Decided by ``Kalman::Routers::Threshold`` when libKalman is installed;
31 :meth:`_reference_route` is the Python reference.
32 """

57 def _reference_route(self, nis_arr, c, likelihoods):
58 if len(nis_arr) == 0: # nothing observed yet
59 return np.full(len(c), 1.0 / len(c))
60 block = nis_arr[-self.window:]
