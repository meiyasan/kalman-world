29def jones_signal(a, psi, theta, phi):
30 """Jones bivariate complex signal: [a·exp(iψ)·cos(θ/2), a·exp(iψ)·sin(θ/2)·exp(iφ)]"""
31 carrier = a * np.exp(1j * psi)
32 return np.array([carrier * np.cos(theta / 2),
33 carrier * np.sin(theta / 2) * np.exp(1j * phi)])

38class PolarizationModel(NativeTwin, ExtendedKalmanModel):
39 """
40 EKF for online estimation of gravitational-wave polarization parameters.
41
42 Implements the Amblard & Michel (Jones formalism) model for tracking
43 a bivariate GW signal in the presence of Gaussian observation noise.
44
45 State (8-D): x = [log(a), ȧ/a, u_ψ, v_ψ, u_θ, v_θ, u_φ, v_φ]
46 Observation (4-D real): z = [Re(z1), Im(z1), Re(z2), Im(z2)]
47
48 Why log-amplitude
49 -----------------
50 The observation z = a·exp(iψ)·Jones is invariant under (a,ψ)→(-a,ψ+π).
51 Tracking ``a`` directly allows it to flip negative, causing the EKF to
52 lose phase coherence. Using ``log(a)`` constrains a = exp(x[0]) > 0
53 always and removes the sign ambiguity.
54
55 Why ``delta`` ≠ ``dt`` for Q
56 ----------------------------
57 ``J`` (state transition) uses ``dt = 1/fs`` (correct physics at sample
58 rate). ``Q`` uses ``delta`` (seconds) as a bandwidth-tuning knob::
59
60 Q[vel, vel] = sigma_p² × delta (larger delta → wider bandwidth)
61
62 Required tracking condition: ``ωn = sqrt(sigma_psi² × delta / sigma_n²)``
63 must satisfy ``ωn² > d²ψ/dt² / max_phase_lag``.
64
65 Why ``L()`` returns ``None``
66 ---------------------------
67 The default base-class ``L()`` numerically computes ∂f/∂w. Since
68 ``f()`` ignores ``w`` (additive noise model), L = 0 → Q = L@Q@L^T = 0,
69 silently zeroing all process noise. Returning ``None`` tells the
70 framework to use Q directly.
71
72 Parameters
73 ----------
74 sensor_id : str
75 Unique identifier for the sensor.
76 fs : float
77 Sampling frequency [Hz]. Default: 100.
78 delta : float
79 Bandwidth tuning parameter [s]. Default: 0.5.
80 sigma_a : float
81 Log-amplitude process noise std. Default: 0.1.
82 sigma_psi : float
83 Phase/frequency process noise std. Default: 0.5.
84 sigma_theta : float
85 θ-polarization process noise std. Default: 0.1.
86 sigma_phi : float
87 φ-polarization process noise std. Default: 0.1.
88 sigma_n : float
89 Observation noise std. Default: 1.0.
90 **kwargs
91 Passed through to ``ExtendedKalmanModel.__init__``
92 (e.g. ``P0``, ``x0``).
93 """
94

127 # ── State transition: constant-velocity at sample rate dt ─
128 J = np.array([[1.0, dt], [0.0, 1.0]])

130 for i in range(4):

134 # ── Process noise: bandwidth via delta (NOT dt) ───────────
135 def _Qblock(sig, d):
136 return (sig ** 2) * np.array([[d**3/3, d**2/2],
137 [d**2/2, d ]])

139 Q = np.zeros((8, 8))
140 for i, sig in enumerate([sigma_a, sigma_psi, sigma_theta, sigma_phi]):
141 Q[2*i:2*i+2, 2*i:2*i+2] = _Qblock(sig, delta)

143 R = (sigma_n ** 2) * np.eye(4)

173 def h(self, x, v=None, t=None):
174 x = np.asarray(x).ravel()
175 a = np.exp(x[0])
176 psi, theta, phi = x[2], x[4], x[6]
177 z_c = jones_signal(a, psi, theta, phi)
178 z = np.array([z_c[0].real, z_c[0].imag,
179 z_c[1].real, z_c[1].imag]).reshape(4, 1)
180 if v is not None:
181 z += np.asarray(v).reshape(4, 1)
182 return z

184 def H(self, x=None, t=None):
185 """Analytic Jacobian ∂h/∂x (4×8)."""
186 if x is None:
187 x = self.x0x0x0
188 x = np.asarray(x).ravel()
189 a = np.exp(x[0])
190 psi, theta, phi = x[2], x[4], x[6]
191 c2 = np.cos(theta / 2)
192 s2 = np.sin(theta / 2)
193 eip = np.exp(1j * psi)
194 eif = np.exp(1j * phi)
195
196 dz1_dloga = a * eip * c2
197 dz2_dloga = a * eip * s2 * eif
198
199 dz1_dpsi = 1j * a * eip * c2
200 dz2_dpsi = 1j * a * eip * s2 * eif
201
202 dz1_dth = -0.5 * a * eip * s2
203 dz2_dth = 0.5 * a * eip * c2 * eif
204
205 dz1_dphi = 0 + 0j
206 dz2_dphi = 1j * a * eip * s2 * eif
207
208 Jac = np.zeros((4, 8))
209
210 def fill(r_re, r_im, dz, col):
211 Jac[r_re, col] = dz.real
212 Jac[r_im, col] = dz.imag
213
214 fill(0, 1, dz1_dloga, 0); fill(2, 3, dz2_dloga, 0)
215 fill(0, 1, dz1_dpsi, 2); fill(2, 3, dz2_dpsi, 2)
216 fill(0, 1, dz1_dth, 4); fill(2, 3, dz2_dth, 4)
217 fill(0, 1, dz1_dphi, 6); fill(2, 3, dz2_dphi, 6)
218 return Jac
