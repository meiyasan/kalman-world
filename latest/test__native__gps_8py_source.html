32def apart(a, b):
33 """Largest difference of two records: on the state in metres, on the covariance relative."""
34 xa, xb = np.asarray(a.xk, float), np.asarray(b.xk, float)
35 Pa, Pb = np.asarray(a.Pk, float), np.asarray(b.Pk, float)
36 return float(np.nanmax(np.abs(xa - xb))), float(np.nanmax(np.abs(Pa - Pb)) / np.nanmax(np.abs(Pa)))
