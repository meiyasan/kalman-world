33def both(make, z, **kw):
34 """(record with the Python physics, record on the C++ twin)."""
35 with native.python_reference():
36 python = make().filter(z, **kw)
37 model = make()
38 assert model.native_handle() is not None, "the twin was expected to be used"
39 return python, model.filter(z, **kw)
40
41
42# How close "the same answer" is. One call of f agrees to the last digit or two
43# (numpy and the C math library round a cosine differently), and both sides then
44# difference f with a step of 1e-6, which turns that last digit into ~1e-9 of
45# noise in each Jacobian. Over a few hundred steps that is ~1e-8 on the state and
46# ~1e-7 on the covariance, relative to the largest entry.
47STATE_RTOL = 1e-7
48COVARIANCE_RTOL = 1e-6
49

51def close(test, a, b, rtol, what):
52 a, b = np.asarray(a, float), np.asarray(b, float)
53 scale = max(1.0, float(np.nanmax(np.abs(a))))
54 test.assertLess(float(np.nanmax(np.abs(a - b))) / scale, rtol, what)

57def trajectory(n=400, fs=100.0, seed=0, drop=17):
58 rng = np.random.default_rng(seed)
59 t = np.arange(n) / fs
60 z = np.c_[30 * t, 10 * t, 120 * t - 0.5 * 9.81 * t**2] + rng.normal(0, 0.5, (n, 3))
61 if drop:
62 z[::drop] = np.nan
63 return z

322def steering(n, fs=50.0):
323 """A lane-change steering program, past the actuator limit for half a second."""
324 t = np.arange(n) / fs
325 u = 0.01 * np.sin(0.8 * t) + 0.15 * ((t > 3.0) & (t < 3.5)) - 0.15 * ((t > 3.5) & (t < 4.0))
326 return u.reshape(-1, 1)

642# --- the gravitational-wave polarization (kalman.apps.gravitational.gravwave) ------
643def polarization_run(fs=100.0, T=40.0, seed=0):
644 """The notebook's synthetic chirp: amplitude, carrier 1 -> 2 Hz and the two
645 polarization angles all moving, sigma_n = 0.3 on each real component."""
646 from modules.apps.gravitational.gravwave import jones_signal
647 t = np.arange(int(T * fs)) / fs
648 rng = np.random.default_rng(seed)
649 a = 1.0 + 0.1 * np.cos(2 * np.pi * 0.2 * t)
650 psi = 2 * np.pi * (t + 0.5 * (1.0 / T) * t ** 2)
651 theta = (np.pi / 2) * (1.0 + 0.5 * np.sin(2 * np.pi * 0.05 * t))
652 phi = np.pi * (1.0 + 0.5 * np.sin(2 * np.pi * 0.10 * t + 1.0))
653 z = np.array([jones_signal(a[k], psi[k], theta[k], phi[k]) for k in range(len(t))])
654 z = z + 0.3 * (rng.standard_normal(z.shape) + 1j * rng.standard_normal(z.shape))
655 zk = np.column_stack([z[:, 0].real, z[:, 0].imag, z[:, 1].real, z[:, 1].imag])
656
657 def make(**kw):
658 from modules.apps.gravitational.gravwave import PolarizationModel
659 model = PolarizationModel("gw", **{"fs": fs, "delta": 0.5, "sigma_a": 0.05, "sigma_psi": 0.6,
660 "sigma_theta": 0.05, "sigma_phi": 0.05, "sigma_n": 0.3, **kw})
661 model.initial_state([np.log(a[0]), 0.0, psi[0] - 2 * np.pi / fs, 2 * np.pi, theta[0], 0.0, phi[0], 0.0])
662 return model
663 return make, zk
