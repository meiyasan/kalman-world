44 def build(self, d_in: int, d_out: int, seed=None) -> "CNN":
45 if seed is not None:
46 torch.manual_seed(seed)
47 M = d_out
48 W = d_in // M
49 k = min(self.kernel_size, W)
50 self._M, self._W = M, W
51 h = self.out_channels
52 self._net = nn.Sequential(
53 nn.Conv1d(M, h, kernel_size=k, padding=k // 2),
54 nn.Tanh(),
55 nn.AdaptiveAvgPool1d(1), # → (batch, h, 1)
56 nn.Flatten(), # → (batch, h)
57 nn.Linear(h, d_out),
58 ).to(_DEVICE)
59 self._opt = torch.optim.Adam(self._net.parameters())
60 self.loss_history = []
61 return self

63 def _reshape(self, X: np.ndarray) -> np.ndarray:
64 """(N, M*W) → (N, M, W)."""
65 return X.reshape(-1, self._M, self._W)

67 def fit(self, X: np.ndarray, y: np.ndarray,
68 epochs: int = 300, lr: float = 3e-3,
69 batch_size: int = 64, **_) -> "CNN":
70 self._opt.param_groups[0]["lr"] = lr
71 Xt = torch.tensor(self._reshape(X), dtype=torch.float32, device=_DEVICE)
72 yt = torch.tensor(y, dtype=torch.long, device=_DEVICE)
73 loss_fn = nn.CrossEntropyLoss()
74 N = len(Xt)
75 for _ in range(epochs):
76 idx = torch.randperm(N, device=_DEVICE)
77 epoch_loss, n_batches = 0.0, 0
78 for s in range(0, N, batch_size):
79 b = idx[s:s + batch_size]
80 self._opt.zero_grad()
81 loss = loss_fn(self._net(Xt[b]), yt[b])
82 loss.backward()
83 self._opt.step()
84 epoch_loss += float(loss.item()); n_batches += 1
85 self.loss_history.append(epoch_loss / max(n_batches, 1))
86 return self

88 def __call__(self, x: np.ndarray) -> np.ndarray:
89 with torch.no_grad():
90 xt = torch.tensor(self._reshape(x[None]), dtype=torch.float32, device=_DEVICE)
91 logits = self._net(xt).squeeze(0).cpu().numpy()
92 e = np.exp(logits - logits.max())
93 return e / e.sum()
