1"""Reduced storm models — typhoon track/intensity and tornado-environment regime.
2
3Two low-dimensional models built to sit *on top of* the field-scale EnKF
4(``kalman.apps.weather.flow``): the EnKF estimates the environment, these estimate the
5local phenomenon and — crucially — its **covariance**.
6
7- :class:`StormModel` — a reduced tropical-cyclone tracker (typhoon, hurricane
8 and cyclone are regional names for one storm, so one model covers all). State
9 ``[lon, lat, v_e, v_n, I]`` (centre, translation velocity, intensity); the
10 dynamics are the textbook **steering flow + beta drift + coordinated turn**,
11 so the motion has genuine *regimes* (straight-running vs. recurving). That
12 regime structure is exactly what an :class:`~kalman.framework.filters.InteractiveKalmanFilter`
13 (IKF) is for — recurvature is the "manoeuvre". Use
14 :meth:`StormModel.ikf_bank` to get the ready-made model bank + transition
15 matrix. The ensemble/covariance gives the track **cone of uncertainty**
16 (:meth:`forecast_cone`) and an intensity error bar for free.
17
18- :class:`TornadoEnvironmentModel` — an honest tornado model is *not* a
19 deterministic tracker (a tornado is sub-grid and minutes-predictable); the
20 actionable quantity is the **probability the environment is in the tornadic
21 regime**. This is a mean-reverting significant-tornado-parameter latent with
22 a benign and a tornadic regime; IKF returns the regime probability, and the
23 **seasonal climatology enters as the IKF transition matrix** (``season=``).
24
25Neither needs training data; both deliver calibrated covariance/probability
26from physics + the filter, which is the point (see the notebook 13 discussion
27of EnKF-vs-ML).
28"""

60class StormModel(NativeTwin, ExtendedKalmanModel):
61 """
62 Reduced tropical-cyclone track + intensity model (steering + beta + turn).
63
64 A typhoon, a hurricane and a cyclone are the same phenomenon under
65 different regional names, so this one model covers all of them; the only
66 thing that differs between basins is the ``hemisphere``, which sets the
67 beta-drift and recurvature directions.
68
69 State ``x = [lon, lat, v_e, v_n, I]``. Over one step ``dt`` (hours):
70
71 .. code-block:: text
72
73 (v_e, v_n) rotate at turn rate ω (coordinated turn: ω=0 straight)
74 centre advected by (v_e+β_e, v_n+β_n) (steering velocity + beta drift)
75 lon += displacement_east / (111.32 cos lat), lat += disp_north / 110.574
76 I persists (random walk; intensity change is Q-driven — RI is hard)
77
78 **Beta drift** ``β`` is the quasi-steady poleward-westward self-advection
79 from the planetary-vorticity gradient (~1–2 m/s toward WNW in the NH); its
80 sign flips by ``hemisphere``. The **turn rate** ``ω`` is what distinguishes
81 the motion regimes an IKF switches between (``ω=0`` straight vs.
82 ``ω=ω_recurve`` recurving) — see :meth:`ikf_bank`.
83
84 Observation: a centre fix ``[lon, lat]`` (best-track / satellite), plus
85 intensity ``I`` when ``observe_intensity`` (default True). ``n=5``.
86
87 Parameters
88 ----------
89 omega : float
90 Turn rate [rad/h]; 0 = straight-running. ``ikf_bank`` sets this per
91 regime. Sign convention: negative = clockwise in the (east, north)
92 plane = NH recurvature (poleward then eastward).
93 beta_drift : float
94 Beta-drift speed [km/h] (default ~5 km/h ≈ 1.4 m/s).
95 hemisphere : {"N", "S"}
96 Sets the beta-drift direction (WNW in NH, WSW in SH).
97 fix_noise : float
98 Centre-fix std [deg] (default 0.3° ≈ 33 km, a realistic satellite fix).
99 intensity_noise : float
100 Intensity measurement std [kt] (default 10).
101 """

114 def __init__(self, sensor_id="TC", *,
115 omega=0.0,
116 beta_drift=5.0,
117 hemisphere="N",
118 dt_hours=6.0,
119 observe_intensity=True,
120 fix_noise=0.3,
121 intensity_noise=10.0,
122 q_pos=0.25, q_vel=6.0, q_intensity=8.0,
123 steering=None, intensity_fn=None,
124 x0=None, **kw):
125 self.dtypedtype = np.float64
126 self.omega = float(omega)
127 self.beta_drift = float(beta_drift)
128 self.hemisphere = str(hemisphere).upper()
129 if self.hemisphere not in ("N", "S"):
130 raise ValueError(f"hemisphere must be 'N' or 'S', not {hemisphere!r}.")
131 self._dt = float(dt_hours)
132 self.observe_intensity = bool(observe_intensity)
133 # Optional physics couplings (both None → the pure kinematic model):
134 # steering(lon, lat, t) -> (v_e, v_n) km/h — environmental deep-layer
135 # flow the vortex is advected by (from the field EnKF, or analytic).
136 # When set, recurvature EMERGES from the flow, so ω is unused.
137 # intensity_fn(lon, lat, I, t) -> dI/dt kt/h — intensity tendency
138 # (physics leaves this None = persistence; the ML elaboration is a
139 # LEARNED intensity_fn, mirroring operational statistical intensity).
140 self.steering = steering
141 self.intensity_fn = intensity_fn
142 self.nnnnn = 5
143 self.mmmm = 3 if self.observe_intensity else 2
144 self.lll = 0
145
146 # Beta drift vector [km/h]: WNW in NH (poleward+westward), WSW in SH.
147 pole = 1.0 if self.hemisphere == "N" else -1.0
148 ang = np.deg2rad(20.0) # ~20° off due-west, toward the pole
149 self.beta_e = -self.beta_drift * np.cos(ang)
150 self.beta_n = pole * self.beta_drift * np.sin(ang)
151
152 Q = np.diag([q_pos**2, q_pos**2, q_vel**2, q_vel**2, q_intensity**2]).astype(float)
153 if self.observe_intensity:
154 R = np.diag([fix_noise**2, fix_noise**2, intensity_noise**2]).astype(float)
155 else:
156 R = np.diag([fix_noise**2, fix_noise**2]).astype(float)
157
158 if x0 is None:
159 x0 = np.array([[135.0], [15.0], [-20.0], [6.0], [80.0]])
160 super().__init__(sensor_id, Q=Q, R=R, L=np.eye(self.nnnnn), fs=1.0 / self._dt,
161 x0=x0, **kw)

186 def f(self, x, u=None, w=None, t=None):
187 x = np.asarray(x, float).reshape(-1)
188 lon, lat, ve, vn, I = x[0], x[1], x[2], x[3], x[4]
189 dt, w_ = self._dt, self.omega
190
191 if self.steering is not None:
192 # Beta-advection model: translation = environmental steering + beta.
193 # Recurvature is not a prescribed turn rate — it emerges when the
194 # storm drifts poleward into a steering flow that reverses. This is
195 # the physically realistic predictor (no ML, no fitted turn rate).
196 se, sn = self.steering(lon, lat, t)
197 vte, vtn = se + self.beta_e, sn + self.beta_n
198 de, dn = vte * dt, vtn * dt
199 ve_n, vn_n = se, sn # velocity state = diagnosed steering
200 elif abs(w_) < 1e-9:
201 vte, vtn = ve + self.beta_e, vn + self.beta_n # advecting velocity (+ beta)
202 de, dn = vte * dt, vtn * dt
203 ve_n, vn_n = ve, vn
204 else:
205 vte, vtn = ve + self.beta_e, vn + self.beta_n
206 s, c = np.sin(w_ * dt), np.cos(w_ * dt)
207 de = (vte * s - vtn * (1 - c)) / w_ # curved displacement [km]
208 dn = (vte * (1 - c) + vtn * s) / w_
209 ve_n = ve * c - vn * s # rotate steering velocity
210 vn_n = ve * s + vn * c
211 lat_n = lat + dn / KM_PER_DEG_LAT
212 lon_n = lon + de / (KM_PER_DEG_LON * np.cos(np.deg2rad(lat)))
213 I_n = I if self.intensity_fn is None else I + self.intensity_fn(lon, lat, I, t) * dt
214 return np.array([[lon_n], [lat_n], [ve_n], [vn_n], [I_n]])

216 def h(self, x, v=None, t=None):
217 x = np.asarray(x, float).reshape(-1)
218 z = np.array([[x[0]], [x[1]], [x[4]]]) if self.observe_intensity else np.array([[x[0]], [x[1]]])
219 return z if v is None else z + np.asarray(v, float).reshape(-1, 1)

222 # IKF bank: straight-running vs recurving (vs stalling) motion regimes

224 @classmethod
225 def ikf_bank(cls, sensor_id="TC", *, omega_recurve=0.022, stall=True,
226 stay=0.90, hemisphere="N", **kw):
227 """Build the IKF model bank + Markov transition matrix.
228
229 Returns ``(models, transition, initial_probs)`` ready for
230 :class:`~kalman.framework.filters.InteractiveKalmanFilter`. Regimes:
231
232 - **straight** — ``ω=0`` (trade-wind steering, WNW motion);
233 - **recurve** — ``ω`` set for the hemisphere (poleward-then-eastward);
234 - **stall** — ``ω=0`` with inflated velocity process noise (erratic /
235 slow-moving), included when ``stall=True``.
236
237 ``stay`` is the self-transition probability (regime persistence);
238 off-diagonal mass is split among the other regimes. Feed a season
239 prior by adjusting ``stay`` or passing your own transition later.
240 """
241 hemi = str(hemisphere).upper()
242 pole_sign = -1.0 if hemi == "N" else 1.0
243 w_rec = pole_sign * abs(omega_recurve)
244 models = [
245 cls(sensor_id, omega=0.0, hemisphere=hemi, **kw),
246 cls(sensor_id, omega=w_rec, hemisphere=hemi, **kw),
247 ]
248 if stall:
249 models.append(cls(sensor_id, omega=0.0, hemisphere=hemi,
250 q_vel=18.0, **kw))
251 M = len(models)
252 off = (1.0 - stay) / (M - 1)
253 Pi = np.full((M, M), off) + (stay - off) * np.eye(M)
254 probs = np.full(M, 1.0 / M)
255 return models, Pi, probs

257 imm_bank = ikf_bank # the literature's name (Interacting Multiple Model)

262 def forecast_cone(self, x0, P0, steps, *, k_sigma=1.0):
263 """Free-run the model ``steps`` cycles from ``(x0, P0)`` and return the
264 track **cone of uncertainty**.
265
266 Pure prediction (``engine._predict`` only, no observations), so the
267 position covariance grows exactly as the model says it should. Returns
268 a dict with ``lon``/``lat`` (mean track), ``I``/``I_sigma`` (intensity ±σ),
269 ``sigma_km`` (standard deviation of the position error distance [km]) and
270 ``ellipses`` — per-step ``(cx, cy, a, b, angle_deg)`` at ``k_sigma``
271 for the lon/lat error ellipse.
272 """
273 from ...framework.engines.extended import ExtendedKalmanEngine
274 eng = ExtendedKalmanEngine()
275 x = np.asarray(x0, float).reshape(-1, 1)
276 P = np.asarray(P0, float)
277 lon, lat, Ivals, Isig, ell, sig_km = [], [], [], [], [], []
278 t = 0.0
279 for _ in range(int(steps)):
280 x, P, _ = eng._predict(self, x, P, t=t)
281 t += self._dt
282 xr = x.ravel()
283 lon.append(xr[0]); lat.append(xr[1])
284 Ivals.append(xr[4]); Isig.append(float(np.sqrt(max(P[4, 4], 0.0))))
285 ell.append(self._ellipse(xr[0], xr[1], P[:2, :2], k_sigma))
286 km_lon = KM_PER_DEG_LON * np.cos(np.deg2rad(xr[1]))
287 sig_km.append(float(np.sqrt(max(P[0, 0], 0.0) * km_lon**2
288 + max(P[1, 1], 0.0) * KM_PER_DEG_LAT**2)))
289 return {"lon": np.array(lon), "lat": np.array(lat),
290 "I": np.array(Ivals), "I_sigma": np.array(Isig),
291 "sigma_km": np.array(sig_km), "ellipses": ell}

293 def ensemble_forecast(self, x0, P0, steps, *, n_samples=300, seed=0):
294 """Monte-Carlo forecast: draw ``n_samples`` initial states from
295 ``N(x0, P0)`` and propagate each through the (nonlinear) dynamics with
296 per-step process-noise draws — the honest **spaghetti ensemble** the
297 Gaussian :meth:`forecast_cone` summarises. Returns ``(n_samples, steps,
298 5)``. Feed to :func:`landfall_probability` for a strike probability."""
299 rng = np.random.default_rng(seed)
300 x0 = np.asarray(x0, float).reshape(-1)
301 P0 = np.asarray(P0, float)
302 Qc = np.linalg.cholesky(self.QQQ() + 1e-12 * np.eye(self.nnnnn))
303 X = rng.multivariate_normal(x0, 0.5 * (P0 + P0.T) + 1e-9 * np.eye(self.nnnnn), size=n_samples)
304 out = np.empty((n_samples, int(steps), self.nnnnn))
305 for i in range(n_samples):
306 x = X[i].reshape(-1, 1)
307 for k in range(int(steps)):
308 x = np.asarray(self.ffff(x), float) + (Qc @ rng.standard_normal((self.nnnnn, 1)))
309 out[i, k] = x.ravel()
310 return out

312 @staticmethod
313 def _ellipse(cx, cy, P2, k_sigma):
314 vals, vecs = np.linalg.eigh(0.5 * (P2 + P2.T))
315 vals = np.maximum(vals, 0.0)
316 order = np.argsort(vals)[::-1]
317 a, b = k_sigma * np.sqrt(vals[order])
318 ang = np.degrees(np.arctan2(vecs[1, order[0]], vecs[0, order[0]]))
319 return (float(cx), float(cy), float(a), float(b), float(ang))

462def synthetic_track(n_steps=40, *, recurve_at=18, dt_hours=6.0, hemisphere="N",
463 speed=22.0, omega_recurve=0.022, fix_noise=0.3,
464 intensity_noise=10.0, peak_intensity=120.0, seed=0):
465 """Generate a synthetic best-track: straight WNW run, then recurvature, with
466 a rise-and-decay intensity — plus noisy centre/intensity fixes.
467
468 Returns ``(truth (T, 5), fixes (T, 3))`` — feed ``truth`` as the OSSE truth
469 and ``fixes`` as observations to an IKF tracker. Exercises the regime
470 switch at ``recurve_at``.
471 """
472 rng = np.random.default_rng(seed)
473 pole = 1.0 if str(hemisphere).upper() == "N" else -1.0
474 lon, lat = 138.0, 12.0
475 ang = np.deg2rad(20.0)
476 ve, vn = -speed * np.cos(ang), pole * speed * np.sin(ang)
477 w = pole * -abs(omega_recurve)
478 truth, fixes = [], []
479 for k in range(n_steps):
480 frac = k / max(1, n_steps - 1)
481 I = peak_intensity * np.sin(np.pi * min(frac / 0.7, 1.0)) + 40.0
482 truth.append([lon, lat, ve, vn, I])
483 fixes.append([lon + rng.normal(0, fix_noise),
484 lat + rng.normal(0, fix_noise),
485 I + rng.normal(0, intensity_noise)])
486 turn = w if k >= recurve_at else 0.0
487 if abs(turn) < 1e-9:
488 de, dn = ve * dt_hours, vn * dt_hours
489 else:
490 s, c = np.sin(turn*dt_hours), np.cos(turn*dt_hours)
491 de = (ve*s - vn*(1-c))/turn; dn = (ve*(1-c) + vn*s)/turn
492 ve, vn = ve*c - vn*s, ve*s + vn*c
493 lat += dn / KM_PER_DEG_LAT
494 lon += de / (KM_PER_DEG_LON * np.cos(np.deg2rad(lat)))
495 return np.array(truth), np.array(fixes)

499# Realistic physics coupling: environmental steering + statistical intensity
500# ======================================================================
501def analytic_steering(lon, lat, t=None, *, ridge_lat=27.0, easterly=-26.0,
502 westerly=32.0, poleward=3.0, poleward_mid=12.0,
503 hemisphere="N", width=5.0):
504 """Idealised deep-layer steering flow [km/h] for a beta-advection tracker.
505
506 Trade-wind **easterlies** equatorward of the subtropical ridge
507 (``ridge_lat``) smoothly giving way to mid-latitude **westerlies** poleward
508 of it (``tanh`` transition of scale ``width``). The westerlies steer
509 **poleward-eastward** (``poleward_mid``), so a recurving storm accelerates
510 toward higher latitudes — the NE sweep that carries Northwest-Pacific
511 typhoons toward Japan. A storm born in the tropics runs west, drifts
512 poleward (beta), crosses the ridge into the westerlies and **recurves** —
513 recurvature as physics, not a fitted turn rate. In the real system this
514 callable is replaced by the deep-layer mean flow diagnosed from the field
515 EnKF (``kalman.apps.weather.flow``); the signature is identical.
516 """
517 pole = 1.0 if str(hemisphere).upper() == "N" else -1.0
518 frac = 0.5 * (1.0 + np.tanh(pole * (lat - ridge_lat) / width)) # 0 tropics → 1 mid-lat
519 ve = (1 - frac) * easterly + frac * westerly
520 vn = pole * (poleward * (1 - frac) + poleward_mid * frac) # NE sweep in the westerlies
521 return float(ve), float(vn)

524def analytic_environment(lon, lat, t=None):
525 """Idealised intensity environment at a point → ``(sst_warmth, shear)``.
526
527 ``sst_warmth`` is a 2-D **warm pool** (peaks near 135°E, 10°N and falls off
528 in both lon and lat), ``shear`` grows toward mid-latitudes. The two vary
529 along different directions, so they are *not* collinear along a track — a
530 learned intensity model can separate their effects, as a real one must.
531 Stand-in for the satellite SST / analysis wind-shear predictors that
532 operational statistical intensity schemes consume.
533 """
534 sst_warmth = float(30.0 * np.exp(-((lon - 135.0) / 45.0) ** 2
535 - ((lat - 10.0) / 14.0) ** 2))
536 shear = float(3.0 + 0.9 * max(abs(lat) - 15.0, 0.0))
537 return sst_warmth, shear

540def _true_intensity_tendency(lon, lat, I, t=None, env=analytic_environment):
541 """Ground-truth intensity physics used to synthesise data: relaxation toward
542 a maximum potential intensity set by SST, penalised by shear."""
543 sst_warmth, shear = env(lon, lat, t)
544 mpi = 25.0 + 3.0 * sst_warmth # max potential intensity [kt]
545 # Slow relaxation (~1.5-day e-folding) so intensification is a multi-day
546 # process a persistence forecast visibly fails to capture — realistic RI.
547 return 0.025 * (mpi - I) - 0.30 * shear

550class LearnedIntensityModel:
551 """Data-driven intensity-tendency model — the ML elaboration on the physics.
552
553 Ridge regression of ``dI/dt`` on environmental predictors
554 ``[sst_warmth, shear, I, 1]`` (the toy analogue of the operational **SHIPS**
555 statistical intensity scheme, which regresses on ~20 predictors). The
556 physics tracker leaves intensity as persistence; this *learns* the
557 SST/shear→intensification relationship from past storms and plugs into
558 :class:`StormModel` via :meth:`as_intensity_fn`.
559
560 Deliberately dependency-free (numpy least squares) so it runs anywhere; a
561 torch MLP is the drop-in scale-up when the relationship is nonlinear.
562 """
563
564 def __init__(self, ridge=1e-3, env=analytic_environment):
565 self.ridge = float(ridge)
566 self.env = env
567 self.coef_ = None
568 self._mu = self._sd = None # feature standardisation
569
570 @staticmethod
571 def _features(sst_warmth, shear, I):
572 return np.array([sst_warmth, shear, I])
573
574 def fit(self, tracks, dt_hours=6.0, floor=15.0):
575 """Fit on a list of truth tracks ``(T, 5)`` — builds ``(features, dI/dt)``
576 pairs from consecutive intensities and their environment. Features are
577 **standardised** before the ridge solve (they span very different
578 scales: SST ~0–30, shear ~3–30, intensity ~15–120), so no predictor is
579 implicitly penalised more than another.
580
581 Steps that end at or below ``floor`` [kt] are left out: a dissipated
582 storm is held at that value (see :func:`synthetic_steered_track`), so
583 its intensity change there says nothing about the tendency."""
584 X, y = [], []
585 for tr in tracks:
586 tr = np.asarray(tr, float)
587 for k in range(len(tr) - 1):
588 if tr[k + 1, 4] <= floor:
589 continue
590 lon, lat, I = tr[k, 0], tr[k, 1], tr[k, 4]
591 sst, shear = self.env(lon, lat)
592 X.append(self._features(sst, shear, I))
593 y.append((tr[k + 1, 4] - I) / dt_hours)
594 X, y = np.asarray(X), np.asarray(y)
595 self._mu, self._sd = X.mean(0), X.std(0) + 1e-9
596 Xs = np.column_stack([(X - self._mu) / self._sd, np.ones(len(X))])
597 A = Xs.T @ Xs + self.ridge * np.eye(Xs.shape[1])
598 self.coef_ = np.linalg.solve(A, Xs.T @ y)
599 return self
600
601 def tendency(self, lon, lat, I, t=None):
602 sst, shear = self.env(lon, lat, t)
603 fs = (self._features(sst, shear, I) - self._mu) / self._sd
604 return float(np.append(fs, 1.0) @ self.coef_)
605
606 def as_intensity_fn(self):
607 """Return the ``intensity_fn(lon, lat, I, t)`` closure for StormModel."""
608 if self.coef_ is None:
609 raise RuntimeError("fit() the LearnedIntensityModel first.")
610 return self.tendency
611
612 @staticmethod
613 def _native_arrays(fn):
614 """The arrays of the C++ storm model (``Kalman::Apps::Weather::Storm``) for
615 an intensity tendency ``fn``, or ``None`` when it has no C++ counterpart.
616 Only this class's own ``tendency`` of a fitted model on
617 :func:`analytic_environment` has one: another environment, another
618 function, or a subclass that changes the regression stays in Python."""
619 owner = getattr(fn, "__self__", None)
620 if not isinstance(owner, LearnedIntensityModel) or getattr(fn, "__func__", None) is not LearnedIntensityModel.tendency:
621 return None
622 if owner.env is not analytic_environment or owner.coef_ is None:
623 return None
624 cls = type(owner)
625 if any(getattr(cls, name) is not getattr(LearnedIntensityModel, name) for name in ("tendency", "_features")):
626 return None
627 coef, mu, sd = (np.asarray(a, dtype=float).ravel() for a in (owner.coef_, owner._mu, owner._sd))
628 if (coef.size, mu.size, sd.size) != (4, 3, 3):
629 return None
630 return {"intensity_coef": coef, "intensity_mean": mu, "intensity_scale": sd}

633def synthetic_steered_track(n_steps=44, *, lon0=165.0, lat0=9.0, hemisphere="N",
634 steering=analytic_steering, intensity_env=analytic_environment,
635 eddy=6.0, fix_noise=0.4, intensity_noise=8.0,
636 dt_hours=6.0, I0=35.0, seed=0):
637 """Truth track advected by a **steering field** (recurvature emerges) with
638 environment-driven intensity, plus noisy fixes.
639
640 Unlike :func:`synthetic_track` (which prescribes a turn), motion here is
641 ``steering + beta + eddy`` and intensity follows
642 :func:`_true_intensity_tendency` — so a steering-coupled physics tracker can
643 predict the recurvature, and a learned intensity model has a real
644 SST/shear signal to capture. Returns ``(truth (T,5), fixes (T,3))``.
645 """
646 rng = np.random.default_rng(seed)
647 pole = 1.0 if str(hemisphere).upper() == "N" else -1.0
648 beta = 5.0
649 be, bn = -beta * np.cos(np.deg2rad(20)), pole * beta * np.sin(np.deg2rad(20))
650 lon, lat, I = lon0, lat0, I0
651 truth, fixes = [], []
652 for k in range(n_steps):
653 se, sn = steering(lon, lat, hemisphere=hemisphere) if steering.__name__ == "analytic_steering" else steering(lon, lat)
654 ve, vn = se, sn
655 truth.append([lon, lat, ve, vn, I])
656 fixes.append([lon + rng.normal(0, fix_noise), lat + rng.normal(0, fix_noise),
657 I + rng.normal(0, intensity_noise)])
658 vte = se + be + rng.normal(0, eddy)
659 vtn = sn + bn + rng.normal(0, eddy)
660 lat += vtn * dt_hours / KM_PER_DEG_LAT
661 lon += vte * dt_hours / (KM_PER_DEG_LON * np.cos(np.deg2rad(lat)))
662 I = max(15.0, I + _true_intensity_tendency(lon, lat, I, env=intensity_env) * dt_hours)
663 return np.array(truth), np.array(fixes)

681def field_steering(velocity_sampler, *, base=japan_steering, weight=1.0):
682 """Compose a climatological steering with a **field-diagnosed flow anomaly**.
683
684 ``velocity_sampler`` is a ``(lon, lat) → (v_e, v_n)`` callable — in practice
685 ``KolmogorovFlowModel.velocity_sampler(field, region)`` built from the
686 field-EnKF analysis (or one ensemble member) of notebook 13. The returned
687 ``steering(lon, lat, t)`` adds ``weight × anomaly`` to the mean ridge
688 ``base``, so:
689
690 - the **mean** flow still gives physical recurvature, and
691 - flow-dependent steering **anomalies from the field EnKF perturb the
692 track** — feed one sampler per field-ensemble member to turn the
693 *environmental* uncertainty into **track-cone** uncertainty, the real
694 coupling between the field filter and the vortex tracker.
695 """
696 def steering(lon, lat, t=None):
697 be, bn = base(lon, lat, t)
698 ae, an = velocity_sampler(lon, lat)
699 return be + weight * ae, bn + weight * an
700 return steering

773class TornadoEnvironmentModel(NativeTwin, ExtendedKalmanModel):
774 """
775 Mean-reverting tornado-environment latent — one regime of an IKF.
776
777 A single latent ``x = [STP]`` (a significant-tornado-parameter proxy —
778 combine CAPE × shear × storm-relative helicity from local mesonet/radar)
779 that relaxes toward a regime equilibrium::
780
781 x_{k+1} = μ + a (x_k − μ) + noise (a = persistence, μ = regime mean)
782
783 A tornado is not deterministically trackable, so the deliverable is the
784 **probability the environment sits in the tornadic regime** — obtained by
785 running a benign and a tornadic instance of this model in an
786 :class:`~kalman.framework.filters.InteractiveKalmanFilter`; the IKF mode probability
787 *is* that probability, and it comes with the latent's covariance.
788 ``ikf_bank(season=...)`` sets the Markov transition matrix from the local
789 **seasonal climatology** — the physically-honest way "seasoning" enters.
790
791 Parameters
792 ----------
793 mu : float
794 Regime equilibrium STP (benign ≈ 0.3, tornadic ≈ 3).
795 persistence : float
796 AR(1) coefficient ``a`` (0–1); higher = longer regime memory.
797 q_std, obs_noise : float
798 Process / observation std.
799 """
800
801 NAME = "Tornado Environment"
802 DESCRIPTION = "Mean-reverting significant-tornado-parameter latent (one IKF regime)."
803 VERSION = "1.0"

808 def observables(self): return TornadoObservable

811 q_std=0.15, obs_noise=0.4, dt_hours=1.0, x0=None, **kw):
812 self.dtypedtype = np.float64
813 self.mu = float(mu)
814 self.a = float(persistence)
815 self.nnnnn = self.mmm = 1
816 self.lll = 0
817 Q = np.array([[q_std**2]], float)
818 R = np.array([[obs_noise**2]], float)
819 if x0 is None:
820 x0 = np.array([[mu]])

822 x0=x0, **kw)

840 @classmethod
841 def ikf_bank(cls, sensor_id="ENV", *, season="shoulder",
842 mu_benign=0.3, mu_tornadic=3.0, **kw):
843 """Benign + tornadic instances + the **seasonal** transition matrix.
844
845 Returns ``(models, transition, initial_probs)`` for the IKF; the
846 transition matrix is drawn from ``_SEASON_PI[season]`` so the local
847 severe-weather season raises both the tornadic-regime persistence and
848 the prior probability of switching into it.
849 """
850 stay_b, stay_t = _SEASON_PI.get(season, _SEASON_PI["shoulder"])
851 benign = cls(sensor_id, mu=mu_benign, persistence=0.7, q_std=0.10, **kw)
852 tornadic = cls(sensor_id, mu=mu_tornadic, persistence=0.9, q_std=0.5, **kw)
853 Pi = np.array([[stay_b, 1 - stay_b], [1 - stay_t, stay_t]])
854 probs = np.array([0.9, 0.1])
855 return [benign, tornadic], Pi, probs

860def synthetic_environment(n_steps=200, *, onsets=((70, 110),), season="shoulder",
861 mu_benign=0.3, mu_tornadic=3.0, obs_noise=0.4, seed=0):
862 """Synthetic STP time series that spends ``onsets`` cycle-windows in the
863 tornadic regime, plus noisy observations and the true regime labels.
864
865 Returns ``(obs (T, 1), labels (T,))`` — feed ``obs`` to the IKF tracker and
866 compare ``mode_probabilities[:, 1]`` against ``labels``.

877 obs.append([x + rng.normal(0, obs_noise)])

900def brazil_tornado_scenario(n_days=360, *, start_month=7, seed=0,
901 mu_benign=0.3, mu_tornadic=3.0, obs_noise=0.4,
902 base_rate=0.006, peak_rate=0.05):
903 """A year of daily tornado-environment observations over southern Brazil.
904
905 Tornadic episodes (2–4 days each, front passages) are sampled with a
906 **seasonally varying rate** — frequent in the Sep–Feb peak, rare off-season
907 — so the same STP excursion carries a very different climatological prior
908 depending when it lands. Returns ``(obs (T,1), labels (T,), months (T,))``.
909 """
910 rng = np.random.default_rng(seed)
911 months = ((start_month - 1 + np.arange(n_days) * 12 // n_days) % 12) + 1
912 rate = {"peak": peak_rate, "shoulder": 0.5 * (peak_rate + base_rate), "off": base_rate}
913 labels = np.zeros(n_days, dtype=int)
914 k = 0
915 while k < n_days:
916 if rng.random() < rate[brazil_season(months[k])]:
917 dur = rng.integers(2, 5)
918 labels[k:k + dur] = 1
919 k += dur
920 else:
921 k += 1
922 x = mu_benign
923 obs = []
924 for k in range(n_days):
925 mu, a, q = ((mu_tornadic, 0.9, 0.5) if labels[k] else (mu_benign, 0.7, 0.10))
926 x = mu + a * (x - mu) + rng.normal(0, q)
927 obs.append([x + rng.normal(0, obs_noise)])

931def seasonal_tornado_probability(obs, months, *, season_fn=brazil_season,
932 model_cls=None, **ikf_kw):
933 """Per-step P(tornadic) with a **season-dependent prior**.
934
935 Runs the IKF once per season (each with that season's transition matrix from
936 :meth:`TornadoEnvironmentModel.ikf_bank`) and, at every step, returns the
937 probability from the filter whose season matches that step's month. So an
938 identical environmental signal reads as a confident warning in the peak
939 season and a hedged one off-season — the Bayesian-correct way to fold the
940 Brazil climatology into a nowcast. Returns the ``(T,)`` probability array.
941 """
942 from ...framework.filters import InteractiveKalmanFilter
943 cls = model_cls or TornadoEnvironmentModel
944 obs = np.asarray(obs, float)
945 months = np.asarray(months)
946 seasons = np.array([season_fn(mo) for mo in months])
947 prob = np.zeros(len(obs))
948 for s in np.unique(seasons):
949 models, Pi, p0 = cls.ikf_bank(season=str(s), **ikf_kw)
950 ikf = InteractiveKalmanFilter(models, Pi, p0)
951 ikf.forward(obs)
952 p_s = ikf.mode_probabilities[:, 1]
953 prob[seasons == s] = p_s[seasons == s]
954 return prob
