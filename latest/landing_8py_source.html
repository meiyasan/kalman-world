1"""Rocket landing — powered descent and touchdown."""
2from __future__ import annotations
3"""
4Rocket science — getting off the pad and back: **takeoff** and **landing**.
5
6This module is the *near-surface, powered-flight* half of the rocketry models (no third
7bodies). The space-travel half — flying to the Moon and interplanetary transfers — lives in
8:mod:`kalman.apps.space`.
9
10The "rocket equation" family (powered flight where what differs is the **phase**):
11
12* :class:`RocketAscentModel` — a planar **guided ascent**: gravity versus **thrust supplied
13 as the control input** ``u`` (constant mass). Demonstrates control channels (``l>0``).
14 *Scaled units.* Legacy alias: ``RocketLaunchModel``.
15* :class:`RocketLiftoffModel` — a 1-D **vertical liftoff** whose thrust is internal and whose
16 **mass burns down** (Tsiolkovsky): acceleration rises until **burnout**. *SI units.*
17 Legacy alias: ``RocketTakeoffModel``.
18* :class:`RocketLandingModel` — the same 1-D mass-burning dynamics run in reverse: a
19 SpaceX-style **hoverslam**, free-falling then firing a closed-loop landing burn to touch
20 down near zero velocity. *SI units.*
21
22All are :class:`ExtendedKalmanModel` subclasses whose ``f()`` advances the state by one
234th-order Runge–Kutta step; the EKF linearises ``f`` with a numerical Jacobian.
24
25The ascent model uses *scaled units* (``length = 1 Mm = 1000 km``, ``time = 1 ks = 1000 s``)
26to stay well-conditioned; the near-surface liftoff/landing models use plain **SI** (m, s, kg).
27Physical constants come from :mod:`kalman.runtime.astro` (astropy-sourced where available).
28"""

52class RocketLandingModel(NativeTwin, ExtendedKalmanModel):
53 """
54 Vertical **propulsive landing** of a booster — the SpaceX "hoverslam" / suicide burn.
55
56 Same 1-D mass-burning dynamics as :class:`RocketLiftoffModel`, run the other way: the
57 stage falls engine-off (``v < 0``), then a **closed-loop landing burn** ignites late
58 and throttles to arrive at the pad near zero velocity — the late, single-shot burn
59 that *is* a hoverslam (there is no fuel to hover)::
60
61 ḣ = v
62 v̇ = a_T(h,v,m) − g − ½ ρ(h) (C_d A) v|v| / m
63 ṁ = −a_T·m / v_e (propellant spent ∝ thrust; 0 once tanks are dry)
64
65 The onboard guidance ``a_T`` (see :meth:`guidance`) holds the descent to a reference
66 profile ``v_ref(h) = −√(2·decel·h) − v_touch`` that vanishes at the pad: above it the
67 engine stays off and the booster free-falls; once it catches the profile the engine
68 fires and tracks it down. A ground radar measures **altitude only**; the EKF recovers
69 the unmeasured descent rate and the dwindling propellant. ``n = 3``, ``m = 1``,
70 ``l = 0``. Seed it descending, e.g. ``initial_state([h0, -v0, m0])``.
71 """
72

80 def __init__(self, name="LANDING", *,
81 m0=1000.0, # mass at the start of descent [kg] (incl. landing fuel)
82 m_dry=700.0, # dry mass [kg] (no propellant left below this)
83 ve=2500.0, # effective exhaust velocity [m/s]
84 thrust_max=30_000.0, # maximum engine thrust [N]
85 drag_area=0.25, # C_d·A [m²]
86 drag=True, # include atmospheric drag
87 g=G0, # gravity [m/s²]
88 decel=12.0, # reference deceleration of the descent profile [m/s²]
89 v_touch=1.0, # target touchdown speed [m/s]
90 kp=1.5, # descent-rate tracking gain [1/s]
91 fs=10.0, # radar sample rate [Hz]
92 track_m=20.0, # altitude measurement std [m]
93 q_alt=1e-3, q_vel=1e-2, q_mass=1e-4, # process-noise variances
94 **kw):
95 self.dtypedtype = np.float64
96 self.m0 = float(m0)
97 self.m_dry = float(m_dry)
98 self.ve = float(ve)
99 self.thrust_max = float(thrust_max)
100 self.drag_area = float(drag_area)
101 self.drag = bool(drag)
102 self.g = float(g)
103 self.decel = float(decel)
104 self.v_touch = float(v_touch)
105 self.kp = float(kp)
106
107 self._dt = 1.0 / fs
108 self.nnnnn, self.mmmm, self.lll = 3, 1, 0
109 self.fsfsfsfsfs = fs
110
111 Q = np.diag([q_alt, q_vel, q_mass]).astype(float)
112 R = np.array([[track_m ** 2]], dtype=float)
113 # L = I: Q enters additively on every state. f() does not route an
114 # additive `w` through, so without it the engine would difference f with
115 # respect to w, find no sensitivity, and multiply Q by a zero matrix at
116 # every prediction: the covariance then collapses to zero.
117 # Start fuelled. The base default is a state of zeros, which for this
118 # model means no mass at all: the engine would cut off in the first step.
119 if kw.get("x0") is None:
120 x0 = np.zeros((3, 1))
121 x0[2, 0] = self.m0
122 kw["x0"] = x0
123 super().__init__(name, Q=Q, R=R, L=np.eye(self.nnnnn), fs=fs, **kw)

139 # --- atmosphere / guidance --------------------------------------------
140 def _rho(self, h):
141 return RHO0 * np.exp(-max(h, 0.0) / H_ATM)
142
143 def guidance(self, x):
144 """Commanded thrust acceleration ``a_T`` [m/s²] (≥ 0, upward) for state
145 ``x = [h, v, m]``: zero while above the reference descent profile (free fall)
146 or out of propellant, otherwise throttled to track it down to the pad."""
147 h, v, m = np.asarray(x, float).reshape(-1)[:3]
148 if m <= self.m_dry or h <= 0.0:
149 return 0.0
150 m_eff = max(m, self.m_dry)
151 a_max = self.thrust_max / m_eff
152 v_ref = -np.sqrt(2.0 * self.decel * max(h, 0.0)) - self.v_touch
153 a_drag = -0.5 * self._rho(h) * self.drag_area * v * abs(v) / m_eff
154 a_cmd = self.kp * (v_ref - v) + self.g - a_drag # so that v̇ → kp·(v_ref − v)
155 return float(np.clip(a_cmd, 0.0, a_max))
156
157 def accel(self, x):
158 """Net vertical acceleration [m/s²] for state ``x = [h, v, m]``."""
159 return self._deriv(np.asarray(x, float).reshape(-1)[:3], 0.0)[1]
160
161 # --- dynamics ----------------------------------------------------------
162 def _deriv(self, s, t):
163 h, v, m = s
164 if h <= 0.0: # on the pad: frozen (landed)
165 return np.zeros(3)
166 a_T = self.guidance(s)
167 m_eff = max(m, self.m_dry)
168 a = a_T - self.g
169 if self.drag:
170 a -= 0.5 * self._rho(h) * self.drag_area * v * abs(v) / m_eff
171 mdot = a_T * m_eff / self.ve if m > self.m_dry else 0.0
172 return np.array([v, a, -mdot])
173
174 DYNAMICS = KalmanDynamicsKind.CONTINUOUS # f is ds/dt; the framework integrates it (RK4)
175
176 def f(self, x, u=None, w=None, t=None):
177 return self._deriv(np.asarray(x, float).reshape(-1), float(t or 0.0)).reshape(np.shape(x))
178
179 def after_step(self, x, u=None, t=None):
180 s = np.asarray(x, float).reshape(-1)
181 if s[0] <= 0.0: # touchdown: clamp to the pad
182 s[0] = 0.0
183 s[1] = 0.0
184 s[2] = max(s[2], self.m_dry) # never burn past the dry mass
185 return s.reshape(-1, 1)
186
187 def h(self, x, v=None, t=None):
188 z = np.asarray(x, float).reshape(-1)[:1].reshape(1, 1)
189 return z if v is None else z + np.asarray(v, float).reshape(1, 1)

191 # --- landing metrics from a record ------------------------------------
192 def metrics(self, record):
193 """Headline landing figures from a sampled/filtered ``record`` (states
194 ``[h, v, m]``): ignition time & altitude, touchdown speed, propellant burned and
195 the peak deceleration of the landing burn."""
196 t = np.asarray(record.time, float).reshape(-1)
197 x = np.asarray(record.xk)
198 h, v, mass = x[:, 0], x[:, 1], x[:, 2]
199 burning = mass < self.m0 - 1e-6
200 i_ig = int(np.argmax(burning)) if burning.any() else len(t) - 1
201 i_td = int(np.argmin(np.abs(h))) # closest to the pad
202 a = np.array([self.accel(xi) for xi in x])
203 return {
204 "ignition time [s]": float(t[i_ig]),
205 "ignition altitude [m]": float(h[i_ig]),
206 "touchdown speed [m/s]": float(abs(v[i_td])),
207 "propellant burned [kg]": float(self.m0 - mass.min()),
208 "max decel [g]": float(np.nanmax(a) / G0),
209 }

217 """Signature figure: altitude vs time (truth, EKF, noisy radar fixes), the descent

220 import matplotlib.pyplot as plt
221 truth = np.asarray(sample.xk) if sample is not None else np.asarray(record.xk)
222 estx = np.asarray(record.xk)
223 n = min(len(truth), len(estx))
224 truth, estx = truth[:n], estx[:n]
225 t = np.arange(n) * self._dt
226 meas = (np.asarray(sample.zk).reshape(-1) if sample is not None
227 else np.asarray(record.zk).reshape(-1))
228 meas = meas[:n] if meas.size else None
229 # throttle = thrust / weight, recomputed from the estimated state
230 a_T = np.array([self.guidance(xi) for xi in estx])
231 throttle = a_T * np.clip(estx[:, 2], self.m_dry, None) / self.thrust_max
232
233 fig, ax = plt.subplots(1, 3, figsize=figsize)
234 if meas is not None:
