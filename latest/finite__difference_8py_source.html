1"""Finite-difference 2D field solver — the first non-spectral discretization.
2
3Where :class:`~kalman.framework.solvers.SpectralSolver` needs a *regular
4periodic* mesh, this solver works on any regular rectangular grid with a choice
5of **boundary condition** — so bounded domains (walls, fixed values) are
6representable. Second-order central stencils for ∂/∂x, ∂/∂y and ∇²; the time
7march is delegated to the composed
8:class:`~kalman.framework.integrator.KalmanIntegrator` exactly as the spectral
9solver does (same Butcher-tableau path), so integrators remain swappable.
10
11Subclasses supply the physics via ``rhs(field, t) → ∂field/∂t`` in *real* space
12(contrast the spectral solver's spectral-space ``rhs_hat``). The built-in
13default is the **diffusion** equation ``∂f/∂t = ν∇²f``, which makes the class
14usable (and testable) on its own.
15
16Boundaries (``bc``): ``"periodic"`` (matches spectral), ``"dirichlet"`` (field
17pinned to ``bc_value`` outside), or ``"neumann"`` (zero-gradient / reflective).
18"""
19from __future__ import annotations
20import numpy as np
21
22from ..solver import KalmanSolver
23from ..constants import KalmanSolverKind

26class FiniteDifferenceSolver(KalmanSolver):
27
28 NAME = "Finite-Difference Solver"
29 DESCRIPTION = ("Second-order finite-difference 2D field solver (periodic / dirichlet / "
30 "neumann boundaries); time march via the composed integrator.")
31 VERSION = "1.0"
32 KIND = KalmanSolverKind.FINITE_DIFFERENCE

34 def __init__(self, n=64, dt=0.01, *, length=2*np.pi, bc="periodic", bc_value=0.0,
35 nu=0.01, integrator=None, backend=None, device=None, seed=0):
36 super().__init__(integrator=integrator, backend=backend, device=device)
37 ny, nx = (int(n), int(n)) if np.isscalar(n) else (int(n[0]), int(n[1]))
38 Ly, Lx = ((float(length), float(length)) if np.isscalar(length)
39 else (float(length[0]), float(length[1])))
40 self.ny, self.nx = ny, nx
41 self.shape = (ny, nx)
42 self.n = nx if nx == ny else None
43 self.Ly, self.Lx = Ly, Lx
44 self.dy, self.dx = Ly / ny, Lx / nx
45 self.dt = float(dt)
46 self.bc = str(bc).lower()
47 self.bc_value = float(bc_value)
48 self.nu = float(nu)
49 self.rng = np.random.default_rng(seed)
50 if self.bc not in ("periodic", "dirichlet", "neumann"):
51 raise ValueError("bc must be 'periodic', 'dirichlet' or 'neumann'.")
52
53 # ------------------------------------------------------------------
54 def _as_grid(self, w):
55 arr = np.asarray(w, dtype=float)
56 ny, nx = self.shape
57 if arr.ndim == 1:
58 if arr.size != ny * nx:
59 raise ValueError(f"field of size {arr.size} does not fit this solver's "
60 f"{ny}x{nx} grid ({ny*nx} cells).")
61 return arr.reshape(ny, nx)
62 if arr.shape[-2:] != (ny, nx):
63 raise ValueError(f"field grid {arr.shape[-2:]} != solver grid {(ny, nx)}.")
64 return arr
65
66 def _pad(self, f):
67 """One ghost cell on each side, per the boundary condition."""
68 if self.bc == "periodic":
69 return np.pad(f, 1, mode="wrap")
70 if self.bc == "neumann":
71 return np.pad(f, 1, mode="edge") # zero-gradient
72 return np.pad(f, 1, mode="constant", constant_values=self.bc_value)
73
74 # ------------------------------------------------------------------
75 # Differential operators (second-order central differences)
76 # ------------------------------------------------------------------
77 def discretize(self, field):
78 """Spatial discretization: pad with ghost cells per the BC (this is the
79 solver's own job; the integrator then marches it in time)."""
80 return self._pad(self._as_grid(field))
81
82 def gradient(self, w):
83 """∇f = (∂f/∂x, ∂f/∂y) by central differences (BC-aware)."""
84 p = self.discretizediscretize(w)
85 fx = (p[1:-1, 2:] - p[1:-1, :-2]) / (2 * self.dx)
86 fy = (p[2:, 1:-1] - p[:-2, 1:-1]) / (2 * self.dy)
87 return fx, fy
88
89 def laplacian(self, w):
90 """∇²f by the 5-point stencil (BC-aware)."""
91 p = self.discretizediscretize(w)
92 c = p[1:-1, 1:-1]
93 return ((p[1:-1, 2:] - 2 * c + p[1:-1, :-2]) / self.dx ** 2 +
94 (p[2:, 1:-1] - 2 * c + p[:-2, 1:-1]) / self.dy ** 2)
95
96 # ------------------------------------------------------------------
97 # Physics hook (real space) — default: diffusion
98 # ------------------------------------------------------------------
99 def rhs(self, field, t=None):
100 """``∂f/∂t``. Default is diffusion ``ν∇²f``; subclasses override."""
101 return self.nu * self.laplacian(field)
102
103 # ------------------------------------------------------------------
104 # Time march — the composed integrator (same tableau path as spectral)
105 # ------------------------------------------------------------------
106 def _tableau(self):
107 tab = self.integrator.tableau
108 if tab is None:
109 raise TypeError(f"{self.integrator.name()} has no explicit tableau; this solver "
110 f"needs an explicit RK integrator (Euler/RK4/Dormand-Prince).")
111 return tab
112
113 def _native_scheme(self):
114 """The C++ integrator's name when this solver can run in C++: the default
115 physics (diffusion), a built-in integrator, ``step`` not overridden."""
116 from ..integrator import _BUILTIN_STEPS
117 from ..models import native
118 cls = type(self)
119 if (not native.native_loop_enabled() or cls.rhs is not FiniteDifferenceSolver.rhs
120 or cls.step is not FiniteDifferenceSolver.step or "rhs" in self.__dict__):
121 return None
122 integrator = self.integrator
123 name = getattr(integrator, "NATIVE", None)
124 if name is None or type(integrator)._step is not _BUILTIN_STEPS.get(name):
125 return None
126 return name
127
128 def step(self, field):
129 from .spectral import SpectralSolver
130 return SpectralSolver._rk_step(lambda f: self.rhs(f), field, self.dt, self._tableau())
131
132 def advance(self, w, nsteps):
133 """Advance a real field by ``nsteps``; accepts ``(ny,nx)`` or flat and
134 returns the same layout."""
135 arr = np.asarray(w, dtype=float)
136 flat = arr.ndim == 1
137 f = self._as_grid(arr)
138 scheme = self._native_scheme()
139 if scheme is not None and f.ndim == 2: # C++: Kalman::Solvers::FiniteDifference
140 from ..native import core
141 f = core.solve_diffusion(self.shape, self.dt, (self.Ly, self.Lx), self.bc, self.bc_value, self.nu,
142 scheme, f, int(nsteps))
143 return f.ravel() if flat else f
144 for _ in range(int(nsteps)):
145 f = self.step(f)
146 return f.ravel() if flat else f
147
148 def solve(self, field, steps=1, u=None, t=None):
149 return self.advance(field, int(steps))
150
151 def random_ic(self, amp=1.0):
152 """Smooth random initial field, unit std."""
153 f = self.rng.standard_normal(self.shape)
154 for _ in range(8): # smooth by diffusion sweeps
155 f = f + 0.2 * self.laplacian(f) * min(self.dx, self.dy) ** 2
156 return amp * f / (f.std() + 1e-12)
