1"""Calibrating an instrument while it measures.
2
3A room: its temperature and its CO2 level, both slowly wandering. A CO2 meter
4reads the second. A thermometer of unknown offset reads the first, as an
5instrument with a calibration, ``z = gain * T + offset``.
6
71. With that thermometer alone, its offset and the temperature cannot be told
8 apart: ``observability`` says so before any data.
92. A second, calibrated thermometer settles it: ``joint`` then estimates the
10 offset along with the state.
11
12Everything is computed by the C++ library (``Kalman::Instrument``,
13``Models::Fusion``, ``Models::Joint``); the same scenario is
14``tests/cxx/25_instrument.cc``.
15
16 python examples/python/instrument.py
17"""

37def room():
38 return KalmanModel("co2-meter", fs=1.0, F=np.eye(2), H=np.array([[0.0, 1.0]]), Q=np.diag([1e-4, 1e-2]),
39 R=np.array([[4.0]]), x0=np.array([[21.0], [450.0]]), P0=np.diag([4.0, 100.0]))

42def thermometer(offset=0.0):
43 probe = KalmanSensor("probe", G=[[1.0, 0.0]], R=0.04, labels={"T_PROBE": "degC"})
44 return compose(probe, name="thermo").calibrate(gain=1.0, offset=offset)

66truth = room().add_instrument(thermometer(OFFSET)).add_sensor(reference)
67rng = np.random.default_rng(20261117)
68Lq, Lr = np.linalg.cholesky(truth.Q()), np.linalg.cholesky(truth.R())
69x, zk = truth.x0.copy(), []
70for _ in range(400):
71 x = truth.f(x) + Lq @ rng.standard_normal((2, 1))
72 zk.append((truth.h(x) + Lr @ rng.standard_normal((3, 1))).ravel())

74record = model.filter(np.array(zk), smooth=False, reset=True)
75estimate, sigma = record.xk[-1, 2], np.sqrt(record.Pk[-1, 2, 2])
76print(f" thermo.offset: {estimate:.3f} +/- {sigma:.3f} (truth {OFFSET})")
