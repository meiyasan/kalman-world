116 async def drive():
117 from mcp import ClientSession, StdioServerParameters
118 from mcp.client.stdio import stdio_client
119 from modules.api.auth import mint_token
120 secret = "a-long-enough-integration-test-secret-key"
121 env = dict(os.environ, PYTHONPATH=os.getcwd(), KALMAN_JWT_SECRET=secret)
122 # `modules.mcp`, not `kalman.mcp`: the tests run on the source tree,
123 # where the package is importable only as `modules` (CI does not
124 # install it).
125 params = StdioServerParameters(command=sys.executable,
126 args=["-m", "modules.mcp"], env=env,
127 cwd=os.getcwd())
128 async with stdio_client(params) as (r, w):
129 async with ClientSession(r, w) as s:
130 await s.initialize()
131 tools = {t.name for t in (await s.list_tools()).tools}
132 reader = mint_token("bob", ["read"], secret=secret)
133 sender = mint_token("ada", ["send_measurement"], secret=secret)
134 ok = await s.call_tool("run_filter",
135 {"spec": CV, "data": [[0.1 * i] for i in range(10)],
136 "token": sender})
137 denied = await s.call_tool("run_filter",
138 {"spec": CV, "data": [[0.0]], "token": reader})
139 return tools, ok, denied
140
141 tools, ok, denied = asyncio.run(drive())
142 self.assertIn("run_filter", tools)
143 self.assertIn("register_model", tools)
144 self.assertFalse(ok.is_error)
145 payload = json.loads("".join(c.text for c in ok.content if getattr(c, "type", "") == "text"))
