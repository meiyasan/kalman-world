1"""
2The model operations as functions.
3
4Each builds a model from other models, and is also a method of
5:class:`~kalman.framework.model.KalmanModel` and a class of
6``kalman.framework.models``: ``joint(m, p)``, ``m.joint(p)`` and ``Joint(m, p)``
7are the same model, and likewise ``add_sensor`` and ``Fusion``, ``lag`` and ``Lag``.
8
9``compose`` puts things side by side: models into a composite model
10(``compose(a, b)``, ``a.compose(b)``, ``Composite(a, b)``), sensors into an
11instrument (:class:`~.instrument.KalmanInstrument`).
12"""

16def joint(model, parameters, **kwargs):
17 """``model`` with some of its ``parameters`` estimated along with its state:

48def compose(*parts, name=None, rate=None, readout=None, names=None, **kwargs):
49 """Things side by side.
50
51 Models, each with its own state, as one model (:class:`~.models.compose.Composite`):
52 ``compose(a, b)``, or ``compose(a, b, readout=KalmanReadout.SUMMED)`` when they
53 share one measurement; ``names=`` calls the parts.
54
55 Sensors as one instrument: ``compose(thermometer, co2, name="station", rate=0.5)``.
56 Without a name it is named after its sensors (``"thermometer+co2"``)."""
57 from .instrument import KalmanInstrument
58 from .model import KalmanModel
59 from .sensor import KalmanSensor
60 if len(parts) == 1 and isinstance(parts[0], (list, tuple)):
61 parts = tuple(parts[0])
62 if parts and all(isinstance(part, KalmanModel) for part in parts):
63 if name is not None or rate is not None:
64 raise TypeError("compose: name= and rate= are an instrument's; models take readout= and names=.")
65 from .constants import KalmanReadout
66 from .models.compose import Composite
67 return Composite(*parts, readout=readout or KalmanReadout.STACKED, names=names, **kwargs)
68 if readout is not None or names is not None or kwargs:
69 raise TypeError("compose: readout= and names= are for models; sensors take name= and rate=.")
70 if not parts or not all(isinstance(part, KalmanSensor) for part in parts):
71 raise TypeError("compose: models, or sensors, are expected, got "
72 f"{[type(part).__name__ for part in parts]}.")
73 return KalmanInstrument(name or "+".join(part.name for part in parts), parts, rate=rate)
