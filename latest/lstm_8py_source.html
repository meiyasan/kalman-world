1"""LSTM — single-layer recurrent router over the NIS window.
2
3The NIS history is processed step by step (one of the ``window`` timesteps at
4a time). Unlike MLP's fixed concatenated window or CNN's local receptive
5field, the LSTM carries hidden/cell state forward — useful when the
6discriminating signal is a temporal pattern (e.g. burst timing) rather than
7the instantaneous NIS level.
8
9Plug into NeuralRouter::
10
11 from kalman.framework.routers import NeuralRouter
12 from kalman.framework.routers.nn import LSTM
13
14 router = NeuralRouter(model=LSTM(hidden=16), window=10)
15 ikf.fit(training_zk_list, labels_list)
16"""

35 def __init__(self, hidden: int = 16):
36 self.hidden = int(hidden)
37 self._M: int | None = None
38 self._W: int | None = None
39 self._lstm: nn.LSTM | None = None
40 self._head: nn.Linear | None = None
41 self._opt: torch.optim.Optimizer | None = None
42 self.loss_history: list[float] = []
43
44 def build(self, d_in: int, d_out: int, seed=None) -> "LSTM":
45 if seed is not None:
46 torch.manual_seed(seed)
47 M = d_out
48 W = d_in // M
49 self._M, self._W = M, W
50 self._lstm = nn.LSTM(M, self.hidden, batch_first=True).to(_DEVICE)
51 self._head = nn.Linear(self.hidden, d_out).to(_DEVICE)
52 params = list(self._lstm.parameters()) + list(self._head.parameters())
53 self._opt = torch.optim.Adam(params)
54 self.loss_history = []
55 return self
56
57 def _to_seq(self, X: np.ndarray) -> torch.Tensor:
58 """(N, M*W) → (N, W, M) float32 tensor on device."""
59 return torch.tensor(
60 X.reshape(-1, self._W, self._M), dtype=torch.float32, device=_DEVICE
61 )
62
63 def fit(self, X: np.ndarray, y: np.ndarray,
64 epochs: int = 300, lr: float = 2e-3,
65 batch_size: int = 64, **_) -> "LSTM":
66 self._opt.param_groups[0]["lr"] = lr
67 Xt = self._to_seq(X) # (N, W, M)
68 yt = torch.tensor(y, dtype=torch.long, device=_DEVICE)
69 loss_fn = nn.CrossEntropyLoss()
70 N = len(Xt)
71 for _ in range(epochs):
72 idx = torch.randperm(N, device=_DEVICE)
73 epoch_loss, n_batches = 0.0, 0
74 for s in range(0, N, batch_size):
75 b = idx[s:s + batch_size]
76 self._opt.zero_grad()
77 out, _ = self._lstm(Xt[b]) # (batch, W, hidden)
78 logits = self._head(out[:, -1, :]) # last step → (batch, d_out)
79 loss = loss_fn(logits, yt[b])
80 loss.backward()
81 self._opt.step()
82 epoch_loss += float(loss.item()); n_batches += 1
83 self.loss_history.append(epoch_loss / max(n_batches, 1))
84 return self
85
86 def __call__(self, x: np.ndarray) -> np.ndarray:
87 with torch.no_grad():
88 xt = self._to_seq(x[None]) # (1, W, M)
89 out, _ = self._lstm(xt)
90 logits = self._head(out[:, -1, :]).squeeze(0).cpu().numpy()
91 e = np.exp(logits - logits.max())
92 return e / e.sum()
