6class KalmanController:
7 """Callable controller: ``u = ctrl(x̂, P, t, nu=ν)``.
8
9 The KalmanFilter receives it via ``uk=`` and treats it like any other
10 control sequence. The controller may update ``model._B`` as a side
11 effect before returning ``u``.
12
13 Subclasses override ``_policy`` (required) and ``_update`` (optional).
14
15 The shipped controllers compute their policy in C++ (``Kalman::Controllers``,
16 the kind named by ``NATIVE``) when libKalman is installed: ``_policy`` and
17 ``_update`` forward to a C++ controller built on the model
18 (:meth:`_native_handle`). The Python arithmetic that remains
19 (``_reference_policy``, ``_reference_update``) is the reference the tests
20 compare against, and what runs inside ``native.python_reference()``, for a
21 model that is not real double precision, and without the library. The logs
22 (``cost_log``, ``u_log``, ``scale_log``) are kept here either way.
23 """

29 def __init__(self, model, x_ref=None, Qcost=None, Rcost=None):
30 self.model = model
31 self.x_ref = (np.zeros((model.n, 1)) if x_ref is None
32 else np.asarray(x_ref, dtype=float).reshape(-1, 1))
33 n, l = model.n, max(model.l, 1)
34 self.Qcost = np.eye(n) if Qcost is None else np.asarray(Qcost, dtype=float)
35 self.Rcost = np.eye(l) if Rcost is None else np.asarray(Rcost, dtype=float)
36 self.reset()

38 def __call__(self, x_hat, P, t, *, nu=None):
39 x_hat = np.asarray(x_hat, dtype=float).reshape(-1, 1)
40 u = np.asarray(self._policy(x_hat, P, t, nu=nu), dtype=float).reshape(-1, 1)
41
42 B_new = self._update(x_hat, P, t, nu=nu)
43 if B_new is not None:
44 self.model._B = np.asarray(B_new, dtype=float)
45 self.B_log.append((t, B_new.copy()))
46
47 # Cost against the command the actuator can actually deliver (model.clip_u), not
48 # the controller's raw ask -- a fixed-gain controller with no notion of an actuator
49 # limit (e.g. LQR) can ask for far more than the plant ever receives, and charging
50 # Rcost against that fictitious, never-applied effort would inflate its cost against
51 # a controller (e.g. MPPI) whose raw output already respects the limit. u_log still
52 # keeps the raw command for diagnostics -- only the logged cost uses the clipped one.
53 u_applied = np.asarray(self.model.clip_u(u), dtype=float).reshape(-1, 1)
54 e = x_hat - self.x_ref
55 cost = float((e.T @ self.Qcost @ e + u_applied.T @ self.Rcost @ u_applied).item())
56 self.cost_log.append((t, cost))
57 self.u_log.append((t, u.copy()))
58 return u

98 def _native_handle(self):
99 """The C++ controller for this call, or ``None`` when the Python reference
100 runs: no ``NATIVE`` kind, the reference path, no library, a model that is
101 not real double precision, or a configuration the C++ class does not take.
102
103 The model reaches C++ as itself when it is a :class:`NativeModel` (a model
104 written in C++), and otherwise as a callback model whose ``f``, ``h``,
105 Jacobians, noise and actuator limit (``clip_u``) call the Python model.
106 A controller that reads the noise (Boost, AdaptiveQR) always gets the
107 callback model, which asks for ``Q`` and ``R`` at every call: the retuning
108 is made on the Python model, by this class, and the C++ side must see it.
109 """
110 if self.NATIVE is None:
111 return None
112 from .models import native
113 if not native.native_loop_enabled() or not native.callback_eligible(self.model):
114 return None
115 try:
116 key = self._native_key()
117 except (TypeError, ValueError, AttributeError):
118 return None
119 if key is None:
120 return None
121 key = (id(self.model),) + tuple(key)
122 state = self.__dict__.get("_native_state")
123 if state is not None and state[0] == key:
124 return state[1]
125 model_handle, keep = self._native_model()
126 handle = self._native_create(model_handle)
127 self._native_state = (key, handle, model_handle, keep)
128 return handle

144 def _clip_callback(self, handle, errors):
145 from .native import core
146 ffi, model, l = core.ffi, self.model, int(self.model.l)
147
148 @ffi.callback("int(void *, const double *, double *)")
149 def clip(_user, u, out):
150 try:
151 command = np.frombuffer(ffi.buffer(u, l * 8), dtype=np.float64).reshape(l, 1).copy()
152 applied = np.ascontiguousarray(np.asarray(model.clip_u(command), dtype=np.float64).reshape(-1))
153 if applied.size != l:
154 raise ValueError(f"clip_u returned {applied.size} values for a command of {l}.")
155 ffi.memmove(out, applied, l * 8)
156 return 0
157 except BaseException as exc: # must not unwind through C++
158 errors.append(exc)
159 return 1
160
161 core.check(core.lib.kalman_model_callback_set_clip(handle, clip), "set_clip")
162 return clip
