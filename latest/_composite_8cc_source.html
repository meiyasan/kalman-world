52 Composite::Composite(std::vector<std::shared_ptr<Model>> parts, Readout readout, std::vector<std::string> names)
53 : fParts(std::move(parts)), fPartNames(std::move(names)), fReadout(readout)
54 {
55 if (fParts.empty()) throw std::invalid_argument("Models::Composite: give at least one model.");
56 for (const auto &part : fParts)
57 if (!part) throw std::invalid_argument("Models::Composite: a part is null.");
58 if (!fPartNames.empty() && fPartNames.size() != fParts.size())
59 throw std::invalid_argument("Models::Composite: " + std::to_string(fParts.size()) + " parts and "
60 + std::to_string(fPartNames.size()) + " names.");
61 const bool summed = fReadout == Readout::Summed;
62
63 if (fPartNames.empty())
64 {
65 // Called by their model's name; numbered from the second that shares one.
66 std::map<std::string, int> seen;
67 for (const auto &part : fParts)
68 {
69 const std::string base = part->Name().empty() ? "part" : part->Name();
70 const int k = ++seen[base];
71 fPartNames.push_back(k == 1 ? base : base + std::to_string(k));
72 }
73 }
74 {
75 std::set<std::string> seen;
76 for (const std::string &name : fPartNames)
77 if (name.empty() || !seen.insert(name).second)
78 throw std::invalid_argument("Models::Composite: every part needs a name of its own ('" + name + "').");
79 }
80
81 std::size_t n = 0, m = 0, l = 0;
82 const double fs = fParts.front()->SampleRate();
83 for (const auto &part : fParts)
84 {
85 if (part->SampleRate() != fs)
86 throw std::invalid_argument("Models::Composite: the parts must share one sample rate.");
87 if (summed && part->NMeasure() != fParts.front()->NMeasure())
88 throw std::invalid_argument("Models::Composite: summed parts must have the same number of measurements.");
89 fState.push_back(n); fMeasure.push_back(summed ? 0 : m); fControl.push_back(l);
90 n += part->NState(); m += part->NMeasure(); l += part->NControl();
91 }
92 if (summed) m = fParts.front()->NMeasure();
93 fState.push_back(n); fMeasure.push_back(m); fControl.push_back(l);
94
95 SetName(fParts.size() == 1 ? fParts.front()->Name() : "composite");
96 Allocate(n, m, l);
97 SetSampleRate(fs);
98 SetT0(fParts.front()->T0());
99 SetTimeUnit(fParts.front()->TimeUnit());
100
101 // What the parts are, together: matrices when all are; code as soon as one is.
102 StructureKind structure = StructureKind::Matrices;
103 EngineKind engine = EngineKind::Linear;
104 const Model *differenced = nullptr;
105 for (const auto &part : fParts)
106 {
107 if (part->Structure() == StructureKind::Maps) structure = StructureKind::Maps;
108 else if (part->Structure() == StructureKind::TimeVarying && structure == StructureKind::Matrices)
109 structure = StructureKind::TimeVarying;
110 if (engine == EngineKind::Linear && part->DefaultEngine() != EngineKind::Linear) engine = part->DefaultEngine();
111 if (!differenced && !part->HasAnalyticJacobian()) differenced = part.get();
112 }
113 SetStructure(structure);
114 SetDefaultEngine(engine);
115 if (differenced) SetDifferentiator(differenced->Differentiator(), differenced->Epsilon());
116 else SetAnalyticJacobian(true);
117
118 AdoptInitialState();
119
120 std::vector<std::vector<Component>> states, measurements;
121 for (const auto &part : fParts)
122 {
123 states.push_back(part->StateComponents());
124 if (!summed) measurements.push_back(part->MeasurementComponents());
125 }
126 DeclareState(Named(states, fPartNames));
127 DeclareMeasurement(summed ? fParts.front()->MeasurementComponents() : Named(measurements, fPartNames));
128 if (summed) SetBuiltInSensorName(fParts.front()->BuiltInSensorName());
129 }

131 void Composite::AdoptInitialState()
132 {
133 Vector x0(fNState, 0.);
134 Matrix P0(fNState, fNState, 0.);
135 for (std::size_t p = 0; p < fParts.size(); ++p)
136 {
137 const std::size_t n = fParts[p]->NState(), at = fState[p];
138 const Vector &s0 = fParts[p]->X0();
139 const Matrix &Ps = fParts[p]->P0();
140 for (std::size_t i = 0; i < n; ++i)
141 {
142 x0(at + i) = s0.Size() == n ? s0(i) : 0.;
143 for (std::size_t j = 0; j < n; ++j)
144 P0(at + i, at + j) = (Ps.Rows() == n && Ps.Cols() == n) ? Ps(i, j) : (i == j ? 1. : 0.);
145 }
146 }
147 SetInitialState(x0);
148 SetInitialCovariance(P0);
149 }

299 Matrix Composite::B(double t) const
300 {
301 if (fNControl == 0) return Matrix();
302 Matrix out(fNState, fNControl, 0.);
303 for (std::size_t p = 0; p < fParts.size(); ++p)
304 {
305 const std::size_t l = fControl[p + 1] - fControl[p], n = fState[p + 1] - fState[p];
306 if (l == 0) continue;
307 const Matrix Bp = fParts[p]->B(t); // empty: u is added to the state directly
308 if (!Bp.Empty()) Place(out, Bp, fState[p], fControl[p]);
309 else
310 for (std::size_t i = 0; i < std::min(n, l); ++i) out(fState[p] + i, fControl[p] + i) = 1.;
311 }

355 {
356 if (fReadout == Readout::Summed) return Model::Sensors();
357 std::vector<Sensor> out;
358 std::map<std::string, int> seen;
359 for (std::size_t p = 0; p < fParts.size(); ++p)
360 for (const Sensor &s : fParts[p]->Sensors()) ++seen[s.name];
361 std::size_t row = 0;
362 const std::vector<Component> channels = MeasurementComponents();
363 for (std::size_t p = 0; p < fParts.size(); ++p)
364 for (Sensor sensor : fParts[p]->Sensors())
365 {
366 const std::size_t k = sensor.Channels(), at = fState[p], n = fState[p + 1] - at;
367 if (seen[sensor.name] > 1) sensor.name = fPartNames[p] + "_" + sensor.name;
368 if (row + k <= channels.size())
369 sensor.channels.assign(channels.begin() + static_cast<std::ptrdiff_t>(row),
370 channels.begin() + static_cast<std::ptrdiff_t>(row + k));
371 row += k;
372 if (sensor.IsLinear())
373 {
374 Matrix G(k, fNState, 0.);
375 Place(G, sensor.G, 0, at);
376 sensor.G = G;
377 }
378 else
379 {
380 const Sensor::Measure g = sensor.g;
381 sensor.g = [this, g, p](const Vector &x, double t) { return g(Cut(x, fState, p, "state"), t); };
382 if (sensor.jacobian)
383 {
384 const Sensor::MeasureJacobian dg = sensor.jacobian;
385 sensor.jacobian = [this, dg, p, k, at, n](const Vector &x, double t) {
386 Matrix G(k, fNState, 0.);
387 (void)n;
388 Place(G, dg(Cut(x, fState, p, "state"), t), 0, at);
389 return G;
390 };
391 }
392 }
393 out.push_back(std::move(sensor));
394 }
395 return out;

405 std::pair<std::size_t, std::string> Composite::Resolve(const std::string &name) const
406 {
407 // The longest part name that fits: a part may itself carry dotted names.
408 std::size_t best = fParts.size(), length = 0;
409 for (std::size_t p = 0; p < fParts.size(); ++p)
410 {
411 const std::string &part = fPartNames[p];
412 if (name.size() > part.size() + 1 && name.compare(0, part.size(), part) == 0 && name[part.size()] == '.'
413 && part.size() >= length && fParts[p]->HasParameter(name.substr(part.size() + 1)))
414 {
415 best = p;
416 length = part.size();
417 }
418 }
419 if (best == fParts.size())
420 {
421 std::string known;
422 for (const std::string &n : ParameterNames()) known += (known.empty() ? "" : ", ") + n;
423 throw std::out_of_range("Models::Composite: no parameter '" + name + "'; it has: " + (known.empty() ? "none" : known) + ".");
424 }
425 return {best, name.substr(length + 1)};
426 }
