19 const Matrix *Tabulated::At(const char *which, double t) const
20 {
21 const auto it = fTables.find(which);
22 if (it == fTables.end() || it->second.empty()) return nullptr;
23 const double k = std::round((t - T0()) * SampleRate());
24 const double last = static_cast<double>(it->second.size() - 1);
25 return &it->second[static_cast<std::size_t>(std::min(std::max(k, 0.), last))];
26 }

28 void Tabulated::SetTable(const std::string &which, std::vector<Matrix> table)
29 {
30 if (which != "F" && which != "H" && which != "Q" && which != "R")
31 throw std::invalid_argument("Kalman::Models::Tabulated: a table is for F, H, Q or R, not '" + which + "'.");
32 if (table.empty()) throw std::invalid_argument("Kalman::Models::Tabulated: the table of " + which + " is empty.");
33 const std::size_t rows = (which == "F" || which == "Q") ? NState() : NMeasure();
34 const std::size_t cols = (which == "R") ? NMeasure() : NState();
35 for (const auto &M : table)
36 if (M.Rows() != rows || M.Cols() != cols)
37 throw std::invalid_argument("Kalman::Models::Tabulated: the table of " + which + " needs matrices of "
38 + std::to_string(rows) + " x " + std::to_string(cols) + ".");
39 fTables[which] = std::move(table);
40 SetStructure(StructureKind::TimeVarying);
41 }

57 void Tabulated::SetComponent(bool state, std::size_t i, const std::string &name, const std::string &unit)
58 {
59 std::vector<Component> all = state ? StateComponents() : MeasurementComponents();
60 if (i >= all.size())
61 throw std::out_of_range("Kalman::Models::Tabulated: there is no component " + std::to_string(i) + ".");
62 all[i] = Component{name, unit};
63 if (state) DeclareState(std::move(all));
