13class BayesFilter(KalmanFilterMetadata, KalmanFilterMixin):
14 """
15 A recursive Bayesian **filter**: the workflow (WHEN), whatever the belief looks
16 like. Composes a ``model`` (WHAT) and an ``engine`` (HOW), owns the run state
17 (``x/P`` + prior/posterior/transition/smooth logs) and drives ``forward`` /
18 ``backward``. It implements no arithmetic: that lives in the engine. The C++
19 counterpart is ``Kalman::Filter``.
20
21 Two families derive from it, by what the belief is taken to be:
22
23 - Gaussian (a mean and a covariance): :class:`KalmanFilter` and its leaves,
24 linear, extended, unscented, ensemble, combinatorial;
25 - not Gaussian (a cloud of weighted samples): ``ParticleFilter``. A particle
26 filter is not a Kalman filter; it reports the mean and covariance of its
27 cloud so that records and plots read alike.
28
29 Engine resolution is declarative: if no engine is passed, one is built from
30 ``self.KIND`` (set by public leaves like :class:`LinearKalmanFilter`) or, failing
31 that, the model's ``DEFAULT_ENGINE``; the chosen engine's ``KIND`` is validated
32 against ``model.ALLOWED_ENGINES``.
33 """
34

36 DESCRIPTION = "Forward/backward workflow over a composed model + engine."
37 VERSION = "1.0"

67 @staticmethod
68 def _linear_model(*, n=None, m=None, F=None, H=None, Q=None, R=None,
69 fs=1.0, x0=None, P0=None, name="filter", **kw):
70 """Build a generic linear :class:`KalmanModel` from dimensions + coefficients.
71
72 ``n`` = state dimension, ``m`` = measurement dimension. Any matrix left as
73 ``None`` gets an editable placeholder: ``F`` = identity, ``H`` = observe the
74 first ``m`` states, ``Q`` = 1e-3·identity, ``R`` = identity. ``n``/``m`` are
75 inferred from ``F``/``H`` when those are given.
76 """
77 from .model import KalmanModel
78 F = None if F is None else np.asarray(F, dtype=float)
79 H = None if H is None else np.asarray(H, dtype=float)
80 if n is None and F is not None: n = F.shape[0]
81 if m is None and H is not None: m = H.shape[0]
82 if n is None:
83 raise ValueError("Provide a state dimension n (or an F matrix) to size the filter.")
84 if m is None: m = n
85 if F is None: F = np.eye(n)
86 if H is None:
87 H = np.zeros((m, n))
88 for i in range(min(m, n)): H[i, i] = 1.0
89 Q = np.eye(n) * 1e-3 if Q is None else np.asarray(Q, dtype=float)
90 R = np.eye(m) * 1.0 if R is None else np.asarray(R, dtype=float)
91 return KalmanModel(name, F=F, H=H, Q=Q, R=R, fs=fs, x0=x0, P0=P0, **kw)

129 # A whole-pass engine (e.g. the combinatorial track finder) is handed the data
130 # and returns finished results; there is no per-step loop to run, and none of
131 # the time-series normalisation below applies. See KalmanEngine.OWNS_PASS.
132 if self.engineengine.OWNS_PASS:
133 self.pass_resultpass_result = self.engineengine.forward(self.modelmodelmodel, gk, **kwargs)
134 self.pass_recordspass_records = self._pass_records(KalmanStateFlavor.POSTERIOR)
135 return self.pass_recordspass_records if record else self
136
137 indexing = kwargs.pop("indexing", "xy")
138 preprocess = kwargs.pop("preprocess", True)
139 inprocess = kwargs.pop("inprocess", True)
140 postprocess = kwargs.pop("postprocess", True)
141
142 reset = kwargs.pop("reset", True)
143 buffer = kwargs.pop("buffer", 60) # seconds
144
145 # An explicit reset wins: start a clean run from the model's x0/P0. Only when
146 # *appending* (reset=False, streaming) do we roll the window forward once it
147 # grows past `buffer` seconds — that path calls reset_to_latest(), which
148 # rebases the model's x0/P0 onto the latest estimate, so it must NOT fire on a
149 # plain reset=True re-run (doing so silently corrupts x0/P0 across calls).
150 if reset:
151 self.reset()
152 elif len(self.posteriorposteriorposterior) > buffer * self.sample_rate():
153 self.reset_to_latest()
154
155 # --- detect callable controller (closed-loop) before normalisation ---
156 uk_fn = uk if callable(uk) else None
157
158 # --- normalise measurements / control ---
159 gk = gk.zk if isinstance(gk, KalmanRecord) else gk
160 gk = KalmanRecord._normalize_in(gk, indexing=indexing, dtype=self.modelmodelmodel.dtype)
161 assert gk.shape[1] >= 1, f"gk must have at least 1 column, got {format_shape(gk)}."
162
163 if uk_fn is None:
164 uk = uk.zk if isinstance(uk, KalmanRecord) else uk
165 if uk is None or len(uk) < 1:
166 uk = np.zeros((len(gk), self.l))
167 uk = KalmanRecord._normalize_in(uk, indexing=indexing, dtype=self.modelmodelmodel.dtype)
168 if self.l > 0:
169 assert uk is not None and uk.shape[1] == self.l, \
170 f"Control input uk has wrong number of columns. Expected ({self.l},1), got {format_shape(uk)}."
171
172 # --- preprocess (may alter shapes) ---
173 if uk_fn is not None:
174 zk, _ = self.modelmodelmodel.preprocess(gk, None) if preprocess else (gk, None)
175 else:
176 zk, uk = self.modelmodelmodel.preprocess(gk, uk) if preprocess else (gk, uk)
177 assert zk.shape[1] == self.m, \
178 f"Preprocessed zk has incorrect number of columns. Expected ({self.m},1), got {format_shape(zk)}."
179 if uk_fn is None and self.l > 0:
180 assert uk is not None and uk.shape[1] == self.l, \
181 f"Preprocessed uk has incorrect number of columns. Expected ({self.l},1), got {format_shape(uk)}."
182
183 # --- main loop (engine does predict + masked update); a constant-matrix
184 # linear model with no control callback can run the whole pass fused in C ---
185 twin = self._native_twin(uk_fn)
186 if twin is not None:
187 self._forward_twin(twin, zk, uk, uk_fn)
188 elif self._native_forward_eligible(uk_fn):
189 self._forward_native(zk, uk)
190 elif self._matrices_eligible(uk_fn):
191 self._forward_matrices(zk, uk, uk_fn)
192 elif self._callback_eligible(uk_fn):
193 self._forward_callback(zk, uk, uk_fn)
194 else:
195 t = float(self.modelmodelmodel.t0) if self.modelmodelmodel.t0 is not None else 0.0
196 dt = 1.0 / self.sample_rate() if self.sample_rate() else 1.0
197 nu = np.zeros((self.modelmodelmodel.m, 1))
198 u = np.zeros((max(self.l, 1), 1))
199
200 for idx, z in enumerate(zk):
201 if uk_fn is not None:
202 u_raw = uk_fn(self.xxxx, self.PPP, t, nu=nu)
203 u = np.asarray(u_raw, dtype=float).reshape(-1, 1)
204 else:
205 u = uk[idx]
206
207 self.measurementmeasurementmeasurement.append(z)
208 self.controlcontrol.append(u.ravel() if np.ndim(u) > 1 else u)
209
210 self.xxxx, self.PPP, F = self.engineengine._predict(self.modelmodelmodel, self.xxxx, self.PPP, u=u, t=t)
211 self.priorpriorprior.append((self.xxxx, self.PPP))
212 self.transitiontransition.append(F)
213
214 h_pred = self.modelmodelmodel.h(self.xxxx, t=t)
215 nu = np.asarray(z, dtype=float).reshape(-1, 1) - np.asarray(h_pred, dtype=float).reshape(-1, 1)
216 self.innovation.append(nu)
217
218 self.xxxx, self.PPP = self.engineengine._update(self.modelmodelmodel, self.xxxx, self.PPP, z, t=t)
219 self.posteriorposteriorposterior.append((self.xxxx, self.PPP))
220
221 t += dt
222
223 return self.get_record(indexing=indexing, inprocess=inprocess, postprocess=postprocess) if record else self

332 # A controller that commands nothing and retunes the model (adaptive Q
333 # and R, the violin-bank controller): it runs at each step, before the

531 # ------------------------------------------------------------------
532 # Backward smoothing (RTS) — drives engine._smooth
533 # ------------------------------------------------------------------
534 def backward(self, **kwargs):
535 if self.engineengine.OWNS_PASS:
536 if self.pass_resultpass_result is None:
537 raise RuntimeError(
538 f"No results to smooth; run {type(self).__name__}.forward() first.")
539 self.pass_resultpass_result = self.engineengine.backward(self.modelmodelmodel, self.pass_resultpass_result, **kwargs)
540 self.pass_recordspass_records = self._pass_records(KalmanStateFlavor.SMOOTHED)
541 return self.pass_recordspass_records
542
543 N = len(self.posteriorposteriorposterior)
544 assert N > 0, "No posterior estimates available for smoothing. (run forward() first?)"
545 assert len(self.priorpriorprior) == N, "Prior estimates must match the number of posterior estimates."
546
547 self.smoothsmoothsmoothsmooth = np.zeros((N, 2), dtype=object)
548 self.smoothsmoothsmoothsmooth[-1] = self.posteriorposteriorposterior[-1]
549
550 if N > 1 and self._native_backward_eligible():
551 self._backward_native(N)
552 else:
553 # transition[k + 1] is the one that took posterior[k] to prior[k + 1]:
554 # the F of the RTS gain P[k] F^T P_prior[k + 1]^-1.
555 for k in reversed(range(N - 1)):
556 self.smoothsmoothsmoothsmooth[k] = self.engineengine._smooth(
557 self.modelmodelmodel, self.posteriorposteriorposterior[k], self.priorpriorprior[k + 1], self.smoothsmoothsmoothsmooth[k + 1], self.transitiontransition[k + 1]
558 )
559
560 kwargs.pop("smooth", None)
561 return self.get_record(smooth=True, **kwargs)

563 # ------------------------------------------------------------------
564 # The RTS sweep in C++, over the transitions the forward pass recorded:
565 # ``kf_backward_rts`` when one F served the whole run, and
566 # ``kf_backward_rts_steps`` when each step had its own (a nonlinear
567 # model's Jacobians, a time-varying F).
568 # ------------------------------------------------------------------
569 def _native_backward_eligible(self) -> bool:
570 from .native import HAS_C_ACCEL
571 from .models import native
572 from .engine import KalmanEngine
573 if not HAS_C_ACCEL or not self.transitiontransition or not native.native_loop_enabled():
574 return False
575 # The sweep in C++ is the RTS step of KalmanEngine: an engine with a
576 # smoothing step of its own keeps it.
577 return type(self.engineengine)._smooth is KalmanEngine._smooth

579 def _backward_native(self, N):
580 from .native import lib, as_contiguous, dptr
581
582 n = self.modelmodelmodel.n
583 # One transition for the whole run (a constant-matrix model), or one per
584 # step: the Jacobians an extended pass recorded, a time-varying F.
585 first = self.transitiontransition[0]
586 constant = all(F is first for F in self.transitiontransition)
587 F = as_contiguous(first if constant else np.array([np.asarray(F, dtype=float) for F in self.transitiontransition]))
588 sweep = lib.kf_backward_rts if constant else lib.kf_backward_rts_steps
589 x_post = as_contiguous(np.array([x.reshape(-1) for x, _ in self.posteriorposteriorposterior]))
590 P_post = as_contiguous(np.array([P for _, P in self.posteriorposteriorposterior]))
591 x_prior = as_contiguous(np.array([x.reshape(-1) for x, _ in self.priorpriorprior]))
592 P_prior = as_contiguous(np.array([P for _, P in self.priorpriorprior]))
593
594 x_smooth = np.empty((N, n), dtype=np.float64)
595 P_smooth = np.empty((N, n, n), dtype=np.float64)
596
597 sweep(
598 n, N,
599 dptr(x_post), dptr(P_post),
600 dptr(x_prior), dptr(P_prior),
601 dptr(F),
602 dptr(x_smooth), dptr(P_smooth),
603 )
604
605 for k in range(N):
606 self.smoothsmoothsmoothsmooth[k] = (x_smooth[k].reshape(n, 1), P_smooth[k])
