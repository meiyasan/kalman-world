15class OscillatorBankModel(CompositeKalmanModel):
16 """
17 A bank of :class:`OscillatorModel` windows analysing **one** real signal across
18 several distinct centers ``fc_i``.
19
20 Internally this is a :class:`CompositeKalmanModel` with ``obs_mode=INDEPENDENT``:
21 each window tracks its own demodulated band and the observations are stacked.
22 The standard :class:`CompositeKalmanModel` preprocess would try to split the input
23 by channel count, which cannot broadcast one real signal to N demodulators. This
24 class overrides preprocess with :meth:`_broadcast_preprocess` to send the single
25 real strain to every window's own Hilbert demodulator.
26
27 Composition pattern::
28
29 OscillatorBankModel(w1, w2, w3)
30 = CompositeKalmanModel(w1, w2, w3, obs_mode=INDEPENDENT)
31 + preprocess: broadcast real strain → each window's own _demod_preprocess
32
33 Each window's postprocess remodulates to a real band, so ``record.zk`` has one
34 real column per window. ``reconstruct(record)`` sums them to give the total
35 extracted signal; ``strain − reconstruct(record)`` is the cleaned residual.
36 """
37
38 NAME = "Oscillator Bank (window bank)"
39 DESCRIPTION = "INDEPENDENT bank of windowed OscillatorModels over one real signal."
40 VERSION = "1.1"

56 def _broadcast_preprocess(self, zk, uk=None, **kwargs):
57 """Preprocess: broadcast one real strain to every window, each demodulates + downsamples.
58
59 Returns ``(N_dec, 2·N_win)`` — columns ``[Re_ψ1, Im_ψ1, Re_ψ2, Im_ψ2, …]``.
60 Symmetric counterpart: ``_reconstruct_postprocess``.
61 """
62 x = np.asarray(zk, dtype=float).reshape(np.shape(zk)[0], -1)[:, :1] # (T, 1) real strain
63 outs = [w._demod_preprocess(x)[0] for w in self.windows] # each (N_dec, 2)
64 return np.concatenate(outs, axis=1), uk # (N_dec, 2·N_win)

66 def _reconstruct_postprocess(self, zk_hat, **kwargs):
67 """Postprocess: N window phasors (decimated) → one real signal at original rate.
68
69 ``zk_hat`` = ``h(xk)`` from the INDEPENDENT composite, shape ``(N_dec, 2·N_win)``.
70 Columns ``2i, 2i+1`` are ``[Re(Σψ̂_win_i), Im(Σψ̂_win_i)]`` — already the FUSED
71 phasor sum within window ``i`` (computed by each window's FUSED ``h``).
72 Each window's phasor is upsampled to ``fs_original``, remodulated at ``fc_i``,
73 and summed across windows. Returns ``(N, 1)`` real.
74 Symmetric counterpart: ``_broadcast_preprocess``.
75 """
76 z = np.asarray(zk_hat, dtype=float).reshape(len(zk_hat), -1)
77 w0 = self.windows[0]
78 N = w0._preprocess_N
79 t = w0.t0 + np.arange(N) / self._fs_original
80 total = np.zeros(N)
81 for i, w in enumerate(self.windows):
82 psi = z[:, 2 * i] + 1j * z[:, 2 * i + 1]
83 psi_up = w._resample(psi, w._fs_dec, self._fs_original)
84 if len(psi_up) > N:
85 psi_up = psi_up[:N]
86 elif len(psi_up) < N:
87 psi_up = np.concatenate([psi_up, np.zeros(N - len(psi_up))])
88 total += psi_up.real * np.cos(w._wc * t) - psi_up.imag * np.sin(w._wc * t)
89 return total.reshape(-1, 1)

91 def reconstruct(self, record, *, per_window=False, band_limit=False):
92 """Upsample per-window decimated phasors from ``record.zk`` and sum to a
93 single real signal at the original sample rate — shape ``(N_original,)``.
94
95 Parameters
96 ----------
97 per_window : bool, default False
98 If ``True``: return a list of per-window ``(N_original,)`` arrays without
99 re-filtering (delegates to :meth:`reconstruct_window`).
100 band_limit : bool, default False
101 If ``True``: clip each window's output to its ``[fc ± 2·dfc]`` band,
102 removing the residual ``fc ± fs_dec`` upsampling images.
103 """
104 if per_window:
105 return [self.reconstruct_window(record, i, band_limit=band_limit)
106 for i in range(len(self.windows))]
107 z = np.asarray(record.zk, dtype=float).reshape(len(record.zk), -1)
108 w0 = self.windows[0]
109 N = w0._preprocess_N
110 t = w0.t0 + np.arange(N) / self._fs_original
111 total = np.zeros(N)
112 for i, w in enumerate(self.windows):
113 psi = z[:, 2 * i] + 1j * z[:, 2 * i + 1]
114 psi_up = w._resample(psi, w._fs_dec, self._fs_original)
115 if len(psi_up) > N:
116 psi_up = psi_up[:N]
117 elif len(psi_up) < N:
118 psi_up = np.concatenate([psi_up, np.zeros(N - len(psi_up))])
119 if band_limit:
120 psi_up = w._band_limit_phasor(psi_up, 2.0 * w.dfc)
121 total += psi_up.real * np.cos(w._wc * t) - psi_up.imag * np.sin(w._wc * t)
122 return total

124 def reconstruct_window(self, record, i, *, band_limit=False):
125 """Return the real-band reconstruction for window ``i`` without re-filtering.
126
127 Slices window ``i``'s state from ``record.xk``, computes the FUSED phasor sum,
128 upsamples, and remodulates. Does not touch the other windows.
129 """
130 offset = sum(w.n for w in self.windows[:i])
131 n = self.windows[i].n
132 xk_w = np.asarray(record.xk, dtype=float)[:, offset:offset + n]
133 zk_hat_w = np.column_stack([xk_w[:, 0::2].sum(axis=1),
134 xk_w[:, 1::2].sum(axis=1)])
135 return self.windows[i]._reconstruct_postprocess(zk_hat_w, band_limit=band_limit).ravel()
