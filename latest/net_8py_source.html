1"""
2kalman.framework.native.net -- libKalmanNet from Python.
3
4``libKalmanNet`` is the C++ JSON API (``Kalman::Net::Api``) behind the C++
5REST and MCP servers (``kalman-serve``). From Python it is one function,
6:func:`call`: a verb of :mod:`kalman.api`, its arguments as a dict, and the
7C++ answer as a dict -- the same answer ``kalman-serve`` gives over HTTP. It
8serves the models compiled into libKalman and its plugins; Python-written
9models stay with the Python facade.
10
11The library is looked for like libKalmanLearn (see :mod:`.learn`), under the
12name ``libKalmanNet``; ``KALMAN_NET_LIBRARY`` names the file itself. It is
13loaded on first use. Loaded next to the core, it shares the core's model
14factory, so a plugin loaded through :mod:`.core` is known to it too.
15"""

48def candidate_paths() -> list[str]:
49 """Where libKalmanNet is looked for, in order."""
50 out = []
51 explicit = os.environ.get("KALMAN_NET_LIBRARY")
52 if explicit:
53 out.append(explicit)
54 if core.LIBRARY_PATH: # next to the core that is loaded
55 out.extend(os.path.join(os.path.dirname(core.LIBRARY_PATH), "lib" + LIBRARY_NAME + ext)
56 for ext in core._EXTS)
57 for prefix in (os.environ.get("KALMAN_ROOT"), os.environ.get("KALMANFRAMEWORK_ROOT"),
58 str(core._REPO / "var" / "install"), str(core._REPO / "var" / "build")):
59 if prefix:
60 out.extend(os.path.join(prefix, "lib", "lib" + LIBRARY_NAME + ext) for ext in core._EXTS)
61 system = ctypes.util.find_library(LIBRARY_NAME)
62 if system:
63 out.append(system)
64 return out

67def load():
68 """dlopen libKalmanNet once. Returns ``lib``, or ``None`` (see ``LOAD_ERROR``)."""
69 global ffi, lib, LIBRARY_PATH, LOAD_ERROR, _tried
70 if lib is not None or _tried:
71 return lib
72 _tried = True
73 try:
74 from cffi import FFI
75 core.load() # the core first: one factory
76 path = next((p for p in candidate_paths() if os.path.isfile(p)), None)
77 if path is None:
78 raise OSError("libKalmanNet not found; build it (make build install) or set "
79 "KALMAN_NET_LIBRARY.")
80 _ffi = FFI()
81 _ffi.cdef(_CDEF)
82 lib = _ffi.dlopen(path)
83 ffi, LIBRARY_PATH = _ffi, path
84 except Exception as e: # missing file, no cffi ...
85 LOAD_ERROR = f"{type(e).__name__}: {e}"
86 return lib

100def call(verb: str, arguments: dict | None = None, token: str | None = None) -> dict:
101 """Run ``verb`` of the C++ API with ``arguments``; the answer as a dict.
102
103 Raises :class:`NetError` with the HTTP status of the refusal (400, 401,
104 403, 404, 501)."""
105 if not available():
106 raise RuntimeError(LOAD_ERROR or "libKalmanNet is not available")
107 out = lib.kalman_net_call(verb.encode(), json.dumps(arguments or {}).encode(),
108 token.encode() if token else ffi.NULL)
109 if out == ffi.NULL:
110 raise NetError(int(lib.kalman_net_last_status()),
111 ffi.string(lib.kalman_net_last_error()).decode())
112 return json.loads(ffi.string(out).decode())
