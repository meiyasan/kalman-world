1"""kalman/framework/native/core — the Python side of libKalman's C API.
2
3This module ``dlopen``s the standalone C++ library — the same ``libKalman`` that
4root-plus links — through the flat C surface declared in ``include/Kalman/CAPI.h``,
5using cffi in **ABI mode**, so no compiler is involved on the Python side.
6
7What comes through is the **factory**: every model compiled into the library or
8into a plugin it has loaded, discoverable by name and constructible without the
9Python side knowing its type. :class:`~kalman.framework.models.native.NativeModel`
10wraps such an instance as an ordinary :class:`~kalman.framework.model.KalmanModel`,
11which is how a C++ model reaches the catalogue, the API and the MCP server.
12
13Exposes ``HAS_CXX_CORE`` and, when the library was found, ``ffi``/``lib`` plus the
14thin helpers below. Everything is optional: an import never fails, it only reports.
15
16Locating the library, in order:
17
18 1. ``KALMAN_LIBRARY`` — the file itself;
19 2. ``KALMAN_ROOT`` — an install prefix (``<prefix>/lib``);
20 3. this checkout's ``var/install/lib`` and ``var/build/lib``;
21 4. the system loader (``ctypes.util.find_library``).
22
23The ``cdef`` below is a copy of ``CAPI.h``; ``tests/test_native_core.py`` compares
24the two, so a prototype added to the header must be added here as well.
25"""

350# --- locating and loading ----------------------------------------------------

354def candidate_paths() -> list[str]:
355 """Every place the library is looked for, in order; the first that exists wins."""
356 out = []
357 # KALMAN_LIBRARY / KALMAN_ROOT are the names; the *FRAMEWORK* ones are those
358 # of releases up to 1.0, when the library was libKalmanFramework, and are
359 # still read.
360 for variable in ("KALMAN_LIBRARY", "KALMAN_FRAMEWORK_LIBRARY"):
361 explicit = os.environ.get(variable)
362 if explicit:
363 out.append(explicit)
364 prefixes = [os.environ.get("KALMAN_ROOT"), os.environ.get("KALMANFRAMEWORK_ROOT"),
365 str(_REPO / "var" / "install"), str(_REPO / "var" / "build")]
366 for prefix in prefixes:
367 if prefix:
368 for name in LIBRARY_NAMES:
369 for ext in _EXTS:
370 out.append(os.path.join(prefix, "lib", "lib" + name + ext))
371 for name in LIBRARY_NAMES:
372 system = ctypes.util.find_library(name)
373 if system:
374 out.append(system)
375 return out
376
377
378def find_library() -> str | None:
379 for path in candidate_paths():
380 if os.path.isfile(path):
381 return path
382 return None

385def load(path: str | None = None):
386 """dlopen the library (idempotent). Returns ``lib`` or raises ``OSError``."""
387 global ffi, lib, LIBRARY_PATH, HAS_CXX_CORE, LOAD_ERROR
388 if lib is not None and (path is None or path == LIBRARY_PATH):
389 return lib
390 from cffi import FFI
391 path = path or find_library()
392 if not path:
393 raise OSError("libKalman not found; build it (make build install) or set "
394 "KALMAN_LIBRARY / KALMAN_ROOT. Looked in: "
395 + ", ".join(candidate_paths()))
396 _ffi = FFI()

513def particle_engine_on_numpy(n_particles, threshold, cloud):
514 """A C++ particle engine that starts from ``cloud`` (n_particles x n, the one
515 the Python engine drew when the filter was reset) and takes every later random
516 number from numpy's global generator, in the order the Python engine asks for
517 them: the process noise of all particles at each prediction, a uniform at each
518 resampling. The C++ pass is then the Python run, draw for draw.
519
520 Returns ``(engine, keep)``; ``keep`` holds the callbacks (keep it referenced
521 for as long as the engine runs) and, in ``keep[-1]``, an exception a callback
522 raised."""
523 L = require()
524 raised = []
525
526 @ffi.callback("int(void *, int, int, const double *, double *)")
527 def gaussian(_user, count, dim, cov, out):
528 try:
529 C = np.frombuffer(ffi.buffer(cov, dim * dim * 8), dtype=np.float64).reshape(dim, dim).copy()
530 draws = np.ascontiguousarray(np.random.multivariate_normal(np.zeros(dim), C, size=count), dtype=np.float64)
531 ffi.buffer(out, count * dim * 8)[:] = draws.tobytes()
532 return 0
533 except BaseException as e: # must not cross the C boundary
534 raised.append(e)
535 return -1
536
537 @ffi.callback("int(void *, double *)")
538 def uniform(_user, out):
539 try:
540 out[0] = float(np.random.uniform())
541 return 0
542 except BaseException as e:
543 raised.append(e)
544 return -1
545
546 engine = create_engine("particle", n_particles=int(n_particles), threshold=float(threshold), seed=0)
547 cloud = np.ascontiguousarray(np.asarray(cloud, dtype=np.float64))
548 check(L.kalman_engine_particle_set_cloud(engine, cloud.shape[0], cloud.shape[1], cdptr(cloud)), "set_cloud")
549 check(L.kalman_engine_particle_set_draws(engine, gaussian, uniform, ffi.NULL), "set_draws")
550 return engine, (gaussian, uniform, raised)

1038# --- plugin policy -----------------------------------------------------------
1039# Loading a plugin executes code. The registry does not decide what may be
1040# loaded; this does: only files inside an allow-listed directory, and by *name*
1041# from the API so a caller never chooses a path. ``KALMAN_PLUGIN_PATH`` (os.pathsep-
1042# separated) replaces the default list, which is the two trees of this checkout
1043# and wherever the library itself was found.
1044def plugin_dirs() -> list[str]:
1045 env = os.environ.get("KALMAN_PLUGIN_PATH")
1046 if env:
1047 dirs = [d for d in env.split(os.pathsep) if d]
1048 else:
1049 dirs = [str(_REPO / "var" / "install" / "lib"), str(_REPO / "var" / "build" / "lib")]
1050 if LIBRARY_PATH:
1051 dirs.append(os.path.dirname(LIBRARY_PATH))
1052 out = []
1053 for d in dirs:
1054 real = os.path.realpath(d)
1055 if real not in out:
1056 out.append(real)

1063def resolve_plugin(name_or_path: str) -> str:
1064 """The file to load for ``name_or_path``, or an error naming what was tried.
1065
1066 A bare name (``KalmanPlugin_example``) is looked up as ``lib<name><ext>``,
1067 ``<name><ext>`` or ``<name>`` in every allow-listed directory. A path is
1068 accepted only when its real location lies inside one of those directories —
1069 ``../`` and symlinks out of the tree are resolved before the check.
1070 """
1071 if not isinstance(name_or_path, str) or not name_or_path:
1072 raise ValueError("plugin: give a plugin name or path.")
1073 dirs = plugin_dirs()
1074 if os.sep in name_or_path or (os.altsep and os.altsep in name_or_path) \
1075 or name_or_path.endswith(_EXTS):
1076 real = os.path.realpath(name_or_path)
1077 if not os.path.isfile(real):
1078 raise FileNotFoundError(f"plugin: {name_or_path!r} does not exist.")
1079 if not any(os.path.dirname(real) == d for d in dirs):
1080 raise PermissionError(f"plugin: {name_or_path!r} is outside the allowed directories "
1081 f"{dirs}; set KALMAN_PLUGIN_PATH to allow another.")
1082 return real
1083 if not _NAME_RE.match(name_or_path):
1084 raise ValueError(f"plugin: {name_or_path!r} is not a plugin name.")
1085 tried = []
1086 for d in dirs:
1087 for candidate in ([f"lib{name_or_path}{e}" for e in _EXTS]
1088 + [f"{name_or_path}{e}" for e in _EXTS] + [name_or_path]):
1089 path = os.path.join(d, candidate)
1090 tried.append(path)
1091 if os.path.isfile(path):
1092 return os.path.realpath(path)
1093 raise FileNotFoundError(f"plugin: no {name_or_path!r} in {dirs} (tried {len(tried)} names).")
