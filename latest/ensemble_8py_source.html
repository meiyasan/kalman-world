9def gaspari_cohn(dist: np.ndarray, L: float) -> np.ndarray:
10 """Gaspari–Cohn 5th-order compactly-supported correlation function.
11
12 Standard covariance-localization taper for ensemble filters: ≈ Gaussian of
13 scale ``L`` but exactly zero beyond ``2L``, so spurious long-range sample
14 correlations from a small ensemble are cut off instead of merely damped.
15 ``dist`` is any array of distances; returns the taper with the same shape.
16 """
17 r = np.asarray(dist, dtype=float) / float(L)
18 C = np.zeros_like(r)
19 m1 = r <= 1
20 C[m1] = 1 - 5/3*r[m1]**2 + 5/8*r[m1]**3 + 0.5*r[m1]**4 - 0.25*r[m1]**5
21 m2 = (r > 1) & (r <= 2)
22 rr = r[m2]
23 C[m2] = 4 - 5*rr + 5/3*rr**2 + 5/8*rr**3 - 0.5*rr**4 + 1/12*rr**5 - 2/(3*rr)
24 return np.maximum(C, 0.0)

27class EnsembleKalmanEngine(KalmanEngine):
28 """
29 Perturbed-observation EnKF (Evensen) with multiplicative inflation and
30 optional covariance localization.
31
32 Like :class:`ParticleEngine` this is a **stateful** engine: it holds the
33 ensemble ``(n, N)`` for the duration of a forward pass and (re)seeds it via
34 ``reset(model)`` — from ``model.initial_ensemble(N, rng)`` when the model
35 provides one (e.g. climatology snapshots), else from a Gaussian ``x0/P0``
36 draw. ``_predict``/``_update`` return the ensemble mean/covariance as
37 ``(x, P)`` so records and diagnostics are unchanged; the raw members stay
38 available as ``.members`` for DA-style workflows that never want the dense
39 ``(n, n)`` covariance (see :meth:`forecast` / :meth:`analyze`).
40
41 Analysis step (per cycle, forecast ensemble ``X``, anomalies ``X'``)::
42
43 P Hᵀ = ρ_xy ∘ X' Y'ᵀ / (N−1) Y' = h(X)' (obs-space anomalies)
44 H P Hᵀ= ρ_yy ∘ Y' Y'ᵀ / (N−1)
45 K = P Hᵀ (H P Hᵀ + R)⁻¹
46 Xᵃ = X + K (y + ε − h(X)), ε ~ N(0, R) per member (perturbed obs)
47
48 ``ρ_xy (n,p)`` / ``ρ_yy (p,p)`` are localization tapers (e.g.
49 :func:`gaspari_cohn` of the state↔obs distances), taken from
50 ``model.localization()`` when available, else no localization.
51
52 **RNG discipline** — the perturbed-obs draws ``ε`` come from an
53 engine-owned ``numpy`` Generator (``seed=``), *never* the global stream.
54 When several filter variants run in the same loop for comparison, each
55 engine consuming its own stream guarantees that adding/removing one method
56 cannot silently change the random draws (and hence the results) of
57 another. Keep the *observation* noise on a separate shared stream in the
58 experiment harness so every method sees the same measurement realization.
59
60 Parameters
61 ----------
62 n_members : int
63 Ensemble size ``N`` (typically 10–100; sampling error is tamed by
64 localization + inflation, not by brute-force ``N``).
65 inflation : float
66 Multiplicative prior inflation ``a``: anomalies scaled by ``a`` before
67 each analysis (compensates sampling error and model bias; surrogate
68 forecast models generally need a larger ``a`` than the true model).
69 localization : (rho_xy, rho_yy) or None
70 Explicit taper matrices; default asks the model, else identity.
71 seed : int or numpy.random.Generator, optional
72 Engine-local RNG stream (see RNG discipline above).
73 """

80 def __init__(self, n_members: int = 50, inflation: float = 1.0,
81 localization=None, seed=None):
82 self.n_members = int(n_members)
83 self.inflation = float(inflation)
84 self._localization = localization

91 def reset(self, model):
92 N = self.n_members
93 init = getattr(model, "initial_ensemble", None)
94 if callable(init):
95 X = np.asarray(init(N, rng=self.rng), dtype=np.float64)
96 if X.shape != (model.n, N):
97 raise ValueError(f"initial_ensemble must return ({model.n},{N}); got {X.shape}")
98 else:
99 x0 = np.asarray(model.x0, dtype=np.float64).reshape(-1, 1)
100 P0 = np.asarray(model.P0, dtype=np.float64)
101 if np.allclose(P0, np.diag(np.diag(P0))):
102 X = x0 + np.sqrt(np.diag(P0))[:, None] * self.rng.standard_normal((model.n, N))
103 else:
104 X = x0 + np.linalg.cholesky(P0) @ self.rng.standard_normal((model.n, N))
105 self._members = X
106
107 loc = self._localization
108 if loc is None:
109 hook = getattr(model, "localization", None)
110 loc = hook() if callable(hook) else None
111 self._rho_xy, self._rho_yy = (None, None) if loc is None else loc
112 return self

127 def _estimate(self):
128 x = self._members.mean(axis=1, keepdims=True)
129 A = self._members - x
130 P = (A @ A.T) / max(self.n_members - 1, 1)
131 P = 0.5 * (P + P.T) + 1e-12 * np.eye(P.shape[0])
132 return x, P

137 def forecast(self, model, u=None, t=None):
138 """Propagate every member through the model dynamics (+ ``Q`` draws).
139
140 Uses ``model.forecast(X, t=t)`` when the model offers a batched map
141 (columns → columns; also the hook where an AI surrogate plugs in),
142 else loops ``model.f`` per member. Returns the ensemble mean.
143 """
144 X = self._members
145 n, N = X.shape
146 u = (np.asarray(u).reshape(-1, 1) if (u is not None and model.l > 0) else None)
147
148 batch = getattr(model, "forecast", None)
149 if callable(batch):
150 X = np.asarray(batch(X, t=t), dtype=np.float64)
151 else:
152 out = np.empty_like(X)
153 for i in range(N):
154 out[:, i] = np.asarray(
155 model.f(X[:, i].reshape(-1, 1), u=u, t=t), dtype=np.float64).reshape(-1)
156 X = out
157
158 Q = np.asarray(model.Q(t=t), dtype=np.float64)
159 # A declared noise coupling applies here as in every engine: w ~ N(0, L Q L^T).
160 L = getattr(model, "_L", None)
161 if L is not None and not callable(L) and not np.array_equal(L, np.eye(n)):
162 Q = np.asarray(L, dtype=np.float64) @ Q @ np.asarray(L, dtype=np.float64).T
163 if Q.any():
164 if np.allclose(Q, np.diag(np.diag(Q))):
165 X = X + np.sqrt(np.diag(Q))[:, None] * self.rng.standard_normal((n, N))
166 else:
167 X = X + np.linalg.cholesky(Q) @ self.rng.standard_normal((n, N))
168 self._members = X
169 return self.mean

171 def analyze(self, model, z, t=None):
172 """Perturbed-obs analysis update of the ensemble; returns the mean.
173
174 NaN channels in ``z`` are skipped (masked update, same convention as
175 the other engines). Inflation is applied to the forecast anomalies
176 *before* the update.
177 """
178 z = np.asarray(z, dtype=np.float64).reshape(-1)
179 keep = ~np.isnan(z)
180 if not keep.any():
181 return self.mean
182
183 X = self._members
184 n, N = X.shape
185 idx = np.where(keep)[0]
186
187 if self.inflation != 1.0:
188 m = X.mean(axis=1, keepdims=True)
189 X = m + self.inflation * (X - m)
190
191 # Obs-space ensemble h(X) — batched model hook if offered, else per member.
192 hbatch = getattr(model, "observe", None)
193 if callable(hbatch):
194 Y = np.asarray(hbatch(X, t=t), dtype=np.float64)[idx, :]
195 else:
196 Y = np.empty((len(idx), N))
197 for i in range(N):
198 Y[:, i] = np.asarray(
199 model.h(X[:, i].reshape(-1, 1), t=t), dtype=np.float64).reshape(-1)[idx]
200
201 R = np.asarray(model.R(t=t), dtype=np.float64)[np.ix_(idx, idx)]
202 # The draws come first and from numpy's generator in both paths, so a
203 # seeded run is the same run whoever does the arithmetic.
204 eps = (np.linalg.cholesky(R) @ self.rng.standard_normal((len(idx), N))
205 if not np.allclose(R, np.diag(np.diag(R)))
206 else np.sqrt(np.diag(R))[:, None] * self.rng.standard_normal((len(idx), N)))
207
208 analysed = self._native_analysis(X, Y, R, z[idx], eps, idx)
209 if analysed is not None: # the arithmetic ran in C++ (enkf_analysis)
210 self._members = analysed
211 return self.mean
212
213 Xp = X - X.mean(axis=1, keepdims=True)
214 Yp = Y - Y.mean(axis=1, keepdims=True)
215
216 PHt = (Xp @ Yp.T) / (N - 1)
217 HPHt = (Yp @ Yp.T) / (N - 1)
218 if self._rho_xy is not None:
219 PHt *= self._rho_xy[:, idx]
220 if self._rho_yy is not None:
221 HPHt *= self._rho_yy[np.ix_(idx, idx)]
222
223 K = PHt @ np.linalg.pinv(HPHt + R)
224 self._members = X + K @ (z[idx][:, None] + eps - Y)
225 return self.mean

227 def _native_analysis(self, X, Y, R, z, eps, idx):
228 """The analysis ``X + K (z + eps - Y)`` computed by the C++ kernel, or
229 ``None`` when the Python arithmetic must run (reference path, no library)."""
230 from ..models import native
231 if not native.native_loop_enabled():
232 return None
233 from ..native import core
234 c = lambda a: np.ascontiguousarray(a, dtype=np.float64)
235 n, N = X.shape
236 p = len(idx)
237 Xc, Yc, Rc, zc, ec = c(X), c(Y), c(R), c(z), c(eps)
238 rxy = None if self._rho_xy is None else c(np.asarray(self._rho_xy)[:, idx])
239 ryy = None if self._rho_yy is None else c(np.asarray(self._rho_yy)[np.ix_(idx, idx)])
240 out = np.empty((n, N))
241 rc = core.lib.enkf_analysis(n, N, p, core.cdptr(Xc), core.cdptr(Yc), core.cdptr(Rc), core.cdptr(zc),
242 core.cdptr(ec), core.ffi.NULL if rxy is None else core.cdptr(rxy),
243 core.ffi.NULL if ryy is None else core.cdptr(ryy), core.dptr(out))
244 core.check(rc, "enkf_analysis")
245 return out

256 def _native_twin(self, model, Q=None):
257 """The configured C++ twin of ``model`` when the pass can run in C++, else
258 ``None``: another engine class, the reference path, a model with no twin
259 for its configuration, or a ``Q`` that is not diagonal."""
260 from ..models import native
261 if type(self) is not EnsembleKalmanEngine or not native.native_loop_enabled():
262 return None
263 handle_of = getattr(model, "native_handle", None)
264 handle = handle_of() if callable(handle_of) else None
265 if handle is None:
266 return None
267 if self._process_variance(model, Q) is None:
268 return None

271 @staticmethod
272 def _process_variance(model, Q=None):
273 """The diagonal of ``Q`` (its value at the start), or ``None`` when ``Q``
274 has off-diagonal terms (the Python steps then run)."""
275 Q = np.asarray(model.Q(t=None) if Q is None else Q, dtype=np.float64)
276 variance = np.diag(Q)
277 if np.count_nonzero(Q) != np.count_nonzero(variance):
278 return None
279 return np.ascontiguousarray(variance)

281 def _native_engine(self, model, handle, zk, analyse=True, Q=None):
282 """The C++ engine for one pass over ``zk``: this engine's members, the
283 draws the Python steps would take, and the tapers given to this engine.
284 Pushes the process noise of ``Q`` (the model's by default) to the twin."""
285 from ..native import core
286 if self._members is None:
287 self.resetreset(model)
288 n, N = model.n, self.n_members
289 variance = self._process_variance(model, Q)
290 core.check(core.lib.kalman_model_set_array(handle, b"process_variance", variance.size, core.cdptr(variance)),
291 "set_array('process_variance')")
292 # The order of the Python steps: the forecast takes (n, N) when Q is not
293 # zero, the analysis (p, N) for the p channels that arrived.
294 rows = []
295 for z in np.asarray(zk, dtype=np.float64).reshape(len(zk), -1):
296 if variance.any():
297 rows.append(n)
298 if analyse:
299 p = int(np.count_nonzero(~np.isnan(z)))
300 if p:
301 rows.append(p)
302 normals = self.rng.standard_normal(sum(rows) * N)
303 loc = None if self._localization is None else tuple(self._localization)
304 return core.ensemble_engine(N, self.inflation, self._members, normals, loc)
