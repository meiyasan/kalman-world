1"""NeuralRouter — IKF router backed by a pluggable neural model.
2
3The router owns feature extraction and normalisation.
4The model owns architecture, training, and inference.
5
6Usage::
7
8 from kalman.framework.routers import NeuralRouter
9 from kalman.framework.routers.nn import MLP
10
11 router = NeuralRouter(model=MLP(hidden=32), window=5)
12
13 ikf = InteractiveKalmanFilter(models, Pi, router=router)
14 ikf.fit(training_zk_list, labels_list) # trains the model
15 record = ikf.forward(test_zk)
16
17Any callable ``model`` with signature ``model(features) → probs`` works,
18including PyTorch modules, sklearn classifiers, or custom numpy networks.
19
20For lazy parameter init and training support, the model may optionally expose:
21
22* ``model.build(d_in, d_out, seed=None)`` — called once before fitting
23* ``model.fit(X, y, **kwargs)`` — training on normalised features
24
25If these methods are absent the model is assumed to be pre-trained.
26
27NIS feature extraction
28----------------------
29Input features are the last ``window`` NIS values per model, flattened
30to a vector of size M × window.
31
32NIS_j = ν_j^T S_j^{-1} ν_j — under the correct model NIS_j ~ χ²(m).
33
34NIS is used rather than raw innovations or mode probabilities because:
35- Raw ν_j: different scales across models.
36- Mode probs μ: distribution shift (Bayesian μ at train ≠ neural μ at test).
37- NIS: dimensionless, scale-free, no distribution shift.
38
39Before ``fit()`` is called the router falls back to the Bayesian update.
40"""
41from __future__ import annotations

46class NeuralRouter(KalmanRouter):
47 """IKF router that delegates mode-probability estimation to a neural model.
48
49 Parameters
50 ----------
51 model : callable
52 Any object callable as ``probs = model(feat)`` where ``feat`` is a
53 1-D normalised numpy array of length M × window.
54 Optionally implements ``build(d_in, d_out, seed)`` and
55 ``fit(X, y, **kw)`` for lazy init and training.
56 window : int
57 NIS sliding window length. Padding value 1.0 = χ²(1) mean.
58 """
59
60 def __init__(self, model, window: int = 5):
61 self.model = model
62 self.window = int(window)
63 self._x_mean: np.ndarray | None = None
64 self._x_std: np.ndarray | None = None
65
66 # ── feature extraction ────────────────────────────────────────────────────
67
68 def _window_feat(self, nis_arr: np.ndarray) -> np.ndarray:
69 """Last ``window`` NIS rows, padded with 1.0, flattened (M×window,)."""
70 block = nis_arr[-self.window:]
71 if len(block) < self.window:
72 pad = np.ones((self.window - len(block), nis_arr.shape[1]))
73 block = np.vstack([pad, block])
74 return block.ravel()
75
76 @property
77 def _is_trained(self) -> bool:
78 return self._x_mean is not None
79
80 # ── routing ───────────────────────────────────────────────────────────────
81
82 def route(self, nis_arr, c, likelihoods):
83 if not self._is_trained:
84 return self._normalise(c * likelihoods) # Bayesian fallback
85 feat = self._window_feat(nis_arr)
86 feat_n = (feat - self._x_mean) / self._x_std
87 probs = np.asarray(self.model(feat_n), dtype=float)
88 return self._normalise(probs)
89
90 # ── training ──────────────────────────────────────────────────────────────
91
92 def fit(self, nis_arrays, labels_arrays, seed=None, **kwargs) -> "NeuralRouter":
93 """Extract features, normalise, build model, train model.
94
95 Called by ``InteractiveKalmanFilter.fit()``.
96
97 Parameters
98 ----------
99 nis_arrays : list of ndarray (T_i, M)
100 NIS history per training scenario (from a Bayesian-IKF run).
101 labels_arrays : list of int ndarray (T_i,)
102 True regime 0 … M−1 at every step.
103 seed : int, optional
104 Forwarded to ``model.build()`` if supported.
105 """
106 M = nis_arrays[0].shape[1]
107 d_in = M * self.window
108
109 X_all, y_all = [], []
110 for nis_arr, labels in zip(nis_arrays, labels_arrays):
111 for k, lbl in enumerate(labels):
112 X_all.append(self._window_feat(nis_arr[:k + 1]))
113 y_all.append(int(lbl))
114
115 X = np.array(X_all, dtype=float)
116 y = np.array(y_all, dtype=int)
117
118 self._x_mean = X.mean(0)
119 self._x_std = X.std(0)
120 self._x_std[self._x_std < 1e-8] = 1.0
121 Xn = (X - self._x_mean) / self._x_std
122
123 if hasattr(self.model, "build"):
124 self.model.build(d_in, M, seed=seed)
125 if hasattr(self.model, "fit"):
126 self.model.fit(Xn, y, **kwargs)
127
128 return self

135 def save(self, path):
136 """Store the trained architecture and the feature normalisation at ``path``
137 (a torch pickle), so a notebook can :meth:`load` it instead of training
138 again. For a policy that must leave Python, use :meth:`export_onnx`."""
139 import torch
140 if not self._is_trained:
141 raise RuntimeError("NeuralRouter.save: nothing to save -- call fit() first.")
142 torch.save({"model": self.model, "x_mean": self._x_mean, "x_std": self._x_std,
143 "window": self.window, "arch": type(self.model).__name__}, path)
144 return path

146 def load(self, path):
147 """Restore what :meth:`save` stored. The architecture object is replaced by the
148 saved one; a mismatch with the one this router was built with is refused
149 rather than silently swapped, since the notebook's parameters would then
150 no longer describe what runs."""
151 import torch
152 data = torch.load(path, weights_only=False)
153 if data.get("arch") != type(self.model).__name__:
154 raise ValueError(f"NeuralRouter.load: {path} holds a {data.get('arch')}, this router "
155 f"was built with a {type(self.model).__name__}.")
156 self.model = data["model"]
157 self._x_mean, self._x_std = data["x_mean"], data["x_std"]
158 self.window = int(data.get("window", self.window))
159 return self

161 def export_onnx(self, path, *, opset: int = 17, ir_version: int | None = 9,
162 bake_normalisation: bool = True):
163 """Write the trained policy to ONNX, for execution by ``Kalman::Routers::Neural``.
164
165 This is the seam between the two implementations of the framework. Training
166 happens here, where the data loaders and the optimiser live; inference
167 happens in C++, which then needs neither Python nor Torch.
168
169 The exported graph is **self-contained by default**: the feature
170 standardisation learned in :meth:`fit` and the softmax head are baked in as
171 constants, so the graph maps a raw NIS window straight to probabilities.
172 That is deliberate — a normalisation shipped separately from the weights it
173 belongs to is a silent failure waiting to happen, because the wrong one
174 still produces a valid-looking probability vector.
175
176 A ``<path>.json`` sidecar records the window length and mode count so the
177 C++ side (or a human) can check compatibility rather than discover a
178 mismatch at runtime.
179
180 Parameters
181 ----------
182 path : str
183 Destination ``.onnx`` file.
184 opset : int
185 ONNX opset version.
186 ir_version : int or None
187 Pin the model's IR version after export. Torch emits the newest IR it
188 knows, which *deployed* runtimes routinely reject — ONNX Runtime 1.16
189 refuses anything above 9 with "Unsupported model IR version: 10". The
190 IR version is a container format, not the operator semantics, so
191 lowering it is safe as long as the opset is supported. Pass None to
192 keep whatever Torch produced.
193 bake_normalisation : bool
194 Fold the standardisation into the graph. Turn it off only if the
195 consumer applies its own via ``SetNormalisation``.
196
197 Returns
198 -------
199 dict — the metadata also written to the sidecar.
200 """
201 import json
202 import torch
203
204 net, mean, std, d_in = self._export_prologue("export_onnx")
205 policy = self._policy_module(net, mean, std, bake_normalisation)
206 dummy = torch.zeros(1, d_in, dtype=torch.float32)
207
208 with torch.no_grad():
209 d_out = int(policy(dummy).shape[-1])
210
211 torch.onnx.export(
212 policy, dummy, str(path),
213 input_names=["nis_window"], output_names=["mode_probabilities"],
214 dynamic_axes={"nis_window": {0: "batch"},
215 "mode_probabilities": {0: "batch"}},
216 opset_version=opset,
217 )
218
219 if ir_version is not None:
220 import onnx
221 model = onnx.load(str(path))
222 if model.ir_version > ir_version:
223 model.ir_version = ir_version
224 onnx.save(model, str(path))
225
226 meta = {
227 "format": "onnx",
228 "window": int(self.window),
229 "opset": int(opset),
230 "ir_version": ir_version,
231 "n_models": d_out,
232 "n_features": d_in,
233 "normalisation_baked": bool(bake_normalisation),
234 "feature_layout": "last `window` NIS rows, front-padded with 1.0, flattened row-major",
235 }
236 if not bake_normalisation:
237 meta["mean"] = [float(v) for v in self._x_mean]
238 meta["scale"] = [float(v) for v in self._x_std]
239
240 with open(f"{path}.json", "w") as fh:
241 json.dump(meta, fh, indent=2)
242
243 return meta

245 @staticmethod
246 def _policy_module(net, mean, std, bake):
247 """Standardise → network → softmax, as one traceable module.
248
249 Shared by both export paths so ONNX and TorchScript cannot drift apart in
250 what they consider "the policy".
251 """
252 import copy
253 import torch
254 from torch import nn
255
256 class _Policy(nn.Module):
257 def __init__(self, net, mean, std, bake):
258 super().__init__()
259 self.net = net
260 self.bake = bake
261 self.register_buffer("mean", mean)
262 self.register_buffer("std", std)
263
264 def forward(self, x):
265 if self.bake:
266 x = (x - self.mean) / self.std
267 return torch.softmax(self.net(x), dim=-1)
268
269 # Deep-copy before moving to CPU: `.to()` is in-place for a Module, so
270 # exporting would otherwise strand the caller's model on the wrong device
271 # and every later route() call would fail.
272 return _Policy(copy.deepcopy(net).to("cpu").eval(), mean, std, bake).eval()
273
274 def _export_prologue(self, what):
275 """Checks and tensors both export paths need."""
276 import torch
277
278 if not self._is_trained:
279 raise RuntimeError(
280 f"NeuralRouter.{what}: nothing to export — call fit() "
281 "(or InteractiveKalmanFilter.fit()) first.")
282
283 net = getattr(self.model, "_net", None)
284 if net is None:
285 raise RuntimeError(
286 f"NeuralRouter.{what}: {type(self.model).__name__} exposes no "
287 "torch module (`_net`), so it cannot be traced.")
288
289 return (net,
290 torch.tensor(self._x_mean, dtype=torch.float32),
291 torch.tensor(self._x_std, dtype=torch.float32),
292 int(len(self._x_mean)))

294 def export_torchscript(self, path, *, bake_normalisation: bool = True):
295 """Trace the trained policy to TorchScript, for ``Kalman::Routers::NN::Torch``.
296
297 The counterpart of :meth:`export_onnx`, and the higher-fidelity one: the
298 C++ side runs *this* module rather than a conversion of it, so an
299 architecture the ONNX exporter mangles still deploys. The cost is that the
300 consumer links libTorch instead of ONNX Runtime — roughly 2 GB against 10
301 MB — and is pinned to a compatible Torch version.
302
303 Like :meth:`export_onnx` the traced module is self-contained: the feature
304 standardisation and the softmax head are inside it, so a raw NIS window
305 maps straight to probabilities.
306
307 Parameters
308 ----------
309 path : str
310 Destination ``.pt`` file.
311 bake_normalisation : bool
312 Fold the standardisation into the module.
313
314 Returns
315 -------
316 dict — the metadata also written to ``<path>.json``.
317 """
318 import json
319 import torch
320
321 net, mean, std, d_in = self._export_prologue("export_torchscript")
322 policy = self._policy_module(net, mean, std, bake_normalisation)
323
324 dummy = torch.zeros(1, d_in, dtype=torch.float32)
325 with torch.no_grad():
326 traced = torch.jit.trace(policy, dummy)
327 d_out = int(policy(dummy).shape[-1])
328 traced.save(str(path))
329
330 meta = {
331 "format": "torchscript",
332 "window": int(self.window),
333 "n_models": d_out,
334 "n_features": d_in,
335 "normalisation_baked": bool(bake_normalisation),
336 "feature_layout": "last `window` NIS rows, front-padded with 1.0, flattened row-major",
337 }
338 if not bake_normalisation:
339 meta["mean"] = [float(v) for v in self._x_mean]
340 meta["scale"] = [float(v) for v in self._x_std]
341
342 with open(f"{path}.json", "w") as fh:
343 json.dump(meta, fh, indent=2)
344
345 return meta
