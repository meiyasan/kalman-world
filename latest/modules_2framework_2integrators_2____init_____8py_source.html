1"""
2Concrete time integrators (the *time-stepping* — advance x(t) -> x(t+dt)).
3
4Each shipped integrator tags its
5:class:`~kalman.framework.constants.KalmanIntegratorKind`; ``INTEGRATOR_BY_KIND``
6lets a model resolve its declared ``DEFAULT_INTEGRATOR`` to a concrete class
7without importing an integrator.
8
9The enum is the authority for **shipped** schemes only. A custom stepper does
10not need a member: :func:`register_integrator` binds it to a name, and
11:func:`resolve_integrator` accepts that name, a class, or a ready instance — so
12a downstream package can add a scheme without editing ``constants.py``. An
13unknown kind raises rather than silently degrading to the default.
14"""

41# Custom schemes registered at runtime: lowercase name -> class. Kept apart from
42# INTEGRATOR_BY_KIND so the "registry covers its whole enum" invariant still holds.
43INTEGRATOR_BY_NAME: dict[str, type] = {}

45DEFAULT_INTEGRATOR = RK4Integrator

48def _builtin_names() -> set:
49 return {k.name.lower() for k in INTEGRATOR_BY_KIND}

52def available_integrators() -> list:
53 """Every resolvable integrator name — shipped kinds plus registered ones."""
54 return sorted(_builtin_names() | set(INTEGRATOR_BY_NAME))

57def register_integrator(name, cls):
58 """Register a custom :class:`KalmanIntegrator` subclass under ``name``.
59
60 ``name`` may be a string (a new kind, resolvable everywhere a kind name is
61 accepted — including a ``KalmanSpec``'s ``bindings.integrator``) or a
62 :class:`KalmanIntegratorKind` member, which **rebinds** that shipped kind to
63 your class. Returns ``cls``, so it also works as a decorator.
64 """
65 if not (isinstance(cls, type) and issubclass(cls, KalmanIntegrator)):
66 raise TypeError(f"{cls!r} is not a KalmanIntegrator subclass.")
67 if isinstance(name, KalmanIntegratorKind):
68 INTEGRATOR_BY_KIND[name] = cls
69 return cls
70 key = str(name).strip().lower()
71 if not key:
72 raise ValueError("an integrator name must be a non-empty string.")
73 if key in _builtin_names():
74 raise ValueError(f"{key!r} is a shipped kind; pass KalmanIntegratorKind.{key.upper()} "
75 f"to rebind it deliberately.")
76 INTEGRATOR_BY_NAME[key] = cls
77 return cls
78
79
80def unregister_integrator(name) -> bool:
81 """Drop a custom registration (no-op for shipped kinds). Returns whether it existed."""
82 return INTEGRATOR_BY_NAME.pop(str(name).strip().lower(), None) is not None

85def resolve_integrator(kind=None, **kwargs) -> KalmanIntegrator:
86 """Instantiate the integrator for ``kind``.
87
88 Accepts a :class:`KalmanIntegratorKind`, a registered name, a
89 ``KalmanIntegrator`` subclass, or a ready instance (returned as-is).
90 ``None`` gives the default (RK4). An unrecognised kind raises
91 :class:`ValueError` — it is never silently replaced by the default, because a
92 typo that quietly runs a different scheme is worse than a failure.
93
94 Enum lookup falls back to matching by ``name`` so the registry survives
95 ``KalmanIntegratorKind`` identity changes from ``%autoreload`` or duplicate
96 imports (same discipline as ``resolve_engine``).
97 """
98 if isinstance(kind, KalmanIntegrator):
99 return kind
100 if isinstance(kind, type) and issubclass(kind, KalmanIntegrator):
101 return kind(**kwargs)
102 if kind is None:
103 return DEFAULT_INTEGRATOR(**kwargs)
104
105 if isinstance(kind, str):
106 key = kind.strip().lower()
107 cls = INTEGRATOR_BY_NAME.get(key) or next(
108 (c for k, c in INTEGRATOR_BY_KIND.items() if k.name.lower() == key), None)
109 else:
110 cls = INTEGRATOR_BY_KIND.get(kind)
111 if cls is None: # %autoreload identity fallback
112 name = getattr(kind, "name", None)
113 cls = next((c for k, c in INTEGRATOR_BY_KIND.items()
114 if getattr(k, "name", None) == name), None)
115
116 if cls is None:
117 raise ValueError(f"unknown integrator kind {kind!r}; "
118 f"choose from {available_integrators()} or register_integrator(...).")
119 return cls(**kwargs)
