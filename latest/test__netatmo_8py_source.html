73class FakeNetatmo:
74 """The web API as far as the adapter uses it: a token, then pages of readings."""
75
76 def __init__(self, n=2500):
77 self.calls, self.n = [], n
78
79 def __call__(self, request):
80 url = request.full_url
81 self.calls.append(request)
82 if url.endswith("/oauth2/token"):
83 return io.BytesIO(json.dumps({"access_token": "a", "expires_in": 10800,
84 "refresh_token": "rotated"}).encode())
85 from urllib.parse import parse_qs, urlparse
86 q = {k: v[0] for k, v in parse_qs(urlparse(url).query).items()}
87 assert request.get_header("Authorization") == "Bearer a"
88 times = [t for t in range(0, self.n * 300, 300) if int(q["date_begin"]) <= t <= int(q["date_end"])]
89 times = times[:int(q["limit"])]
90 return io.BytesIO(json.dumps({"status": "ok", "body": {str(t): [400 + t // 300 % 7] for t in times}}).encode())

146class TestFiveModules(unittest.TestCase):
147 """Two days, five modules, four of them off by an unknown offset; the room is
148 empty at night. The offsets are found and the people counted."""
149
150 def test_offsets_and_count(self):
151 warnings.simplefilter("ignore")
152 step, T = 300.0, 2 * 288
153 t = T0 + step * np.arange(T)
154 hour = np.array([datetime.fromtimestamp(x, ZoneInfo(TZ)).hour for x in t])
155 people = np.where((hour >= 9) & (hour < 12), 5, np.where((hour >= 14) & (hour < 18), 3, 0)).astype(float)
156 spots = {"main": (2, 2), "desk": (2, 9), "door": (9, 2), "centre": (5, 5), "window": (4, 1)}
157 offsets = {"main": 0.0, "desk": 60.0, "door": -40.0, "centre": 25.0, "window": 90.0}
158
159 room = RoomCO2Model("main", at=spots["main"], fs=1 / step)
160 truth = room
161 for name in list(spots)[1:]:
162 truth = truth.add_instrument(compose(room.sensor(name, spots[name], sigma=20.0), name=name)
163 .calibrate(1.0, offsets[name]))
164 sim = truth.sample(T * step, uk=np.column_stack([people, np.ones(T)]), seed=3)
165
166 with tempfile.TemporaryDirectory() as d:
167 files = {}
168 for j, name in enumerate(spots):
169 files[name] = os.path.join(d, f"{name}.csv")
170 lost = np.random.default_rng(j).random(T) < 0.05
171 with open(files[name], "w") as f:
172 f.write("Timestamp;Timezone : Europe/Paris;CO2\n")
173 for k in np.flatnonzero(~lost):
174 f.write(f"{int(t[k]) + (k * 37) % 81 - 40};x;{sim.zk[k, j]:.0f}\n")
175 source = NetatmoCSV(files)
176 counting = RoomCO2Model("unread", count_people=True, fs=1 / step)
177 modules = [NetatmoModule(counting, n, spots[n], station="main", module=None if n == "main" else n,
178 source=source) for n in spots]
179 watched = counting
180 for module in modules:
181 watched = watched.add_instrument(module)
182 watched = watched.add_instrument(empty_at_night(counting, tz=TZ))
183 zk = watched.measurements(start=T0, end=T0 + T * step)
184
185 self.assertEqual(zk.shape, (T, 7))
186
187 # the ventilation, from the fall after the last person left at 6 pm of the first day (truth: 2 per hour)
188 left = int(np.flatnonzero(hour == 18)[0])
189 ach, sigma = ventilation_from_decay(t[left:left + 48], zk[left:left + 48, 1:6])
190 self.assertLess(abs(ach - 2.0), max(3 * sigma, 0.1))
191 self.assertLess(sigma, 0.2)
192 estimated = [n for n in spots if n != "main"]
193 run = watched.joint({f"{n}.offset": (60.0, 0.5) for n in estimated}).filter(zk, uk=np.ones((T, 1)), reset=True)
194 n = counting.n
195 for i, name in enumerate(estimated):
196 sigma = np.sqrt(run.Pk[-1, n + i, n + i])
197 self.assertLess(abs(run.xk[-1, n + i] - offsets[name]), max(3 * sigma, 8.0), name)
198 error = run.xk[:, n - 1] - people
199 self.assertLess(np.sqrt(np.mean(error[people > 0] ** 2)), 0.6)
200 self.assertLess(np.sqrt(np.mean(error[people == 0] ** 2)), 0.4)
