28 double VelocityRMSE(const Record &rec, const std::vector<Vector> &truth, std::size_t from)
29 {
30 double sum = 0.;
31 for (std::size_t k = from; k < truth.size(); ++k) sum += std::pow(rec.State(k)(1) - truth[k](1), 2);
32 return std::sqrt(sum / static_cast<double>(truth.size() - from));
33 }
34}

103 // --- the second sensor helps: the velocity is no longer inferred from the angle alone ---
104 // A gyro sharper than the velocity error left by the angle alone (about 0.005 rad/s):
105 // the one above, at 0.2 rad/s, adds too little to show over the noise of a record.
106 // Over several records: std::normal_distribution draws another stream with each
107 // standard library (libstdc++, libc++), so one record is not the same everywhere.
108 {
109 Matrix Rsharp(1, 1, 0.); Rsharp(0, 0) = 1e-6; // 0.001 rad/s
110 Models::Fusion sharp(pendulum, Sensor("gyro", rate, Rsharp, nullptr, {{"rate", "rad/s"}}));
111 const std::size_t N = 1500;
112 double nis = 0., with = 0., without = 0.;
113 const std::vector<std::uint64_t> seeds = {99, 1, 2, 3, 4, 5, 6, 7};
114 for (const std::uint64_t seed : seeds)
115 {
116 Sampler sampler(&sharp, seed);
117 sampler.Generate(N, x0);
118 Filters::EKF both(&sharp);
119 const Record rec = both.Forward(sampler.Measurements());
120 nis += both.MeanNIS() / seeds.size();
121
122 std::vector<Vector> angles;
123 for (const Vector &z : sampler.Measurements()) { Vector a(1); a(0) = z(0); angles.push_back(a); }
124 Filters::EKF alone(pendulum.get());
125 const Record ref = alone.Forward(angles);
126 const double w = VelocityRMSE(rec, sampler.Truth(), 300), wo = VelocityRMSE(ref, sampler.Truth(), 300);
127 std::printf(" seed %2llu: velocity RMSE angle only %.4f, angle + rate %.4f\n",
128 static_cast<unsigned long long>(seed), wo, w);
129 with += w / seeds.size();
130 without += wo / seeds.size();
131 }
132 std::printf(" mean NIS with two channels %.3f; mean velocity RMSE: angle only %.4f, angle + rate %.4f\n",
133 nis, without, with);
134 CHECK(nis > 1.7 && nis < 2.3, "EKF: mean NIS consistent with m = 2");
135 CHECK(with < 0.7 * without, "EKF: the rate channel lowers the velocity error");
136 }
