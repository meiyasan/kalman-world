1"""
2``KalmanSpec`` authoring — the validated pipeline behind :func:`register_model`.
3
4An authored spec goes through: **(1)** structural validation → **(2)** framework
5validation (shapes / symmetric-PSD Q,R / dim-consistency, via instantiation +
6``_check_shapes``) → **(3)** a one-step smoke gate ("does it actually run") →
7**(4)** registration in the shared store (:mod:`.store`), so ``catalog()`` and
8``make_model`` see it in every process on the same state directory, and
9subscribers get a ``model.created`` event → **(5)** optional :func:`save_spec`
10to a ``specs/`` dir (reloadable). Authoring is gated
11by the ``write_models`` scope (:mod:`kalman.api.auth`).
12
13**No code execution**: specs are pure data; only names already in the catalog
14(or the ``matrices``/``continuous`` declarative bases) are accepted; anything
15else raises ``ValueError``.
16"""
17from __future__ import annotations
18import json
19import os
20import numpy as np

22_DECLARATIVE = ("matrices", "continuous")

25def _psd_ok(M, name):
26 M = np.asarray(M, dtype=float)
27 if M.ndim != 2 or M.shape[0] != M.shape[1]:
28 raise ValueError(f"{name} must be square; got shape {M.shape}.")
29 if not np.allclose(M, M.T, atol=1e-8):
30 raise ValueError(f"{name} must be symmetric.")
31 if np.linalg.eigvalsh(0.5 * (M + M.T)).min() < -1e-8:
32 raise ValueError(f"{name} must be positive semi-definite.")

35def _validate_structure(spec: dict):
36 if not isinstance(spec, dict):
37 raise ValueError("spec must be a dict.")
38 base = spec.get("base", "matrices")
39 from . import _model_catalog, _MODELS
40 if base not in _DECLARATIVE and base not in _model_catalog() and base not in _MODELS:
41 raise ValueError(f"unknown base {base!r}; use one of {_DECLARATIVE} or a catalog model.")
42 noise = spec.get("noise", {}) or {}
43 for k in ("Q", "R"):
44 if noise.get(k) is not None:
45 _psd_ok(noise[k], k)
46 # reject obviously non-spec / code-bearing values in the declarative dynamics
47 if base in _DECLARATIVE:
48 for k, v in (spec.get("dynamics", {}) or {}).items():
49 if v is not None:
50 try:
51 np.asarray(v, dtype=float)
52 except (TypeError, ValueError):
53 raise ValueError(f"dynamics[{k!r}] must be numeric arrays, not {type(v).__name__}.")

56def register_model(spec: dict, name: str | None = None, *, replace: bool = False,
57 credential: str | None = None) -> dict:
58 """Validate, smoke-test, and register an authored model spec in the shared
59 registry.
60
61 Returns ``{"name", "n", "m", "l", "owner", "action"}``; ``action`` is
62 ``created``, ``replaced`` or ``unchanged`` (the same spec again is a no-op).
63 A different spec under a taken name needs ``replace=True``, and then the
64 caller must own the entry or be an admin. A name may not shadow a catalog
65 model: everybody who asks for ``constant-velocity`` must get the same one.
66 Requires the ``write_models`` scope when auth is enabled."""
67 from .auth import authorize, identity, require_owner
68 authorize(credential, "write_models")
69
70 _validate_structure(spec)
71 from . import make_model, _model_catalog, _check_name
72 # An unnamed spec is named after its content: the same spec always lands
73 # on the same entry, where a counter would differ from process to process.
74 import hashlib
75 name = name or spec.get("name") or "authored-" + hashlib.sha1(
76 json.dumps(spec, sort_keys=True, default=str).encode()).hexdigest()[:10]
77 _check_name(name, "model")
78 if name in _DECLARATIVE or name in _model_catalog():
79 raise ValueError(f"{name!r} is a catalog model; choose another name for yours.")
80 model = make_model(spec) # framework validation via _check_shapes
81 # smoke gate: the dynamics and observation must actually evaluate
82 x0 = np.asarray(model.x0, dtype=float)
83 _ = model.f(x0)
84 _ = model.h(x0)
85 if model.fs:
86 _ = model.sample(2.0 / model.fs) # a couple of steps
87
88 from . import store
89 info = {"n": int(model.n), "m": int(model.m), "l": int(model.l)}
90 rec, action = store.put_model(
91 name, dict(spec), info, identity(credential), replace=replace,
92 may_replace=lambda r: require_owner(r, credential, f"model {name!r}"))
93 return {"name": name, **info, "owner": rec["owner"], "action": action}

96def save_spec(name: str, directory: str | None = None, *, credential: str | None = None) -> str:
97 """Persist a registered spec as JSON under ``directory`` (default ``./specs``).
98 Reloadable via :func:`load_spec`. Requires ``write_models``."""
99 from .auth import authorize
100 authorize(credential, "write_models")
101 from . import _MODELS
102 if name not in _MODELS:
103 raise KeyError(f"no registered spec named {name!r}; register_model first.")
104 directory = directory or os.path.join(os.getcwd(), "specs")
105 os.makedirs(directory, exist_ok=True)
106 path = os.path.join(directory, f"{name}.json")
107 with open(path, "w") as fh:
108 json.dump(_MODELS[name], fh, indent=2)
109 return path

112def load_spec(path: str, *, register: bool = True, credential: str | None = None) -> dict:
113 """Load a spec JSON from disk; optionally re-register it (``write_models``)."""
114 with open(path) as fh:
115 spec = json.load(fh)
116 if register:
117 register_model(spec, name=os.path.splitext(os.path.basename(path))[0], credential=credential)
118 return spec
