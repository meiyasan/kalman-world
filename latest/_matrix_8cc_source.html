182 Matrix operator*(const Matrix &a, const Matrix &b)
183 {
184 RequireSame(a.Cols(), b.Rows(), "inner dimensions differ in matrix product");
185 Matrix out(a.Rows(), b.Cols(), 0.);
186
187#if KALMAN_WITH_BLAS
188 // Measured crossover on this class of hardware (double, square, hot cache):
189 // the hand-rolled kernel wins below ~6 because a BLAS call costs ~20 ns of
190 // argument handling, and loses badly above it (4x at n=16, 17x at n=32).
191 // Dispatching on size takes the better of the two rather than betting on one.
192 if (a.Rows() >= KALMAN_BLAS_MIN_DIM || a.Cols() >= KALMAN_BLAS_MIN_DIM ||
193 b.Cols() >= KALMAN_BLAS_MIN_DIM)
194 {
195 cblas_dgemm(CblasRowMajor, CblasNoTrans, CblasNoTrans,
196 static_cast<int>(a.Rows()), static_cast<int>(b.Cols()),
197 static_cast<int>(a.Cols()), 1.0,
198 a.Data(), static_cast<int>(a.Cols()),
199 b.Data(), static_cast<int>(b.Cols()), 0.0,
200 out.Data(), static_cast<int>(b.Cols()));
201 return out;
202 }
203#endif
204
205 for (std::size_t i = 0; i < a.Rows(); ++i)
206 for (std::size_t k = 0; k < a.Cols(); ++k)
207 {
208 const double aik = a(i, k);
209 if (aik == 0.) continue;
210 for (std::size_t j = 0; j < b.Cols(); ++j)
211 out(i, j) += aik * b(k, j);
212 }
213 return out;
214 }
