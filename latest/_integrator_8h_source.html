64 template <typename State, typename RHS>
65 State RungeKuttaStep(const Tableau &tableau, const RHS &rhs, const State &x, double t, double dt)
66 {
67 std::vector<State> k;
68 k.reserve(tableau.Stages());
69 for (std::size_t i = 0; i < tableau.Stages(); ++i)
70 {
71 State xi = x;
72 for (std::size_t j = 0; j < i; ++j)
73 if (tableau.a[i][j] != 0.) xi += (dt * tableau.a[i][j]) * k[j];
74 k.push_back(rhs(xi, t + tableau.c[i] * dt));
75 }
76 State out = x;
77 for (std::size_t i = 0; i < tableau.Stages(); ++i)
78 if (tableau.b[i] != 0.) out += (dt * tableau.b[i]) * k[i];
79 return out;
80 }
