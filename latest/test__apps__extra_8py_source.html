341 def _signal(self, fs=100.0, T=8.0, seed=0):
342 t = np.arange(int(T * fs)) / fs
343 a = 1.0 + 0.1 * np.cos(2 * np.pi * 0.2 * t)
344 psi = 2 * np.pi * 1.0 * t
345 theta = (np.pi / 2) * np.ones_like(t)
346 phi = np.pi * np.ones_like(t)
347 z = np.array([jones_signal(a[k], psi[k], theta[k], phi[k]) for k in range(len(t))])
348 zk = np.column_stack([z[:, 0].real, z[:, 0].imag, z[:, 1].real, z[:, 1].imag])
349 rng = np.random.RandomState(seed)
350 zk = zk + 0.05 * rng.randn(*zk.shape)
351 return t, a, psi, theta, phi, zk

379 np.random.seed(0)

383 # Clean odd-harmonic mains: fundamental + a small 3rd harmonic, NO 2nd harmonic.

385 v_noisy = v + 0.5 * np.random.randn(N)

395 self.assertAlmostEqual(h1, peak, delta=0.1 * peak) # fundamental recovered
396 self.assertLess(h2, 0.1 * h1) # absent even harmonic ~ 0
397 self.assertGreater(h3, h2) # present 3rd harmonic above the even
398
399 v_hat = pg.reconstruct_voltage(rec)
400 self.assertEqual(v_hat.shape, (N,))

402 self.assertLess(rms_err, 0.1 * peak) # reconstruction tracks the clean wave

429 def _orbit_step_error(self, substeps):
430 """Position error, m, of one 86.4 s propagation on a circular orbit (r = 5,600 km
431 round GM = 1.06252e14 m^3/s^2, a 2.2 h period) against the exact circle."""
432 G, gm, r, dt = 6.674e-11, 1.06252e14, 5.6e6, 86.4
433 v = np.sqrt(gm / r)
434 m = BallisticModel("XYZ", order=3, fs=1.0 / dt, G=G, M=gm / G, g=None, c_drag=0,
435 substeps=substeps)
436 x = np.array([r, 0, -gm / r**2, 0, 0, 0, 0, v, 0], dtype=float)
437 nxt = m.f(x).reshape(-1)
438 theta = v / r * dt
439 exact = np.array([r * np.cos(theta), 0.0, r * np.sin(theta)])
440 return float(np.linalg.norm(nxt[[0, 3, 6]] - exact))
441
442 def test_orbit_substeps_follow_the_orbit(self):
443 # The model's integrator (RK4) follows a 2.2 h orbit sampled every 86 s to
