46 """
47 States per measurement channel = n // dim.
48 Used as the stride in the state-index formula idx = _spc * axis + comp_id,
49 which is correct for both single models and composite models:
50 - BallisticModel (n=9, m=3): _spc = 3 (pos/vel/acc per axis)
51 - OscillatorComposite (n=2N, m=N): _spc = 2 (pos+vel per oscillator)
52 - AnalyticOscillator (n=2, m=2): _spc = 1 (one state per obs channel)
53 """

252 def _normalize_component(self, component, *, preserve_shape: bool = False, count: int | None = None):
253 """
254 Normalize `component` with ambiguity-safe enum resolution.
255
256 Returns:
257 - preserve_shape=True:
258 * scalar -> Enum (KalmanState or KalmanObservable)
259 * flat list/tuple -> list[Enum]
260 * nested list/tuple -> list[tuple[Enum, ...]] (one nesting level preserved)
261 - preserve_shape=False (default / plotting path):
262 * returns a tuple[Enum, ...] of length `count` (pad by repeating last;
263 truncate if longer). `count` defaults to `self.dim`, but plotters that
264 need a fixed number of components regardless of the sensor count pass it
265 explicitly — e.g. a 2-D phase plot passes ``count=2`` so a single-sensor
266 model (``self.dim == 1``) keeps both phase components instead of
267 truncating to one.
268 """
269 need = max(1, count if count is not None else (self.dimdim or 1))
270 # --- local resolver that yields an Enum or raises ---
271 def _norm_one(v):
272 if isinstance(v, _PyEnum):
273 return v
274 if isinstance(v, str):
275 return self._resolve_name_to_enum(v)
276 if isinstance(v, (int, np.integer)):
277 return self._resolve_index_to_enum(int(v))
278 # anything else is unsupported for components
279 raise TypeError(f"Unsupported component type: {type(v).__name__}")
280
281 # --- scalar path ------------------------------------------------------------
282 if not isinstance(component, (list, tuple, np.ndarray)):
283 member = _norm_one(component)
284 if preserve_shape:
285 return member
286 # flattening mode: repeat to the requested count (defaults to self.dim)
287 return tuple([member] * need)
288
289 # --- sequence path: detect one-level nesting --------------------------------
290 seq = list(component)
291
292 def _is_seq(x):
293 return isinstance(x, (list, tuple, np.ndarray))
294
295 has_nested = any(_is_seq(x) for x in seq)
296
297 if preserve_shape:
298 if not has_nested:
299 # flat -> list[Enum]
300 return [ _norm_one(x) for x in seq ]
301 # nested -> list[tuple[Enum,...]]
302 out = []
303 for grp in seq:
304 if not _is_seq(grp):
305 raise TypeError("Mixed nested/flat component structure is not supported.")
306 out.append(tuple(_norm_one(x) for x in grp))
307 return out
308
309 # flattening mode for plotting: produce tuple of length self.dim
310 flat = []
311 if not has_nested:
312 flat = [ _norm_one(x) for x in seq ]
313 else:
314 # flatten one level (left-to-right) then use pad/truncate
315 for grp in seq:
316 if not _is_seq(grp):
317 raise TypeError("Mixed nested/flat component structure is not supported.")
318 for x in grp:
319 flat.append(_norm_one(x))
320
321 if len(flat) == 0:
322 # fallback: one arbitrary member replicated (but this should rarely happen)
323 return tuple([_norm_one(0)] * max(1, self.dimdim or 1))
324
325 if len(flat) < need:
326 flat += [flat[-1]] * (need - len(flat))
327 elif len(flat) > need:
328 flat = flat[:need]
329 return tuple(flat)

678 # ---------- Arrow helpers ----------
679 def _arrow_indices(self, N: int, arrows: int | None = None) -> list[int]:
680 """
681 Choose ~uniform indices along a curve to place arrows.
682 If `arrows` is None:
683 - use 5 for longer series (N >= 100), else 4.
684 Positions at fractions i/(arrows+1), i = 1..arrows (≈ N/5 or N/4).
685 """
686 if N < 2:
687 return []
688 if arrows is None:
689 arrows = 5 if N >= 100 else 4
690 arrows = max(1, int(arrows))
691 idxs = []
692 for i in range(1, arrows + 1):
693 j = int(round(i * (N - 1) / (arrows + 1)))
694 if 0 <= j < N - 1:
695 idxs.append(j)
696 return sorted(set(idxs))

698 def _smooth_dir_pts(self, arrays, j, w):
699 """
700 Return (tail_coords, head_coords) for an arrow centered at index j.
701 Each coordinate is the mean over a window of width w on each side,
702 which suppresses noise-driven direction flips.
703
704 tail = mean(a[max(0, j-w) : j+1])
705 head = mean(a[j : min(N-1, j+w)+1])
706 """
707 N = len(arrays[0])
708 j0 = max(0, j - w)
709 j1 = min(N - 1, j + w)
710 tail = [float(np.mean(a[j0 : j + 1])) for a in arrays]
711 head = [float(np.mean(a[j : j1 + 1])) for a in arrays]
712 return tail, head

742 def _draw_arrows_3d(self, ax, x, y, z, color='r', arrows=None, lw=1, size=15):
743 """
744 Draw '−|>' arrowheads along a 3D curve using projected FancyArrowPatch.
745 No tails — just visible heads aligned to curve direction.
746 """
747 import numpy as np
748 from mpl_toolkits.mplot3d import proj3d
749 from matplotlib.patches import FancyArrowPatch
750
751 class Arrow3D(FancyArrowPatch):
752 def __init__(self, xs, ys, zs, *args, **kwargs):
753 super().__init__((0, 0), (0, 0), *args, **kwargs)
754 self._verts3d = xs, ys, zs
755
756 def do_3d_projection(self, renderer=None):
757 xs3d, ys3d, zs3d = map(np.ravel, self._verts3d)
758 xs, ys, zs = proj3d.proj_transform(xs3d, ys3d, zs3d, self.axes.M)
759 self.set_positions((xs[0], ys[0]), (xs[1], ys[1]))
760 return np.min(zs)
761
762 x, y, z = np.asarray(x), np.asarray(y), np.asarray(z)
763 N = len(x)
764 idxs = self._arrow_indices(N, arrows=arrows)
765 if not idxs:
766 return
767
768 w = max(1, min(N // 15, 50))
769 for j in idxs:
770 if j + 1 >= N:
771 continue
772 (x0, y0, z0), (x1, y1, z1) = self._smooth_dir_pts((x, y, z), j, w)
773 if np.allclose([x0, y0, z0], [x1, y1, z1]):
774 continue
775 arrow = Arrow3D([x0, x1], [y0, y1], [z0, z1],
776 mutation_scale=size,
777 lw=lw,
778 arrowstyle='-|>',
779 color=color,
780 shrinkA=0, shrinkB=0)
781 ax.add_artist(arrow)

783 def plot(self, **kwargs):
784 """
785 Unified plotting entrypoint using kwargs only.
786
787 kwargs:
788 - view: "auto" | "1d" | "2d" | "3d" (default: "auto")
789 - component: int | Enum | sequence of them (optional)
790 - dim: int | sequence of ints (optional)
791 - ... any other kwargs forwarded to _plot1D/_plot2D/_plot3D
792 """
793 if self.timetime is None or len(self.timetime) < 1:
794 return None, None
795
796 # ----------------------- pull kwargs with defaults ------------------------
797 component = kwargs.pop("component", None)
798 dim = kwargs.pop("dim", None)
799
800 view = kwargs.pop("view", "auto")
801 slider = kwargs.pop("slider", False) or kwargs.pop("interactive", False) or kwargs.pop("replay", False)
802 error = kwargs.pop("error", False)
803
804 dim_explicit = dim is not None
805 is_multi_comp = isinstance(component, (list, tuple)) and len(component) > 1
806
807 def _is_single(x):
808 if x is None: return False
809 return not isinstance(x, (list, tuple)) or (isinstance(x, (list, tuple)) and len(x) == 1)
810
811 if view == "auto" and _is_single(component) and _is_single(dim):
812 view = "1d"
813 # A multi-component request without an explicit axis is a per-component
814 # *dashboard* (one panel per component vs time), not a cross-axis phase
815 # plot: resolve it to 1d so each component indexes its own state slot
816 # (idx = _spc*axis + comp_id). Otherwise auto-filling dim to the available
817 # sensor axes (e.g. [0, 1]) forces a 2d phase plot whose second axis
818 # overflows on models that carry one component per axis (AnalyticOscillator
819 # re/im, _spc=1 → iy = _spc*1 + 1 = 2 on a 2-wide state vector).
820 elif view == "auto" and is_multi_comp and not dim_explicit:
821 view = "1d"
822
823 # ----------------------- small local helpers ------------------------------
824 def _need_by_view(v: str) -> int:
825 if v == "1d": return 1
826 if v == "2d": return 2
827 if v == "3d": return 3
828 if v == "auto": return None
829 raise ValueError(f"view must be 'auto'|'1d'|'2d'|'3d', got {v}")
830
831 def _decide_view(v, dim_len):
832 if v in ("1d","2d","3d"): return v
833 # auto
834 return "3d" if dim_len >= 3 else ("2d" if dim_len == 2 else "1d")
835
836 def _norm_dim(d, dim_available):
837 if d is None:
838 # default to XYZ (trim to available)
839 base = [0, 1, 2]
840 k = 3 if (dim_available or 0) >= 3 else (2 if (dim_available or 0) == 2 else 1)
841 return base[:k]
842 if not isinstance(d, (list, tuple)): d = [int(d)]
843 out = [int(x) for x in d]
844 for x in out:
845 if not (0 <= x < max(1, dim_available or 1)):
846 raise ValueError(f"Axis {x} out of range [0,{max(1, dim_available or 1)-1}]")
847 return out
848
849 # ----------------------- normalize dim/component/view -------------------
850 dim_available = self.dimdim or 1
851 dim = _norm_dim(dim, dim_available)
852 view = _decide_view(view, len(dim))
853 need = _need_by_view(view) or len(dim)
854
855 # if dim shorter than need, extend with 0,1,2…; if longer, trim.
856 # Clamp extra axes to the available spatial axes so a 2D *phase* plot on a
857 # single-sensor model (e.g. position-vs-speed) reuses axis 0 for both
858 # components instead of indexing a non-existent axis.
859 base_axes = [0, 1, 2]
860 if len(dim) < need:
861 extra = [min(a, dim_available - 1) for a in base_axes[len(dim):need]]
862 dim = list(dim) + extra
863 elif len(dim) > need:
864 dim = list(dim[:need])
865
866 # An explicit multi-component 2d/3d plot with no axis given is a *phase /
867 # state-space* portrait: the distinct components share one sensor axis
868 # (idx = _spc*axis + comp_id), so collapse the auto-filled axis list onto
869 # dim[0] rather than spreading components across separate sensor axes
870 # (which overflows on one-component-per-axis models, e.g. _spc=1).
871 if is_multi_comp and not dim_explicit and view in ("2d", "3d"):
872 dim = [dim[0]] * need
873
874 # ----------------------- interactive (time-slider) variant ----------------
875 if slider:
876 sigma = kwargs.pop("sigma", None) # ±sigma·√Pkk band on the 1d panels
877 # multi-panel dashboard: a list of components and/or a measurement-error pad
878 is_list = isinstance(component, (list, tuple)) and len(component) > 1
879 if view != "2d" and (is_list or error):
880 return self._replay(component=list(component) if is_list else [component],
881 dim=dim[0], view="1d", sample=kwargs.get("sample"), error=error, sigma=sigma,
882 jshtml=kwargs.pop("jshtml", False),
883 frames=kwargs.pop("frames", 30), interval=kwargs.pop("interval", 90))
884 if view == "2d" or len(dim) >= 2:
885 cc = list(component) if isinstance(component, (list, tuple)) else [component, component]
886 return self._replay(component=cc[:2], dim=dim[:2], view="2d",
887 sample=kwargs.get("sample"), jshtml=kwargs.pop("jshtml", False),
888 frames=kwargs.pop("frames", 40), interval=kwargs.pop("interval", 80))
889 c = component[0] if isinstance(component, (list, tuple)) else (component if component is not None else 0)
890 return self._replay(component=c, dim=dim[0], view="1d", sigma=sigma,
891 sample=kwargs.get("sample"), jshtml=kwargs.pop("jshtml", False),
892 frames=kwargs.pop("frames", 40), interval=kwargs.pop("interval", 80))
893
894 # `need` is the view's component count (1d→1, 2d→2, 3d→3); pass it so phase
895 # plots keep all components even when the model has fewer sensor axes.
896 component = self._normalize_component(component, count=need)
897
898 # ----------------------- dispatch -----------------------------------------
899 if view == "3d":
900 fig, ax = self._plot3D(component, dim=dim, **kwargs)
901 elif view == "2d":
902 fig, ax = self._plot2D(component, dim=dim, **kwargs)
903 else:
904 fig, ax = self._plot1D(component, dim=dim, **kwargs)
905
906 if ax.get_title() == "":
907 ax.set_title(kwargs.pop("title", self.modelmodel.name() if self.modelmodel else "Generic Kalman Model"))
908
909 return fig, ax

926 estimate as a function of time, optionally overlaying the **truth** and the noisy
927 **measurements** (pass the sampled record as ``sample=``) so you can watch the
928 filter track the data.
929
930 Works both in a plain **Python window** (a GUI backend → ``plt.show()`` plays it)
931 and in **Jupyter** (``jshtml=True`` returns an embedded play/scrub-slider player).
932 It is also reachable as ``record.plot(slider=True, ...)`` with the same
933 ``component`` / ``dim`` / ``sample`` arguments as :meth:`plot`.
934
935 rec = model.filter(z)
936 rec.replay(jshtml=True) # estimate of component 0 vs time

938 rec.replay(component=["position","speed"], view="2d", jshtml=True) # phase plane

940 Parameters

943 State component(s) (same meaning as in :meth:`plot`); a pair is needed for ``view="2d"``.

945 Spatial/sensor axis (or pair of axes for 2D).
946 sample : KalmanRecord | None
947 A sampled record (truth ``xk`` + measurements ``zk``) to overlay.
