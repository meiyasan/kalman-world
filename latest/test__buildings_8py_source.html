91def fall(room, steps=180, start=1200.0):
92 """The room's own fall from ``start`` ppm with nobody in: times [s] and the mean level."""
93 x = np.full((room.n, 1), start)
94 levels = []
95 for k in range(steps):
96 levels.append(float(np.mean(x)))
97 x = room.f(x, u=np.array([[0.0], [1.0]]), t=k * 60.0)
98 return 60.0 * np.arange(steps), np.asarray(levels)

173class TestCountingWithAWindow(unittest.TestCase):
174 """Four people; the window is open for an hour. The count holds when the window
175 is reported, and when it is not but the filter weighs both regimes; it falls
176 when the window is left out."""
177
178 SPOTS = {"B": (2, 9), "C": (9, 2), "D": (5, 5)}
179
180 def build(self, count, opened):
181 room = RoomCO2Model("A", count_people=count,
182 openings=[window("window", [(4, 0), (5, 0)], flow=0.1, open=opened)])
183 model = room
184 for name, at in self.SPOTS.items():
185 model = model.add_sensor(room.sensor(name, at))
186 return room, model
187
188 def test_known_ignored_and_routed(self):
189 warnings.simplefilter("ignore")
190 people = np.where(MINUTE < 30, 0, 4).astype(float)
191 opened = ((MINUTE >= 120) & (MINUTE < 180)).astype(float)
192 _, truth = self.build(False, opened)
193 data = truth.sample(T * 60, uk=np.column_stack([people, np.ones(T)]), seed=2)
194 ones = np.ones((T, 1))
195 while_open = slice(150, 180)
196
197 room, known = self.build(True, opened)
198 count = known.filter(data.zk, uk=ones, reset=True).xk[:, room.n - 1]
199 self.assertAlmostEqual(count[while_open].mean(), 4.0, delta=0.3)
200
201 _, ignored = self.build(True, False)
202 count = ignored.filter(data.zk, uk=ones, reset=True).xk[:, room.n - 1]
203 self.assertLess(count[while_open].mean(), 3.0)
204
205 _, shut = self.build(True, False)
206 _, wide = self.build(True, True)
207 modes = route([shut, wide], np.array([[0.98, 0.02], [0.02, 0.98]]))
208 count = modes.filter(data.zk, uk=ones, reset=True).xk[:, room.n - 1]
209 self.assertAlmostEqual(count[while_open].mean(), 4.0, delta=0.3)
210 is_open = modes.mode_probabilities[:, 1]
211 self.assertGreater(is_open[while_open].mean(), 0.9) # the filter says when it was open
212 self.assertLess(is_open[60:115].mean(), 0.1)
213 self.assertLess(is_open[200:].mean(), 0.1)
