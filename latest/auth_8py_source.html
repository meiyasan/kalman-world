1"""
2Scoped-token auth for the API facade / MCP / REST — fail-closed.
3
4Two layers: a ``.htpasswd`` **account store** (Apache ``name:bcrypt``, with the
5scopes an account may hold in the optional third field that Apache and nginx
6treat as a comment) that authenticates users to *mint* tokens — on the admin's
7machine with the CLI, or by the user over HTTP with :func:`login` — and
8**signed JWT bearer tokens** callers present at runtime carrying fine-grained
9**scopes**:
10
11 - ``read`` — introspection (catalog/describe/diagnostics/plots)
12 - ``send_measurement``— execution (run_filter/forecast/osse/control, fetch)
13 - ``write_models`` — authoring (register_model/save_spec/scaffold)
14 - ``load_plugins`` — loading compiled model libraries (load_plugin); this
15 one runs code, which is why it is not ``write_models``
16 - ``admin`` — everything + account management
17
18Each verb declares its required scope; :func:`authorize` verifies the JWT
19(HS256, ``KALMAN_JWT_SECRET``) and the scope. **Auth disabled** (no secret
20configured) allows in-process Python use (the local user already has the box);
21when a secret *is* set, a valid token is required and anything missing/invalid/
22expired is denied. Optional deps ``pyjwt`` + ``bcrypt`` (``kalman[auth]``),
23lazily imported; secrets are never logged.
24
25Tokens carry ``sub`` (the user), ``scopes``, ``iat``, ``exp`` and ``jti`` (the
26token's own id). A token can be revoked before it expires — one token by its
27``jti``, or every token of a user issued so far (:func:`revoke`); the list is
28in the shared registry, so every server on it refuses the token at once. Set
29``KALMAN_JWT_ISSUER`` and/or ``KALMAN_JWT_AUDIENCE`` to also stamp and require
30``iss``/``aud``, so a token minted for one deployment is refused by another
31that shares the secret. A server that listens on a network interface refuses
32to start without auth (:func:`check_server`).
33
34CLI (``python -m kalman.api.auth``): ``adduser`` / ``token`` / ``verify`` /
35``revoke`` / ``unlock``.
36"""

70def _decode(credential: str) -> dict:
71 """Verify signature, expiry and (when configured) issuer/audience."""
72 try:
73 import jwt
74 except ImportError as e: # fail-closed
75 raise AuthError("auth is enabled but PyJWT is not installed "
76 "(pip install 'kalman[auth]').") from e
77 iss = os.environ.get("KALMAN_JWT_ISSUER") or None
78 aud = os.environ.get("KALMAN_JWT_AUDIENCE") or None
79 try:
80 payload = jwt.decode(credential, _secret(), algorithms=["HS256"],
81 issuer=iss, audience=aud,
82 options={"require": ["exp", "sub"]})
83 except Exception as e: # invalid signature / expired / malformed
84 raise InvalidToken(f"invalid or expired token: {type(e).__name__}.") from e
85 try:
86 from . import store
87 revoked = store.is_revoked(payload.get("jti"), payload.get("sub"), payload.get("iat"))
88 except Exception as e: # fail-closed: no list, no entry
89 raise InvalidToken(f"cannot read the revocation list: {type(e).__name__}.") from e
90 if revoked:
91 raise InvalidToken("this token was revoked.")
92 return payload

118def require_owner(record: dict, credential: str | None, what: str):
119 """Allow a change to a shared record only for its owner or an admin.

137def check_server(host: str, *, service: str, verified_elsewhere: bool = False):
138 """Refuse to serve an open API on a network interface.
139
140 A server bound to loopback is the local user's own; one bound to anything
141 else (``0.0.0.0``, a LAN address, a container port) must authenticate,
142 unless ``KALMAN_ALLOW_ANONYMOUS=1`` says the network itself is trusted.
143 ``verified_elsewhere`` is for the MCP server in front of the REST API: the
144 REST API checks the tokens, so the MCP server needs no secret of its own."""
145 if enabled():
146 if len(_secret().encode()) < MIN_SECRET_BYTES:
147 raise AuthError(f"{service}: KALMAN_JWT_SECRET is shorter than "
148 f"{MIN_SECRET_BYTES} bytes; HS256 needs at least that "
149 "(python -c 'import secrets; print(secrets.token_urlsafe(48))').")
150 return
151 if verified_elsewhere or host in _LOOPBACK:
152 return
153 if os.environ.get("KALMAN_ALLOW_ANONYMOUS") == "1":
154 import sys
155 print(f"WARNING: {service} listens on {host} WITHOUT authentication "
156 "(KALMAN_ALLOW_ANONYMOUS=1). Anybody who can reach it can register "
157 "models and load plugins.", file=sys.stderr)
158 return
159 raise AuthError(f"{service}: refusing to listen on {host} without authentication. "
160 "Set KALMAN_JWT_SECRET (>= 32 bytes) and give clients a token from "
161 "`make api-token`, bind to 127.0.0.1, or set KALMAN_ALLOW_ANONYMOUS=1 "
162 "on a network you trust.")

168def authorize(credential: str | None, scope: str) -> bool:
169 """Authorize ``credential`` (a JWT) for ``scope``; raise :class:`AuthError`
170 if denied. No-op (returns True) when auth is disabled."""
171 if scope not in SCOPES:
172 raise ValueError(f"unknown scope {scope!r} (choose from {SCOPES}).")
173 secret = _secret()
174 if secret is None:
175 return True # auth disabled — local dev
176 if not credential:
177 raise InvalidToken(f"a bearer token with scope '{scope}' is required.")
178 payload = _decode(credential)
179 scopes = set(payload.get("scopes", []))
180 if scope in scopes or "admin" in scopes:
181 return True
182 raise AuthError(f"token lacks required scope '{scope}'.")

188def mint_token(user: str, scopes, *, secret: str | None = None, exp_hours: float = 24.0) -> str:
189 """Sign a scoped JWT for ``user``. ``secret`` defaults to ``KALMAN_JWT_SECRET``."""
190 import jwt
191 secret = secret or _secret()
192 if not secret:
193 raise AuthError("set KALMAN_JWT_SECRET to mint tokens.")
194 scopes = [s.strip() for s in (scopes.split(",") if isinstance(scopes, str) else scopes) if s]
195 bad = [s for s in scopes if s not in SCOPES]
196 if bad:
197 raise ValueError(f"unknown scopes {bad}; choose from {SCOPES}.")
198 import math, uuid
199 t = time.time()
200 now = int(t)
201 # iat to the millisecond (rounded down, so it is never in the future): a
202 # "revoke everything until now" must not catch the token of the login that
203 # follows it in the same second.
204 payload = {"sub": user, "scopes": scopes, "iat": math.floor(t * 1000) / 1000,
205 "exp": now + int(exp_hours * 3600), "jti": uuid.uuid4().hex}
206 if os.environ.get("KALMAN_JWT_ISSUER"):
207 payload["iss"] = os.environ["KALMAN_JWT_ISSUER"]
208 if os.environ.get("KALMAN_JWT_AUDIENCE"):
209 payload["aud"] = os.environ["KALMAN_JWT_AUDIENCE"]
210 return jwt.encode(payload, secret, algorithm="HS256")

236def _accounts(path: str | None = None) -> dict:
237 """``{user: (digest, granted scopes or None)}`` from the account store."""
238 path = path or _htpasswd_path()
239 out = {}
240 if os.path.exists(path):
241 with open(path) as fh:
242 for line in fh.read().splitlines():
243 if ":" not in line or line.lstrip().startswith("#"):
244 continue
245 name, digest, *rest = line.split(":", 2)
246 granted = [s for s in rest[0].split(",") if s] if rest and rest[0].strip() else None
247 out[name] = (digest, granted)
248 return out

251def add_user(user: str, password: str, path: str | None = None, scopes=None) -> str:
252 """Add/update a bcrypt ``.htpasswd`` account; returns the file path.
253
254 Written in the Apache ``user:$2b$…`` format that httpd, nginx and every
255 htpasswd tool understand. Uses the maintained ``bcrypt`` package directly
256 — ``passlib`` is unmaintained and breaks against bcrypt >= 5.
257
258 ``scopes`` is what the account may ask for when it logs in
259 (:func:`login`), kept in the third field, which those tools read as a
260 comment. ``None`` keeps the account's present grant.
261 """
262 bc = _bcrypt()
263 if ":" in user or "\n" in user or not user:
264 raise ValueError("username must be non-empty and must not contain ':' or a newline")
265 path = path or _htpasswd_path()
266 parent = os.path.dirname(path)
267 if parent:
268 os.makedirs(parent, exist_ok=True)
269 digest = bc.hashpw(password.encode()[:_BCRYPT_MAX], bc.gensalt()).decode()
270 granted = _clean_scopes(scopes) if scopes is not None else _accounts(path).get(user, (None, None))[1]
271
272 lines = []
273 if os.path.exists(path):
274 with open(path) as fh:
275 lines = [ln for ln in fh.read().splitlines()
276 if ln.strip() and ln.split(":", 1)[0] != user]
277 lines.append(f"{user}:{digest}" + (f":{','.join(granted)}" if granted else ""))
278 with open(path, "w") as fh:
279 fh.write("\n".join(lines) + "\n")
280 os.chmod(path, 0o600) # credentials file: owner-only
281 return path

284def verify_user(user: str, password: str, path: str | None = None) -> bool:
285 """Check ``password`` against the account store (constant-time via bcrypt)."""
286 bc = _bcrypt()
287 digest = _accounts(path).get(user, (None, None))[0]
288 if not digest:
289 return False
290 # Apache writes $2y$; it is bit-identical to $2b$ for verification.
291 if digest.startswith("$2y$"):
292 digest = "$2b$" + digest[4:]
293 if not digest.startswith("$2"):
294 raise AuthError(f"account {user!r} uses a non-bcrypt hash; "
295 f"re-add it with `make api-user USER={user}`.")
296 try:
297 return bc.checkpw(password.encode()[:_BCRYPT_MAX], digest.encode())
298 except ValueError:
299 return False

318class RateLimited(AuthError):
319 """Too many failed logins; ``retry_after`` seconds until the next try."""
320
321 def __init__(self, retry_after: float):
322 super().__init__(f"too many failed logins; try again in {int(retry_after) + 1} s.")
323 self.retry_after = retry_after

326#: Failed logins allowed per (user, client) inside the window, in seconds.
327LOGIN_ATTEMPTS, LOGIN_WINDOW = 5, 300.0
328_DUMMY: list = []

331def _throttle(user: str, client: str):
332 """Refuse when the pair is locked out. The count is in the shared registry,
333 so it holds across uvicorn workers, across the REST and MCP processes and
334 across a restart -- a counter in memory would give every worker its own
335 five tries and forget them all on a restart."""
336 from . import store
337 fails = store.login_failures(user, client, LOGIN_WINDOW)
338 if len(fails) >= LOGIN_ATTEMPTS:
339 raise RateLimited(LOGIN_WINDOW - (time.time() - fails[0]))

348def login(user: str, password: str, scopes=None, *, exp_hours: float | None = None,
349 client: str = "") -> dict:
350 """Exchange an account's password for a scoped JWT.
351
352 The token carries the scopes asked for, which must all be in the account's
353 grant (an ``admin`` grant covers any); with ``scopes=None`` it carries the
354 whole grant. Its lifetime is at most ``KALMAN_TOKEN_MAX_HOURS`` (default
355 24). Five failed tries per user and client lock that pair out for five
356 minutes -- counted in the shared registry, so on every worker and across a
357 restart -- and an unknown user costs the same bcrypt check as a wrong
358 password, so the answer's timing does not say which accounts exist."""
359 if not enabled():
360 raise ValueError("authentication is off (no KALMAN_JWT_SECRET); no token is needed.")
361 if not isinstance(user, str) or not isinstance(password, str) or not user:
362 raise ValueError("login needs a user and a password.")
363 from . import store
364 _throttle(user, client)
365 ok = verify_user(user, password)
366 if not ok and user not in _accounts():
367 bc = _bcrypt()
368 if not _DUMMY:
369 _DUMMY.append(bc.hashpw(b"no-such-account", bc.gensalt()))
370 bc.checkpw(password.encode()[:_BCRYPT_MAX], _DUMMY[0])
371 if not ok:
372 store.add_login_failure(user, client, LOGIN_WINDOW)
373 raise BadCredentials("wrong user or password.")
374 store.clear_login_failures(user, client)
375
376 granted = user_scopes(user)
377 asked = _clean_scopes(scopes) if scopes else granted
378 if "admin" not in granted:
379 denied = [s for s in asked if s not in granted]
380 if denied:
381 raise AuthError(f"account {user!r} is not granted {denied}; it has {granted}.")
382 if not asked:
383 raise AuthError(f"account {user!r} has no scopes.")
384 cap = float(os.environ.get("KALMAN_TOKEN_MAX_HOURS", "24"))
385 hours = cap if exp_hours is None else max(min(float(exp_hours), cap), 1.0 / 60.0)
386 token = mint_token(user, asked, exp_hours=hours)
387 return {"access_token": token, "token_type": "bearer", "expires_in": int(hours * 3600),
388 "scopes": asked, "user": user}

391def revoke(credential: str | None = None, *, user: str | None = None) -> dict:
392 """Stop a token from working before it expires.
393
394 With no ``user``: the presented token itself (a logout; any valid token may
395 do this). With ``user``: every token of that user issued until now — the
396 user's own, or anybody's for an ``admin``. The list is in the shared
397 registry, so every server on it refuses the token from the next request."""
398 from . import store
399 c = claims(credential)
400 if c is None:
401 raise ValueError("authentication is off; there are no tokens to revoke.")
402 if user is None:
403 if not c.get("jti"):
404 raise ValueError("this token has no id (jti); revoke by user instead.")
405 store.revoke_token(c["jti"], c.get("sub"), c.get("exp"), c.get("sub"))
406 return {"revoked": "token", "user": c.get("sub")}
407 if user != c.get("sub") and "admin" not in c.get("scopes", []):
408 raise AuthError("only an admin may revoke another user's tokens.")
409 cutoff = store.revoke_user(user, c.get("sub"))
410 return {"revoked": "user", "user": user, "before": cutoff}

413def _api_post(base_url: str, path: str, token: str, body: dict) -> dict:
414 """POST JSON to the REST API with a bearer token (stdlib only: the CLI must
415 work where httpx is not installed)."""
416 import json
417 import urllib.error
418 import urllib.request
419 req = urllib.request.Request(
420 base_url.rstrip("/") + path, data=json.dumps(body).encode(), method="POST",
421 headers={"Content-Type": "application/json", "Authorization": f"Bearer {token}"})
422 try:
423 with urllib.request.urlopen(req, timeout=30) as resp:
424 return json.loads(resp.read().decode())
425 except urllib.error.HTTPError as e:
426 detail = e.read().decode(errors="replace")
427 raise RuntimeError(f"HTTP {e.code}: {detail}") from None

433def _main(argv=None):
434 import argparse, getpass, sys
435 p = argparse.ArgumentParser(prog="kalman.api.auth", description="Kalman API auth")
436 sub = p.add_subparsers(dest="cmd", required=True)
437 a = sub.add_parser("adduser"); a.add_argument("--user", required=True); a.add_argument("--htpasswd")
438 a.add_argument("--scopes", help="what the account may ask for at login, e.g. read,send_measurement")
439 r = sub.add_parser("revoke", help="revoke one token, or every token of a user issued so far")
440 r.add_argument("--token"); r.add_argument("--user")
441 r.add_argument("--api", default=os.environ.get("KALMAN_API_URL"),
442 help="the REST API whose registry holds the list (default: $KALMAN_API_URL; "
443 "without it, the registry in $KALMAN_STATE_DIR on this machine)")
444 u = sub.add_parser("unlock", help="forget an account's failed logins"); u.add_argument("--user", required=True)
445 u.add_argument("--api", default=os.environ.get("KALMAN_API_URL"), help="as for revoke")
446 t = sub.add_parser("token"); t.add_argument("--user", required=True)
447 t.add_argument("--scopes", default="read"); t.add_argument("--exp", type=float, default=24.0)
448 t.add_argument("--no-verify", action="store_true", help="mint without checking the htpasswd account")
449 v = sub.add_parser("verify"); v.add_argument("--token", required=True); v.add_argument("--scope", default="read")
450 args = p.parse_args(argv)
451
452 if args.cmd == "adduser":
453 pw = getpass.getpass(f"password for {args.user}: ")
454 path = add_user(args.user, pw, args.htpasswd, scopes=args.scopes or None)
455 print(f"account '{args.user}' written to {path} "
456 f"(login scopes: {','.join(user_scopes(args.user, path))})")
457 elif args.cmd == "unlock" and args.api:
458 try:
459 _api_post(args.api, "/token/unlock", mint_token("cli", ["admin"], exp_hours=1 / 60),
460 {"user": args.user})
461 except Exception as e:
462 print(f"unlock through {args.api} failed: {e}", file=sys.stderr); return 1
463 print(f"failed logins of '{args.user}' forgotten through {args.api}")
464 elif args.cmd == "unlock":
465 unlock(args.user)
466 print(f"failed logins of '{args.user}' forgotten")
467 elif args.cmd == "revoke" and args.api:
468 # The servers' registry is not on this machine (a container volume, another
469 # host): ask the API. A user's tokens need an admin, so mint one that lives
470 # a minute with the secret this machine has; one token revokes itself.
471 if not (args.user or args.token):
472 print("give --token or --user", file=sys.stderr); return 1
473 try:
474 bearer = args.token if not args.user else mint_token("cli", ["admin"], exp_hours=1 / 60)
475 out = _api_post(args.api, "/token/revoke", bearer, {"user": args.user} if args.user else {})
476 except Exception as e:
477 print(f"revoke through {args.api} failed: {e}", file=sys.stderr); return 1
478 print(f"revoked through {args.api}: {out}")
479 elif args.cmd == "revoke":
480 # The list is in the registry: this writes $KALMAN_STATE_DIR on this machine.
481 from . import store
482 if args.user:
483 store.revoke_user(args.user, "cli")
484 print(f"every token of '{args.user}' issued until now is revoked")
485 elif args.token:
486 c = claims(args.token)
487 if not c or not c.get("jti"):
488 print("this token has no id (jti); use --user", file=sys.stderr); return 1
489 store.revoke_token(c["jti"], c.get("sub"), c.get("exp"), "cli")
490 print(f"token {c['jti']} of '{c.get('sub')}' is revoked")
491 else:
492 print("give --token or --user", file=sys.stderr); return 1
493 elif args.cmd == "token":
494 if not args.no_verify:
495 pw = getpass.getpass(f"password for {args.user}: ")
496 if not verify_user(args.user, pw):
497 print("authentication failed", file=sys.stderr); return 1
498 print(mint_token(args.user, args.scopes, exp_hours=args.exp))
499 elif args.cmd == "verify":
500 try:
501 authorize(args.token, args.scope); print("OK")
502 except AuthError as e:
503 print(f"DENIED: {e}", file=sys.stderr); return 1
504 return 0
