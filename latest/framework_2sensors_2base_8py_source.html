46class NativeSensor(KalmanSensor):
47 """A :class:`KalmanSensor` whose readings, Jacobian and noise are a C++ sensor's
48 (``Kalman::SensorFactory``): nothing is computed in Python.
49
50 ``kind`` and ``numbers`` are what the C++ kind is built from; a subclass turns
51 its own arguments into them and documents what it reads. What does not fit is
52 refused by the C++ sensor, and its message is raised as a ``ValueError``.
53 """
54
55 def __init__(self, kind, name, n, numbers, *, labels=None, rate=None, delay=None):
56 lib, ffi = core.require(), core.ffi
57 numbers = np.ascontiguousarray(numbers, dtype=np.float64).reshape(-1)
58 handle = lib.kalman_sensor_create(kind.encode(), str(name).encode(), int(n), core.cdptr(numbers), numbers.size)
59 if handle == ffi.NULL:
60 raise ValueError(core.last_error())
61 handle = ffi.gc(handle, lib.kalman_sensor_free)
62 self.n = int(n)
63 k = core.check(lib.kalman_sensor_channels(handle), "sensor_channels")
64 R = np.zeros((k, k))
65 core.check(lib.kalman_sensor_R(handle, core.dptr(R)), "sensor_R")
66 G = np.zeros((k, self.n))
67 if core.check(lib.kalman_sensor_G(handle, core.dptr(G)), "sensor_G"):
68 super().__init__(name, G=G, R=R, labels=labels, rate=rate, delay=delay)
69 else:
70 super().__init__(name, self._read_read, R=R, jacobian=self._jacobian, labels=labels, rate=rate, delay=delay)
71 self.nativenative = handle
72 self._k = k
73
74 def _state(self, x):
75 x = np.ascontiguousarray(x, dtype=np.float64).reshape(-1)
76 if x.size != self.n:
77 raise ValueError(f"{type(self).__name__} {self.name!r} reads a state of {self.n} components, got {x.size}.")
78 return x

80 def _read(self, x, t=0.0):
81 z = np.zeros(self._k)
82 core.check(core.lib.kalman_sensor_read(self.nativenative, core.cdptr(self._state(x)), float(t or 0.0), core.dptr(z)),
83 "sensor_read")
84 return z

86 def _jacobian(self, x, t=0.0):
87 J = np.zeros((self._k, self.n))
88 core.check(core.lib.kalman_sensor_jacobian(self.nativenative, core.cdptr(self._state(x)), float(t or 0.0),
89 core.dptr(J)), "sensor_jacobian")
