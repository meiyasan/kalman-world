1"""The model operations, one after the other, on a cart.
2
3Each operation takes models and gives back a model; the filter is always the
4ordinary one. Everything is computed by the C++ library.
5

40def cart(name="cart", q=1.0):
41 """Position and velocity; a noisy position sensor."""
42 Q = q * np.array([[DT**4 / 4, DT**3 / 2], [DT**3 / 2, DT**2]])
43 return KalmanModel(name, fs=FS, F=np.array([[1, DT], [0, 1]]), H=np.array([[1.0, 0.0]]), Q=Q,
44 R=np.array([[0.25]]), x0=np.array([[0.0], [1.0]]), P0=np.eye(2))

47def simulate(model, steps, n=2):
48 """Truth and readings from the model's own f and h (noise on the first ``n`` components)."""
49 Lq = np.linalg.cholesky(model.Q()[:n, :n] + 1e-15 * np.eye(n))
50 Lr = np.linalg.cholesky(model.R())
51 x, xs, zs = np.asarray(model.x0, dtype=float).copy(), [], []
52 for _ in range(steps):
53 x = model.f(x)
54 x[:n] += Lq @ rng.standard_normal((n, 1))
55 xs.append(x.ravel().copy())
56 zs.append((model.h(x) + Lr @ rng.standard_normal((model.m, 1))).ravel())
57 return np.array(xs), np.array(zs)

60def states(record, steps):
61 return np.asarray(record.xk).reshape(steps, -1)

66# --- add_sensor, then lag: a precise sensor that reports 0.3 s late -------------------------
67camera = KalmanSensor("camera", G=[[1.0, 0.0]], R=0.01, delay=0.3)
68late = cart().add_sensor(camera).lag(3)
69truth, zk = simulate(late, N)
70with_camera = states(late.filter(zk, reset=True), N)
71without = states(cart().filter(zk[:, :1], reset=True), N)
72print("add_sensor + lag: position error %.3f with the late camera, %.3f without"
73 % (rmse(with_camera[:, 0], truth[:, 0]), rmse(without[:, 0], truth[:, 0])))
74print(" the state three steps ago is known to %.3f (fixed-lag smoothing)"
75 % rmse(with_camera[:, late.block(3)][:, 0], truth[:, 6]))

77# --- joint: the offset of an uncalibrated thermometer-like sensor ------------------------------
78probe = KalmanSensor("probe", G=[[1.0, 0.0]], R=0.04)

94# --- route: a cart that is quiet, then shaken ---------------------------------------------------------
95quiet, shaken = cart("quiet", q=0.01), cart("shaken", q=100.0)
96_, z_quiet = simulate(cart("quiet", q=0.01), N // 2)
97restart = cart("shaken", q=100.0)
98restart.initial_state(np.array([[z_quiet[-1, 0]], [1.0]]), np.eye(2))
99_, z_shaken = simulate(restart, N // 2)
100Pi = np.array([[0.98, 0.02], [0.02, 0.98]])
101modes = route([quiet, shaken], Pi)
102modes.filter(np.vstack([z_quiet, z_shaken]))
103mu = modes.mode_probabilities[:, 1]
104print("route: probability of the shaken regime: %.2f in the first half, %.2f in the second"
105 % (mu[: N // 2].mean(), mu[N // 2:].mean()))
