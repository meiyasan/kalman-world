55 // Largest difference between the estimates of two models on the same data.
56 double Apart(Model &a, Model &b, std::size_t N = 60)
57 {
58 const std::vector<Vector> zk = Measurements(N);
59 Filter fa(&a), fb(&b);
60 const IO::Record ra = fa.Forward(zk), rb = fb.Forward(zk);
61 double worst = 0.;
62 for (std::size_t k = 0; k < N; ++k)
63 for (std::size_t i = 0; i < a.NState(); ++i)
64 {
65 worst = std::max(worst, std::fabs(ra.State(k)(i) - rb.State(k)(i)));
66 worst = std::max(worst, std::fabs(ra.Covariance(k)(i, i) - rb.Covariance(k)(i, i)));
67 }
68 return worst;
69 }
