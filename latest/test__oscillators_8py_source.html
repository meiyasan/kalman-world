262 def test_violin_mode_suppression(self):
263 """End-to-end KAGRA-style check: window-bank subtraction suppresses each
264 injected mode by at least 10 dB at its own frequency bin."""
265 np.random.seed(0)
266 f0v, Bv, nv = 180.1, 0.105, 3
267 fsv, Tv = 4096.0, 4.0
268 Nv = int(Tv * fsv)
269 tv = np.arange(Nv) / fsv
270 sigma_noise = 5e-19
271 Qs = [1.3e4, 1.8e4, 2.1e4]
272 A = [1.0e-18, 3.0e-19, 1.5e-19]
273 freqs = partial_frequencies(f0v, Bv, nv)
274 strain = sigma_noise * np.random.randn(Nv)
275 for a, f, Q in zip(A, freqs, Qs):
276 strain += a * np.exp(-tv / (Q / (np.pi * f))) * np.cos(2 * np.pi * f * tv)
277
278 windows = [OscillatorModel("KAGRA", freqs, fc=f, dfc=5.0, Q=Q, fs=fsv,
279 measurement_uncertainty=40 * sigma_noise,
280 process_uncertainty=1e-21)
281 for f, Q in zip(freqs, Qs)]
282 bank = OscillatorBankModel(*windows)
283 for w, a in zip(bank.windows, A):
284 w.initial_state([[a, 0.0]])
285 bank.initial_state(np.vstack([w.x0 for w in bank.windows]))
286 rec = bank.filter(strain, smooth=True)
287 cleaned = strain - bank.reconstruct(rec)
288
289 from numpy.fft import rfft, rfftfreq
290 fr = rfftfreq(Nv, 1 / fsv)
291 Sr = np.abs(rfft(strain))
292 Sc = np.abs(rfft(cleaned))
293 for fk in freqs:
294 mask = (fr > fk - 3) & (fr < fk + 3)
295 suppression_db = 20 * np.log10(Sr[mask].max() / Sc[mask].max())
296 self.assertGreater(suppression_db, 10.0,
297 f"mode @ {fk:.1f} Hz suppressed only {suppression_db:.1f} dB")
