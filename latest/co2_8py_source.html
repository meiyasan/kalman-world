1"""CO2 in a room: a mass balance on a grid of cells.
2
3The room is a set of cells of a rectangular grid (an L-shaped room is the grid
4without one quarter). In each cell the CO2 concentration ``c`` [ppm] follows
5
6 dc/dt = D lap(c) - lambda (c - c_out) + S / V
7
8- ``D lap(c)``: mixing between neighbouring cells of the room (no flow through
9 the walls: a cell exchanges only with its neighbours inside the room);
10- ``lambda (c - c_out)``: ventilation, ``lambda`` the air changes per second,
11 bringing outdoor air at ``c_out``;
12- ``S / V``: the CO2 the occupants breathe out, spread over the seats.
13
14The equation is linear, so the model is a model of matrices: ``F`` and ``B``
15are its exact discretization over one step (matrix exponential), and it runs on
16the linear engine. The known inputs ``u = [occupants, 1]`` are the number of
17people in the room and a constant 1 that carries the outdoor air.
18
19With ``count_people=True`` the number of occupants is not given but estimated:
20it becomes the last component of the state, a number that drifts slowly
21(people come and go), and ``u = [1]``. The model stays linear.
22
23Openings change the balance while they are open (:class:`Opening`): a window
24exchanges the air of its cells with the outdoors, a door with the space beyond
25it, and an air conditioner brings fresh air to the whole room, or only stirs
26it. An opening that is always open or always shut leaves the model a model of
27matrices; one that follows a known schedule (a contact sensor, the unit's own
28signal) makes ``F`` and ``B`` functions of time, each the exact discretization
29of the balance in that configuration. An opening nobody reports is two models,
30shut and open, under ``route``.
31
32The model's own sensor reads one cell. Other sensors are added with the
33framework's operations; a sensor that drifts is an instrument with a
34calibration, whose offset ``joint`` estimates with the field.
35"""

45class Opening:
46 """Something that changes the air of the room while it is open: a window, a
47 door, an air conditioner.
48
49 Parameters
50 ----------
51 name : what it is called.
52 cells : the cells it acts on, ``(row, column)`` each; ``None`` for the whole room.
53 flow : the air it exchanges while open [m^3/s], shared by its cells.
54 ach : that exchange as air changes per hour of the whole room, in place of ``flow``.
55 level : the CO2 of the air that comes in [ppm]; ``None`` for the outdoor level.
56 mixing : what it adds to the room's mixing coefficient while open [m^2/s].
57 open : ``True`` or ``False`` for always; or how open it is at each step, an
58 array (0 shut, 1 open, one value per step of the model) or a function of
59 the time [s].

120class RoomCO2Model(LinearKalmanModel):
121 """CO2 concentration in every cell of a room.
122
123 Parameters
124 ----------
125 sensor : the name of the model's own CO2 sensor.
126 at : ``(row, column)`` of the cell that sensor reads.
127 room : boolean grid, True for the cells of the room; an L by default.
128 cell : the side of a cell [m]; height : the height of the room [m].
129 mixing : the mixing coefficient D [m^2/s].
130 ach : ventilation, in air changes per hour.
131 outdoor : the outdoor concentration [ppm].
132 seats : the cells where the occupants sit; their breath is spread over them.
133 emission : the CO2 one person breathes out [L/s].
134 sigma : the noise of the model's own sensor [ppm].
135 process_uncertainty : what the balance leaves out (a door opened, a person
136 moving), as a standard deviation per cell and per step [ppm].
137 count_people : estimate the number of occupants as the last state component
138 instead of taking it as an input.
139 people_drift : how much that number may change in a step, a standard
140 deviation [people]; used with ``count_people``.
141 openings : the windows, doors and air conditioners of the room
142 (:class:`Opening`, or :func:`window`, :func:`door`, :func:`air_conditioner`).
143 fs : steps per second; one step a minute by default.
144 """
145
146 NAME = "Room CO2 Model"
147 DESCRIPTION = "CO2 mass balance on the cells of a room: mixing, ventilation and occupants."
148 VERSION = "1.0"

150 def __init__(self, sensor: str = "co2", at=(2, 2), *, room=None, cell: float = 0.5, height: float = 2.5,
151 mixing: float = 0.02, ach: float = 2.0, outdoor: float = 420.0, seats=None,
152 emission: float = 0.005, sigma: float = 15.0, process_uncertainty: float = 2.0,
153 count_people: bool = False, people_drift: float = 0.3, openings=(), fs: float = 1 / 60, **kwargs):
154 self._made_with = dict(sensor=sensor, at=at, room=room, cell=cell, height=height, mixing=mixing, ach=ach,
155 outdoor=outdoor, seats=seats, emission=emission, sigma=sigma,
156 process_uncertainty=process_uncertainty, count_people=count_people,
157 people_drift=people_drift, openings=tuple(openings), fs=fs, **kwargs)
158 self.room = l_shaped_room() if room is None else np.asarray(room, dtype=bool)
159 rows, cols = self.room.shape
160 self.cells = [(r, c) for r in range(rows) for c in range(cols) if self.room[r, c]]
161 index = {rc: i for i, rc in enumerate(self.cells)}
162 self.index = index
163 cells_n = len(self.cells)
164 self.count_people = bool(count_people)
165 n = cells_n + (1 if self.count_people else 0)
166 self.cell, self.height, self.outdoor = float(cell), float(height), float(outdoor)
167 seats = [(1, 1), (1, 2), (2, 1), (2, 2)] if seats is None else [tuple(s) for s in seats]
168 for s in [tuple(at), *seats]:
169 if s not in index:
170 raise ValueError(f"RoomCO2Model: the cell {s} is not in the room.")
171 self.seats = seats
172 self.openings = list(openings)
173 names = [o.name for o in self.openings]
174 if len(set(names)) != len(names):
175 raise ValueError(f"RoomCO2Model: two openings of one name ({names}).")
176
177 # The continuous balance dc/dt = A c + Bc u, u = [occupants, 1].
178 lam = float(ach) / 3600.0
179 volume = self.cell * self.cell * self.height # of one cell [m^3]
180 neighbours = np.zeros((cells_n, cells_n)) # the exchange between cells, per unit k
181 for (r, c), i in index.items():
182 for nb in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
183 j = index.get(nb)
184 if j is not None: # a wall exchanges nothing
185 neighbours[i, i] -= 1.0
186 neighbours[i, j] += 1.0
187 per_person = float(emission) * 1e-3 / volume * 1e6 # ppm/s in one cell
188 breath = np.zeros(n)
189 for s in seats:
190 breath[index[s]] += per_person / len(seats)
191 rates = [] # per opening: its exchange in each cell [1/s], fully open
192 for o in self.openings:
193 rate = np.zeros(cells_n)
194 if o.cells is None:
195 rate[:] = o.ach / 3600.0 if o.ach is not None else o.flow / (volume * cells_n)
196 else:
197 for c in o.cells:
198 if c not in index:
199 raise ValueError(f"RoomCO2Model: the cell {c} of {o.name!r} is not in the room.")
200 per_cell = o.ach / 3600.0 * cells_n if o.ach is not None else o.flow / volume
201 rate[[index[c] for c in o.cells]] = per_cell / len(o.cells)
202 rates.append(rate)
203 dt = 1.0 / float(fs)
204 made = {}
205
206 def matrices(states):
207 """``F`` and ``B`` over one step with each opening as open as ``states`` says: the
208 exact discretization, the inputs held, exp([[A, Bc], [0, 0]] dt)."""
209 if states not in made:
210 exchange = np.full(cells_n, lam)
211 incoming = np.full(cells_n, lam * self.outdoor)
212 k = float(mixing)
213 for o, rate, s in zip(self.openings, rates, states):
214 exchange = exchange + s * rate
215 incoming = incoming + s * rate * (self.outdoor if o.level is None else o.level)
216 k += s * o.mixing
217 A = np.zeros((n, n))
218 A[:cells_n, :cells_n] = k / self.cell ** 2 * neighbours - np.diag(exchange)
219 outdoor_air = np.zeros(n)
220 outdoor_air[:cells_n] = incoming
221 if self.count_people:
222 A[:, cells_n] = breath # the occupants are the last state
223 Bc = outdoor_air.reshape(n, 1)
224 else:
225 Bc = np.column_stack([breath, outdoor_air])
226 width = Bc.shape[1]
227 block = np.zeros((n + width, n + width))
228 block[:n, :n], block[:n, n:] = A, Bc
229 E = expm(block * dt)
230 made[states] = (E[:n, :n], E[:n, n:])
231 return made[states]
232
233 fs_ = float(fs)
234 states_at = lambda t: tuple(o.state(t, fs_) for o in self.openings)
235 if any(o.scheduled for o in self.openings): # a known schedule: F and B follow it
236 F, B = (lambda t=None: matrices(states_at(t))[0]), (lambda t=None: matrices(states_at(t))[1])
237 else:
238 F, B = matrices(states_at(None))
239 H = np.zeros((1, n))
240 H[0, index[tuple(at)]] = 1.0
241 q = np.full(n, float(process_uncertainty) ** 2)
242 x0 = np.full((n, 1), self.outdoor)
243 P0 = np.full(n, 50.0 ** 2)
244 if self.count_people:
245 q[-1] = float(people_drift) ** 2
246 x0[-1], P0[-1] = 0.0, 10.0 ** 2
247 kwargs.setdefault("x0", x0)
248 kwargs.setdefault("P0", np.diag(P0))
249 super().__init__(sensor, F=F, B=B, H=H, Q=np.diag(q), R=np.array([[float(sigma) ** 2]]), fs=fs, **kwargs)
250 self.at = tuple(at)

271 def schedule(self, name="schedule", sigma=0.1, *, rate=None):
272 """What is known of the number of occupants, as a sensor that reads it.

293def empty_at_night(room, hours=(0, 6), *, tz=None, sigma=0.1, name="schedule"):
294 """The room is empty between two hours of the night: an instrument made of the
295 room's ``schedule`` sensor, whose readings are 0 people at those steps and NaN
296 at the others.
297
298 ``empty_at_night(room)`` is added to the room model; its readings
299 come with the others from ``model.measurements(start=..., end=...)`` [unix s].
300 ``hours=(0, 6)``: from midnight to 6 am, in the time zone ``tz`` (the
301 computer's when ``None``).
302 """
303 from datetime import datetime
304 from zoneinfo import ZoneInfo
305 from ...framework.instrument import KalmanInstrument
306 first, last = (float(h) for h in hours)
307 step = 1.0 / float(room.fs)
308
309 def readings(*, start, end, **query):
310 zone = ZoneInfo(tz) if tz else None
311 n = max(int(np.floor((float(end) - float(start)) / step)), 0)
312 out = np.full((n, 1), np.nan)
313 for k in range(n):
314 moment = datetime.fromtimestamp(float(start) + k * step, zone)
315 hour = moment.hour + moment.minute / 60.0
316 if (first <= hour < last) if first <= last else (hour >= first or hour < last):
317 out[k, 0] = 0.0
318 return out
319
320 return KalmanInstrument(name, [room.schedule(name, sigma=sigma)], provider=readings)

323def ventilation_from_decay(times, ppm):
324 """The ventilation of a room [air changes per hour] from the fall of its CO2
325 once everybody has left, and its standard deviation.
326
327 With nobody breathing, the balance leaves ``c(t) = a + b exp(-lambda t)``:
328 the rate of the fall is the ventilation. The level ``a`` it falls to is
329 fitted too, so a sensor's offset and the outdoor level do not enter.
330
331 ``times`` [s] and ``ppm`` are the readings of the fall only (from the moment
332 the room empties until the level is flat, or the next person comes in);
333 ``ppm`` holds one column per sensor, each fitted alone and the rates
334 averaged by their uncertainties. NaN readings are left out.
335 """
336 from scipy.optimize import curve_fit
337 t = np.asarray(times, dtype=float).reshape(-1)
338 c = np.asarray(ppm, dtype=float).reshape(len(t), -1)
339 t = t - t[0]
340 fall = lambda t, a, b, lam: a + b * np.exp(-lam * t)
341 rates, variances = [], []
342 for column in c.T:
343 read = np.isfinite(column)
344 if read.sum() < 6:
345 continue
346 tr, cr = t[read], column[read]
347 start = (cr[-3:].mean(), max(cr[:3].mean() - cr[-3:].mean(), 1.0), 3.0 / max(tr[-1], 1.0))
348 try:
349 best, cov = curve_fit(fall, tr, cr, p0=start, bounds=([-np.inf, 0.0, 0.0], np.inf))
350 except RuntimeError:
351 continue
352 if np.isfinite(cov[2, 2]) and cov[2, 2] > 0.0 and best[2] > 0.0:
353 rates.append(best[2])
354 variances.append(cov[2, 2])
355 if not rates:
356 raise ValueError("ventilation_from_decay: no fall of the CO2 could be fitted; give the readings from the "
357 "moment the room empties, at least six per sensor.")
358 weights = 1.0 / np.asarray(variances)
359 rate = float(np.sum(weights * np.asarray(rates)) / np.sum(weights))
360 return rate * 3600.0, float(np.sqrt(1.0 / np.sum(weights))) * 3600.0
