93 std::vector<double> Record::Sigma(std::size_t i) const

115 double Record::RMSE(std::size_t i, const std::vector<Vector> &truth) const
116 {
117 if (truth.size() != NSteps())
118 throw std::runtime_error("Kalman::Record::RMSE: truth must have one state per step.");
119
120 double sum = 0.;
121 for (std::size_t k = 0; k < NSteps(); ++k)
122 {
123 const double d = fState[k](i) - truth[k](i);
124 sum += d * d;
125 }
126 return NSteps() ? std::sqrt(sum / static_cast<double>(NSteps())) : 0.;
127 }
