45 Vector Controller::Command(const Vector &xHat, const Matrix &P, double t, const Vector *nu)
46 {
47 Vector u = Policy(xHat, P, t, nu);
48 if (u.Size() != NControl())
49 throw std::runtime_error("Kalman::Controller: the policy returned " + std::to_string(u.Size())
50 + " commands, expected " + std::to_string(NControl()) + ".");
51
52 Update(xHat, P, t, nu);
53
54 // Cost against what the actuator delivers, not the raw ask (see the header).
55 const Vector applied = fModel->ClipControl(u);
56 Vector e = xHat;
57 e -= fXRef;
58 double cost = 0.;
59 for (std::size_t i = 0; i < e.Size(); ++i)
60 for (std::size_t j = 0; j < e.Size(); ++j) cost += e(i) * fQCost(i, j) * e(j);
61 for (std::size_t i = 0; i < applied.Size(); ++i)
62 for (std::size_t j = 0; j < applied.Size(); ++j) cost += applied(i) * fRCost(i, j) * applied(j);
63
64 fCostLog.emplace_back(t, cost);
65 fControlLog.emplace_back(t, u);
66 return u;
67 }
