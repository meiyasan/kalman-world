104def moliere_radius_mm(material: str) -> float:
105 """Estimate Molière radius [mm]: R_M ≈ 21.2 MeV · X₀ / Ec."""
106 mat = MATERIALS.get(material, {})
107 return 0.0212 * mat.get("X0_mm", 100.) / _EC_GEV.get(material, 0.040)

112@dataclass
113class CaloLayer:
114 """One slab in a calorimeter stack."""
115 material: str # key in MATERIALS
116 thickness_mm: float # along z
117 is_active: bool # True = sensitive / readout
118 cell_size_mm: float = 30.0 # transverse cell/pad pitch [mm]
119 sigma_sampling: Optional[float] = None # override material default
120
121 @property
122 def X0_mm(self) -> float:
123 return MATERIALS[self.material]["X0_mm"]
124
125 @property
126 def lambda_int_mm(self) -> float:
127 return MATERIALS[self.material]["lambda_int_mm"]
128
129 @property
130 def t_X0(self) -> float:
131 return self.thickness_mm / self.X0_mm
132
133 @property
134 def t_lambda(self) -> float:
135 return self.thickness_mm / self.lambda_int_mm
136
137 @property
138 def sigma_s(self) -> float:
139 if self.sigma_sampling is not None:
140 return self.sigma_sampling
141 return MATERIALS[self.material].get("sigma_s") or 0.10
142
143 @property
144 def pos_resolution_mm(self) -> float:
145 """Intrinsic position resolution from cell pitch: sigma = cell/sqrt(12)."""
146 return self.cell_size_mm / (12 ** 0.5)

149class CalorimeterGeometry:
150 """
151 Calorimeter geometry defined by a layer stack.
152
153 Parameters
154 ----------
155 layers : list of CaloLayer
156 z_start : float Upstream face z [mm].
157 aperture_mm : float Transverse full-width [mm].
158 """
159
160 def __init__(self, layers: List[CaloLayer],
161 z_start: float = 0., aperture_mm: float = 200.):
162 self._layers = list(layers)
163 self._z_start = float(z_start)
164 self.aperture_mm = float(aperture_mm)
165 self._build()
166
167 # ── internal build ────────────────────────────────────────────────────────
168
169 def _build(self):
170 z = self._z_start
171 cum_X0 = 0.; cum_lam = 0.
172 z_iface = [z]
173
174 z_act, dz_act = [], []
175 t_X0_act, dt_X0_act = [], []
176 t_lam_act, dt_lam_act = [], []
177 act_layers = []
178
179 for layer in self._layers:
180 dX0 = layer.t_X0
181 dlam = layer.t_lambda
182 if layer.is_active:
183 z_ctr = z + 0.5 * layer.thickness_mm
184 z_act.append(z_ctr)
185 dz_act.append(layer.thickness_mm)
186 t_X0_act.append(cum_X0 + 0.5 * dX0)
187 dt_X0_act.append(dX0)
188 t_lam_act.append(cum_lam + 0.5 * dlam)
189 dt_lam_act.append(dlam)
190 act_layers.append(layer)
191 cum_X0 += dX0
192 cum_lam += dlam
193 z += layer.thickness_mm
194 z_iface.append(z)
195
196 self._z_interfaces = np.asarray(z_iface)
197 self.z_active = np.asarray(z_act)
198 self.dz_active = np.asarray(dz_act)
199 self.t_X0_active = np.asarray(t_X0_act)
200 self.dt_X0_active = np.asarray(dt_X0_act)
201 self.t_lam_active = np.asarray(t_lam_act)
202 self.dt_lam_active = np.asarray(dt_lam_act)
203 self._active_layers = act_layers
204 self.total_X0 = cum_X0
205 self.total_lambda = cum_lam
206 self.total_length_mm = z - self._z_start
207
208 # ── derived physics ───────────────────────────────────────────────────────
209
210 @property
211 def n_active(self) -> int:
212 return len(self.z_active)
213
214 @property
215 def X0_eff_mm(self) -> float:
216 """mm per radiation length (for shower profile scaling)."""
217 return self.total_length_mm / max(self.total_X0, 1e-9)
218
219 @property
220 def lambda_eff_mm(self) -> float:
221 """mm per interaction length."""
222 return self.total_length_mm / max(self.total_lambda, 1e-9)
223
224 @property
225 def moliere_mm(self) -> float:
226 """Effective Molière radius of the whole stack [mm].
227
228 A shower spreads in every layer it crosses, absorber included, so the layers
229 combine like the mass fractions of a mixture, ``1/R_M = Σ w_i / R_M,i``. With
230 lengths in mm this is the thickness-weighted harmonic mean over all layers.
231 """
232 if not self._layers:
233 return 20.
234 total = sum(l.thickness_mm for l in self._layers)
235 return float(total / sum(l.thickness_mm / moliere_radius_mm(l.material)
236 for l in self._layers))
237
238 @property
239 def sigma_t_em_mm(self) -> float:
240 """1-σ transverse EM spread [mm] from 2D-Gaussian 90% containment at R_Molière."""
241 return self.moliere_mm / np.sqrt(2. * np.log(10.)) # = R_M / 2.145
242
243 @property
244 def sigma_t_had_mm(self) -> float:
245 """1-σ transverse hadronic spread [mm]. 90% lat. containment ≈ 0.5 λ_int."""
246 return self.lambda_eff_mm * 0.5 / np.sqrt(2. * np.log(10.))
247
248 @property
249 def hadronic_spread_mm(self) -> float:
250 return self.sigma_t_had_mm
251
252 @property
253 def sigma_sampling_em(self) -> float:
254 if not self._active_layers:
255 return 0.020
256 return float(np.mean([l.sigma_s for l in self._active_layers]))
257
258 @property
259 def sigma_sampling_had(self) -> float:
260 # Use HCAL_PRESETS reference (~50%/√E for Fe/Scint) as floor;
261 # the active-layer sigma_s is typically set for EM showers.
262 return max(self.sigma_sampling_emsigma_sampling_em, 0.50)
263

265 return (
266 f"CalorimeterGeometry\n"
267 f" Layers : {len(self._layers)} total, {self.n_active} active\n"
268 f" z range: {self._z_start:.0f} – {self._z_start + self.total_length_mm:.0f} mm\n"
269 f" Depth : {self.total_X0:.1f} X₀, {self.total_lambda:.2f} λ_int\n"

344 def plot_xy(self, ax=None, z_slice_mm: Optional[float] = None,
345 hits: Optional[List[dict]] = None):
346 """
347 Transverse (x–y) view at *z_slice_mm*.
348 If *hits* is provided the cells fired by those hits are filled
349 proportionally to energy (hot colormap). Empty cells drawn as outlines.
350 """
351 import matplotlib.pyplot as plt
352 import matplotlib.patches as mpatches
353 import matplotlib.colors as mcolors
354 if ax is None:
355 _, ax = plt.subplots(figsize=(5, 5))
356
357 half = self.aperture_mm * 0.5
358
359 if z_slice_mm is None and self.n_active > 0:
360 z_slice_mm = self.z_active[self.n_active // 2]
361
362 # find layer at z_slice
363 active_layer = None
364 if z_slice_mm is not None:
365 z = self._z_start
366 for layer in self._layers:
367 if z <= z_slice_mm < z + layer.thickness_mm:
368 active_layer = layer
369 break
370 z += layer.thickness_mm
371
372 mat = active_layer.material if active_layer else "air"
373 color = MATERIAL_COLORS.get(mat, "#4FC3F7")
374 label = MATERIALS.get(mat, {}).get("label", "")
375
376 # aperture outline
377 ax.add_patch(mpatches.Rectangle(
378 (-half, -half), 2 * half, 2 * half,
379 facecolor=color, alpha=0.12, edgecolor="k", linewidth=1.5))
380
381 if active_layer:
382 cs = active_layer.cell_size_mm
383 xs_c = np.arange(-half, half, cs) + cs / 2
384 ys_c = np.arange(-half, half, cs) + cs / 2
385
386 # build hit energy map keyed by (snapped_x, snapped_y)
387 hit_map: dict = {}
388 if hits:
389 z_vals = np.array([h["z"] for h in hits])
390 # pick hits at this z layer (within ±half thickness)
391 tol = (active_layer.thickness_mm / 2 + 1.) if active_layer else 5.
392 for h in hits:
393 if abs(h["z"] - z_slice_mm) <= tol:
394 sx = float(np.round((h["x"] - xs_c[0]) / cs) * cs + xs_c[0])
395 sy = float(np.round((h["y"] - ys_c[0]) / cs) * cs + ys_c[0])
396 hit_map[(sx, sy)] = hit_map.get((sx, sy), 0.) + h["E"]
397
398 E_max = max(hit_map.values()) if hit_map else 1.
399 cmap = plt.cm.hot_r
400
401 for xc in xs_c:
402 for yc in ys_c:
403 key = (float(xc), float(yc))
404 E = hit_map.get(key, 0.)
405 fc = cmap(E / E_max) if E > 0 else "none"
406 ec = color if E > 0 else "lightgray"
407 lw = 0.8 if E > 0 else 0.25
408 ax.add_patch(mpatches.Rectangle(
409 (xc - cs/2, yc - cs/2), cs, cs,
410 facecolor=fc, edgecolor=ec, linewidth=lw, zorder=3 if E > 0 else 1))
411
412 if hit_map:
413 sm = plt.cm.ScalarMappable(cmap=cmap,
414 norm=mcolors.Normalize(0, E_max))
415 plt.colorbar(sm, ax=ax, label="E [GeV]", fraction=0.046, pad=0.04)
416
417 z_tag = f"z = {z_slice_mm:.0f} mm" if z_slice_mm is not None else ""
418 ax.set_xlim(-half * 1.05, half * 1.05)
419 ax.set_ylim(-half * 1.05, half * 1.05)
420 ax.set_xlabel("x [mm]"); ax.set_ylabel("y [mm]")
421 ax.set_title(f"x–y Pad Map ({label}, {z_tag})")
422 ax.set_aspect("equal")
423 return ax

425 def plot_hits_projections(self, hits: List[dict], particle: str = "",
426 smoothed_states: Optional[list] = None,
427 z_slice_mm: Optional[float] = None):
428 """
429 Three-panel hit visualization:
430 Left : z–x projection — energy aggregated by (z, x) column
431 Middle: z–y projection — energy aggregated by (z, y) row
432 Right : x–y pad map — cells colored by energy at z_slice layer
433
434 Energy at each (z, pad-row/column) is summed across all hits in that
435 row/column before plotting, so overlapping hits from the same z layer
436 are correctly represented as one marker per row/column.
437 Marker area is proportional to deposited energy.
438
439 Parameters
440 ----------
441 hits : list of {x, y, z, E} from CalorimeterSampler.sample()
442 particle : label string for the title
443 smoothed_states : list from engine.smooth — overlays the KF track
444 z_slice_mm : z for the XY pad map (default: shower-maximum layer)
445 """
446 import matplotlib.pyplot as plt
447 import matplotlib.colors as mcolors
448 from collections import defaultdict
449 if not hits:
450 raise ValueError("hits list is empty")
451
452 zs = np.array([h["z"] for h in hits])
453 xs = np.array([h["x"] for h in hits])
454 ys = np.array([h["y"] for h in hits])
455 es = np.array([h["E"] for h in hits])
456
457 # default slice = layer with max total energy
458 if z_slice_mm is None:
459 E_by_z = defaultdict(float)
460 for h in hits:
461 E_by_z[h["z"]] += h["E"]
462 z_slice_mm = max(E_by_z, key=E_by_z.get)
463
464 # ── aggregate energy by (z, x-column) and (z, y-row) ─────────────────
465 agg_zx: dict = defaultdict(float) # (z, x) → E
466 agg_zy: dict = defaultdict(float) # (z, y) → E
467 for h in hits:
468 agg_zx[(h["z"], h["x"])] += h["E"]
469 agg_zy[(h["z"], h["y"])] += h["E"]
470
471 def _unpack(agg):
472 keys = list(agg.keys())
473 zz = np.array([k[0] for k in keys])
474 cc = np.array([k[1] for k in keys])
475 ee = np.array([agg[k] for k in keys])
476 return zz, cc, ee
477
478 zz_x, xx, ee_x = _unpack(agg_zx)
479 zz_y, yy, ee_y = _unpack(agg_zy)
480
481 E_max = max(ee_x.max(), ee_y.max(), 1e-9)
482 norm = mcolors.Normalize(vmin=0, vmax=E_max)
483 cmap = plt.cm.hot_r
484
485 # marker size proportional to energy (area ~ E)
486 def _sz(ee):
487 return np.clip(20 + 280 * ee / E_max, 8, 300)
488
489 half = self.aperture_mm * 0.5
490 z_lo = self._z_start - 5
491 z_hi = self._z_start + self.total_length_mm + 5
492
493 fig, axes = plt.subplots(1, 3, figsize=(15, 4.5))
494
495 # ── z–x ──────────────────────────────────────────────────────────────
496 ax_zx = axes[0]
497 sc = ax_zx.scatter(zz_x, xx, c=ee_x, cmap=cmap, norm=norm,
498 s=_sz(ee_x), edgecolors="k", lw=0.3, zorder=4)
499 if smoothed_states is not None:
500 z_sm = self.z_active[:len(smoothed_states)]
501 x_sm = np.array([np.asarray(s[0]).ravel()[0] for s in smoothed_states])
502 ax_zx.plot(z_sm, x_sm, "b-o", ms=4, lw=1.8, label="KF track", zorder=5)
503 ax_zx.legend(fontsize=8)
504 ax_zx.axvline(z_slice_mm, color="lime", lw=1.2, ls="--")
505 ax_zx.set_xlabel("z [mm]"); ax_zx.set_ylabel("x [mm]")
506 ax_zx.set_xlim(z_lo, z_hi)
507 ax_zx.set_ylim(-half * 1.05, half * 1.05)
508 ax_zx.set_title("z–x projection (Σ E per column)")
509 plt.colorbar(sc, ax=ax_zx, label="E [GeV]", fraction=0.046, pad=0.04)
510
511 # ── z–y ──────────────────────────────────────────────────────────────
512 ax_zy = axes[1]
513 sc2 = ax_zy.scatter(zz_y, yy, c=ee_y, cmap=cmap, norm=norm,
514 s=_sz(ee_y), edgecolors="k", lw=0.3, zorder=4)
515 if smoothed_states is not None:
516 y_sm = np.array([np.asarray(s[0]).ravel()[1] for s in smoothed_states])
517 ax_zy.plot(z_sm, y_sm, "b-o", ms=4, lw=1.8, zorder=5)
518 ax_zy.axvline(z_slice_mm, color="lime", lw=1.2, ls="--")
519 ax_zy.set_xlabel("z [mm]"); ax_zy.set_ylabel("y [mm]")
520 ax_zy.set_xlim(z_lo, z_hi)
521 ax_zy.set_ylim(-half * 1.05, half * 1.05)
522 ax_zy.set_title("z–y projection (Σ E per row)")
523 plt.colorbar(sc2, ax=ax_zy, label="E [GeV]", fraction=0.046, pad=0.04)
524
525 # ── x–y pad map ───────────────────────────────────────────────────────
526 self.plot_xy(ax=axes[2], z_slice_mm=z_slice_mm, hits=hits)
527
528 ptitle = f"{particle} — " if particle else ""
529 fig.suptitle(f"{ptitle}Hit Projections (z–x | z–y | x–y pad map)",
530 fontsize=12, fontweight="bold")
531 plt.tight_layout()
532 return fig, axes
