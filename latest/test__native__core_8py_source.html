33def _prototypes(text: str) -> set:
34 text = re.sub(r"/\*.*?\*/", "", text, flags=re.S) # block comments

39class TestHeaderContract(unittest.TestCase):
40 """``native/core.py``'s cdef is a copy of ``CAPI.h``; they must not drift."""
41
42 def test_cdef_matches_header(self):
43 with open(HEADER) as fh:
44 header = _prototypes(fh.read())
45 cdef = _prototypes(core._CDEF)
46 self.assertTrue(header, "no prototypes parsed from CAPI.h")
47 self.assertEqual(header - cdef, set(), "in CAPI.h but missing from core._CDEF")
48 self.assertEqual(cdef - header, set(), "in core._CDEF but not in CAPI.h")
49
50 def test_info_struct_fields_match(self):
51 with open(HEADER) as fh:
52 header = fh.read()
53 def fields(text):
54 body = re.search(r"typedef struct kalman_model_info\s*\{(.*?)\}", text, re.S).group(1)
55 body = re.sub(r"/\*.*?\*/", "", body, flags=re.S)
56 return [ln.strip().rstrip(";").split()[-1].lstrip("*") for ln in body.splitlines() if ln.strip()]
57 self.assertEqual(fields(header), fields(core._CDEF))

175 def test_nonlinear_model_uses_numerical_jacobian(self):
176 p = self.NativeModel("pendulum")

179 F = p.F(x)
180 self.assertEqual(F.shape, (2, 2))
181 # d(theta_next)/d(omega) ~ dt for a small step; d(omega_next)/d(theta) ~ -omega^2 cos(theta) dt
