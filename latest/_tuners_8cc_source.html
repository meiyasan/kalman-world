39 void Boost::Update(const Vector &xHat, const Matrix &P, double t, const Vector *nu)
40 {
41 if (!nu) return;
42 if (fRNominal.Empty()) fRNominal = fModel->R(t);
43
44 const std::vector<std::size_t> keep = Utils::ValidChannels(*nu);
45 if (keep.empty()) return;
46
47 const Matrix H = Utils::CompressRows(fModel->JacobianH(xHat, t), keep);
48 Matrix S = Utils::Similarity(H, P);
49 S += Utils::CompressBoth(fModel->R(t), keep);
50 fNis.push_back(Utils::Mahalanobis(Utils::Compress(*nu, keep), Utils::Inverse(S)));
51
52 const std::size_t first = fNis.size() > fWindow ? fNis.size() - fWindow : 0;
53 double mean = 0.;
54 for (std::size_t i = first; i < fNis.size(); ++i) mean += fNis[i];
55 mean /= static_cast<double>(fNis.size() - first);
56
57 const double scale = mean > fThreshold * static_cast<double>(keep.size()) ? 1. / fBoostFactor : 1.;
58 Matrix R = fRNominal;
59 R *= scale;
60 fModel->SetR(R);
61 fScaleLog.emplace_back(t, scale);
62 }

81 void AdaptiveQR::Update(const Vector &xHat, const Matrix &P, double t, const Vector *nu)
82 {
83 if (!nu) return;
84 if (fQNominal.Empty()) fQNominal = fModel->Q(t);
85 if (!Utils::IsComplete(*nu)) return; // a partial innovation has no full outer product
86
87 fInnovations.push_back(*nu);
88 if (fInnovations.size() < fWindow) return;
89
90 const std::size_t m = nu->Size();
91 Matrix empirical(m, m, 0.);
92 for (std::size_t s = fInnovations.size() - fWindow; s < fInnovations.size(); ++s)
93 for (std::size_t i = 0; i < m; ++i)
94 for (std::size_t j = 0; j < m; ++j) empirical(i, j) += fInnovations[s](i) * fInnovations[s](j);
95 empirical *= 1. / static_cast<double>(fWindow);
96
97 Matrix S = Utils::Similarity(fModel->JacobianH(xHat, t), P);
98 S += fModel->R(t);
99
100 const double lambda = std::min(fQMax, std::max(fQMin, Trace(empirical) / std::max(Trace(S), 1e-12)));
101 Matrix Q = fQNominal;
102 Q *= lambda;
103 fModel->SetQ(Q);
104 fScaleLog.emplace_back(t, lambda);
105 }
