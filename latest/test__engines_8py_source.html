36def _assert_valid_cov(P, n, label="P"):
37 P = np.asarray(P)
38 assert P.shape == (n, n), f"{label} shape {P.shape} != ({n},{n})"
39 assert not np.any(np.isnan(P)), f"{label} contains NaN"
40 assert not np.any(np.isinf(P)), f"{label} contains Inf"
41 eigv = np.linalg.eigvalsh(P)
42 assert np.all(eigv > -1e-8), f"{label} not PSD (min eig {eigv.min():.2e})"

61 xp, _, _ = self.e._predict(self.m, x, P0, t=0.0)

65 def test_predict_with_control(self):
66 m = LinearKalmanModel("cv_ctrl", F=F_CV, H=H_POS, Q=Q0, R=R0, fs=10,
67 B=np.array([[0.0], [1.0]]))

74 def test_predict_covariance_grows(self):
75 _, Pp, _ = self.e._predict(self.m, X0, P0, t=0.0)
76 # Prior covariance must be larger than initial (process noise added)
77 self.assertGreater(np.trace(Pp), np.trace(P0))
78
79 def test_update_reduces_uncertainty(self):
80 xp, Pp, _ = self.e._predict(self.m, X0, P0, t=0.0)
81 z = np.array([[0.5]])
82 x_post, P_post = self.e._update(self.m, xp, Pp, z, t=0.0)
83 self.assertEqual(x_post.shape, (2, 1))
84 _assert_valid_cov(P_post, 2, "P_post")
85 self.assertLess(P_post[0, 0], Pp[0, 0]) # position variance reduced
86
87 def test_update_moves_towards_measurement(self):
88 x_prior = np.array([[3.0], [0.0]]) # predicted position = 3
89 P_prior = np.eye(2) * 10 # high uncertainty
90 z = np.array([[0.0]]) # observed position = 0
91 x_post, _ = self.e._update(self.m, x_prior, P_prior, z, t=0.0)
92 # posterior should be pulled toward z=0
93 self.assertLess(x_post[0, 0], x_prior[0, 0])
94
95 def test_nan_measurement_skipped(self):

97 xp, Pp, _ = self.e._predict(self.m, X0, P0, t=0.0)
98 z_nan = np.array([[float("nan")]])

100 np.testing.assert_array_equal(x_post, xp)

111class _NonlinearModel(ExtendedKalmanModel):
112 """Angle-only observation (nonlinear h)."""
113
114 def __init__(self):
115 self.n, self.m, self.l = 2, 1, 0
116 super().__init__("angle", Q=Q0, R=np.array([[0.01]]), fs=10)
117
118 def f(self, x, u=None, w=None, t=None):
119 return F_CV @ x
120
121 def h(self, x, v=None, t=None):
122 angle = np.arctan2(float(x[1, 0]), float(x[0, 0]))
123 if v is not None:
124 angle += float(np.asarray(v).reshape(-1)[0])
125 return np.array([[angle]])

266class TestCrossEngineFullStep(unittest.TestCase):
267 """All deterministic engines produce the same result on a linear/Gaussian model."""
268
269 def _run_engine(self, engine):
270 m = _cv_model()
271 x = X0.copy()
272 P = P0.copy()
273 zk = [np.array([[k * DT + 0.01]]) for k in range(20)]
274 ests = []
275 for z in zk:
276 x, P, _ = engine._predict(m, x, P, t=0.0)
277 x, P = engine._update(m, x, P, z, t=0.0)
278 ests.append(float(x[0, 0]))
279 return np.array(ests)
280
281 def test_linear_ekf_ckf_agree(self):
282 xk_lin = self._run_engine(LinearKalmanEngine())
283 xk_ek = self._run_engine(ExtendedKalmanEngine())
284 xk_ck = self._run_engine(CombinatorialKalmanEngine())
285 np.testing.assert_allclose(xk_lin, xk_ek, atol=1e-8)
286 np.testing.assert_allclose(xk_lin, xk_ck, atol=1e-8)
287
288 def test_ukf_close_to_kf(self):
289 xk_lin = self._run_engine(LinearKalmanEngine())
290 xk_uk = self._run_engine(UnscentedKalmanEngine())
