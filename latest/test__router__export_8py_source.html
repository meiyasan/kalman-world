1"""The ONNX export contract: a router trained here must reproduce itself elsewhere.
2
3``NeuralRouter.export_onnx()`` is the seam between the Python framework and
4libKalman: Python owns training, C++ owns inference, and a *trained
5policy* — not a language binding — is what crosses between them. That only works
6if the exported graph computes exactly what ``router.route()`` computes, so this
7suite pins the three things that can silently break it:
8
9* the feature layout (window, front-padding, row-major flatten),
10* the standardisation, which is baked into the graph so it cannot be shipped
11 separately from the weights it belongs to,
12* the IR version, which deployed runtimes reject when it is too new.
13
14The equivalence is checked with the ``onnxruntime`` Python package, which is the
15same engine ``Kalman::Routers::Neural`` uses in C++.
16"""

34def _training_set(n_scenarios=12, T=100, M=2, seed=0):
35 """Labelled NIS histories where the correct model sits at chi2(1) and the other far above."""
36 rng = np.random.default_rng(seed)
37 nis_arrays, label_arrays = [], []
38 for _ in range(n_scenarios):
39 labels = (np.arange(T) >= T // 2).astype(int)
40 nis = np.empty((T, M))
41 for k, lbl in enumerate(labels):
42 nis[k, lbl] = rng.chisquare(1)
43 nis[k, 1 - lbl] = rng.chisquare(1) * 8 + 5
44 nis_arrays.append(nis)
45 label_arrays.append(labels)
46 return nis_arrays, label_arrays

91 def test_ir_version_is_pinned_for_deployed_runtimes(self):
92 # Torch emits the newest IR it knows; ONNX Runtime 1.16 rejects anything

112 def test_graph_reproduces_the_python_router(self):
113 """The whole point: same NIS history in, same probabilities out."""
114 import tempfile
115 import onnxruntime as ort
116
117 c, likelihoods = np.array([0.5, 0.5]), np.ones(2)
118
119 with tempfile.TemporaryDirectory() as tmp:
120 path, _ = self._export(tmp)
121 session = ort.InferenceSession(path, providers=["CPUExecutionProvider"])
122 name = session.get_inputs()[0].name
123
124 worst = 0.0
125 # Short histories exercise the front-padding, long ones the window.
126 for nis in (self.nis_arrays[0][:1], self.nis_arrays[0][:3],
127 self.nis_arrays[0][:40], self.nis_arrays[3][:100]):
128 expected = self.router.route(nis, c, likelihoods)
129
130 # Reproduce the C++ feature extraction: raw window, unnormalised,
131 # because the standardisation is inside the graph.
132 block = nis[-self.window:]
133 if len(block) < self.window:
134 block = np.vstack([np.ones((self.window - len(block), nis.shape[1])), block])
135 feat = block.ravel().astype(np.float32)[None, :]
136
137 got = session.run(None, {name: feat})[0].ravel()
138 worst = max(worst, float(np.max(np.abs(got - expected))))
139
140 self.assertLess(worst, 1e-5, f"graph and router disagree by {worst:.2e}")
