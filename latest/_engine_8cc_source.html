31 std::size_t Engine::Update(const Model &model, const Vector &x, const Matrix &P,
32 const Vector &z, double t,
33 Vector &xOut, Matrix &POut) const
34 {
35 xOut = x;
36 POut = P;
37
38 const std::vector<std::size_t> keep = Utils::ValidChannels(z);
39 if (keep.empty()) return 0; // predict-only step
40
41 Vector zpred;
42 Matrix H;
43 Measure(model, x, t, zpred, H);
44
45 // Residual against the true h(x) — not H x — so the same code is correct
46 // for a nonlinear observation map.
47 Vector y(z);
48 y -= zpred;
49
50 Utils::Correct(Utils::CompressRows(H, keep), Utils::CompressBoth(model.R(t), keep),
51 Utils::Compress(y, keep), x, P, xOut, POut);
52 return keep.size();
53 }
