21 double MaxAbsDiff(const Field &a, const Field &b)
22 {
23 double d = 0.;
24 for (std::size_t i = 0; i < a.Size(); ++i) d = std::max(d, std::abs(a.Data()[i] - b.Data()[i]));

28 // Direct O(n^2) DFT, the reference the fast transforms are checked against.
29 std::vector<cd> DirectDFT(const std::vector<cd> &x)
30 {
31 const std::size_t n = x.size();
32 std::vector<cd> out(n);
33 for (std::size_t k = 0; k < n; ++k)
34 for (std::size_t j = 0; j < n; ++j)
35 out[k] += x[j] * std::exp(cd(0., -2. * kPi * static_cast<double>(k * j) / static_cast<double>(n)));
36 return out;
37 }

39 Field SinCos(std::size_t n, double L)
40 {
41 Field f(n, n);
42 for (std::size_t i = 0; i < n; ++i)
43 for (std::size_t j = 0; j < n; ++j)
44 f(i, j) = std::sin(L * j / n) * std::cos(L * i / n); // sin(x) cos(y), x along columns
45 return f;
46 }
