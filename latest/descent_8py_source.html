1"""Rocket descent — re-entry and guided descent (autopilot)."""
2from __future__ import annotations
3"""
4Rocket science — getting off the pad and back: **takeoff** and **landing**.
5
6This module is the *near-surface, powered-flight* half of the rocketry models (no third
7bodies). The space-travel half — flying to the Moon and interplanetary transfers — lives in
8:mod:`kalman.apps.space`.
9
10The "rocket equation" family (powered flight where what differs is the **phase**):
11
12* :class:`RocketAscentModel` — a planar **guided ascent**: gravity versus **thrust supplied
13 as the control input** ``u`` (constant mass). Demonstrates control channels (``l>0``).
14 *Scaled units.* Legacy alias: ``RocketLaunchModel``.
15* :class:`RocketLiftoffModel` — a 1-D **vertical liftoff** whose thrust is internal and whose
16 **mass burns down** (Tsiolkovsky): acceleration rises until **burnout**. *SI units.*
17 Legacy alias: ``RocketTakeoffModel``.
18* :class:`RocketLandingModel` — the same 1-D mass-burning dynamics run in reverse: a
19 SpaceX-style **hoverslam**, free-falling then firing a closed-loop landing burn to touch
20 down near zero velocity. *SI units.*
21
22All are :class:`ExtendedKalmanModel` subclasses whose ``f()`` advances the state by one
234th-order Runge–Kutta step; the EKF linearises ``f`` with a numerical Jacobian.
24
25The ascent model uses *scaled units* (``length = 1 Mm = 1000 km``, ``time = 1 ks = 1000 s``)
26to stay well-conditioned; the near-surface liftoff/landing models use plain **SI** (m, s, kg).
27Physical constants come from :mod:`kalman.runtime.astro` (astropy-sourced where available).
28"""

42def _rocket_shape(x, y, th, length, width, *, flame=0.0, nose=None, anchor="base"):
43 """Corner arrays for a rocket drawn at ``(x, y)`` tilted ``th`` rad from vertical.
44
45 Returns a dict of ``(N, 2)`` point arrays — ``body`` (rectangle), ``nose`` (triangle) and,
46 when ``flame > 0``, a ``flame`` triangle out of the base. The body axis (nose direction)
47 maps to ``(−sin θ, cos θ)``, so positive ``θ`` leans the vehicle toward +x. ``anchor="base"``
48 (default) plants the **base** of the vehicle at ``(x, y)`` so it stands on the ground/pad at
49 that point (the physical offset from the centre of mass); ``anchor="com"`` centres it instead.
50 """
51 hw = 0.5 * width
52 nose = 1.5 * hw if nose is None else nose
53 y0, y1 = (0.0, length) if anchor == "base" else (-0.5 * length, 0.5 * length)
54 c, s = np.cos(th), np.sin(th)
55 R = np.array([[c, -s], [s, c]])
56 def world(pts):
57 P = (R @ np.asarray(pts, float).T)
58 return np.column_stack([P[0] + x, P[1] + y])
59 parts = {
60 "body": world([(-hw, y0), (hw, y0), (hw, y1), (-hw, y1)]),
61 "nose": world([(-hw, y1), (hw, y1), (0.0, y1 + nose)]),
62 }
63 if flame > 0:
64 parts["flame"] = world([(-0.6 * hw, y0), (0.6 * hw, y0), (0.0, y0 - flame)])
65 return parts

150class RocketDescentModel(NativeTwin, ExtendedKalmanModel):
151 """
152 Planar **powered descent & landing** of a rigid booster — the accurate, *steerable*
153 cousin of :class:`RocketLandingModel` (which is 1-D vertical only).
154
155 The vehicle is a **rigid body** (a rectangle of length ``L`` and width ``W``) with a
156 **gimballed** engine. Mass burns as it fires. The state carries horizontal position and
157 **orientation**, so the booster can tilt to kill sideways drift and must arrive **upright**
158 on the pad::
159
160 ẋ = vx v̇x = −(T/m) sinθ − ½ρ C_dA |v| vx / m
161 ẏ = vy v̇y = (T/m) cosθ − g − ½ρ C_dA |v| vy / m
162 θ̇ = ω ω̇ = T sinδ · ℓ / I (engine gimbal torque)
163 ṁ = −T / v_e
164
165 An onboard **control system** (see :meth:`guidance`) closes three loops every step:
166 a vertical throttle loop tracking the descent profile ``v_ref(y)``; an outer horizontal
167 loop that commands a target tilt ``θ_des`` to null cross-range position/velocity; and an
168 inner attitude loop that gimbals the engine (``δ``) to track ``θ_des`` and right the vehicle
169 for touchdown. A radar/altimeter + IMU measure ``[x, y, θ]``; the EKF reconstructs the
170 velocities, tilt rate and remaining propellant. ``n = 7``, ``m = 3``.
171
172 **Control input.** The two actuators — engine **throttle** and **gimbal** (the side
173 adjustment that steers the booster) — are exposed as the control vector
174 ``u = [throttle ∈ [0,1], gimbal δ ∈ rad]`` when constructed with ``control=True``
175 (``l = 2``): supply your own ``uk`` program, or call :meth:`autopilot` to roll the guidance
176 law out into one (just like the Apollo ascent's thrust program). With the default
177 ``control=False`` (``l = 0``) the same guidance runs **closed-loop inside** the model, so a
178 plain ``sample()`` / ``filter()`` flies and lands it.
179 """
180
181 NAME = "Rocket Descent (2-D, gimballed, lands upright)"; VERSION = "1.0"

188 def __init__(self, name="DESCENT", *,
189 m0=1000.0, # mass at start of descent [kg]
190 m_dry=700.0, # dry mass [kg]
191 ve=2800.0, # exhaust velocity [m/s]
192 thrust_max=40_000.0, # max engine thrust [N] (TWR ≈ 4 at liftoff mass)
193 length=40.0, # body length [m] (rectangle, for inertia & drawing)
194 width=3.7, # body width [m]
195 drag_area=2.0, # C_d·A [m²]
196 drag=True,
197 g=G0,
198 decel=8.0, # deceleration of the near-ground flare [m/s²]
199 v_cruise=55.0, # capped descent speed away from the ground [m/s]
200 v_touch=1.5, # target touchdown descent speed [m/s]
201 theta_max=0.35, # max commanded tilt [rad] (~20°)
202 gimbal_max=0.20, # max engine gimbal [rad] (~11°)
203 kp_v=0.7, # vertical descent-rate gain [1/s]
204 kx=6.0e-3, kvx=4.5e-2, # cross-range → desired-tilt gains (ωn≈0.24, ζ≈0.8)
205 kth=8.0, kom=4.0, # attitude (tilt-tracking) gains
206 fs=20.0, # sensor sample rate [Hz]
207 control=False, # expose u=[throttle, gimbal] as a control input (l=2)
208 track_m=15.0, track_th=0.01, # position [m] / attitude [rad] noise std
209 q_pos=1e-3, q_vel=1e-2, q_ang=1e-5, q_rate=1e-4, q_mass=1e-3,
210 **kw):
211 self.dtypedtype = np.float64
212 self.m0, self.m_dry, self.ve = float(m0), float(m_dry), float(ve)
213 self.thrust_max = float(thrust_max)
214 self.length, self.width = float(length), float(width)
215 self.drag_area, self.drag, self.g = float(drag_area), bool(drag), float(g)
216 self.decel, self.v_cruise, self.v_touch = float(decel), float(v_cruise), float(v_touch)
217 self.theta_max, self.gimbal_max = float(theta_max), float(gimbal_max)
218 self.kp_v, self.kx, self.kvx = float(kp_v), float(kx), float(kvx)
219 self.kth, self.kom = float(kth), float(kom)
220 self.lever = 0.5 * self.length # base-to-CoM lever arm [m]
221
222 self._dt = 1.0 / fs
223 self.nnnnn, self.mmmm, self.lll = 7, 3, (2 if control else 0)
224 self.fsfsfsfsfs = fs
225 Q = np.diag([q_pos, q_pos, q_vel, q_vel, q_ang, q_rate, q_mass]).astype(float)
226 R = np.diag([track_m ** 2, track_m ** 2, track_th ** 2]).astype(float)
227 # L = I: Q enters additively on every state. f() does not route an
228 # additive `w` through, so without it the engine would difference f with
229 # respect to w, find no sensitivity, and multiply Q by a zero matrix at
230 # every prediction: the covariance then collapses to zero.
231 # Start fuelled. The base default is a state of zeros, which for this
232 # model means no mass at all: the engine would cut off in the first step.
233 if kw.get("x0") is None:
234 x0 = np.zeros((7, 1))
235 x0[6, 0] = self.m0
236 kw["x0"] = x0
237 super().__init__(name, Q=Q, R=R, L=np.eye(self.nnnnn), fs=fs, **kw)

261 # --- atmosphere / inertia ---------------------------------------------
262 def _rho(self, y):
263 return RHO0 * np.exp(-max(y, 0.0) / H_ATM)

265 def inertia(self, m):
266 """Moment of inertia of the booster (slender rectangle) [kg·m²]."""
267 return m * self.length ** 2 / 12.0

269 # --- the onboard control system ---------------------------------------
270 def guidance(self, s):
271 """Closed-loop commands for state ``s = [x, y, vx, vy, θ, ω, m]``:
272 returns ``(T, δ)`` — engine thrust [N] and gimbal angle [rad]."""
273 x, y, vx, vy, th, om, m = np.asarray(s, float).reshape(-1)[:7]
274 if m <= self.m_dry or y <= 0.0:
275 return 0.0, 0.0
276 # vertical loop: hold a capped descent speed, then brake at the constant deceleration
277 # `decel` along v² = v_touch² + 2·decel·y, which reaches v_touch at the pad. On that
278 # braking arc the rate of change of the reference, dv_ref/dt = decel·(−vy)/v_brake,
279 # is fed forward: feedback on the speed error alone lags the profile and would hit
280 # the pad at a large fraction of v_cruise.
281 v_brake = np.sqrt(self.v_touch ** 2 + 2.0 * self.decel * max(y, 0.0))
282 braking = v_brake < self.v_cruise
283 v_ref = -(v_brake if braking else self.v_cruise)
284 a_ff = self.decel * max(-vy, 0.0) / v_brake if braking else 0.0
285 a_vert = self.g + a_ff + self.kp_v * (v_ref - vy) # needed vertical accel
286 T = m * a_vert / max(np.cos(th), 0.3)
287 T = float(np.clip(T, 0.0, self.thrust_max))
288 # outer horizontal loop: desired tilt to null cross-range (x>0 ⇒ lean +θ ⇒ thrust −x)
289 th_des = float(np.clip(self.kx * x + self.kvx * vx, -self.theta_max, self.theta_max))
290 # inner attitude loop: gimbal to track θ_des and damp the tilt rate
291 delta = float(np.clip(self.kth * (th_des - th) - self.kom * om,
292 -self.gimbal_max, self.gimbal_max))
293 return T, delta
294
295 def _control_from_u(self, u):
296 """Map a control input ``u = [throttle, gimbal]`` to physical ``(T, δ)`` (clamped)."""
297 thr, delta = np.asarray(u, float).reshape(-1)[:2]
298 return (float(np.clip(thr, 0.0, 1.0)) * self.thrust_max,
299 float(np.clip(delta, -self.gimbal_max, self.gimbal_max)))

301 # --- dynamics ----------------------------------------------------------
302 def _deriv(self, s, t, ctrl=None):
303 x, y, vx, vy, th, om, m = s
304 if y <= 0.0: # landed: frozen on the pad
305 return np.zeros(7)
306 T, delta = self.guidance(s) if ctrl is None else ctrl
307 m_eff = max(m, self.m_dry)
308 ax = -(T / m_eff) * np.sin(th)
309 ay = (T / m_eff) * np.cos(th) - self.g
310 if self.drag:
311 v = np.hypot(vx, vy)
312 k = 0.5 * self._rho(y) * self.drag_area * v / m_eff
313 ax -= k * vx
314 ay -= k * vy
315 domega = T * np.sin(delta) * self.lever / self.inertia(m_eff)
316 mdot = -T / self.ve if m > self.m_dry else 0.0
317 return np.array([vx, vy, ax, ay, om, domega, mdot])

321 def f(self, x, u=None, w=None, t=None):
322 # ds/dt. u supplied (control=True) ⇒ fly that throttle/gimbal command (zero-order hold over the
323 # step); u absent ⇒ the autopilot closes the loop inside _deriv at each RK4 substage.
324 ctrl = self._control_from_u(u) if (self.lll > 0 and u is not None) else None
325 return self._deriv(np.asarray(x, float).reshape(-1), float(t or 0.0), ctrl).reshape(np.shape(x))

327 def after_step(self, x, u=None, t=None):
328 s = np.asarray(x, float).reshape(-1)
329 if s[1] <= 0.0: # touchdown: clamp to the pad
330 s[1] = 0.0
331 s[2] = s[3] = s[5] = 0.0 # zero velocities & tilt rate
332 s[6] = max(s[6], self.m_dry)
333 return s.reshape(-1, 1)

335 def autopilot(self, x0, n, *, t0=0.0):
336 """Roll the guidance law out into an explicit **control program**.
337
338 Flies the closed-loop controller from ``x0`` for ``n`` steps and returns
339 ``(uk, truth)`` where ``uk[i] = [throttle, gimbal]`` is the command applied at step ``i``
340 (ready to pass to ``sample(uk=uk)`` / ``filter(zk, uk)`` on a ``control=True`` model) and
341 ``truth`` is the resulting ``(n, 7)`` state history."""
342 x = np.asarray(x0, float).reshape(-1).copy()
343 uk = np.zeros((n, 2)); truth = np.zeros((n, self.nnnnn))
344 for i in range(n):
345 truth[i] = x
346 T, delta = self.guidance(x)
347 uk[i] = (T / self.thrust_max, delta)
348 x = self.ffff(x, u=uk[i], t=t0 + i * self._dt).reshape(-1)
349 return uk, truth

351 def h(self, x, v=None, t=None):
352 s = np.asarray(x, float).reshape(-1)
353 z = np.array([[s[0]], [s[1]], [s[4]]]) # [x, y, θ]

356 # --- landing metrics ---------------------------------------------------
357 def metrics(self, record):
358 """Headline landing figures: touchdown miss distance & speed, final tilt, ignition
359 time/altitude and propellant burned, from a sampled/filtered ``record``."""
360 t = np.asarray(record.time, float).reshape(-1)
361 s = np.asarray(record.xk).reshape(len(t), -1)
362 y = s[:, 1]
363 # Touchdown is the last sample still in the air. Once on the pad, f() zeroes the
364 # velocities, and process noise can relight the engine, so later samples would
365 # report a zero touchdown speed and propellant burned after landing.
366 landed = np.where(y <= 0.5)[0]
367 i_td = max(int(landed[0]) - 1, 0) if len(landed) else int(np.argmin(np.abs(y)))
368 T = np.array([self.guidance(si)[0] for si in s])
369 burning = T > 1e-6
370 i_ig = int(np.argmax(burning)) if burning.any() else 0
371 return {
372 "touchdown miss [m]": float(abs(s[i_td, 0])),
373 "touchdown speed [m/s]": float(np.hypot(s[i_td, 2], s[i_td, 3])),
374 "final tilt [deg]": float(np.degrees(abs(s[i_td, 4]))),
375 "ignition time [s]": float(t[i_ig]),
376 "ignition altitude [m]": float(s[i_ig, 1]),
377 "propellant burned [kg]": float(s[0, 6] - s[i_td, 6]),
378 }

401 def _plot_static(self, record, sample=None, *, component=None, error=True,
402 nmax=None, scale=6.0, n_pose=7, figsize=(15, 6.6), **kwargs):
403 """Signature figure — matches the interactive layout: booster trajectory (left),
404 position / attitude+gimbal / throttle time series (centre), compact propellant
405 gauge + attitude reticle (narrow right column)."""
406 import matplotlib.pyplot as plt
407 truth = np.asarray(sample.xk) if sample is not None else np.asarray(record.xk)
408 estx = np.asarray(record.xk)
409 n = min(len(truth), len(estx)) if nmax is None else int(nmax)
