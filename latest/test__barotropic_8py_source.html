1"""Barotropic beta-plane: the physics, checked against theory before anything trusts it.
2
3The reason this model exists is that KolmogorovFlowModel is not a model of the atmosphere —
4it has no rotation, and at 50N the beta term it omits is the same order as the ones it keeps.
5So the thing worth testing is not that the code runs; it is that the one new term produces the
6one behaviour it was added for: westward-propagating Rossby waves at c = -beta / k^2.
7
8That dispersion relation is also the sharpest available test of a sign error, which is the
9most likely way to get this wrong and the hardest to notice by eye — an eastward-propagating
10"Rossby wave" still looks like weather.
11"""

43@pytest.mark.parametrize("mode", [1, 2])
44def test_rossby_waves_propagate_westward_at_the_theoretical_speed(mode):
45 """c = -beta / k^2 for a zonal wave, measured by tracking the spectral phase.
46
47 Amplitude is tiny on purpose: the dispersion relation is linear theory, so the test has to
48 stay in the regime where advection is negligible beside the beta term.
49 """
50 solver = _solver()
51 n = 64
52 x = LX * np.arange(n) / n
53 k = 2.0 * math.pi * mode / LX
54
55 field = (1e-7 * np.cos(k * x))[None, :] * np.ones((n, 1))
56 phase = lambda w: np.angle(np.fft.fft2(w)[0, mode])
57
58 steps = 360 # six hours at dt = 60 s
59 evolved = solver.advance(field.reshape(-1), steps).reshape(n, n)
60
61 delta = (phase(evolved) - phase(field) + math.pi) % (2 * math.pi) - math.pi
62 measured = -delta / (k * steps * 60.0)
63 theory = -solver.beta / (k * k)
64
65 assert measured < 0.0, "Rossby waves must travel WESTWARD; a positive speed is a sign error"
66 assert measured == pytest.approx(theory, rel=0.02)
67
