67 def test_matches_linear_kf_statistically(self):
68 """Large-N EnKF posterior mean tracks the exact KF on a linear model."""
69 model = _cv_model()
70 kf, enkf = LinearKalmanEngine(), EnsembleKalmanEngine(n_members=2000, seed=1)
71 enkf.reset(model)
72 rng = np.random.default_rng(0)
73 x, P = X0.copy(), P0.copy()
74 truth = np.array([[0.0], [1.0]])
75 for k in range(10):
76 truth = F_CV @ truth
77 z = (H_POS @ truth + rng.normal(0, 0.5, (1, 1))).ravel()
78 x, P, _ = kf._predict(model, x, P, t=k * DT)
79 x, P = kf._update(model, x, P, z, t=k * DT)
80 xe, Pe, _ = enkf._predict(model, None, None, t=k * DT)
81 xe, Pe = enkf._update(model, None, None, z, t=k * DT)
82 self.assertLess(np.abs(x - xe).max(), 0.05)
83 self.assertLess(np.abs(P - Pe).max(), 0.05)
84 eig = np.linalg.eigvalsh(Pe)
85 self.assertTrue(np.all(eig > -1e-10))
