23 Observability::Observability(const Model &model, std::size_t steps, const Vector *x0, double tolerance)
24 : fTolerance(tolerance)
25 {
26 const std::size_t n = model.NState();
27 if (steps == 0) throw std::invalid_argument("Kalman::Observability: consider at least one step.");
28 Vector x = x0 ? *x0 : model.X0();
29 if (x.Size() != n)
30 throw std::invalid_argument("Kalman::Observability: the initial state has " + std::to_string(x.Size())
31 + " components, " + std::to_string(n) + " expected.");
32
33 const Vector u(model.NControl(), 0.);
34 const double dt = model.DeltaT();
35 double t = model.T0();
36
37 Matrix G(n, n, 0.);
38 Matrix Phi = Matrix::Identity(n);
39 for (std::size_t k = 0; k < steps; ++k)
40 {
41 // The filter measures after it steps: the first measurement is of x_1.
42 const Matrix F = model.Jacobian(x, u, t);
43 x = model.f(x, u, t);
44 Phi = F * Phi;
45 t += dt;
46
47 const Matrix HPhi = model.JacobianH(x, t) * Phi;
48 G += HPhi.Transpose() * (Algebra::Inverse(model.R(t)) * HPhi);
49 }
50
51 // Dimensionless: in units of the initial standard deviations.
52 const Matrix &P0 = model.P0();
53 Vector d(n, 1.);
54 for (std::size_t i = 0; i < n; ++i)
55 if (P0.Rows() == n && P0(i, i) > 0.) d(i) = std::sqrt(P0(i, i));
56 fGramian = Matrix(n, n, 0.);
57 for (std::size_t i = 0; i < n; ++i)
58 for (std::size_t j = 0; j < n; ++j) fGramian(i, j) = d(i) * G(i, j) * d(j);
59 Algebra::Symmetrize(fGramian);
60
61 Vector values;
62 Matrix vectors;
63 if (!Algebra::SymmetricEigen(fGramian, values, vectors))
64 throw std::runtime_error("Kalman::Observability: the eigendecomposition did not converge.");
65
66 // Smallest first, whatever order the decomposition returns.
67 std::vector<std::size_t> order(n);
68 std::iota(order.begin(), order.end(), std::size_t{0});
69 std::sort(order.begin(), order.end(), [&](std::size_t a, std::size_t b) { return values(a) < values(b); });
70 fValues = Vector(n);
71 fVectors = Matrix(n, n, 0.);
72 for (std::size_t c = 0; c < n; ++c)
73 {
74 fValues(c) = values(order[c]);
75 for (std::size_t i = 0; i < n; ++i) fVectors(i, c) = vectors(i, order[c]);
76 }
77
78 const double largest = n ? fValues(n - 1) : 0.;
79 fRank = 0;
80 for (std::size_t c = 0; c < n; ++c)
81 if (largest > 0. && fValues(c) > fTolerance * largest) ++fRank;
82 }

84 double Observability::Weakness(std::size_t i) const
85 {
86 if (i >= fValues.Size()) throw std::out_of_range("Kalman::Observability: no such state component.");
87 // The squared length of the component's axis projected on the weak
88 // subspace: the columns are orthonormal, so the squares add up.
89 double share = 0.;
90 for (std::size_t c = 0; c < NWeak(); ++c) share += fVectors(i, c) * fVectors(i, c);
91 return std::min(1., share);
92 }
