43 void Particle::Reset(const Model &model)
44 {
45 if (!fGiven.empty())
46 {
47 for (const Vector &p : fGiven)
48 if (p.Size() != model.NState())
49 throw std::invalid_argument("Engines::Particle: the given cloud has states of the wrong size.");
50 fParticles = fGiven;
51 fWeights.assign(fNParticles, 1. / static_cast<double>(fNParticles));
52 fResamples = 0;
53 return;
54 }
55 const Matrix factor = Utils::CholeskyFactor(model.P0(), true);
56 if (factor.Empty())
57 throw std::runtime_error("Engines::Particle: P0 has no Cholesky factor; cannot seed the cloud.");
58
59 fParticles.assign(fNParticles, model.X0());
60 for (Vector &p : fParticles) p += Draw(factor);
61 fWeights.assign(fNParticles, 1. / static_cast<double>(fNParticles));
62 fResamples = 0;
63 }

65 void Particle::Estimate(Vector &mean, Matrix &cov) const
66 {
67 if (fParticles.empty()) throw std::runtime_error("Engines::Particle: no cloud; call Reset() first.");
68 const std::size_t n = fParticles.front().Size();
69
70 mean = Vector(n, 0.);
71 for (std::size_t s = 0; s < fParticles.size(); ++s)
72 for (std::size_t i = 0; i < n; ++i) mean(i) += fWeights[s] * fParticles[s](i);
73
74 cov = Matrix(n, n, 0.);
75 for (std::size_t s = 0; s < fParticles.size(); ++s)
76 for (std::size_t i = 0; i < n; ++i)
77 {
78 const double di = fWeights[s] * (fParticles[s](i) - mean(i));
79 for (std::size_t j = 0; j < n; ++j) cov(i, j) += di * (fParticles[s](j) - mean(j));
80 }
81 Utils::Symmetrize(cov);
82 for (std::size_t i = 0; i < n; ++i) cov(i, i) += 1e-9; // the Python engine's ridge
83 }

92 void Particle::Resample() const
93 {
94 // Systematic resampling: one uniform draw, N evenly spaced positions.
95 const std::size_t N = fParticles.size();
96 std::uniform_real_distribution<double> uniform(0., 1.);
97 std::vector<int> indices(N);
98 const double u = fDraws.uniform ? fDraws.uniform() : uniform(fRandom);
99 Utils::SystematicResample(N, fWeights.data(), u, indices.data());
100
101 std::vector<Vector> next;
102 next.reserve(N);
103 for (std::size_t i = 0; i < N; ++i) next.push_back(fParticles[static_cast<std::size_t>(indices[i])]);
104 fParticles.swap(next);
105 fWeights.assign(N, 1. / static_cast<double>(N));
106 ++fResamples;
107 }

125 void Particle::Predict(const Model &model, const Vector &, const Matrix &,
126 const Vector *u, double t,
127 Vector &xk, Matrix &Pk, Matrix &F) const
128 {
129 if (fParticles.empty()) const_cast<Particle *>(this)->Reset(model);
130
131 Vector uu(model.NControl(), 0.);
132 if (u && model.NControl() > 0 && u->Size() == model.NControl()) uu = *u;
133
134 Matrix Q = model.Q(t);
135 const bool through = model.HasProcessNoiseInput(); // w goes into f, not after it
136 const Matrix L = model.L(t);
137 if (!through && !L.Empty()) Q = Utils::Similarity(L, Q);
138
139 if (fDraws.gaussian)
140 {
141 // The draws come from outside, all at once, in the order of the particles.
142 std::vector<Vector> noise;
143 fDraws.gaussian(Q, fParticles.size(), noise);
144 if (noise.size() != fParticles.size())
145 throw std::runtime_error("Engines::Particle: the noise source returned the wrong number of draws.");
146 for (std::size_t s = 0; s < fParticles.size(); ++s)
147 {
148 if (noise[s].Size() != Q.Rows())
149 throw std::runtime_error("Engines::Particle: a noise draw has the wrong size.");
150 if (through) fParticles[s] = model.f(fParticles[s], uu, noise[s], t);
151 else
152 {
153 fParticles[s] = model.f(fParticles[s], uu, t);
154 fParticles[s] += noise[s];
155 }
156 }
157 }
158 else
159 {
160 bool noisy = false;
161 for (std::size_t i = 0; i < Q.Rows() * Q.Cols() && !noisy; ++i) noisy = Q.Data()[i] != 0.;
162 const Matrix factor = noisy ? Utils::CholeskyFactor(Q, true) : Matrix();
163
164 for (Vector &p : fParticles)
165 {
166 if (through)
167 p = model.f(p, uu, factor.Empty() ? Vector(Q.Rows(), 0.) : Draw(factor), t);
168 else
169 {
170 p = model.f(p, uu, t);
171 if (!factor.Empty()) p += Draw(factor);
172 }
173 }
174 }
175
176 Estimate(xk, Pk);
177 F = Matrix::Identity(model.NState());
178 }

180 std::size_t Particle::Update(const Model &model, const Vector &, const Matrix &,
181 const Vector &z, double t,
182 Vector &xOut, Matrix &POut) const
183 {
184 if (fParticles.empty()) const_cast<Particle *>(this)->Reset(model);
185
186 const std::vector<std::size_t> keep = Utils::ValidChannels(z);
187 if (keep.empty())
188 {
189 Estimate(xOut, POut);
190 return 0;
191 }
192
193 const Vector zc = Utils::Compress(z, keep);
194 const Matrix Rinv = Utils::Inverse(Utils::CompressBoth(model.R(t), keep));
195
196 std::vector<double> logw(fParticles.size());
197 double top = -INFINITY;
198 for (std::size_t s = 0; s < fParticles.size(); ++s)
199 {
200 Vector dz = zc;
201 dz -= Utils::Compress(model.h(fParticles[s], t), keep);
202 logw[s] = std::log(fWeights[s] + 1e-300) - 0.5 * Utils::Mahalanobis(dz, Rinv);
203 top = std::max(top, logw[s]);
204 }
205
206 double total = 0.;
207 for (std::size_t s = 0; s < logw.size(); ++s)
208 {
209 fWeights[s] = std::exp(logw[s] - top);
210 total += fWeights[s];
211 }
212 for (double &w : fWeights) w /= total;
213
214 if (EffectiveSampleSize() < fThreshold * static_cast<double>(fParticles.size())) Resample();
215
216 Estimate(xOut, POut);
217 return keep.size();
218 }
