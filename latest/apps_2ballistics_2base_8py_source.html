21# Process-noise type — the member VALUE is the physical unit of
22# ``process_uncertainty``, which is exactly what distinguishes the two: feeding a
23# value in m/s² means "acceleration noise" (constant-acceleration model), m/s³
24# means "jerk noise" (nearly-constant-acceleration model). Picking the unit picks
25# the model. Compares equal to its name too (e.g. ``noise == "acceleration"``).
26class BallisticNoise(EnumBis):
27 ACCELERATION = "m/s²" # white-noise-acceleration → accel stays flat (constant-accel)
28 JERK = "m/s³" # white-noise-jerk → accel random-walks (nearly-constant-accel)

30 @classmethod
31 def of(cls, noise):
32 """Resolve a member from itself, its name ('acceleration'/'jerk'), or its
33 UNIT ('m/s²'/'m/s³') — so the unit of process_uncertainty selects the model."""
34 if isinstance(noise, cls):
35 return noise
36 for m in cls:
37 if m == noise or m.value == noise: # match by name (EnumBis.__eq__) or by unit
38 return m
39 raise ValueError(f"Unknown noise {noise!r}; use {cls.repr()} or a unit in {cls.values()}.")
