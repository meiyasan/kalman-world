1"""CNN forecast surrogate for :class:`~kalman.apps.weather.flow.KolmogorovFlowModel`.
2
3Imports torch at module level, so it is **not** re-exported from
4``kalman.apps`` — import it explicitly (same convention as
5``kalman.framework.routers.nn``)::
6
7 from kalman.apps.weather.surrogate import (make_trajectory, train_surrogate,
8 as_forecaster, rollout_rmse)
9
10The prototype that preceded this module showed that a surrogate trained on
11**one-step** pairs with plain MSE diverges exponentially once cycled inside an
12EnKF (nothing in the objective penalizes multi-cycle drift, and the bias
13compounds every cycle). Two rules are therefore built in here:
14
151. :func:`train_surrogate` uses a **rollout loss** — the network is applied
16 ``rollout`` times and penalized at every intermediate cycle, which is what
17 actually constrains cycle-to-cycle bias.
182. :func:`rollout_rmse` measures **open-loop** error growth against the true
19 solver. Verify it stays bounded (well below the climatological field std)
20 over at least as many cycles as the DA experiment will run *before* wiring
21 the surrogate into a filter — and still budget a larger inflation for the
22 surrogate branch than for the true-model branch (their error statistics
23 differ).
24"""

31class PeriodicConv2d(nn.Module):
32 """Conv2d with periodic (torus) padding — matches the solver's domain."""
33 def __init__(self, cin, cout, k=3):
34 super().__init__()
35 self.pad = k // 2
36 self.conv = nn.Conv2d(cin, cout, k)
37 def forward(self, x):
38 p = self.pad
39 x = torch.cat([x[..., -p:, :], x, x[..., :p, :]], dim=-2)
40 x = torch.cat([x[..., -p:], x, x[..., :p]], dim=-1)
41 return self.conv(x)

44class ResBlock(nn.Module):
45 def __init__(self, c, k=3):
46 super().__init__()
47 self.c1 = PeriodicConv2d(c, c, k)
48 self.c2 = PeriodicConv2d(c, c, k)
49 self.act = nn.GELU()
50 def forward(self, x):
51 h = self.act(self.c1(x))
52 h = self.c2(h)
53 return x + h

56class FlowSurrogate(nn.Module):
57 """Predicts the increment ``Δω = ω(t+cycle) − ω(t)``; residual formulation
58 (stabilizes training and keeps the identity map as an easy fallback)."""
59 def __init__(self, width=16, nblocks=3):
60 super().__init__()
61 self.width, self.nblocks = width, nblocks
62 self.inp = PeriodicConv2d(1, width, 3)
63 self.blocks = nn.ModuleList([ResBlock(width) for _ in range(nblocks)])
64 self.out = PeriodicConv2d(width, 1, 3)
65 def forward(self, w): # w: (B, 1, n, n), normalized
66 h = torch.tanh(self.inp(w))
67 for b in self.blocks:
68 h = b(h)
69 return w + self.out(h)

72def make_trajectory(model, n_cycles: int, burn: int = 2000) -> np.ndarray:
73 """Roll the true solver and return snapshots one cycle apart, ``(T+1, n, n)``."""
74 w = model.solver.advance(model.solver.random_ic(), burn)
75 traj = [w.copy()]
76 for _ in range(n_cycles):
77 w = model.solver.advance(w, model.cycle)
78 traj.append(w.copy())
79 return np.stack(traj).astype(np.float32)

82def train_surrogate(traj: np.ndarray, *, width=16, nblocks=3, epochs=60, lr=2e-3,
83 batch=16, rollout=4, val_frac=0.15, device="cpu", seed=0,
84 verbose=True):
85 """Train :class:`FlowSurrogate` on a solver trajectory with a rollout loss.
86
87 ``traj`` is ``(T, n, n)`` snapshots one cycle apart (from
88 :func:`make_trajectory`). Each sample is a window ``w_i … w_{i+rollout}``;
89 the loss is the mean MSE of ``net^k(w_i)`` against ``w_{i+k}`` for
90 ``k = 1 … rollout`` (backprop through the whole rollout). The validation
91 split is the trajectory *tail* (contiguous in time, no window leakage).
92
93 Returns ``(net, (mu, sd), best_val)`` — normalization constants travel
94 with the network (see :func:`as_forecaster` / :func:`save_surrogate`).
95 """
96 torch.manual_seed(seed)
97 T = traj.shape[0]
98 if T < rollout + 2:
99 raise ValueError(f"trajectory too short ({T}) for rollout={rollout}")
100 n_val = max(rollout + 1, int(val_frac * T))
101 tr, va = traj[:T - n_val], traj[T - n_val:]
102
103 mu, sd = float(tr.mean()), float(tr.std())
104 trn = torch.tensor((tr - mu) / sd, device=device).unsqueeze(1) # (Ttr, 1, n, n)
105 van = torch.tensor((va - mu) / sd, device=device).unsqueeze(1)
106
107 def rollout_loss(data, starts):
108 x = data[starts]
109 loss = 0.0
110 for k in range(1, rollout + 1):
111 x = net(x)
112 loss = loss + nn.functional.mse_loss(x, data[starts + k])
113 return loss / rollout
114
115 net = FlowSurrogate(width, nblocks).to(device)
116 opt = torch.optim.Adam(net.parameters(), lr=lr)
117 sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, epochs)
118 tr_starts = np.arange(trn.shape[0] - rollout)
119 va_starts = torch.arange(van.shape[0] - rollout)
120 rng = np.random.default_rng(seed)
121
122 best_val, best_state = np.inf, None
123 for ep in range(epochs):
124 net.train()
125 perm = rng.permutation(tr_starts)
126 tot = 0.0
127 for i in range(0, len(perm), batch):
128 b = torch.tensor(perm[i:i + batch])
129 opt.zero_grad()
130 loss = rollout_loss(trn, b)
131 loss.backward()
132 opt.step()
133 tot += loss.item() * len(b)
134 sched.step()
135 net.eval()
136 with torch.no_grad():
137 vloss = rollout_loss(van, va_starts).item()
138 if vloss < best_val:
139 best_val = vloss
140 best_state = {k: v.clone() for k, v in net.state_dict().items()}
141 if verbose and (ep + 1) % max(1, epochs // 6) == 0:
142 print(f" epoch {ep+1:3d}: train={tot/len(perm):.5f} val={vloss:.5f}")
143 net.load_state_dict(best_state)
144 return net, (mu, sd), best_val

147def as_forecaster(net, mu, sd):
148 """Wrap a trained net as a batched forecast map (d, N) → (d, N) — plug it
149 into ``model.surrogate`` or ``TwinExperiment.add_method(forecast=...)``."""
150 net.eval()
151 def forecast(X):
152 d, N = X.shape
153 n = int(round(np.sqrt(d)))
154 xin = torch.tensor(X.T.reshape(N, 1, n, n).astype(np.float32))
155 xin = (xin - mu) / sd
156 with torch.no_grad():
157 out = net(xin) * sd + mu
158 return out.numpy().reshape(N, d).T.astype(np.float64)
159 return forecast

162def rollout_rmse(forecast, model, w0=None, n_cycles=10) -> np.ndarray:
163 """Open-loop rollout error of a forecaster vs the true solver, per cycle.
164
165 The mandatory pre-DA check: if this grows past the climatological field
166 std within the horizon of the planned DA experiment, the surrogate will
167 drag the filter along with it — retrain (longer rollout, more data)
168 before touching the filter.
169 """
170 w0 = model.equilibrium_state() if w0 is None else np.asarray(w0, dtype=float).ravel()
171 truth, pred = w0.copy(), w0.copy()[:, None]
172 errs = []
173 for _ in range(n_cycles):
174 truth = model.f(truth)
175 pred = forecast(pred)
176 errs.append(float(np.sqrt(np.mean((pred.ravel() - truth) ** 2))))
177 return np.array(errs)

180class ReanchoredForecast:
181 """Hybrid forecaster: the surrogate runs most cycles, the true solver every
182 ``every``-th cycle — the cheap first mitigation when a surrogate is not yet
183 stable under cycling (its bias is flushed by the solver before it can
184 compound, at ``1/every`` of the full forecast cost). Plug into
185 ``TwinExperiment.add_method(forecast=...)`` like any forecaster; call
186 :meth:`reset` before reusing it in a new experiment.
187 """
188 def __init__(self, model, forecast, every=3):
189 self.model, self.forecast, self.every = model, forecast, int(every)
190 self.k = 0
191
192 def reset(self):
193 self.k = 0
194 return self
195
196 def __call__(self, X):
197 self.k += 1
198 if self.k % self.every == 0:
199 out = np.empty_like(X)
200 for i in range(X.shape[1]):
201 out[:, i] = self.model.solver.advance(X[:, i], self.model.cycle)
202 return out
203 return self.forecast(X)
