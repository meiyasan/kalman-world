1"""
2``kalman.api`` — one JSON-friendly facade over the framework, for humans, agents,
3and the MCP/REST transports.
4
5It *harvests* the existing metadata and ``*_BY_KIND`` registries rather than
6inventing a parallel description layer, and exposes plain-dict/-list verbs (numpy
7arrays ↔ nested lists) so any transport can carry them:
8
9 - **introspect** — :func:`catalog`, :func:`describe`
10 - **execute** — :func:`make_filter`, :func:`sample`, :func:`run_filter`,
11 :func:`forecast`, :func:`run_osse`, :func:`run_control`, :func:`diagnostics`,
12 :func:`render_plot`
13 - **author** — :func:`register_model`, :func:`save_spec`, :func:`scaffold`

15 - **share** — :func:`models`, :func:`delete_model`, :func:`register_source`,

90def jsonify(obj):
91 """Recursively convert numpy arrays/scalars to JSON-native types."""
92 if isinstance(obj, np.ndarray):
93 return obj.tolist()
94 if isinstance(obj, (np.floating, np.integer)):
95 return obj.item()
96 if isinstance(obj, dict):
97 return {k: jsonify(v) for k, v in obj.items()}
98 if isinstance(obj, (list, tuple)):
99 return [jsonify(v) for v in obj]
100 return obj

141def _sync_plugins():
142 """Load every plugin the registry records and this process has not loaded.
143
144 A plugin loaded through any process (or before a restart) is part of the
145 shared catalog; this is what makes that true here. It is resolved again by
146 *name* inside this process's allow-listed directories, never by the
147 recorded path, so a record cannot point a server at a file outside them."""
148 try:
149 recorded = store.list_plugins()
150 except Exception:
151 return
152 if not recorded:
153 return
154 from ..framework.native import core
155 if not core.HAS_CXX_CORE:
156 return
157 loaded = set(core.plugins())
158 for rec in recorded:
159 if rec["name"] in _PLUGIN_ERRORS:
160 continue
161 try:
162 path = core.resolve_plugin(rec["name"])
163 if path not in loaded:
164 core.load_plugin(path)
165 except Exception as e: # missing here; report, do not break the catalog
166 _PLUGIN_ERRORS[rec["name"]] = str(e)

280#: Where each KalmanSpec key lives. A spec is nested, and the overwhelmingly
281#: common authoring mistake — especially from an agent writing JSON by hand — is
282#: to put the matrices at the top level. Mapping inner key -> owning section is
283#: what lets :func:`_check_spec` say "F belongs in dynamics" instead of letting a
284#: dimensionless model fail deep in the framework.
285_SPEC_SECTIONS = {
286 "F": "dynamics", "H": "dynamics", "B": "dynamics", "L": "dynamics",
287 "M": "dynamics", "A": "dynamics", "Qc": "dynamics", "Lc": "dynamics",
288 "params": "dynamics",
289 "Q": "noise", "R": "noise",
290 "x0": "initial", "P0": "initial",
291 "states": "labels", "observables": "labels",
292 "engine": "bindings", "integrator": "bindings", "solver": "bindings",
293 "cost": "control", "Qcost": "control", "Rcost": "control", "x_ref": "control",
294}
295
296_SPEC_TOP_LEVEL = {"base", "name", "sensor_id", "fs", "dynamics", "noise",
297 "initial", "labels", "control", "bindings", "description"}
298
299#: Names callers reach for that the spec calls something else. ``model`` is the
300#: big one: every other verb in this facade takes a model *name*, so a spec
301#: argument invites ``{"model": "SomeModel"}`` — which is a spec with no base
302#: and no dynamics, i.e. the dimensionless-model failure again.
303_SPEC_ALIASES = {"model": "base", "class": "base", "model_name": "base",
304 "state_names": "labels.states", "observable_names": "labels.observables",
305 "measurements": "(pass measurements to run_filter's data argument)",
306 "data": "(pass measurements to run_filter's data argument)"}

309def _check_spec(spec: dict):
310 """Reject a malformed spec while the caller can still see their own mistake.
311
312 Without this a spec whose matrices sat at the top level simply produced a
313 model with no dimensions, and the failure surfaced from deep inside the
314 framework as ``array must be (d,1) or (d,), got 0`` — a message that names
315 nothing the caller wrote, and that reads like a broken server rather than a
316 misplaced key. (It cost a downstream integration an afternoon.) Checked
317 here, in :func:`make_model`, so every verb that builds a model —
318 ``run_filter``, ``run_osse``, ``forecast``, ``register_model`` — gets the
319 same guidance.
320 """
321 if not isinstance(spec, dict):
322 raise ValueError("spec must be a dict; see catalog()/describe() for the shape.")
323
324 problems = [f"{k!r} belongs in {_SPEC_SECTIONS[k]}[{k!r}]"
325 for k in sorted(spec) if k in _SPEC_SECTIONS]
326 problems += [f"{k!r} is named {_SPEC_ALIASES[k]!r} in a spec"
327 for k in sorted(spec) if k in _SPEC_ALIASES]
328 problems += [f"{k!r} is not a spec key" for k in sorted(spec)
329 if k not in _SPEC_TOP_LEVEL and k not in _SPEC_SECTIONS
330 and k not in _SPEC_ALIASES]
331 if problems:
332 raise ValueError(
333 "malformed KalmanSpec — " + "; ".join(problems) + ". A spec is nested: "
334 '{"base": "matrices", "fs": 1.0, "dynamics": {"F": .., "H": ..}, '
335 '"noise": {"Q": .., "R": ..}, "initial": {"x0": .., "P0": ..}}; for a '
336 'catalog model use {"base": "<ClassName>", "dynamics": {"params": {..}}}.')
337
338 base = spec.get("base", "matrices")
339 if base in ("matrices", "continuous"):
340 dyn = spec.get("dynamics", {}) or {}
341 if not any(dyn.get(k) is not None
342 for k in ("F", "H", "A", "Qc", "B", "L", "M", "Lc")):
343 raise ValueError(
344 f"a {base!r} spec carries its own dynamics, but none were given. "
345 'Provide dynamics.F and dynamics.H (base "matrices"), or '
346 'dynamics.A and dynamics.Qc (base "continuous"). To use a '
347 "catalog model instead, set base to its class name — see "
348 "catalog()['models'].")

351def make_model(spec: dict) -> KalmanModel:
352 """Instantiate a :class:`KalmanModel` from a :class:`KalmanSpec` dict."""
353 spec = dict(spec or {})
354 _check_spec(spec)
355 base = spec.get("base", "matrices")
356 if base in _MODELS: # authored spec referenced by name
357 return make_model(_MODELS[base])
358 sensor = spec.get("sensor_id", "api")
359 fs = spec.get("fs", 1.0)
360 noise = spec.get("noise", {}) or {}
361 initial = spec.get("initial", {}) or {}
362 binds = spec.get("bindings", {}) or {}
363 kw = {}
364 if "engine" in binds:
365 pass # engine is chosen by the filter, not the model ctor
366 if binds.get("integrator"):
367 kw["integrator"] = _kind(KalmanIntegratorKind, binds["integrator"])
368 if binds.get("solver"):
369 kw["solver"] = _kind(KalmanSolverKind, binds["solver"])
370 if "x0" in initial:
371 kw["x0"] = _arr(initial["x0"])
372 if "P0" in initial:
373 kw["P0"] = _arr(initial["P0"])
374
375 if base in ("matrices", "continuous"):
376 dyn = spec.get("dynamics", {}) or {}
377 for k in ("F", "H", "B", "L", "M", "A", "Qc", "Lc"):
378 if dyn.get(k) is not None:
379 kw[k] = _arr(dyn[k])
380 for k in ("Q", "R"):
381 if noise.get(k) is not None:
382 kw[k] = _arr(noise[k])
383 return KalmanModel(sensor, fs=fs, **kw)
384
385 cls = _model_catalog().get(base)
386 if cls is None:
387 _sync_plugins() # a plugin another process loaded
388 cls = _model_catalog().get(base)
389 if cls is None:
390 raise ValueError(f"unknown model base {base!r}; see catalog()['models'].")
391 params = dict((spec.get("dynamics", {}) or {}).get("params", {}) or {})
392 # bind params against the class signature; pass sensor_id/fs when accepted
393 try:
394 plist = list(inspect.signature(cls.__init__).parameters) # ordered
395 except (TypeError, ValueError):
396 plist = []
397 accepted = set(plist)
398 # A class taking **kwargs accepts x0/P0 even though they are not named parameters --
399 # KalmanModel's base __init__ reads them straight out of kwargs, which is exactly how the
400 # matrices path passes them. Filtering on the NAMED signature alone silently dropped
401 # initial.x0 for every catalog model, and silently is the problem: the filter then starts
402 # at the origin with a default covariance and produces confident nonsense rather than an
403 # error. A satellite spec carrying its true initial state diverged by 892 km this way.
404 try:
405 takes_var_kw = any(p.kind is inspect.Parameter.VAR_KEYWORD
406 for p in inspect.signature(cls.__init__).parameters.values())
407 except (TypeError, ValueError):
408 takes_var_kw = False
409 is_native = hasattr(cls, "NATIVE_NAME")
410 if is_native:
411 # A C++ model has its own sample rate, noise and parameters: a spec that
412 # does not give fs keeps the model's, and its noise is pushed into the
413 # C++ instance (the catalogue's Python classes define their noise through
414 # their own parameters instead).
415 if "fs" in spec:
416 params["fs"] = fs
417 for k in ("Q", "R"):
418 if noise.get(k) is not None:
419 params[k] = _arr(noise[k])
420 try:
421 return cls(sensor, **{**kw, **params})
422 except KeyError as e:
423 raise ValueError(f"model {base!r}: {e.args[0] if e.args else e}") from None
424 if "fs" in accepted and "fs" not in params:
425 params["fs"] = fs
426 for k, v in list(kw.items()):
427 if k in accepted or takes_var_kw:
428 params.setdefault(k, v)
429 ordered = [p for p in plist if p != "self"]
430 if ordered and ordered[0] in ("sensor_id", "name"):
431 return cls(sensor, **params)
432 return cls(**params)

435def make_filter(spec: dict) -> BayesFilter:
436 """Build a filter (model + resolved engine, of whatever family) from a spec."""
437 model = make_model(spec)
438 binds = spec.get("bindings", {}) or {}
439 engine_kind = _kind(KalmanEngineKind, binds.get("engine"))
440 if str(binds.get("backend", "python")).lower() == "native":
441 # The C++ engine behind the Python filter. Only a native (C++) model can be
442 # driven this way -- NativeEngine says so itself if the model is Python.
443 from ..framework.engines import NativeEngine
444 kind = engine_kind or getattr(model, "DEFAULT_ENGINE", KalmanEngineKind.EXTENDED)
445 return BayesFilter(model, engine=NativeEngine(kind, **(binds.get("engine_params", {}) or {})))
446 if engine_kind is not None:
447 from ..framework.engines import resolve_engine
448 return BayesFilter(model, engine=resolve_engine(engine_kind))
449 return BayesFilter(model)

455def _stash(record, model, spec, truth=None, measurements=None) -> str:
456 """Store a run for later diagnostics/plotting.
457
458 ``measurements`` is kept SEPARATELY from ``record`` on purpose: a filtered
459 record's ``zk`` holds the filter's own output, not the data that went in, so
460 a diagnostic that reaches for ``record.zk`` ends up re-filtering estimates
461 (see :func:`diagnostics`)."""
462 rid = f"run-{uuid.uuid4().hex[:12]}" # unique across processes and restarts
463 while len(_RUNS) >= max(_MAX_RUNS, 1):
464 _RUNS.popitem(last=False)
465 _RUNS[rid] = {"record": record, "model": model, "spec": spec, "truth": truth,
466 "measurements": None if measurements is None
467 else np.asarray(measurements, dtype=float)}
468 return rid

479def run_filter(spec: dict, data=None, *, smooth: bool = False, source: str | None = None,
480 query: dict | None = None, credential: str | None = None, **kw) -> dict:
481 """Filter ``data`` (measurements, list of lists) through a spec's filter.
482 Returns ``{run_id, xk, Pk}`` (posterior mean + covariance).
483
484 Give ``source`` (a registered source name) instead of ``data`` to measure
485 and filter in one call: the measurements are fetched first, with ``query``,
486 under the ``send_measurement`` scope."""
487 if source is not None:
488 if data is not None:
489 raise ValueError("give data or source, not both.")
490 data = fetch({"source": source}, query, credential=credential)["zk"]
491 if data is None:
492 raise ValueError("run_filter needs measurements: data=[[...], ...] or source=<name>.")
493 filt = make_filter(spec)
494 zk = np.asarray(data, dtype=float)
495 if zk.size == 0:
496 raise ValueError("no measurements to filter.")
497 rec = filt.filter(zk, smooth=smooth, **kw)
498 rid = _stash(rec, filt.model, spec, measurements=zk)
499 return {"run_id": rid, "xk": jsonify(rec.xk), "Pk": jsonify(rec.Pk),
500 "time": jsonify(rec.time)}

518def run_control(model_spec: dict, controller_spec: dict, duration: float) -> dict:
519 """Close the loop: run ``controller`` over the model for ``duration``.
520 Returns ``{run_id, xk, cost}`` (state trace + total control cost)."""
521 model = make_model(model_spec)
522 cs = dict(controller_spec or {})
523 cls = _controller_catalog().get(_ctrl_name(cs.get("controller")))
524 if cls is None:
525 raise ValueError(f"unknown controller {cs.get('controller')!r}.")
526 cost = cs.get("cost", {}) or {}
527 ctrl = cls(model,
528 Qcost=_arr(cost.get("Qcost")), Rcost=_arr(cost.get("Rcost")),
529 x_ref=_arr(cost.get("x_ref")), **(cs.get("params", {}) or {}))
530 if hasattr(ctrl, "fit"): # LQR/Neural controllers train before use
531 ctrl.fit()
532 filt = BayesFilter(model)
533 truth = model.sample(duration)
534 rec = filt.forward(truth.zk, uk=ctrl)
535 rid = _stash(rec, model, model_spec)
536 total = float(sum(c for _, c in ctrl.cost_log))
537 return {"run_id": rid, "xk": jsonify(rec.xk), "cost": total,
538 "cost_log": jsonify([c for _, c in ctrl.cost_log])}

550def diagnostics(run_id: str) -> dict:
551 """Filter-consistency diagnostics (NIS + innovation) for a stored run.
552
553 NIS answers the only question a filter can ask without ground truth: are the
554 innovations the size the claimed noise says they should be (≈ m on average,
555 χ²(m))? That makes it the measurement contract — and therefore a diagnostic
556 that must be fed the MEASUREMENTS.
557
558 It previously reached for ``record.zk``, which on a filtered run holds the
559 filter's own output rather than its input; the pass then re-filtered
560 estimates and reported NIS ≈ 0.1 instead of ≈ 1 — a filter flagged as
561 over-conservative when it was in fact consistent, and a check that failed
562 quietly in the direction of looking fine. Stored measurements are used when
563 present, and a run that has none says so rather than silently diagnosing
564 the wrong array."""
565 r = _RUNS.get(run_id)
566 if r is None:
567 raise KeyError(f"unknown run_id {run_id!r}.")
568
569 zk = r.get("measurements")
570 if zk is None:
571 record = r["record"]
572 zk = getattr(record, "zk", None)
573 # A sample() run IS its own measurement set; a filtered run is not.
574 if zk is None or r.get("truth") is not record:
575 raise ValueError(
576 f"run {run_id!r} carries no stored measurements to diagnose; "
577 "re-run it through run_filter (which now keeps them) or pass a "
578 "sample() run.")

580 filt.forward(np.asarray(zk, dtype=float), record=False)
581 nu, S = filt.innovations()

583 "innovation_covariance": jsonify(S)}

587 """Render a stored run's signature figure to PNG bytes (base64-encode for JSON).

613def forecast(spec: dict, x0, P0=None, steps: int = 1, **kw) -> dict:
614 """Forecast ahead from ``x0``/``P0`` for ``steps``.
615
616 Uses the model's ``forecast_cone`` when it defines one (the storm family
617 returns a track cone: ``{lon, lat, I, I_sigma, ellipses}``); otherwise the
618 state and its covariance are propagated with the model's own ``f``/``F``, so
619 every model answers something rather than only the storms.
620 """
621 model = make_model(spec)
622 x = _arr(x0).reshape(model.n, 1)
623 P = _arr(P0) if P0 is not None else np.asarray(model.P0, dtype=float)
624 steps = int(steps)
625
626 cone = getattr(model, "forecast_cone", None)
627 if callable(cone):
628 return jsonify(cone(x, P, steps, **kw))
629
630 Q = model.Q()
631 xs, Ps = [], []
632 for k in range(steps):
633 F = model.F(x=x, t=None)
634 x = np.asarray(model.f(x), dtype=float).reshape(model.n, 1)
635 P = F @ P @ F.T + Q
636 xs.append(x); Ps.append(P)
637 return {"xk": jsonify(xs), "Pk": jsonify(Ps)}

649 """Pull data and return its JSON payload; requires ``send_measurement``.

670def _fetch_source(name: str, query: dict) -> dict:
671 rec = store.get_source(name)
672 if rec is None:
673 raise KeyError(f"no source named {name!r}; see sources().")
674 if rec["kind"] == "push":
675 since = int(query.pop("since", 0) or 0)
676 limit = int(query.pop("limit", 10000) or 10000)
677 if query:
678 raise ValueError(f"a push source takes only since= and limit=, not {sorted(query)}.")
679 rows = store.read_measurements(name, since, limit)
680 out = {"zk": [r["row"] for r in rows],
681 "meta": {"provider": "push", "source": name,
682 "first": rows[0]["seq"] if rows else None,
683 "last": rows[-1]["seq"] if rows else None,
684 "next": rows[-1]["seq"] if rows else since}}
685 if rows and all(r["t"] is not None for r in rows):
686 out["time"] = [r["t"] for r in rows]
687 return out
688 # The source fixes where the data comes from; the query may only narrow it.
689 fixed = set(rec.get("init") or {}) | {"path", "source"}
690 if rec["kind"] == "remote":
691 fixed.discard("path") # a sub-path under the fixed base_url
692 clash = sorted(k for k in query if k in fixed)
693 if clash:
694 raise ValueError(f"source {name!r} fixes {clash}; a query cannot change them.")
695 from ..providers import resolve_provider
696 prov = resolve_provider(rec["kind"], **(rec.get("init") or {}))
697 out = jsonify(prov.fetch(**query))
698 if isinstance(out, dict):
699 out.setdefault("meta", {})["source"] = name
700 return out

749def register_source(name: str, provider: dict, *, replace: bool = False,
750 credential: str | None = None) -> dict:
751 """Set up a named measurement source that everybody can fetch by name.
752
753 ``provider`` is ``{"kind": "push", "m": <width>?, "description": ..}`` --
754 a buffer that devices, simulations or game engines write rows into with
755 :func:`push_measurements` -- or a provider ``{"kind": "<kind>", "init":
756 {...}}``. A ``file``/``remote``/``sensor`` source reads server files or
757 uses the server's outbound credentials, so it needs ``admin``; the others
758 need ``write_models``."""
759 from .auth import authorize, identity, require_owner
760 _check_name(name, "source")
761 prov = dict(provider or {})
762 kind = str(prov.get("kind", "")).lower()
763 if not kind:
764 raise ValueError('a source needs a kind: {"kind": "push"} or a provider kind.')
765 authorize(credential, "admin" if kind in _PRIVILEGED_KINDS else "write_models")
766 if kind == "push":
767 m = prov.get("m")
768 if m is not None and (not isinstance(m, int) or m < 1):
769 raise ValueError("a push source's m (the row width) must be a positive integer.")
770 spec = {"kind": "push", "m": m, "description": str(prov.get("description", ""))}
771 else:
772 from ..providers import resolve_provider
773 init = dict(prov.get("init") or {})
774 resolve_provider(kind, **init) # fails now, not on the first fetch
775 spec = {"kind": kind, "init": init, "description": str(prov.get("description", ""))}
776 rec, action = store.put_source(
777 name, spec, identity(credential), replace=replace,
778 may_replace=lambda r: require_owner(r, credential, f"source {name!r}"))
779 return {**rec, "action": action}

808def push_measurements(name: str, rows, time=None, *, credential: str | None = None) -> dict:
809 """Append measurement rows to a ``push`` source; requires ``send_measurement``.
810
811 ``rows`` is a list of rows (one measurement vector each) or one row;
812 ``time`` an optional timestamp per row. Subscribers get a
813 ``measurement.received`` event naming the new sequence numbers."""
814 from .auth import authorize, identity
815 authorize(credential, "send_measurement")
816 rec = store.get_source(name)
817 if rec is None:
818 raise KeyError(f"no source named {name!r}; register_source first.")
819 if rec["kind"] != "push":
820 raise ValueError(f"source {name!r} is a {rec['kind']!r} source; only a push "
821 "source takes measurements.")
822 try:
823 z = np.asarray(rows, dtype=float)
824 except (TypeError, ValueError):
825 raise ValueError("rows must be numbers: [[z1, z2, ...], ...].")
826 if z.ndim == 1:
827 z = z.reshape(1, -1)
828 if z.ndim != 2 or z.shape[0] == 0 or z.shape[1] == 0:
829 raise ValueError(f"rows must be a non-empty list of rows; got shape {z.shape}.")
830 if z.shape[0] > MAX_PUSH_ROWS:
831 raise ValueError(f"at most {MAX_PUSH_ROWS} rows per push; got {z.shape[0]}.")
832 if not np.all(np.isfinite(z)):
833 raise ValueError("rows must be finite (no NaN or infinity).")
834 if rec.get("m") is not None and z.shape[1] != rec["m"]:
835 raise ValueError(f"source {name!r} takes rows of width {rec['m']}; got {z.shape[1]}.")
836 times = None
837 if time is not None:
838 times = [float(t) for t in np.asarray(time, dtype=float).reshape(-1)]
839 if len(times) != z.shape[0]:
840 raise ValueError(f"{len(times)} timestamps for {z.shape[0]} rows.")
841 return store.append_measurements(name, z.tolist(), times, identity(credential))
