28class Oscillator(KalmanModel):
29 """A damped oscillator stepped by explicit Euler; the damping is a parameter
30 read at every step. Only the position is measured."""
31
32 NAME = "Test Oscillator"
33
34 def __init__(self, sensor="osc", *, omega=2.0, damping=0.0, fs=50.0, **kwargs):
35 self.nnnn, self.mmmm, self.lll = 2, 1, 0
36 self["omega"], self["damping"] = omega, damping
37 super().__init__(sensor, fs=fs, Q=np.diag([1e-6, 1e-4]), R=np.array([[1e-2]]),
38 x0=np.array([[1.0], [0.0]]), P0=np.eye(2), **kwargs)
39
40 def f(self, x, u=None, w=None, t=None):
41 x = np.asarray(x, dtype=float).reshape(2, 1)
42 dt = 1.0 / self.fsfsfs
43 out = np.array([[x[0, 0] + dt * x[1, 0]],
44 [x[1, 0] + dt * (-self["omega"] ** 2 * x[0, 0] - self["damping"] * x[1, 0])]])
45 return out if w is None else out + np.asarray(w, dtype=float).reshape(2, 1)
46
47 def h(self, x, v=None, t=None):
48 z = np.asarray(x, dtype=float).reshape(-1, 1)[:1]
49 return z if v is None else z + np.asarray(v, dtype=float).reshape(1, 1)

164class Gauge(KalmanModel):
165 """A constant amplitude read through a gain and an offset: ``z = gain * a + offset``.
166 The three enter the one measurement together, so the data cannot tell them apart."""
167
168 def __init__(self, sensor="gauge", *, gain=2.0, offset=0.0, **kwargs):
169 self.nnnn, self.mmmm, self.lll = 1, 1, 0
170 self["gain"], self["offset"] = gain, offset
171 super().__init__(sensor, fs=10.0, Q=np.array([[1e-8]]), R=np.array([[1e-2]]),
172 x0=np.array([[3.0]]), P0=np.eye(1), **kwargs)
173
174 def f(self, x, u=None, w=None, t=None):
175 x = np.asarray(x, dtype=float).reshape(1, 1)
176 return x if w is None else x + np.asarray(w, dtype=float).reshape(1, 1)
177
178 def h(self, x, v=None, t=None):
179 z = self["gain"] * np.asarray(x, dtype=float).reshape(1, 1) + self["offset"]
180 return z if v is None else z + np.asarray(v, dtype=float).reshape(1, 1)

218 def test_spec_forms(self):
219 for spec in ((1.0, 0.1, 2.0), {"sigma": 1.0, "drift": 0.1, "tau": 2.0}):
220 self.assertEqual(Gauge().joint({"offset": spec}).estimated["offset"], (1.0, 0.1, 2.0))
221 self.assertEqual(Gauge().joint({"offset": (1.0, 0.1)}).estimated["offset"], (1.0, 0.1, 0.0))

224
225 def test_it_decays_towards_its_starting_value(self):
226 phi = np.exp(-0.1 / 2.0)
227 x = np.array([[3.0], [0.8]]) # the offset started at 0
228 walk, markov = Gauge().joint({"offset": (1.0, 0.1)}), Gauge().joint({"offset": (1.0, 0.1, 2.0)})
229 self.assertEqual(walk.f(x)[1, 0], 0.8)
230 self.assertAlmostEqual(markov.f(x)[1, 0], phi * 0.8, places=14)
231 self.assertAlmostEqual(markov.F(x)[1, 1], phi, places=14)
232
233 def test_its_variance_settles(self):
234 phi = np.exp(-0.1 / 2.0)
235 markov = Gauge().joint({"offset": (1.0, 0.1, 2.0)})
236 z = np.full((3000, 1), np.nan) # nothing is measured
237 with warnings.catch_warnings():
238 warnings.simplefilter("ignore")
239 record = markov.filter(z, smooth=False, reset=True)
