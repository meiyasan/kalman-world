1"""
2A sensor: what acquires a measurement.
3
4The Python face of ``Kalman::Sensor`` (``include/Kalman/Sensor.h``), with the same
5fields. A sensor reads ``k`` channels of a model's state,
6
7 z_sensor = g(x, t) + v, v ~ N(0, R)
8
9through a function ``g`` or a matrix ``G`` (``g(x) = G x``). Its channels are
10named: each is an *observable*, a quantity that can be measured, with its unit.
11
12 doppler = KalmanSensor("doppler", radial_velocity, R=1.0, labels={"V_RADIAL": "m/s"})
13 radar = shell.add_sensor(doppler) # a new model: the shell, read by one more sensor
14 radar.sensors # [position, doppler]
15
16The words, used the same way in both languages: an **observable** is a quantity
17that can be measured; a **sensor** measures one or more of them; a
18**measurement** is the numbers that come out.
19"""

27class KalmanSensor:
28 """One sensor.
29
30 Parameters
31 ----------
32 name : what the sensor is called.
33 g : ``g(x)`` or ``g(x, t=...)``, returning its ``k`` readings of the state ``x``
34 (a column). Give this or ``G``.
35 G : a ``k x n`` matrix, for readings linear in the state.
36 R : its noise: a ``k x k`` covariance, ``k`` variances, or one variance for
37 every channel.
38 labels : its observables: ``{"X": "m", "Y": "m"}`` (name and unit), a list of
39 names, or a ``KalmanObservable`` enum. Named after the sensor by default
40 (numbered when ``k > 1``).
41 jacobian : ``dg/dx`` as ``jacobian(x)`` or ``jacobian(x, t=...)``, ``k x n``;
42 differenced when not given.
43 rate : how often it reports [Hz, or per unit of the model's time]; ``None``
44 for every step of the model. Between two of its reports its channels are
45 missing (NaN), which the engines skip: see ``model.measurements``.
46 delay : how late it reports [s, or the model's unit of time]: it reads the
47 state as it was ``delay`` ago. The model must carry its past states:
48 ``model.add_sensor(sensor).lag(L)`` with ``L >= delay * fs`` steps.
49 """
50
51 #: The C++ sensor, for one written in C++ (``kalman.framework.sensors``): the
52 #: model then takes it as it is, and ``g`` is not called back during a pass.
53 native = None
54
55 #: For a sensor of a sequential instrument (a detector's layer): where it sits
56 #: along the sequence, the value handed to ``f`` as its time.
57 station = None
58
59 #: The numbers its readings depend on, by name (a camera's focal length, where
60 #: a radar is). The model that takes the sensor has each as its parameter
61 #: ``"<name>.<setting>"``, which ``joint`` can estimate.
62 settings = None

64 def __init__(self, name, g=None, *, G=None, R=None, labels=None, jacobian=None, rate=None, delay=None):
65 if not isinstance(name, str) or not name:
66 raise ValueError("KalmanSensor: a sensor needs a name.")
67 if (g is None) == (G is None):
68 raise ValueError(f"KalmanSensor: the sensor {name!r} needs a function g or a matrix G=, one of them.")
69 if R is None:
70 raise ValueError(f"KalmanSensor: the sensor {name!r} needs R=, its noise covariance.")
71 self.name = name
72 self.g = g
73 self.G = None if G is None else np.ascontiguousarray(np.asarray(G, dtype=np.float64))
74 if self.G is not None and self.G.ndim != 2:
75 raise ValueError(f"KalmanSensor: the G of {name!r} must be a k x n matrix.")
76 self.R = R
77 self.jacobian = jacobian
78 if rate is not None and not float(rate) > 0.0:
79 raise ValueError(f"KalmanSensor: the rate of {name!r} must be positive.")
80 self.rate = None if rate is None else float(rate)
81 if delay is not None and not float(delay) >= 0.0:
82 raise ValueError(f"KalmanSensor: the delay of {name!r} cannot be negative.")
83 self.delay = float(delay) if delay else None
84 self.labels = self._labels(labels)
85 # What the instrument that carries it gives it (see KalmanInstrument); filled
86 # for the sensors a model reports (model.sensors).
87 self.instrument = None
88 self.calibrated, self.gain, self.offset = False, 1.0, 0.0

91 @staticmethod
92 def _labels(labels):
93 """``[(name, unit), ...]`` or ``None``."""
94 if labels is None:
95 return None
96 if isinstance(labels, Mapping):
97 return [(str(k), "" if v is None else str(v)) for k, v in labels.items()]
98 members = getattr(labels, "__members__", None)
99 if members is not None: # a KalmanObservable enum
100 return [(k, v._value_ if isinstance(v._value_, str) else "") for k, v in members.items()]
101 return [(str(k), "") for k in labels]

103 @property
104 def is_linear(self) -> bool:
105 """True when the readings are a matrix."""
106 return self.G is not None

108 @property
109 def channels(self):
110 """The number of channels ``k``, or ``None`` while only ``g`` can tell
111 (a function with one variance for ``R`` and no labels)."""
112 if self.G is not None:
113 return int(self.G.shape[0])
114 R = np.asarray(self.R, dtype=np.float64)
115 if R.ndim >= 1:
116 return int(R.shape[0])
117 return len(self.labels) if self.labels is not None else None

119 def resolved(self, n, x0=None, t0=0.0):
120 """``(k, R, labels)`` against a state of ``n`` components: the number of
121 channels (asked of ``g`` at ``x0`` when nothing else says), the ``k x k``
122 covariance, and the ``(name, unit)`` of each observable it was given
123 (``None`` when it names none: the C++ sensor then names them after itself)."""
124 k = self.channels
125 if k is None:
126 x = np.zeros((n, 1)) if x0 is None else np.asarray(x0, dtype=float).reshape(n, 1)
127 k = int(np.asarray(self.measure(x, t0)).size)
128 R = np.asarray(self.R, dtype=np.float64)
129 R = np.ascontiguousarray(R * np.eye(k) if R.ndim == 0 else np.diag(R) if R.ndim == 1 else R)
130 return k, R, self.labels

132 def measure(self, x, t=None):
133 """The readings at state ``x``."""
134 if self.G is not None:
135 return self.G @ np.asarray(x, dtype=float).reshape(-1, 1)
136 return _timed(self.g)(x, t)
