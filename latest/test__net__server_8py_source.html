73class Server:
74 """``kalman-serve --rest`` on a free port, until stop()."""
75
76 def __init__(self, env=None, *args):
77 e = dict(os.environ)
78 e.pop("KALMAN_JWT_SECRET", None)
79 e.update(env or {})
80 self.proc = subprocess.Popen([SERVE, "--rest", "--port", "0", "--plugin", PLUGIN, *args],
81 stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, env=e, text=True)
82 line = self.proc.stderr.readline()
83 m = re.search(r":(\d+) ", line)
84 if not m:
85 self.proc.kill()
86 raise RuntimeError(f"kalman-serve did not start: {line!r} {self.proc.stderr.read()}")
87 self.url = f"http://127.0.0.1:{m.group(1)}"
88
89 def stop(self):
90 self.proc.terminate()
91 try:
92 self.proc.wait(timeout=10)
93 except subprocess.TimeoutExpired:
94 self.proc.kill()

96 def request(self, method, path, body=None, token=None):
97 data = None if body is None else json.dumps(body).encode()
98 headers = {"Content-Type": "application/json"}
99 if token:
100 headers["Authorization"] = f"Bearer {token}"
101 req = urllib.request.Request(self.url + path, data=data, method=method, headers=headers)
102 try:
103 with urllib.request.urlopen(req, timeout=60) as r:
104 return r.status, json.loads(r.read().decode())
105 except urllib.error.HTTPError as e:
106 return e.code, json.loads(e.read().decode() or "null")

138 a, b = np.asarray(a, dtype=float), np.asarray(b, dtype=float)

166 from modules import api
167 for spec in (CONSTANT_VELOCITY, PENDULUM):
168 zk = self._data(spec)
169 for smooth in (False, True):
170 status, c = self.server.request("POST", "/filter", {"spec": spec, "data": zk, "smooth": smooth})
171 self.assertEqual(status, 200, c)
172 p = api.run_filter(spec, zk, smooth=smooth)
173 if smooth and spec is PENDULUM:
174 # The Python RTS pass on a nonlinear model gains with
175 # transition[k] (the Jacobian at x_post[k-1]); Kalman::Filter
176 # with transition[k+1] (at x_post[k]), the RTS gain. They differ
177 # by ~5e-5 here, so the reference is the C++ pass driven from
178 # Python (NativeModel.filter_native), the same Kalman::Filter.
179 rec = api.make_model(spec).filter_native(zk, engine="extended", smooth=True)
180 p = {"xk": rec.xk, "Pk": rec.Pk, "time": rec.time}
181 self.assertClose(c["xk"], p["xk"], 1e-9, f"{spec['base']} xk smooth={smooth}")
182 self.assertClose(c["Pk"], p["Pk"], 1e-9, f"{spec['base']} Pk smooth={smooth}")

265 from modules import api
266 core.load_plugin(_plugin_path())

272 "params": {"protocolVersion": "2025-06-18", "capabilities": {},
273 "clientInfo": {"name": "pytest", "version": "0"}}},
274 {"jsonrpc": "2.0", "method": "notifications/initialized"},

278 ]

281 self.assertEqual(out.returncode, 0, out.stderr)
282 replies = [json.loads(line) for line in out.stdout.splitlines() if line.strip()]
283 self.assertEqual([r["id"] for r in replies], [1, 2, 3]) # no answer to the notification
284 self.assertEqual(replies[0]["result"]["serverInfo"]["name"], "Kalman-AI")

287 "list_sources", "list_events", "run_filter", "forecast", "load_plugin"}
288 self.assertTrue(python_tools <= names, python_tools - names)
289 result = replies[2]["result"]
290 self.assertFalse(result["isError"], result)

322 c = auth.claims(reader)
323 store.revoke_token(c["jti"], c["sub"], c["exp"], "test")
324 self.assertEqual(server.request("GET", "/catalog", token=reader)[0], 401)
