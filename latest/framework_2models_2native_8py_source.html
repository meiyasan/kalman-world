1"""
2A libKalman model as a :class:`~kalman.framework.model.KalmanModel`.
3
4:class:`NativeModel` holds an instance built by the C++ factory and forwards the
5model contract — ``f/h``, ``F/H/B/L``, ``Q/R``, the initial state — to it through
6the C API (:mod:`kalman.framework.native.core`). Everything above the contract is
7the ordinary Python framework: the engines, the filters, the sampler, the record,
8``plot``, the API verbs and the MCP tools all see just another ``KalmanModel``.
9
10That is the one crossing between the two implementations for *models*: the C++
11side owns the physics, the Python side owns everything a notebook does with it.
12A crossing costs one C call per ``f``/``h``/coefficient evaluation, which is what
13the measured numbers in ``docs/15_Cxx-Library.md`` say is affordable in this
14direction (a C call is cheap; it is a *Python* callback from C++ that is not).
15
16:func:`native_model_classes` turns every registered name into a class — ``NAME``,
17``DESCRIPTION``, ``VERSION`` and ``DEFAULT_ENGINE`` filled from the C++ metadata —
18so the catalogue, ``describe()`` and ``make_model()`` treat a plugin model exactly
19like a class in ``apps/``: ``{"base": "constant-velocity", "sensor": "cv"}``.
20"""

35class NativeModel(KalmanModel):
36 """A C++ model from the factory, driven by the Python engines.
37
38 ``NativeModel("constant-velocity", "sensor-1")`` constructs the registered
39 C++ model and adopts its dimensions, sample rate, engine binding and initial
40 state. ``fs=``, ``x0=``, ``P0=`` and ``t0=`` override those and are pushed
41 back into the C++ instance, so both sides agree on every step length.
42 """
43
44 NAME = "Native Model"
45 DESCRIPTION = "A libKalman model (C++) exposed through the C API."
46 VERSION = "1.0"

48 NATIVE_NAME: str | None = None # set on the classes native_model_classes() builds
49 DEFAULT_ENGINE = KalmanEngineKind.EXTENDED

52 def __init__(self, name: str | None = None, sensor=None, **kwargs):
53 L = core.require()
54 # A model that exists already (read from a file) is wrapped as it is.
55 handle = kwargs.pop("_handle", None)
56 if handle is not None:
57 name = name or core._s(L.kalman_model_name(handle))
58 else:
59 name = name or self.NATIVE_NAME
60 if not name:
61 raise ValueError("NativeModel needs the registered name of a C++ model.")
62 handle = L.kalman_model_create(name.encode())
63 if handle == core.ffi.NULL:
64 raise KeyError(f"NativeModel: {core.last_error()}")
65 self._handle = core.ffi.gc(handle, L.kalman_model_destroy)
66 self.native_name = name
67
68 n, m, l = core.ffi.new("int *"), core.ffi.new("int *"), core.ffi.new("int *")
69 core.check(L.kalman_model_dims(self._handle, n, m, l), "kalman_model_dims")
70 self.nnnn, self.mmmm, self.lll = int(n[0]), int(m[0]), int(l[0])
71
72 # Time base: an explicit fs wins and is pushed down; otherwise the C++ one is adopted.
73 fs = kwargs.pop("fs", None) or kwargs.pop("sample_rate", None)
74 native_fs = float(L.kalman_model_sample_rate(self._handle))
75 if fs is not None:
76 core.check(L.kalman_model_set_sample_rate(self._handle, float(fs)), "set_sample_rate")
77 else:
78 fs = native_fs if native_fs > 0 else None
79 if "t0" in kwargs and kwargs["t0"] is not None:
80 core.check(L.kalman_model_set_t0(self._handle, float(kwargs["t0"])), "set_t0")
81 else:
82 kwargs["t0"] = float(L.kalman_model_t0(self._handle))
83
84 # Initial state: same rule. A C++ P0 that was never set is all zeros,
85 # which the Python side reads as "infer from Q", as it does for any model.
86 x0 = kwargs.pop("x0", None)
87 if x0 is not None:
88 x0 = np.asarray(x0, dtype=float).reshape(-1)
89 core.check(L.kalman_model_set_x0(self._handle, core.cdptr(np.ascontiguousarray(x0))), "set_x0")
90 else:
91 x0 = self._vec(L.kalman_model_x0, "x0")
92 P0 = kwargs.pop("P0", None)
93 if P0 is not None:
94 P0 = np.ascontiguousarray(np.asarray(P0, dtype=float).reshape(self.nnnn, self.nnnn))
95 core.check(L.kalman_model_set_P0(self._handle, core.cdptr(P0)), "set_P0")
96 else:
97 P0 = self._mat(L.kalman_model_P0, (self.nnnn, self.nnnn), "P0")
98 if not np.any(P0):
99 P0 = None
100
101 # Noise and named parameters live in the C++ instance (Q(), R() and
102 # parameters() read them from there), so what is passed here is pushed
103 # down, not kept on the Python side where nothing would read it.
104 for key, setter, size in (("Q", L.kalman_model_set_Q, self.nnnn), ("R", L.kalman_model_set_R, self.mmmm)):
105 value = kwargs.pop(key, None)
106 if value is not None:
107 value = np.ascontiguousarray(np.asarray(value, dtype=float).reshape(size, size))
108 core.check(setter(self._handle, core.cdptr(value)), "set_" + key)
109 # A model that computes its noise from its own parameters does
110 # not take a matrix: say so instead of running with another one.
111 getter = L.kalman_model_Q if key == "Q" else L.kalman_model_R
112 if not np.allclose(self._mat(getter, (size, size), key, 0.0), value, rtol=1e-12, atol=0.0):
113 raise ValueError(
114 f"the C++ model {name!r} computes its own {key} and does not take one; "
115 f"set its parameters instead ({', '.join(sorted(self.parameters())) or 'it declares none'}).")
116 declared = self.parameters()
117 for key in [k for k in kwargs if k in declared]:
118 self.set_parameter(key, kwargs.pop(key))
119
120 self.DEFAULT_ENGINEDEFAULT_ENGINE = ENGINE_BY_INT.get(
121 core.check(L.kalman_model_engine(self._handle), "engine"), KalmanEngineKind.EXTENDED)
122 self._analytic = core.check(L.kalman_model_analytic_jacobian(self._handle), "analytic") == 1
123
124 super().__init__(sensor if sensor is not None else name,
125 fs=fs, x0=x0.reshape(-1, 1), P0=P0, **kwargs)

128 """The C++ model itself, so that a filtering pass runs wholly in
129 Kalman::Filter instead of asking for f, h and the matrices step by step.
130 ``None`` for a subclass that rewrites part of the model in Python."""
131 import os
132 if not USE_NATIVE_COUNTERPARTS or os.environ.get("KALMAN_NATIVE_COUNTERPARTS", "1") == "0":
133 return None
134 for name in ("f", "h", "F", "H", "B", "L", "Q", "R"):

136 return None

140 def _vec(self, fn, what, *args):
141 out = np.empty(self.nnnn if what != "z" else self.mmmm, dtype=np.float64)
142 core.check(fn(self._handle, *args, core.dptr(out)), what)

145 def _mat(self, fn, shape, what, *args):
146 out = np.empty(shape, dtype=np.float64)
147 core.check(fn(self._handle, *args, core.dptr(out)), what)
148 return out

150 def _mat_optional(self, fn, shape, what, t):
151 out = np.empty(shape, dtype=np.float64)
152 return out if core.check(fn(self._handle, t, core.dptr(out)), what) == 1 else None

154 @staticmethod
155 def _t(t):
156 return 0.0 if t is None else float(t)

158 def _flat(self, v, size, what):
159 a = np.ascontiguousarray(np.asarray(v, dtype=np.float64).reshape(-1))
160 if a.size != size:
161 raise ValueError(f"{self.native_name}: {what} has {a.size} elements, expected {size}.")
162 return a

164 def _u(self, u):
165 if u is None or self.lll == 0:
166 return core.ffi.NULL
167 return core.cdptr(self._flat(u, self.lll, "u"))

169 @staticmethod
170 def _like(out, x):
171 # Column in, column out; flat in, flat out — the engines use both.
172 return out.reshape(-1, 1) if np.ndim(x) == 2 else out

174 # --- the contract: coefficients --------------------------------------------
175 def F(self, x=None, u=None, t=None):
176 L = core.lib
177 if self._analytic:
178 return self._mat(L.kalman_model_F, (self.nnnn, self.nnnn), "F", self._t(t))
179 xf = self._flat(self.x0x0x0x0 if x is None else x, self.nnnn, "x")
180 return self._mat(L.kalman_model_jacobian, (self.nnnn, self.nnnn), "df/dx",
181 core.cdptr(xf), self._u(u), self._t(t))
182
183 def H(self, x=None, t=None):
184 L = core.lib
185 if self._analytic:
186 return self._mat(L.kalman_model_H, (self.mmmm, self.nnnn), "H", self._t(t))
187 xf = self._flat(self.x0x0x0x0 if x is None else x, self.nnnn, "x")
188 return self._mat(L.kalman_model_jacobian_h, (self.mmmm, self.nnnn), "dh/dx",
189 core.cdptr(xf), self._t(t))
190
191 def B(self, x=None, u=None, t=None):
192 if self.lll == 0:
193 return None
194 return self._mat_optional(core.lib.kalman_model_B, (self.nnnn, self.lll), "B", self._t(t))
195
196 def L(self, x=None, u=None, w=None, t=None):
197 return self._mat_optional(core.lib.kalman_model_L, (self.nnnn, self.nnnn), "L", self._t(t))
198
199 def M(self, x=None, v=None, t=None):
200 return None # additive measurement noise
201
202 def Q(self, t=None):
203 return self._mat(core.lib.kalman_model_Q, (self.nnnn, self.nnnn), "Q", self._t(t))
204
205 def R(self, t=None):
206 return self._mat(core.lib.kalman_model_R, (self.mmmm, self.mmmm), "R", self._t(t))
207
208 # --- the contract: maps ----------------------------------------------------
209 def f(self, x, u=None, w=None, t=None):
210 xf = self._flat(x, self.nnnn, "x")
211 out = np.empty(self.nnnn, dtype=np.float64)
212 core.check(core.lib.kalman_model_f(self._handle, core.cdptr(xf), self._u(u), self._t(t),
213 core.dptr(out)), "f")
214 if w is not None:
215 Lw = self.LLLL(t=t)
216 wf = self._flat(w, self.nnnn, "w")
217 out = out + (wf if Lw is None else Lw @ wf)
218 return self._like(out, x)
219
220 def h(self, x, v=None, t=None):
221 xf = self._flat(x, self.nnnn, "x")
222 out = np.empty(self.mmmm, dtype=np.float64)
223 core.check(core.lib.kalman_model_h(self._handle, core.cdptr(xf), self._t(t), core.dptr(out)), "h")
224 if v is not None:
225 out = out + self._flat(v, self.mmmm, "v")
226 return self._like(out, x)

229 def run_native(self, zk, uk=None, *, engine="extended", smooth=False, **engine_params) -> dict:
230 """Kalman::Filter over ``zk`` entirely in C++ with a native ``engine`` (name
231 or kind, plus its constructor parameters). Returns the per-step arrays:
232 ``x_prior/P_prior/x_post/P_post/innovations/transitions`` and, with
233 ``smooth=True``, ``x_smooth/P_smooth``."""
234 eng = core.create_engine(engine.name.lower() if hasattr(engine, "name") else str(engine).lower(),
235 **engine_params)
236 return core.run_filter(self._handle, eng, self.nnnn, self.mmmm, self.lll, zk, uk, smooth=smooth)

238 def filter_native(self, zk, uk=None, *, engine="extended", smooth=False, **engine_params):
239 """:meth:`run_native` as a :class:`~kalman.framework.io.record.KalmanRecord` —
240 posterior, or smoothed when ``smooth=True`` — so it plots, records and
241 compares like any Python filter's output."""
242 from ..io.record import KalmanRecord
243 from ..constants import KalmanStateFlavor
244 out = self.run_native(zk, uk, engine=engine, smooth=smooth, **engine_params)
245 N = out["x_post"].shape[0]
246 t0 = float(self.t0t0 or 0.0)
247 time = t0 + np.arange(N, dtype=float) / (self.fsfsfs or 1.0)
248 zk = np.asarray(zk, dtype=float).reshape(N, self.mmmm)
249 if smooth:
250 return KalmanRecord(time=time, xk=out["x_smooth"], Pk=out["P_smooth"], zk=zk,
251 model=self, flavor=KalmanStateFlavor.SMOOTHED)
252 return KalmanRecord(time=time, xk=out["x_post"], Pk=out["P_post"], zk=zk,
253 model=self, flavor=KalmanStateFlavor.POSTERIOR)

388class NativeCounterpart:
389 """Mixin for an application model whose physics also exists in libKalmanApps.
390
391 The Python class stays the model: its constructor, its attributes, ``plot``
392 and the notebooks are unchanged. What changes is who runs the filtering
393 pass. When the configuration has a C++ counterpart, ``KalmanFilter.forward``
394 hands the whole pass to the counterpart (``Kalman::Filter`` in C++, one call), and
395 otherwise the Python loop runs as before.
396
397 A model declares its counterpart with two methods:
398
399 - :meth:`native_name` — the registered C++ name for *this* configuration, or
400 ``None`` when it has none (a Python callback, a feature not ported);
401 - :meth:`native_parameters` — the scalars to push, by the names the C++
402 class declares.
403
404 The sample rate, ``Q`` and ``R`` are pushed from the Python instance on every
405 run, and the initial state by the filter, so nothing is defined twice and a
406 change made to the Python model after construction reaches the counterpart.
407 """
408
409 #: What must be exactly the declaring class's own for the counterpart to stand in.
410 #: A continuous model's class keeps what it wrote as ``f`` under
411 #: ``derivative`` (KalmanModel.__init_subclass__), so its ``f`` is the shared
412 #: decoded step: the derivative, the closing step and the sub-stepping are
413 #: what a subclass changes, and are checked too.
414 _COUNTERPART_CONTRACT = ("f", "h", "F", "H", "B", "L", "Q", "R", "derivative", "after_step", "substeps")
415
416 def native_name(self) -> str | None:
417 return None
418
419 def native_parameters(self) -> dict:
420 return {}
421
422 def native_arrays(self) -> dict:
423 """The named arrays to push (the C++ class's ``DeclareArray`` names). The
424 counterpart may size itself from them."""
425 return {}
426
427 #: Stored matrices the counterpart computes itself from what is pushed (``("_F",)``
428 #: for a model built with ``A=``): they do not keep the counterpart from standing in.
429 _COUNTERPART_DECODES = ()
430
431 def _counterpart_owner(self) -> type:
432 for cls in type(self).__mro__:
433 if "native_name" in cls.__dict__ and cls is not NativeCounterpart:
434 return cls
435 return NativeCounterpart

437 def native_handle(self):
438 """The configured C++ counterpart, or ``None`` when the Python physics must run."""
439 import os
440 if not USE_NATIVE_COUNTERPARTS or os.environ.get("KALMAN_NATIVE_COUNTERPARTS", "1") == "0":
441 return None
442 if not core.HAS_CXX_CORE:
443 return None
444 name = self.native_name()
445 if not name or name not in apps_library_names():
446 return None
447 # A subclass or an instance that redefines the contract is a different
448 # model from the one that was ported.
449 owner = self._counterpart_owner()
450 for attr in self._COUNTERPART_CONTRACT:
451 if attr in self.__dict__ or getattr(type(self), attr, None) is not getattr(owner, attr, None):
452 return None
453 # A control input is fine: the counterpart's f takes u, and the filter hands
454 # the counterpart the control sequence (a controller function keeps the Python
455 # model, see KalmanFilter._native_counterpart).
456 if np.dtype(getattr(self, "dtype", np.float64)) != np.float64:
457 return None
458 if any(getattr(self, attr) is not None for attr in ("_F", "_H") if attr not in self._COUNTERPART_DECODES_COUNTERPART_DECODES):
459 return None
460 if callable(self._Q) or callable(self._R):
461 return None
462 L_ = self._L
463 if L_ is not None and (callable(L_) or not np.array_equal(np.asarray(L_), np.eye(self.n))):
464 return None
465
466 lib = core.lib
467 cached = self.__dict__.get("_native_counterpart")
468 if cached is None or cached[0] != name:
469 handle = lib.kalman_model_create(name.encode())
470 if handle == core.ffi.NULL:
471 raise KeyError(f"{type(self).__name__}: {core.last_error()}")
472 handle = core.ffi.gc(handle, lib.kalman_model_destroy)
473 self._push_arrays(handle) # a counterpart may size itself from them
474 n, m, l = core.ffi.new("int *"), core.ffi.new("int *"), core.ffi.new("int *")
475 core.check(lib.kalman_model_dims(handle, n, m, l), "kalman_model_dims")
476 if (int(n[0]), int(m[0]), int(l[0])) != (self.n, self.m, self.l):
477 raise RuntimeError(
478 f"{type(self).__name__}: the C++ counterpart {name!r} has (n, m, l) = "
479 f"({n[0]}, {m[0]}, {l[0]}), the Python model ({self.n}, {self.m}, {self.l}).")
480 self._native_counterpart = cached = (name, handle)
481 handle = cached[1]
482
483 if self.fs:
484 core.check(lib.kalman_model_set_sample_rate(handle, float(self.fs)), "set_sample_rate")
485 # The counterpart linearises with the scheme the Python model declares.

642def handle_of(model, parameters=()):
643 """``model`` as a C++ model, for an operation (joint, observe ...) to wrap.
644
645 Returns ``(handle, keep, refresh)``: ``keep`` must stay referenced for as long
646 as the handle is used, and ``refresh()`` pushes the Python model's current
647 settings into the handle again (call it before a pass).
648
649 In order: the C++ model itself (a :class:`NativeModel`, which another
650 operation's result is), the model's counterpart in libKalmanApps, a model of its
651 constant matrices, and otherwise a ``Kalman::Models::Callback`` calling its
652 Python methods. ``parameters`` names what the handle must declare: a counterpart that
653 lacks one, or a model of matrices, is passed over for the callback model, on
654 which they are declared and set back on the Python model
655 (``kalman_model_callback_declare_parameters``).
656 """
657 from ..constants import KalmanDifferentiatorKind, KalmanStructureKind
658 lib, ffi = core.require(), core.ffi
659 names = list(parameters)
660
661 def missing(available):
662 return KeyError(f"{type(model).__name__} has no parameter {[n for n in names if n not in available]}; "
663 f"it has {sorted(available)}.")
664
665 if isinstance(model, NativeModel):
666 refresh = getattr(model, "_refresh", lambda: None)
667 refresh()
668 declared = declared_parameters(model._handle)
669 if any(name not in declared for name in names):
670 raise missing(declared)
671 return model._handle, (model,), refresh
672
673 counterpart_of = getattr(model, "native_handle", None)
674 counterpart = counterpart_of() if counterpart_of is not None else None
675 counterpart_declares = declared_parameters(counterpart) if counterpart is not None else {}
676 if counterpart is not None and all(name in counterpart_declares for name in names):
677 _push_initial_state(model, counterpart)
678 _push_labels(model, counterpart)
679 return counterpart, (model,), counterpart_of # native_handle() pushes Q, R, fs, parameters
680
681 if any(name not in model.keys() for name in names):
682 raise missing(set(model.keys()) | set(counterpart_declares))
683
684 if not names and getattr(model, "structure", None) is KalmanStructureKind.MATRICES \
685 and np.dtype(getattr(model, "dtype", np.float64)) == np.float64:
686 from ...native import _stand_in
687 try:
688 stand_in = _stand_in(model, KalmanStructureKind.MATRICES, False)
689 _push_labels(model, stand_in)
690
691 def push_matrices(n=int(model.n), m=int(model.m)):
692 # What may have been retuned since: the noise and the two matrices.
693 flat = lambda a, shape: np.ascontiguousarray(np.asarray(a, dtype=np.float64).reshape(shape))
694 if model.fs:
695 core.check(lib.kalman_model_set_sample_rate(stand_in, float(model.fs)), "set_sample_rate")
696 core.check(lib.kalman_model_set_Q(stand_in, core.cdptr(flat(model.Q(), (n, n)))), "set_Q")
697 core.check(lib.kalman_model_set_R(stand_in, core.cdptr(flat(model.R(), (m, m)))), "set_R")
698 core.check(lib.kalman_model_set_matrix(stand_in, b"F", core.cdptr(flat(model.F(), (n, n)))), "set F")
699 core.check(lib.kalman_model_set_matrix(stand_in, b"H", core.cdptr(flat(model.H(), (m, n)))), "set H")
700 return stand_in, (model,), push_matrices
701 except (TypeError, ValueError, AttributeError):
702 pass # matrices the C++ model does not take
703
704 handle, keep = callback_handle(model, "extended", live_noise=True)
705 # Differenced as the Python model differences its own Jacobians.
706 scheme = "central" if getattr(model, "DIFFERENTIATOR", None) is KalmanDifferentiatorKind.CENTRAL else "forward"
707 core.check(lib.kalman_model_set_differentiator(handle, core.DIFFERENTIATOR_KINDS[scheme],
708 1e-5 if scheme == "central" else 1e-6), "set_differentiator")
709 _push_initial_state(model, handle)
710 _push_labels(model, handle)
711 # Every number the model keeps is declared, not only those asked for: an
712 # operation around this handle may name one later (joint around observe).
713 names = names + [key for key in model.keys() if key not in names
714 and isinstance(model[key], (int, float)) and not isinstance(model[key], bool)]
715 if not names:
716 return handle, (model, keep), lambda: None
717
718 errors = keep[-1]
719
720 @ffi.callback("int(void *, const char *, double)")
721 def set_parameter(user, name, value):
722 try:
723 model[ffi.string(name).decode()] = float(value)
724 return 0
725 except BaseException as exc: # must not unwind through C++
726 errors.append(exc)
727 return 1
728
729 encoded = [ffi.new("char[]", name.encode()) for name in names]
730 values = np.ascontiguousarray([float(model[name]) for name in names], dtype=np.float64)
731 core.check(lib.kalman_model_callback_declare_parameters(
732 handle, len(names), ffi.new("const char *[]", encoded), core.cdptr(values), set_parameter),
733 "declare_parameters")
734 return handle, (model, keep, set_parameter), lambda: None

747# --- a Python-written model on the C++ engines ----------------------------------
748def callback_handle(model, engine="extended", live_noise=False):
749 """A C++ ``Kalman::Models::Callback`` whose ``f``, ``h`` and Jacobians call
750 ``model``'s own Python methods, so the C++ filter can run a model written in
751 Python. Returns ``(handle, keep)``: ``keep`` holds the callback objects and
752 must stay referenced for as long as the handle is used.
753
754 The Jacobians are the model's ``F(x, u, t)`` and ``H(x, t)``, analytic where
755 the model defines them and differenced by the Python framework otherwise, so
756 both sides linearise alike. ``Q`` and ``R`` are pushed as constants (``Q``
757 with the noise coupling ``L`` applied); the sample rate too. A Python
758 exception inside a callback stops the pass and is re-raised by the caller
759 (see :func:`run_callback_filter`).
760 """
761 lib = core.require()
762 ffi = core.ffi
763 n, m, l = int(model.n), int(model.m), int(model.l)
764 errors: list[BaseException] = []
765
766 def col(ptr, size):
767 return np.frombuffer(ffi.buffer(ptr, size * 8), dtype=np.float64).reshape(size, 1).copy()
768
769 def guard(fn):
770 def wrapped(*args):
771 try:
772 fn(*args)
773 return 0
774 except BaseException as exc: # must not unwind through C++
775 errors.append(exc)
776 return 1
777 return wrapped
778
779 def write(out, values, size):
780 ffi.memmove(out, np.ascontiguousarray(np.asarray(values, dtype=np.float64).reshape(-1)), size * 8)
781
782 def control(u):
783 return None if (u == ffi.NULL or l == 0) else col(u, l)
784
785 @ffi.callback("int(void *, const double *, const double *, double, double *)")
786 @guard
787 def step(user, x, u, t, out):
788 write(out, model.f(col(x, n), u=control(u), t=t), n)
789
790 @ffi.callback("int(void *, const double *, double, double *)")
791 @guard
792 def measure(user, x, t, out):
793 write(out, model.h(col(x, n), t=t), m)
794
795 @ffi.callback("int(void *, const double *, const double *, double, double *)")
796 @guard
797 def step_jacobian(user, x, u, t, out):
798 write(out, model.F(x=col(x, n), u=control(u), t=t), n * n)
799
800 @ffi.callback("int(void *, const double *, double, double *)")
801 @guard
802 def measure_jacobian(user, x, t, out):
803 try:
804 H = model.H(x=col(x, n), t=t)
805 except TypeError:
806 # A model whose h takes no noise argument: the base class cannot
807 # difference it (it passes v=). Only the sigma-point engines accept
808 # such a model, and they use H for the reported innovation alone.
809 from ...helpers import numerical_jacobian
810 H = numerical_jacobian(lambda xx: model.h(xx, t=t), col(x, n))
811 write(out, H, m * n)
812
813 handle = lib.kalman_model_create_callback(type(model).__name__.encode(), n, m, l,
814 step, measure, step_jacobian, measure_jacobian, ffi.NULL)
815 if handle == ffi.NULL:
816 raise RuntimeError(f"callback model: {core.last_error()}")
817 handle = ffi.gc(handle, lib.kalman_model_destroy)
818
819 if model.fs:
820 core.check(lib.kalman_model_set_sample_rate(handle, float(model.fs)), "set_sample_rate")
821
822 keep = [step, measure, step_jacobian, measure_jacobian]
823 if engine in ("unscented", "particle"):
824 keep += _unscented_noise(model, handle, guard, col, control, write)
825 elif constant_noise(model) and not live_noise:
826 Lm = np.asarray(model._L, dtype=np.float64)
827 Q = Lm @ np.asarray(model.Q(), dtype=np.float64).reshape(n, n) @ Lm.T
828 R = np.asarray(model.R(), dtype=np.float64).reshape(m, m)
829 core.check(lib.kalman_model_set_Q(handle, core.cdptr(np.ascontiguousarray(Q))), "set_Q")
830 core.check(lib.kalman_model_set_R(handle, core.cdptr(np.ascontiguousarray(R))), "set_R")
831 else:
832 # The noise as the Python engine works it out at each step: Q at the
833 # state the step starts from when Q takes one, then the coupling L.
834 from ..engine import KalmanEngine
835 takes_state = KalmanEngine._accepts_param(model.Q, "x")
836
837 @ffi.callback("int(void *, const double *, const double *, double, double *)")
838 @guard
839 def process_noise(user, x, u, t, out):
840 xc, uc = col(x, n), control(u)
841 Q = model.Q(t=t, x=xc) if takes_state else model.Q(t=t)
842 Lm = KalmanEngine._safe_L(model, xc, uc, t)
843 if Lm is not None:
844 Q = Lm @ Q @ Lm.T
845 write(out, Q, n * n)
846
847 @ffi.callback("int(void *, double, double *)")
848 @guard
849 def measurement_noise(user, t, out):
850 write(out, model.R(t=t), m * m)
851
852 core.check(lib.kalman_model_callback_set_noise(handle, process_noise, measurement_noise), "set_noise")
853 keep += [process_noise, measurement_noise]
854 return handle, (*keep, errors)

857def _unscented_noise(model, handle, guard, col, control, write):
858 """The noise hooks of the unscented engine, as the Python engine applies
859 them: where ``f`` takes ``w`` (or ``h`` takes ``v``) the noise goes through
860 the map and the covariance is handed over as it is; otherwise it is added
861 afterwards, coupled by ``L`` (or ``M``)."""
862 from ..engine import KalmanEngine
863 from ..engines.unscented import UnscentedKalmanEngine as U
864 lib, ffi = core.lib, core.ffi
865 n, m = int(model.n), int(model.m)
866 through_f = KalmanEngine._noise_through_f(model)
867 through_h = KalmanEngine._noise_through_h(model)
868
869 @ffi.callback("int(void *, const double *, const double *, double, double *)")
870 @guard
871 def process_noise(user, x, u, t, out):
872 Q = U._Q(model, t)
873 if not through_f:
874 Lm = U._L(model, col(x, n), control(u), t)
875 Q = Lm @ Q @ Lm.T
876 write(out, Q, n * n)
877
878 @ffi.callback("int(void *, double, double *)")
879 @guard
880 def measurement_noise(user, t, out):
881 R = U._R(model, t)
882 if not through_h:
883 Mm = U._M(model, np.asarray(model.x0, dtype=np.float64).reshape(-1, 1), t)
884 R = Mm @ R @ Mm.T
885 write(out, R, m * m)
886
887 @ffi.callback("int(void *, const double *, const double *, const double *, double, double *)")
888 @guard
889 def noisy_step(user, x, u, w, t, out):
890 write(out, model.f(col(x, n), u=control(u), w=col(w, n), t=t), n)
891
892 @ffi.callback("int(void *, const double *, const double *, double, double *)")
893 @guard
894 def noisy_measure(user, x, v, t, out):
895 write(out, model.h(col(x, n), v=col(v, m), t=t), m)
896
897 core.check(lib.kalman_model_callback_set_noise(handle, process_noise, measurement_noise), "set_noise")
898 core.check(lib.kalman_model_callback_set_noise_inputs(
899 handle, noisy_step if through_f else ffi.NULL, noisy_measure if through_h else ffi.NULL), "set_noise_inputs")
900 return [process_noise, measurement_noise, noisy_step, noisy_measure]

903def run_callback_filter(model, zk, uk=None, *, x0=None, P0=None, smooth=False,
904 engine="extended", control=None, **engine_params) -> dict:
905 """Filter ``zk`` with a C++ engine (``"extended"`` or ``"unscented"``, with
906 its constructor parameters) on a Python-written ``model``. Returns the
907 per-step arrays of :func:`core.run_filter`."""
908 lib = core.require()
909 # A controller may retune the model while it runs (adaptive Q and R), so in
910 # closed loop the noise is asked for at every step, never handed over once.
911 # The engine by name, or a C++ engine already made (the particle engine on numpy's draws).
912 kind = engine if isinstance(engine, str) else {v: k for k, v in core.ENGINE_KINDS.items()}.get(
913 int(core.lib.kalman_engine_kind(engine)), "extended")
914 handle, keep = callback_handle(model, kind, live_noise=control is not None)
915 n = int(model.n)
916 x0 = np.ascontiguousarray(np.asarray(model.x0 if x0 is None else x0, dtype=np.float64).reshape(-1))
917 P0 = np.ascontiguousarray(np.asarray(model.P0 if P0 is None else P0, dtype=np.float64).reshape(n, n))
918 core.check(lib.kalman_model_set_x0(handle, core.cdptr(x0)), "set_x0")
919 core.check(lib.kalman_model_set_P0(handle, core.cdptr(P0)), "set_P0")
920 core.check(lib.kalman_model_set_t0(handle, float(model.t0) if model.t0 is not None else 0.0), "set_t0")
921 try:
922 made = engine if not isinstance(engine, (str, int)) else core.create_engine(engine, **engine_params)
923 return core.run_filter(handle, made, n, int(model.m),
924 int(model.l), zk, uk, smooth=smooth, control=control)
925 except RuntimeError:
926 if keep[-1]:
927 raise keep[-1][0] # the Python exception, not its C++ shadow
928 raise

945 """``{registered name -> NativeModel subclass}`` for every model the C++
946 factory currently knows, including those loaded from plugins since the last
947 call. Empty when the library is not available."""
948 if not core.HAS_CXX_CORE:
949 return {}
950 for name in core.names():
951 # libKalmanApps' models are reached through their Python classes
952 # (StormModel, BallisticModel ...), not listed a second time here.
953 if name in _CLASSES or name in (_apps_names or ()):
954 continue
955 info = core.describe(name)
