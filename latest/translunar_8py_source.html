3"""
4Space science — travelling through space (the companion of :mod:`kalman.apps.rocket`,
5which handles takeoff and landing near the pad).
6
7Two regimes, both tracked with an Extended Kalman Filter:
8
9* **Fly to the Moon** — :class:`TransLunarModel`: a geocentric restricted four-body coast
10 (Earth + Moon + Sun), the Apollo free-return and its lunar **gravitational catapult**.
11 With ``use_ephemeris=True`` the Moon/Sun starting geometry is taken from the real
12 ephemeris (astropy) at a chosen ``epoch``.
13
14* **Interplanetary travel** — :class:`InterplanetaryModel`: a heliocentric **N-body** cruise
15 (Sun + planetary perturbations) general over any *departure → target* planet (Mars is the
16 worked example). Built up **gradually**:
17
18 1. :func:`hohmann_transfer` — analytic two-impulse estimate (Δv, time of flight, phasing,
19 synodic period, next ideal departure);
20 2. :func:`lambert` / :func:`porkchop` / :func:`best_departure` — Lambert targeting and a
21 porkchop scan over the live ephemeris to find the **minimum-Δv launch window**, refined
22 to a time-of-day at a real launch site (Earth rotation);
23 3. :class:`InterplanetaryModel` — the full N-body EKF cruise, seeded by the Lambert
24 solution, with impulsive mid-course Δv maneuvers as the control input.
25
26Positions are in scaled units (Mm), velocities in km/s; physical constants and the ephemeris
27come from :mod:`kalman.runtime.astro`.
28"""

64def _kep_arc(r0, v0, tof_ks, n=200, mu=None):
65 """Two-body RK4 arc: returns (n+1, 3) position array in Mm."""
66 if mu is None:
67 mu = GM_SUN
68 def _deriv(s, t): return np.r_[s[3:], -mu * s[:3] / np.linalg.norm(s[:3]) ** 3]
69 s = np.concatenate([r0, v0]); dt = tof_ks / n
70 pts = [s[:3].copy()]
71 for _ in range(n):
72 s = _rk4(s, 0, dt, _deriv)
73 pts.append(s[:3].copy())
74 return np.array(pts)

77def _orbit_ellipse(body, t_ref=None, n=300):
78 """True heliocentric orbit path for *body* (one full period, two-body RK4).
79
80 Returns an (n+1, 2) array of (x, y) in AU, or ``None`` for hyperbolic orbits.
81 Using the actual ephemeris position+velocity avoids the circular-orbit approximation
82 and gives the correct elliptical shape (eccentricity, inclination projection).
83 """
84 r0, v0 = body_state(body, t_ref)
85 r0_mag = np.linalg.norm(r0)
86 v0_sq = float(np.dot(v0, v0))
87 denom = 2.0 * GM_SUN / r0_mag - v0_sq
88 if denom <= 0:
89 return None
90 a = GM_SUN / denom
91 T_ks = 2.0 * np.pi * np.sqrt(a ** 3 / GM_SUN)
92 pts = _kep_arc(r0, v0, T_ks, n=n)
93 return pts[:, :2] / D_SUN

99def _third_body(r, r_body, GM):
100 """Third-body perturbation in a frame centred on the primary: ``GM[(r_b-r)/|r_b-r|³ −
101 r_b/|r_b|³]``. The first term is the body's direct pull on the spacecraft; the second
102 (indirect) subtracts its pull on the central body, so a distant body contributes only its
103 true (tiny) tidal effect rather than a spurious bulk force."""
104 d = r_body - r
105 return GM * (d / np.linalg.norm(d) ** 3 - r_body / np.linalg.norm(r_body) ** 3)

108def gravity(r, t, moon_phase=0.0, sun_phase=0.0, bodies=("earth", "moon", "sun")):
109 """Geocentric acceleration on a spacecraft at ``r`` (scaled): Earth's central pull plus
110 the **third-body perturbations** of the Moon and Sun (analytic circular orbits)."""
111 r = np.asarray(r, dtype=float).reshape(3)
112 a = np.zeros(3)
113 if "earth" in bodies:
114 a += -GM_EARTH * r / np.linalg.norm(r) ** 3
115 if "moon" in bodies:
116 a += _third_body(r, moon_position(t, moon_phase), GM_MOON)
117 if "sun" in bodies:
118 a += _third_body(r, sun_position(t, sun_phase), GM_SUN)
119 return a

158class TransLunarModel(NativeTwin, ExtendedKalmanModel):
159 """
160 Restricted four-body trans-lunar model (Earth + Moon + Sun).
161
162 State ``x = [rx, ry, rz, vx, vy, vz]`` (position in Mm, velocity in km/s); only the
163 position is measured (ground-station tracking). ``f()`` is one RK4 step of the multi-body
164 gravity ODE; the Moon and Sun ride on circular orbits. Pass ``use_ephemeris=True`` with an
165 ``epoch`` (anything :func:`astropy.time.Time` accepts, or ``None`` = now) to seed the Moon
166 and Sun **phases from the real ephemeris** at that date, so the flyby geometry matches a
167 genuine launch window while keeping the fast analytic propagation.
168 """
169
170 NAME = "Trans-Lunar (Earth+Moon+Sun)"; VERSION = "1.0"

174 @property

177 def __init__(self, moon_phase_deg=125.0, sun_phase_deg=60.0, dt_s=120.0,
178 track_km=5.0, bodies=("earth", "moon", "sun"), n_sub=1,
179 use_ephemeris=False, epoch=None, q_pos=1e-6, q_vel=1e-7, **kw):
180 self.dtypedtype = np.float64
181 if use_ephemeris: # real Moon/Sun geometry at the epoch
182 rm, _ = body_state("moon", epoch, origin="geocentric")
183 rs, _ = body_state("sun", epoch, origin="geocentric")
184 moon_phase_deg = np.degrees(np.arctan2(rm[1], rm[0]))
185 sun_phase_deg = np.degrees(np.arctan2(rs[1], rs[0]))
186 self.use_ephemeris = bool(use_ephemeris)
187 self.epoch = epoch
188 self.moon_phase = np.deg2rad(moon_phase_deg)
189 self.sun_phase = np.deg2rad(sun_phase_deg)
190 self.bodies = tuple(bodies)
191 self._dt = dt_s / TU
192 self.n_sub = max(1, int(n_sub)) # internal RK4 substeps per f() call
193 self.nnnnn, self.mmmm, self.lll = 6, 3, 0
194 self.fsfsfsfsfs = 1.0 / self._dt
195 Q = np.diag([q_pos, q_pos, q_pos, q_vel, q_vel, q_vel]).astype(float)
196 R = (track_km * KM) ** 2 * np.eye(3)
197 super().__init__("ROCKET", Q=Q, R=R, fs=self.fsfsfsfsfs, **kw)
