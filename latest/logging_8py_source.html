25def add_custom_level(levelName, levelNum, methodName=None):
26 """
27 Comprehensively adds a new logging level to the `logging` module and the
28 currently configured logging class. (Stack Overflow)
29
30 `levelName` becomes an attribute of the `logging` module with the value
31 `levelNum`. `methodName` becomes a convenience method for both `logging`
32 itself and the class returned by `logging.getLoggerClass()` (usually just
33 `logging.Logger`). If `methodName` is not specified, `levelName.lower()` is
34 used.
35
36 To avoid accidental clobberings of existing attributes, this method will
37 raise an `AttributeError` if the level name is already an attribute of the
38 `logging` module or if the method name is already present
39
40 Example
41 -------
42 >>> logging.addLevel('TRACE', logging.DEBUG - 5)
43 >>> logging.getLogger(__name__).setLevel("TRACE")
44 >>> logging.getLogger(__name__).trace('that worked')
45 >>> logging.trace('so did this')
46 >>> logging.TRACE
47
48 """
49 if not methodName:
50 methodName = levelName.lower()
51
52 if hasattr(logging, levelName):
53 raise AttributeError('{} already defined in logging module'.format(levelName))
54 if hasattr(logging, methodName):
55 raise AttributeError('{} already defined in logging module'.format(methodName))
56 if hasattr(logging.getLoggerClass(), methodName):
57 raise AttributeError('{} already defined in logger class'.format(methodName))

59 # This method was inspired by the answers to Stack Overflow post
60 # http://stackoverflow.com/q/2183233/2988730, especially
61 # http://stackoverflow.com/a/13638084/2988730
62 def logForLevel(self, message, *args, **kwargs):
63 if self.isEnabledFor(levelNum):
64 self._log(levelNum, message, args, **kwargs)
65 def logToRoot(message, *args, **kwargs):
66 logging.log(levelNum, message, *args, **kwargs)
67
68 logging.addLevelName(levelNum, levelName)
69 setattr(logging, levelName, levelNum)
70 setattr(logging.getLoggerClass(), methodName, logForLevel)
71 setattr(logging, methodName, logToRoot)

119 def wlen(text):
120 """Compute the real display width of a string, accounting for double-width characters."""
121 width = 0
122 for char in text:
123 w = wcwidth.wcwidth(char) # Handle complex emoji properly
124 if w > 0:
125 width += w

128 @staticmethod
129 def wljust(text, width, fillchar=' '):
130 """Left-justify a string based on real display width."""
131 current_width = LoggerFormatter.wlen(text)
132 padding = max(0, width - current_width)
133 return text + fillchar * padding

135 @staticmethod
136 def wrjust(text, width, fillchar=' '):
137 """Right-justify a string based on real display width."""
138 current_width = LoggerFormatter.wlen(text)
139 padding = max(0, width - current_width)
140 return fillchar * padding + text

142 @staticmethod
143 def wcenter(text, width, fillchar=' '):
144 """Center a string based on real display width."""
145 current_width = LoggerFormatter.wlen(text)
146 padding = max(0, width - current_width)
147 left_pad = padding // 2
148 right_pad = padding - left_pad
149 return fillchar * left_pad + text + fillchar * right_pad

151 @staticmethod
152 def spacer(text):
153 """Replace all visible characters with spaces but keep shell colors and tabs."""
154 ansi_escape = re.compile(r'\x1B(?:[@-Z\\-_]|\[[0-?]*[ -/]*[@-~])')
155 ansi_sequences = {match.start(): match.group() for match in ansi_escape.finditer(text)}
156 replaced_text = re.sub(r'[^\t\x1B]', " ", text)
157 for pos, ansi_seq in ansi_sequences.items():
158 replaced_text = replaced_text[:pos] + ansi_seq + replaced_text[pos + len(ansi_seq):]
159
160 return replaced_text

190 def format(self, record):
191
192 formatter = self._fmt
193 formatter = logging.Formatter(formatter)
194
195 record.pid = getattr(record, 'pid', "") # Process ID
196 record.endl = getattr(record, 'endl', '\n')
197 record.emoji = getattr(record, 'emoji', None) # Emoji (default empty)
198 if record.emoji is not None and self.wlen(record.emoji) < 4:
199 record.emoji = record.emoji[:2]
200 record.emoji = "["+self.wcenter(record.emoji,2)+"]" if record.emoji != "" else ""
201 record.emoji = self.wcenter(record.emoji[:4] or "", 4)
202
203 record.levelicon = self.ICONS.get(record.levelno, "")
204 record.location = "%(filename)s:%(lineno)d" % record.__dict__ if record.levelno > logging.ERROR else ""
205 record.levelname = self.NAMES.get(record.levelno, "").rjust(max(len(v) for v in self.NAMES.values()))
206 for key, value in record.__dict__.items():
207 if not isinstance(value, str):
208 continue
209 if key == "endl":
210 continue
211
212 if value != "":
213 record.__dict__[key] = value + " "
214
215 record.levelcolor = self.COLORS.get(record.levelno, "")
216
217 empty = logging.LogRecord(**record.__dict__, level=logging.DEBUG)
218 empty.__dict__.update(record.__dict__)
219 empty.levelicon = self.ICONS.get(logging.MESSAGE, "")
220 empty.msg = ""
221
222 # Multi-line messages
223 message = record.msg.split("\n")
224 if len(message) > 1:
225 message = [message[0].strip()] + [formatter.format(empty) + " " + line.strip() for line in message[1:] if line.strip() != ""]
226 message = [msg for msg in message if msg.strip() != ""]
227
228 record.msg = "\n".join(message)
229
230 return "\r" + formatter.format(record).strip() + (record.endl if record.levelno < logging.WARNING else "")
