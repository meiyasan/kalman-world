18def _majority_truth(raw, event):
19 ids = [int(event.truth[l][j]) for (l, j) in raw.hit_ids]
20 sig = [i for i in ids if i >= 0]
21 return (max(set(sig), key=sig.count) if sig else -1), ids

64 r1 = float(m.radii[1])
65 xp, Pp, _ = eng._predict(m, x0, P0, t=r1)
66 on = ev.tracks[0][1][:2].reshape(2, 1) # true position at layer 1
67 off = on + np.array([[6.0], [6.0]])
68 chi2_on, _, _ = eng.gate(m, xp, Pp, on, t=r1)
