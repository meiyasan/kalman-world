1"""MCP server over the :mod:`kalman.api` facade — one thin adapter.
2
3Tools mirror the facade verbs, each declaring a required **scope**. Where the
4work happens is the backend's business (:mod:`.backend`): in this process, or
5— with ``KALMAN_API_URL`` set — in the REST API, which then owns the registry
6and the runs for everybody. Apps need no per-app MCP code — they're
7auto-cataloged; an authored spec is indistinguishable from a built-in. The
8``mcp`` SDK is imported lazily.
9
10The caller's token, first match wins:
11
12 1. the ``token`` argument of the tool (clients that cannot set headers);
13 2. the ``Authorization: Bearer`` header of the HTTP transports (``sse``,
14 ``streamable-http``), checked by :class:`JwtVerifier` before any tool runs;
15 3. on stdio only, ``KALMAN_MCP_TOKEN`` — the token the client was registered
16 with (``tools/mcp_install.py --token``).
17
18Clients that listen on the resources (``catalog://models``,
19``catalog://sources``, ``registry://events``) are told when the shared registry
20changes, whoever changed it — see :mod:`.watch`.
21"""

45class JwtVerifier:
46 """Bearer-token check for the HTTP transports (the SDK's ``TokenVerifier``).
47
48 Asks the backend who the token belongs to — verified locally with
49 ``KALMAN_JWT_SECRET``, or by the REST API's ``/whoami`` — and refuses the
50 connection when the token is missing, forged or expired. Scopes are then
51 checked per tool."""
52
53 def __init__(self, backend):
54 self.backend = backend
55
56 async def verify_token(self, token: str):
57 import anyio
58 from mcp.server.auth.provider import AccessToken
59 try:
60 info = await anyio.to_thread.run_sync(self.backend.whoami, token)
61 except Exception:
62 return None
63 if not info.get("user"):
64 return None
65 return AccessToken(token=token, client_id=info["user"], subject=info["user"],
66 scopes=list(info.get("scopes", [])), expires_at=info.get("exp"))

78def build_server(backend=None, *, transport: str = "stdio", auth_settings=None,
79 token_verifier=None):
80 """The MCP server. ``backend`` defaults to :func:`.backend.from_env`;
81 ``auth_settings``/``token_verifier`` turn on bearer auth for HTTP transports
82 (see ``__main__``)."""
83 from .backend import from_env
84 backend = backend or from_env()
85 service_token = os.environ.get("KALMAN_MCP_TOKEN") if transport == "stdio" else None
86
87 def tok(token: str) -> str | None:
88 return token or _transport_token() or service_token or None
89
90 cls = _server_class()
91 kw = {}
92 if token_verifier is not None:
93 kw = {"token_verifier": token_verifier, "auth": auth_settings}
94 watch_kw = {}
95 try: # SDK 2.x: the subscriptions/listen fan-out
96 from mcp.server.subscriptions import InMemorySubscriptionBus
97 from .watch import RegistryWatcher
98 bus = InMemorySubscriptionBus()
99 watcher = RegistryWatcher(backend, bus, token=os.environ.get("KALMAN_MCP_TOKEN") or None)
100 watch_kw = {"subscriptions": bus, "lifespan": watcher.lifespan}
101 except ImportError: # SDK 1.x: no listen; clients poll list_events
102 pass
103 try: # SDK 2.x carries a display title/description
104 mcp = cls(SERVER_NAME, title=SERVER_TITLE,
105 description="Kalman estimation framework: models, filters, "
106 "solvers, providers, and spec authoring.", **kw, **watch_kw)
107 if watch_kw:
108 watcher.attach(mcp) # resources/subscribe, for clients before listen
109 except TypeError: # SDK 1.x FastMCP takes the name only
110 mcp = cls(SERVER_TITLE, **kw)
