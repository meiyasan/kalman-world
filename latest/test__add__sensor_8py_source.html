299class TestCameras(unittest.TestCase):
300 """A camera reads two pixel coordinates, by a projection that divides by the
301 depth. One camera gives the direction of a point and not how far it is; two
302 at different places give both."""
303
304 FOCAL = 800.0
305 TRUTH = np.array([[1.5], [-0.8], [40.0]])
306 LEFT, RIGHT = np.array([-1.0, 0.0, 0.0]), np.array([1.0, 0.0, 0.0])
307
308 def camera(self, name, position):
309 def pixels(x):
310 p = np.asarray(x, dtype=float).reshape(-1) - position # looking along +z
311 return [self.FOCAL * p[0] / p[2], self.FOCAL * p[1] / p[2]]
312 return KalmanSensor(name, pixels, R=0.5 ** 2, labels={"U": "px", "V": "px"})
313
314 def point(self):
315 # A point at rest. The first guess is what one look through the left camera
316 # gives: on its line of sight, at an assumed depth of 30 m (the truth is 40 m),
317 # with 20 m of doubt on the depth.
318 guess = self.LEFT + 0.75 * (self.TRUTH.ravel() - self.LEFT)
319 return KalmanModel("point", fs=10.0, F=np.eye(3), H=np.array([[1.0, 0.0, 0.0]]),
320 Q=1e-10 * np.eye(3), R=np.array([[1e4]]),
321 x0=guess.reshape(3, 1), P0=np.diag([25.0, 25.0, 400.0]))
322
323 def track(self, model, seed, steps=200):
324 rng = np.random.default_rng(seed)
325 clean = np.asarray(model.h(self.TRUTH), dtype=float).reshape(-1)
326 z = clean + np.sqrt(np.diag(model.R())) * rng.standard_normal((steps, model.m))
327 return model.filter(z, smooth=False, reset=True)
328
329 def test_observables_of_two_cameras(self):
330 both = self.point().add_sensor(self.camera("left", self.LEFT)).add_sensor(self.camera("right", self.RIGHT))
331 self.assertEqual(list(both.observables.__members__), ["Z0", "U", "V", "right_U", "right_V"])
332
333 def test_one_camera_leaves_the_depth_open_and_two_give_it(self):
334 left, right = self.camera("left", self.LEFT), self.camera("right", self.RIGHT)
335 for seed in (4, 5, 6):
336 one = self.track(self.point().add_sensor(left), seed)
337 two = self.track(self.point().add_sensor(left).add_sensor(right), seed)
338 depth_one, depth_two = np.sqrt(one.Pk[-1, 2, 2]), np.sqrt(two.Pk[-1, 2, 2])
339 # The error bars say it, and they are honest ones.
340 self.assertGreater(depth_one, 5.0) # one camera: metres of doubt left
341 self.assertLess(abs(one.xk[-1, 2] - 40.0), 3 * depth_one)
342 self.assertLess(depth_two, 0.1) # two: triangulated to centimetres
343 self.assertLess(abs(two.xk[-1, 2] - 40.0), 3 * depth_two)
344 self.assertLess(abs(two.xk[-1, 0] - 1.5), 0.05)

356 x = np.asarray(x).reshape(-1)
