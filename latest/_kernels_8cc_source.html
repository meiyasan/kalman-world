88 // A linear model assembled from the matrices a kernel is handed; a NULL B means
89 // the control does not enter (l = 0), a NULL L means identity noise coupling.
90 Kalman::Models::Linear Assemble(int n, int m, int l, const double *F, const double *H,
91 const double *Q, const double *R, const double *B, const double *L)
92 {
93 const std::size_t nn = static_cast<std::size_t>(n), mm = static_cast<std::size_t>(m);
94 const std::size_t ll = (B && l > 0) ? static_cast<std::size_t>(l) : 0;
95 Kalman::Models::Linear model("kernel", nn, mm, ll);
96 if (F) model.SetF(Mat(F, nn, nn));
97 if (H) model.SetH(Mat(H, mm, nn));
98 if (Q) model.SetQ(Mat(Q, nn, nn));
99 if (R) model.SetR(Mat(R, mm, mm));
100 if (ll) model.SetB(Mat(B, nn, ll));
101 if (L) model.SetL(Mat(L, nn, nn));
102 return model;
103 }
104}

108void kf_predict(int n, int l,
109 const double *F, const double *x, const double *P,
110 const double *B, const double *u,
111 const double *L, const double *Q,
112 double *x_out, double *P_out)
113{
114 const std::size_t nn = static_cast<std::size_t>(n);
115 Guard([&] {
116 // A one-row H keeps the model well-formed; the prediction never reads it.
117 std::vector<double> H(nn, 0.);
118 Kalman::Models::Linear model = Assemble(n, 1, l, F, H.data(), Q, nullptr, B, L);
119 const Kalman::Engines::Linear engine;
120 Vector uv;
121 if (B && u && l > 0) uv = Vec(u, static_cast<std::size_t>(l));
122 Vector xk;
123 Matrix Pk, Fk;
124 engine.Predict(model, Vec(x, nn), Mat(P, nn, nn), uv.Empty() ? nullptr : &uv, 0., xk, Pk, Fk);
125 Out(xk, x_out);
126 Out(Pk, P_out);
127 return 0;
128 }, {{x_out, nn}, {P_out, nn * nn}});
129}

132 const double *H, const double *x, const double *P,
133 const double *z, const double *R,

145 return static_cast<int>(used);
146 }, {{x_out, nn}, {P_out, nn * nn}});

149int kf_update_residual(int n, int m,
150 const double *H, const double *x, const double *P,
151 const double *y, const double *R,
152 double *x_out, double *P_out)
153{
154 const std::size_t nn = static_cast<std::size_t>(n), mm = static_cast<std::size_t>(m);
155 return Guard([&] {
156 const Vector yv = Vec(y, mm);
157 const std::vector<std::size_t> keep = Kalman::Utils::ValidChannels(yv);
158 const Vector xv = Vec(x, nn);
159 const Matrix Pm = Mat(P, nn, nn);
160 if (keep.empty())
161 {
162 Out(xv, x_out);
163 Out(Pm, P_out);
164 return 0;
165 }
166 Vector xo;
167 Matrix Po;
168 Kalman::Utils::Correct(Kalman::Utils::CompressRows(Mat(H, mm, nn), keep),
169 Kalman::Utils::CompressBoth(Mat(R, mm, mm), keep),
170 Kalman::Utils::Compress(yv, keep), xv, Pm, xo, Po);
171 Out(xo, x_out);
172 Out(Po, P_out);
173 return static_cast<int>(keep.size());
174 }, {{x_out, nn}, {P_out, nn * nn}});
175}

177void kf_forward_linear(int n, int m, int l, int N,
178 const double *F, const double *H,
179 const double *Q, const double *R,
180 const double *B, const double *L,
181 const double *zk, const double *uk,
182 const double *x0, const double *P0,
183 double *x_prior, double *P_prior,
184 double *x_post, double *P_post,
185 double *innovations)
186{
187 const std::size_t nn = static_cast<std::size_t>(n), mm = static_cast<std::size_t>(m);
188 const std::size_t steps = N > 0 ? static_cast<std::size_t>(N) : 0;
189 Guard([&] {
190 if (N <= 0) throw std::invalid_argument("kf_forward_linear: N must be positive");
191 Kalman::Models::Linear model = Assemble(n, m, l, F, H, Q, R, B, L);
192 model.SetInitialState(Vec(x0, nn));
193 model.SetInitialCovariance(Mat(P0, nn, nn));
194 const std::size_t ll = model.NControl();
195
196 std::vector<Vector> z(steps), u;
197 for (std::size_t k = 0; k < steps; ++k) z[k] = Vec(zk + k * mm, mm);
198 if (uk && ll > 0)
199 {
200 u.resize(steps);
201 for (std::size_t k = 0; k < steps; ++k) u[k] = Vec(uk + k * ll, ll);
202 }
203
204 // The very filter a C++ program runs, on the very engine.
205 Kalman::Engines::Linear engine;
206 Kalman::Filter filter(&model, &engine);
207 filter.Forward(z, u.empty() ? nullptr : &u);
208
209 for (std::size_t k = 0; k < steps; ++k)
210 {
211 Out(filter.PriorStates()[k], x_prior + k * nn);
212 Out(filter.PriorCovariances()[k], P_prior + k * nn * nn);
213 Out(filter.PosteriorStates()[k], x_post + k * nn);
214 Out(filter.PosteriorCovariances()[k], P_post + k * nn * nn);
215 Out(filter.Innovations()[k], innovations + k * mm);
216 }
217 return 0;
218 }, {{x_prior, steps * nn}, {P_prior, steps * nn * nn}, {x_post, steps * nn},
219 {P_post, steps * nn * nn}, {innovations, steps * mm}});
220}

222void kf_backward_rts(int n, int N,
223 const double *x_post, const double *P_post,
224 const double *x_prior, const double *P_prior,
225 const double *F,
226 double *x_smooth, double *P_smooth)
227{
228 const std::size_t nn = static_cast<std::size_t>(n);
229 const std::size_t steps = N > 0 ? static_cast<std::size_t>(N) : 0;
230 Guard([&] {
231 if (N <= 0) throw std::invalid_argument("kf_backward_rts: N must be positive");
232 const Kalman::Engines::Linear engine;
233 const Matrix Fm = Mat(F, nn, nn);
234 // The last posterior is already the smoothed estimate; walk backwards from it.
235 std::memcpy(x_smooth + (steps - 1) * nn, x_post + (steps - 1) * nn, nn * sizeof(double));
236 std::memcpy(P_smooth + (steps - 1) * nn * nn, P_post + (steps - 1) * nn * nn, nn * nn * sizeof(double));
237 for (std::size_t k = steps - 1; k-- > 0;)
238 {
239 Vector xs;
240 Matrix Ps;
241 engine.Smooth(Vec(x_post + k * nn, nn), Mat(P_post + k * nn * nn, nn, nn),
242 Vec(x_prior + (k + 1) * nn, nn), Mat(P_prior + (k + 1) * nn * nn, nn, nn),
243 Vec(x_smooth + (k + 1) * nn, nn), Mat(P_smooth + (k + 1) * nn * nn, nn, nn),
244 Fm, xs, Ps);
245 Out(xs, x_smooth + k * nn);
246 Out(Ps, P_smooth + k * nn * nn);
247 }
248 return 0;
249 }, {{x_smooth, steps * nn}, {P_smooth, steps * nn * nn}});
250}

281void pf_weight_update(int n_particles, int m,
282 const double *z, const double *zp,
283 const double *Rinv, double *logw)
284{
285 const std::size_t np = static_cast<std::size_t>(n_particles), mm = static_cast<std::size_t>(m);
286 Guard([&] {
287 const Matrix Ri = Mat(Rinv, mm, mm);
288 const Vector zv = Vec(z, mm);
289 for (std::size_t i = 0; i < np; ++i)
290 {
291 Vector dz = zv;
292 dz -= Vec(zp + i * mm, mm);
293 logw[i] += -0.5 * Kalman::Utils::Mahalanobis(dz, Ri);
294 }
295 return 0;
296 }, {});
297}

332void diag_nis_batch(int N, int m,
333 const double *nu, const double *S, double *nis)
334{
335 const std::size_t steps = N > 0 ? static_cast<std::size_t>(N) : 0, mm = static_cast<std::size_t>(m);
336 Guard([&] {
337 for (std::size_t k = 0; k < steps; ++k)
338 {
339 const Vector v = Vec(nu + k * mm, mm);
340 const Vector s = Kalman::Utils::Solve(Mat(S + k * mm * mm, mm, mm), v);
341 double q = 0.;
342 for (std::size_t j = 0; j < mm; ++j) q += v(j) * s(j);
343 nis[k] = q;
344 }
345 return 0;
346 }, {{nis, steps}});
347}

361int enkf_analysis(int n, int N, int p,
362 const double *X, const double *Y, const double *R, const double *z, const double *eps,
363 const double *rho_xy, const double *rho_yy, double *X_out)
364{
365 const std::size_t nn = static_cast<std::size_t>(n), NN = static_cast<std::size_t>(N), pp = static_cast<std::size_t>(p);
366 return Guard([&] {
367 if (n <= 0 || N < 2 || p <= 0) throw std::invalid_argument("enkf_analysis: need n, p > 0 and at least two members");
368 // One implementation: the engine's (Engines::Ensemble::Analysis).
369 Matrix Xm(nn, NN), Ym(pp, NN), E(pp, NN), Rm(pp, pp);
370 Kalman::Vector zv(pp);
371 std::copy(X, X + nn * NN, Xm.Data());
372 std::copy(Y, Y + pp * NN, Ym.Data());
373 std::copy(eps, eps + pp * NN, E.Data());
374 std::copy(R, R + pp * pp, Rm.Data());
375 std::copy(z, z + pp, zv.Data());
376 Matrix rxy, ryy;
377 if (rho_xy) { rxy = Matrix(nn, pp); std::copy(rho_xy, rho_xy + nn * pp, rxy.Data()); }
378 if (rho_yy) { ryy = Matrix(pp, pp); std::copy(rho_yy, rho_yy + pp * pp, ryy.Data()); }
379 const Matrix out = Kalman::Engines::Ensemble::Analysis(Xm, Ym, Rm, zv, E, rho_xy ? &rxy : nullptr,
380 rho_yy ? &ryy : nullptr);
381 std::copy(out.Data(), out.Data() + nn * NN, X_out);
382 return 0;
383 }, {{X_out, nn * NN}});
384}
