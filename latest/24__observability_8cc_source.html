51 // --- a pendulum measured in angle: both components are determined ----------------
52 {
53 Fixture::Pendulum pendulum(1.0, 0.15, 100.);
54 Vector x0(2); x0(0) = 0.6; x0(1) = 0.;
55 pendulum.SetInitialState(x0);
56 const Observability obs(pendulum, 100);
57 CHECK(obs.IsObservable() && obs.Rank() == 2, "pendulum: the angle determines the angle and the rate");
58 CHECK(obs.WeakComponents().empty(), "pendulum: no weak component");
59 CHECK(obs.Eigenvalues()(0) <= obs.Eigenvalues()(1), "eigenvalues: smallest first");
60 }
61
62 // --- a constant velocity measured in position: one step is not enough, two are -------
63 {
64 Fixture::ConstantVelocity cv(1, 10.);
65 Vector x0(2); x0(0) = 0.; x0(1) = 1.;
66 const Observability one(cv, 1, &x0), two(cv, 2, &x0);
67 CHECK(one.Rank() == 1 && !one.IsObservable(), "constant velocity: one fix leaves a direction open");
68 CHECK(two.IsObservable(), "constant velocity: two fixes give the velocity");
69 }
70
71 // --- a gain and an amplitude that only appear as their product ---------------------------
72 {
73 auto gauge = std::make_shared<Gauge>();
74 Models::Joint both(gauge, {{"gain", 0.5, 0.}});
75 const Observability obs(both, 50);
76 CHECK(obs.Rank() == 1 && obs.NWeak() == 1, "product: one direction of two is determined");
77 CHECK(obs.Weakness(0) > 0.05 && obs.Weakness(1) > 0.05, "product: the amplitude and the gain share the weak direction");
78 const auto weak = obs.WeakComponents();
79 CHECK(weak.size() == 2, "product: both are reported");
80 // The weak direction leaves gain * a unchanged: gain da + a dgain = 0, in scaled
81 // units (sigma_a = 1, sigma_gain = 0.5) the direction is (a sigma_a, -gain sigma_gain)... up to sign.
82 const double ratio = obs.Directions()(0, 0) / obs.Directions()(1, 0);
83 CHECK_CLOSE(ratio, -(3. * 0.5) / (2. * 1.), 1e-4, "product: the weak direction keeps the product constant");
84
85 // The offset is no better: z = gain a + offset has one number for three unknowns.
86 Models::Joint three(gauge, {{"gain", 0.5, 0.}, {"offset", 1., 0.}});
87 CHECK(Observability(three, 50).Rank() == 1, "product and offset: still one direction");
88
89 // An offset alone, with the gain known, is as hidden: a + offset / gain.
90 Models::Joint shifted(gauge, {{"offset", 1., 0.}});
91 CHECK(!Observability(shifted, 50).IsObservable(), "offset: cannot be told from the amplitude");
92 }
93
94 // --- the damping of an oscillator is determined by the positions ------------------------------
95 {
96 // (the fixture pendulum has no declared parameter; the damped pendulum's
97 // rate is in its state, which is the point: motion reveals it.)
98 Fixture::Pendulum pendulum(1.0, 0.15, 100.);
99 Vector rest(2); rest(0) = 0.; rest(1) = 0.;
100 CHECK(Observability(pendulum, 100, &rest).IsObservable(), "pendulum at rest: still determined (linear in both)");
101 }
102
103 // --- tau: a parameter that returns to where it started ------------------------------------------
104 {
105 auto gauge = std::make_shared<Gauge>();
106 const double dt = 0.1, tau = 2., phi = std::exp(-dt / tau);
107 Models::Joint walk(gauge, {{"offset", 1., 0.1, 0.}});
108 Models::Joint markov(gauge, {{"offset", 1., 0.1, tau}});
109 Vector x(2); x(0) = 3.; x(1) = 0.8; // the offset started at 0
110 CHECK(walk.f(x, Vector(), 0.)(1) == 0.8, "tau = 0: a random walk keeps its value");
111 CHECK_CLOSE(markov.f(x, Vector(), 0.)(1), phi * 0.8, 1e-15, "tau: the value decays towards where it started");
112 CHECK(walk.Jacobian(x, Vector(), 0.)(1, 1) == 1., "tau = 0: d theta' / d theta = 1");
113 CHECK_CLOSE(markov.Jacobian(x, Vector(), 0.)(1, 1), phi, 1e-15, "tau: d theta' / d theta = exp(-dt / tau)");
114 Vector w(2); w(0) = 0.; w(1) = 0.05;
115 CHECK_CLOSE(markov.f(x, Vector(), w, 0.)(1), phi * 0.8 + 0.05, 1e-15, "tau: plus its noise");
116
117 bool refused = false;
118 try { Models::Joint bad(gauge, {{"offset", 1., 0.1, -1.}}); }
119 catch (const std::invalid_argument &) { refused = true; }
120 CHECK(refused, "refused: a negative tau");
121
122 // With nothing measured the variance of a walk grows without bound; with a tau
123 // it settles at drift^2 / (1 - phi^2).
124 Engines::Extended engine;
125 Vector xw = walk.X0(), xm = markov.X0(), xp;
126 Matrix Pw = walk.P0(), Pm = markov.P0(), Pp, F;
127 for (int k = 0; k < 2000; ++k)
128 {
129 engine.Predict(walk, xw, Pw, nullptr, 0., xp, Pp, F); xw = xp; Pw = Pp;
130 engine.Predict(markov, xm, Pm, nullptr, 0., xp, Pp, F); xm = xp; Pm = Pp;
131 }
132 CHECK_CLOSE(Pm(1, 1), 0.01 / (1. - phi * phi), 1e-9, "tau: the variance settles at drift^2 / (1 - phi^2)");
133 CHECK_CLOSE(Pw(1, 1), 1. + 2000 * 0.01, 1e-6, "tau = 0: the variance grows by drift^2 per step");
134 }
135
136 return Check::Summary("24_observability");
