33 double GaspariCohn(double distance, double L)
34 {
35 // The same polynomials as engines/ensemble.py, term for term.
36 const double r = distance / L;
37 double c = 0.;
38 if (r <= 1.)
39 c = 1. - 5. / 3. * (r * r) + 5. / 8. * std::pow(r, 3) + 0.5 * std::pow(r, 4) - 0.25 * std::pow(r, 5);
40 else if (r <= 2.)
41 c = 4. - 5. * r + 5. / 3. * (r * r) + 5. / 8. * std::pow(r, 3) - 0.5 * std::pow(r, 4)
42 + 1. / 12. * std::pow(r, 5) - 2. / (3. * r);
43 return std::max(c, 0.);
44 }

104 std::vector<Vector> Ensemble::Normals(std::size_t rows) const
105 {
106 const std::size_t N = fNMembers;
107 std::vector<Vector> w(N, Vector(rows));
108 if (!fHasNormals)
109 {
110 // The engine's own stream, one member after the other.
111 std::normal_distribution<double> gauss(0., 1.);
112 for (Vector &v : w)
113 for (std::size_t i = 0; i < rows; ++i) v(i) = gauss(fRandom);
114 return w;
115 }
116 if (fNormals.size() - fNormalsUsed < rows * N)
117 {
118 std::ostringstream os;
119 os << "Engines::Ensemble: the draws handed in are used up (" << rows * N << " more needed, "
120 << fNormals.size() - fNormalsUsed << " left).";
121 throw std::runtime_error(os.str());
122 }
123 // numpy's (rows, N) block, row-major: member s reads column s.
124 const double *block = fNormals.data() + fNormalsUsed;
125 for (std::size_t i = 0; i < rows; ++i)
126 for (std::size_t s = 0; s < N; ++s) w[s](i) = block[i * N + s];
127 fNormalsUsed += rows * N;
128 return w;
129 }

140 {

172 void Ensemble::Estimate(Vector &mean, Matrix &cov) const
173 {
174 mean = Mean();
175 const std::size_t n = mean.Size();
176 const std::size_t N = fMembers.size();
177 const double norm = 1. / static_cast<double>(std::max<std::size_t>(N - 1, 1));
178
179 // (A A^T) / (N - 1) with A the anomalies, as one product (BLAS when it pays).
180 Matrix A(n, N), AT(N, n);
181 for (std::size_t s = 0; s < N; ++s)
182 for (std::size_t i = 0; i < n; ++i)
183 {
184 const double a = fMembers[s](i) - mean(i);
185 A(i, s) = a;
186 AT(s, i) = a;
187 }
188 cov = A * AT;
189 cov *= norm;
190 Utils::Symmetrize(cov);
191 for (std::size_t i = 0; i < n; ++i) cov(i, i) += 1e-12;
192 }

194 double Ensemble::Spread() const
195 {
196 const Vector mean = Mean();
197 const std::size_t n = mean.Size();
198 double total = 0.;
199 for (std::size_t i = 0; i < n; ++i)
200 {
201 double v = 0.;
202 for (const Vector &x : fMembers) v += (x(i) - mean(i)) * (x(i) - mean(i));
203 total += std::sqrt(v / static_cast<double>(fMembers.size()));
204 }
205 return total / static_cast<double>(n);
206 }

208 Vector Ensemble::Forecast(const Model &model, const Vector *u, double t) const
209 {
210 if (fMembers.empty()) const_cast<Ensemble *>(this)->Reset(model);
211 const std::size_t n = model.NState();
212
213 Vector uu(model.NControl(), 0.);
214 if (u && model.NControl() > 0 && u->Size() == model.NControl()) uu = *u;
215
216 const auto *hooks = dynamic_cast<const EnsembleModel *>(&model);
217 if (hooks) hooks->Forecast(model, fMembers, uu, t);
218 else
219 for (Vector &x : fMembers) x = model.f(x, uu, t);
220
221 // The process noise, after every member has moved (the order of the draws
222 // is the Python engine's: one (n, N) block).
223 Vector variances;
224 if (hooks && hooks->ProcessNoiseVariances(t, variances) && model.L(t).Empty())
225 {
226 if (variances.Size() != n) throw std::runtime_error("Engines::Ensemble: the process-noise variances are not n long.");
227 bool any = false;
228 for (std::size_t i = 0; i < n; ++i) any = any || variances(i) != 0.;
229 if (any)
230 {
231 Vector sigma(n);
232 for (std::size_t i = 0; i < n; ++i) sigma(i) = std::sqrt(variances(i));
233 const std::vector<Vector> w = Normals(n);
234 for (std::size_t s = 0; s < fMembers.size(); ++s)
235 for (std::size_t i = 0; i < n; ++i) fMembers[s](i) += sigma(i) * w[s](i);
236 }
237 return Mean();
238 }
239
240 Matrix Q = model.Q(t);
241 const Matrix L = model.L(t);
242 if (!L.Empty()) Q = Utils::Similarity(L, Q);
243 if (AnyNonZero(Q))
244 {
245 const Matrix factor = Utils::CholeskyFactor(Q, true);
246 const std::vector<Vector> w = Normals(factor.Cols());
247 for (std::size_t s = 0; s < fMembers.size(); ++s) fMembers[s] += factor * w[s];
248 }
249 return Mean();
250 }

265 Matrix Ensemble::Analysis(const Matrix &X, const Matrix &Y, const Matrix &R, const Vector &z,
266 const Matrix &eps, const Matrix *rhoXY, const Matrix *rhoYY)
267 {
268 const std::size_t n = X.Rows(), N = X.Cols(), p = Y.Rows();
269 if (N < 2 || Y.Cols() != N || eps.Rows() != p || eps.Cols() != N || R.Rows() != p || R.Cols() != p || z.Size() != p)
270 throw std::invalid_argument("Engines::Ensemble::Analysis: inconsistent shapes.");
271 const double norm = 1. / static_cast<double>(N - 1);
272
273 // Anomalies: X' (n x N) and Y'^T (N x p).
274 Matrix Xa(n, N), YaT(N, p);
275 for (std::size_t i = 0; i < n; ++i)
276 {
277 double mean = 0.;
278 for (std::size_t s = 0; s < N; ++s) mean += X(i, s);
279 mean /= static_cast<double>(N);
280 for (std::size_t s = 0; s < N; ++s) Xa(i, s) = X(i, s) - mean;
281 }
282 for (std::size_t j = 0; j < p; ++j)
283 {
284 double mean = 0.;
285 for (std::size_t s = 0; s < N; ++s) mean += Y(j, s);
286 mean /= static_cast<double>(N);
287 for (std::size_t s = 0; s < N; ++s) YaT(s, j) = Y(j, s) - mean;
288 }
289
290 Matrix PHt = Xa * YaT; // (n, p)
291 PHt *= norm;
292 Matrix S = YaT.Transpose() * YaT; // (p, p)
293 S *= norm;
294 if (rhoXY)
295 for (std::size_t i = 0; i < n; ++i)
296 for (std::size_t j = 0; j < p; ++j) PHt(i, j) *= (*rhoXY)(i, j);
297 for (std::size_t i = 0; i < p; ++i)
298 for (std::size_t j = 0; j < p; ++j)
299 S(i, j) = S(i, j) * (rhoYY ? (*rhoYY)(i, j) : 1.) + R(i, j);
300
301 const Matrix K = PHt * Utils::Inverse(S); // (n, p)
302
303 // D = z + eps - Y (p x N), then X + K D.
304 Matrix D(p, N);
305 for (std::size_t j = 0; j < p; ++j)
306 for (std::size_t s = 0; s < N; ++s) D(j, s) = z(j) + eps(j, s) - Y(j, s);
307 Matrix out = K * D;
308 out += X;
309 return out;
310 }

312 Vector Ensemble::Analyze(const Model &model, const Vector &z, double t) const
313 {
314 if (fMembers.empty()) const_cast<Ensemble *>(this)->Reset(model);
315
316 const std::vector<std::size_t> keep = Utils::ValidChannels(z);
317 if (keep.empty()) return Mean();
318
319 const std::size_t n = model.NState();
320 const std::size_t m = model.NMeasure();
321 const std::size_t p = keep.size();
322 const std::size_t N = fMembers.size();
323
324 // Inflate the anomalies before the analysis.
325 const Vector mean = Mean();
326 if (fInflation != 1.)
327 for (Vector &x : fMembers)
328 for (std::size_t i = 0; i < n; ++i) x(i) = mean(i) + fInflation * (x(i) - mean(i));
329
330 // Observation-space ensemble, restricted to the channels that arrived.
331 std::vector<Vector> Yfull;
332 Observations(model, t, Yfull);
333
334 Matrix X(n, N), Y(p, N);
335 for (std::size_t s = 0; s < N; ++s)
336 {
337 for (std::size_t i = 0; i < n; ++i) X(i, s) = fMembers[s](i);
338 if (Yfull[s].Size() != m) throw std::runtime_error("Engines::Ensemble: a member's measurement is not m long.");
339 for (std::size_t j = 0; j < p; ++j) Y(j, s) = Yfull[s](keep[j]);
340 }
341
342 // The tapers: those set on the engine, else the model's.
343 const Matrix &rhoXY = fRhoXY.Empty() && fRhoYY.Empty() ? fModelRhoXY : fRhoXY;
344 const Matrix &rhoYY = fRhoXY.Empty() && fRhoYY.Empty() ? fModelRhoYY : fRhoYY;
345 Matrix rxy, ryy;
346 if (!rhoXY.Empty())
347 {
348 if (rhoXY.Rows() != n || rhoXY.Cols() != m)
349 throw std::runtime_error("Engines::Ensemble: rho_xy must be (n, m).");
350 rxy = Matrix(n, p);
351 for (std::size_t i = 0; i < n; ++i)
352 for (std::size_t j = 0; j < p; ++j) rxy(i, j) = rhoXY(i, keep[j]);
353 }
354 if (!rhoYY.Empty())
355 {
356 if (rhoYY.Rows() != m || rhoYY.Cols() != m)
357 throw std::runtime_error("Engines::Ensemble: rho_yy must be (m, m).");
358 ryy = Matrix(p, p);
359 for (std::size_t i = 0; i < p; ++i)
360 for (std::size_t j = 0; j < p; ++j) ryy(i, j) = rhoYY(keep[i], keep[j]);
361 }
362
363 // The perturbations: one (p, N) block of draws through the factor of R.
364 const Matrix R = Utils::CompressBoth(model.R(t), keep);
365 const Matrix Rfac = Utils::CholeskyFactor(R, true);
366 Matrix eps(p, N, 0.);
367 if (!Rfac.Empty())
368 {
369 const std::vector<Vector> w = Normals(Rfac.Cols());
370 for (std::size_t s = 0; s < N; ++s)
371 {
372 const Vector e = Rfac * w[s];
373 for (std::size_t j = 0; j < p; ++j) eps(j, s) = e(j);
374 }
375 }
376
377 const Matrix Xa = Analysis(X, Y, R, Utils::Compress(z, keep), eps,
378 rxy.Empty() ? nullptr : &rxy, ryy.Empty() ? nullptr : &ryy);
379 for (std::size_t s = 0; s < N; ++s)
380 for (std::size_t i = 0; i < n; ++i) fMembers[s](i) = Xa(i, s);
381 return Mean();
382 }
