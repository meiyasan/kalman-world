29 Matrix Forward::Jacobian(const Map &fn, const Vector &x, const Vector *y0) const
30 {
31 const std::size_t n = x.Size();
32 const Vector base = y0 ? *y0 : fn(x);
33 Matrix J(base.Size(), n, 0.);
34 for (std::size_t j = 0; j < n; ++j)
35 {
36 const double step = fStep * std::max(1., std::abs(x(j)));
37 Vector xp(x);
38 xp(j) += step;
39 const Vector y1 = fn(xp);
40 for (std::size_t i = 0; i < J.Rows(); ++i) J(i, j) = (y1(i) - base(i)) / step;
41 }
42 return J;
43 }

45 Matrix Central::Jacobian(const Map &fn, const Vector &x, const Vector *y0) const
46 {
47 const std::size_t n = x.Size();
48 // The value at x is not part of the formula; it only gives the number of rows.
49 std::size_t rows = y0 ? y0->Size() : 0;
50 Matrix J;
51 for (std::size_t j = 0; j < n; ++j)
52 {
53 const double step = fStep * std::max(1., std::abs(x(j)));
54 Vector xp(x), xm(x);
55 xp(j) += step;
56 xm(j) -= step;
57 const Vector fp = fn(xp), fm = fn(xm);
58 if (J.Empty())
59 {
60 rows = fp.Size();
61 J = Matrix(rows, n, 0.);
62 }
63 for (std::size_t i = 0; i < rows; ++i) J(i, j) = (fp(i) - fm(i)) / (2. * step);
64 }
65 if (J.Empty()) J = Matrix(y0 ? y0->Size() : fn(x).Size(), n, 0.);
66 return J;
67 }
