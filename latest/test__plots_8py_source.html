53 def test_powergrid_signature_figure(self):

60 def test_gravwave_signature_figure(self):
61 fs = 100.0
62 t = np.arange(int(8 * fs)) / fs
63 a = 1 + 0.1 * np.cos(2 * np.pi * 0.2 * t)
64 psi = 2 * np.pi * 1.0 * t
65 theta = (np.pi / 2) * np.ones_like(t)
66 phi = np.pi * np.ones_like(t)
67 z = np.array([jones_signal(a[k], psi[k], theta[k], phi[k]) for k in range(len(t))])
68 zk = np.column_stack([z[:, 0].real, z[:, 0].imag, z[:, 1].real, z[:, 1].imag])
69 zk = zk + 0.05 * np.random.randn(*zk.shape)
70 gw = PolarizationModel("gw", fs=fs, sigma_n=0.5)
71 gw.initial_state([np.log(a[0]), 0.0, psi[0], 2 * np.pi, theta[0], 0.0, phi[0], 0.0])
72 self._assert_figure(gw.plot(gw.filter(zk, smooth=False)))

74 def test_rocket_trajectory_figure(self):
75 tl = TransLunarModel(dt_s=120.0, track_km=5.0)

79 tl.initial_state(np.array([[r0], [0], [0], [0], [v_tli], [0]]),

81 full = tl.sample(40 * 0.12, process_noise=False)
