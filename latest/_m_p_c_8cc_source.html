81 // `x_next ~ f(x, u) + A (dx) + B (du)` at (x, u, t).
82 void MPC::Linearise(const Vector &x, const Vector &u, double t, Matrix &A, Matrix &B) const
83 {
84 const std::size_t n = fModel->NState(), l = NControl();
85 if (fModel->HasAnalyticJacobian())
86 {
87 // A model of matrices: its own F and B, exactly.
88 A = fModel->F(t);
89 B = fModel->B(t);
90 if (B.Empty()) B = Matrix::Identity(n); // the command enters the state directly
91 return;
92 }
93 A = fModel->Jacobian(x, u, t);
94
95 // df/du, differenced towards the inside of the limits: an f that clips its
96 // command is flat outside them, and a plan resting on a limit would
97 // otherwise see a command with no effect.
98 B = Matrix(n, l, 0.);
99 const Vector base = fModel->f(x, u, t);
100 for (std::size_t j = 0; j < l; ++j)
101 {
102 double step = 1e-6 * std::max(1., std::fabs(u(j)));
103 if (u(j) + step > fUMax(j)) step = -step;
104 Vector moved(u);
105 moved(j) += step;
106 const Vector shifted = fModel->f(x, moved, t);
107 for (std::size_t i = 0; i < n; ++i) B(i, j) = (shifted(i) - base(i)) / step;
108 }
109 }

111 const Matrix &MPC::Terminal(double t)
112 {
113 if (!fTerminal.Empty()) return fTerminal;
114 // The cost of running LQR for ever from the end of the horizon: with it, and
115 // no limit in the way, the first command of the plan is the LQR command.
116 try
117 {
118 Matrix A, B;
119 Linearise(fXRef, Vector(NControl(), 0.), t, A, B);
120 fTerminal = Algebra::SolveDARE(A, B, fQCost, fRCost);
121 }
122 catch (const std::exception &)
123 {
124 fTerminal = fQCost; // no stabilising solution: weigh the end as any step
125 }
126 return fTerminal;
127 }

129 void MPC::Solve(const Vector &x0, double t)
130 {
131 const std::size_t n = fModel->NState(), l = NControl(), N = fHorizon;
132 const double dt = fModel->DeltaT();
133 const Matrix &Pf = Terminal(t);
134
135 // The path the current plan leads through, and the model linearised along it.
136 std::vector<Vector> xbar(N + 1);
137 std::vector<Matrix> A(N), B(N);
138 xbar[0] = x0;
139 for (std::size_t k = 0; k < N; ++k)
140 {
141 const double tk = t + static_cast<double>(k) * dt;
142 Linearise(xbar[k], fPlan[k], tk, A[k], B[k]);
143 xbar[k + 1] = fModel->f(xbar[k], fPlan[k], tk);
144 }
145
146 // S: how each command moves each later state. Block (k, j), for the state
147 // after k + 1 steps and command j <= k, is A_k ... A_{j+1} B_j.
148 Matrix S(n * N, l * N, 0.);
149 for (std::size_t j = 0; j < N; ++j)
150 {
151 Matrix M = B[j];
152 for (std::size_t k = j; k < N; ++k)
153 {
154 if (k > j) M = A[k] * M;
155 for (std::size_t a = 0; a < n; ++a)
156 for (std::size_t b = 0; b < l; ++b) S(k * n + a, j * l + b) = M(a, b);
157 }
158 }
159
160 // The states as a function of the commands U: X = xbar + S (U - Ubar), so the
161 // error is E0 + S U with E0 = xbar - x_ref - S Ubar.
162 Vector Ubar(l * N);
163 for (std::size_t k = 0; k < N; ++k)
164 for (std::size_t b = 0; b < l; ++b) Ubar(k * l + b) = fPlan[k](b);
165 const Vector SU = S * Ubar;
166 Vector E0(n * N);
167 for (std::size_t k = 0; k < N; ++k)
168 for (std::size_t a = 0; a < n; ++a) E0(k * n + a) = xbar[k + 1](a) - fXRef(a) - SU(k * n + a);
169
170 // W S and W E0, with W = diag(Qcost, ..., Qcost, Pf).
171 Matrix WS(n * N, l * N, 0.);
172 Vector WE(n * N, 0.);
173 for (std::size_t k = 0; k < N; ++k)
174 {
175 const Matrix &W = (k + 1 == N) ? Pf : fQCost;
176 for (std::size_t a = 0; a < n; ++a)
177 {
178 double e = 0.;
179 for (std::size_t c = 0; c < n; ++c) e += W(a, c) * E0(k * n + c);
180 WE(k * n + a) = e;
181 for (std::size_t col = 0; col < l * N; ++col)
182 {
183 double v = 0.;
184 for (std::size_t c = 0; c < n; ++c) v += W(a, c) * S(k * n + c, col);
185 WS(k * n + a, col) = v;
186 }
187 }
188 }
189
190 // minimise U^T H U / 2 + g^T U, H = S^T W S + Rbar, g = S^T W E0.
191 const Matrix St = S.Transpose();
192 Matrix H = St * WS;
193 for (std::size_t k = 0; k < N; ++k)
194 for (std::size_t a = 0; a < l; ++a)
195 for (std::size_t b = 0; b < l; ++b) H(k * l + a, k * l + b) += fRCost(a, b);
196 Algebra::Symmetrize(H);
197 const Vector g = St * WE;
198 const Matrix Hinv = Algebra::Inverse(H);
199
200 // The limits, as rows C U <= c.
201 struct Row { std::size_t at; double sign; std::size_t before; bool chained; double bound; };
202 std::vector<Row> rows;
203 for (std::size_t k = 0; k < N; ++k)
204 for (std::size_t b = 0; b < l; ++b)
205 {
206 const std::size_t i = k * l + b;
207 if (std::isfinite(fUMax(b))) rows.push_back({i, 1., 0, false, fUMax(b)});
208 if (std::isfinite(fUMin(b))) rows.push_back({i, -1., 0, false, -fUMin(b)});
209 if (!std::isfinite(fDUMax(b))) continue;
210 if (k > 0)
211 {
212 rows.push_back({i, 1., i - l, true, fDUMax(b)});
213 rows.push_back({i, -1., i - l, true, fDUMax(b)});
214 }
215 else if (!fLast.Empty())
216 {
217 rows.push_back({i, 1., 0, false, fDUMax(b) + fLast(b)});
218 rows.push_back({i, -1., 0, false, fDUMax(b) - fLast(b)});
219 }
220 }
221
222 Vector U = Hinv * g;
223 U *= -1.; // the plan with no limit
224 fSweeps = fActive = 0;
225
226 const std::size_t nc = rows.size();
227 if (nc > 0)
228 {
229 // Does the free plan already respect every limit?
230 const auto lhs = [&rows](const Vector &V, std::size_t r) {
231 const Row &row = rows[r];
232 return row.sign * (V(row.at) - (row.chained ? V(row.before) : 0.));
233 };
234 bool feasible = true;
235 for (std::size_t r = 0; r < nc && feasible; ++r) feasible = lhs(U, r) <= rows[r].bound + 1e-12;
236
237 if (!feasible)
238 {
239 // C and C H^-1, row by row (each row of C has one or two entries).
240 Matrix CHinv(nc, l * N, 0.);
241 for (std::size_t r = 0; r < nc; ++r)
242 for (std::size_t col = 0; col < l * N; ++col)
243 CHinv(r, col) = rows[r].sign * (Hinv(rows[r].at, col) - (rows[r].chained ? Hinv(rows[r].before, col) : 0.));
244 // M = C H^-1 C^T, d = c + C H^-1 g.
245 Matrix M(nc, nc, 0.);
246 Vector d(nc);
247 for (std::size_t r = 0; r < nc; ++r)
248 {
249 for (std::size_t q = 0; q < nc; ++q)
250 M(r, q) = rows[q].sign * (CHinv(r, rows[q].at) - (rows[q].chained ? CHinv(r, rows[q].before) : 0.));
251 double cg = 0.;
252 for (std::size_t col = 0; col < l * N; ++col) cg += CHinv(r, col) * g(col);
253 d(r) = rows[r].bound + cg;
254 }
255
256 // Hildreth: one multiplier at a time, each set to its best value given the others.
257 Vector lambda(nc, 0.);
258 for (fSweeps = 1; fSweeps <= fMaxSweeps; ++fSweeps)
259 {
260 double change = 0., size = 0.;
261 for (std::size_t r = 0; r < nc; ++r)
262 {
263 double w = d(r);
264 for (std::size_t q = 0; q < nc; ++q)
265 if (q != r) w += M(r, q) * lambda(q);
266 const double next = M(r, r) > 0. ? std::max(0., -w / M(r, r)) : 0.;
267 change = std::max(change, std::fabs(next - lambda(r)));
268 size = std::max(size, std::fabs(next));
269 lambda(r) = next;
270 }
271 if (change <= fTolerance * std::max(1., size)) break;
272 }
273 fSweeps = std::min(fSweeps, fMaxSweeps);
274
275 // U = -H^-1 (g + C^T lambda).
276 Vector rhs(g);
277 for (std::size_t r = 0; r < nc; ++r)
278 {
279 if (lambda(r) == 0.) continue;
280 ++fActive;
281 rhs(rows[r].at) += rows[r].sign * lambda(r);
282 if (rows[r].chained) rhs(rows[r].before) -= rows[r].sign * lambda(r);
283 }
284 U = Hinv * rhs;
285 U *= -1.;
286 }
287 }
288
289 // The dual solve stops within a tolerance: the limits are then met exactly,
290 // the rate limit from the command applied last onwards.
291 for (std::size_t k = 0; k < N; ++k)
292 for (std::size_t b = 0; b < l; ++b)
293 {
294 double lo = fUMin(b), hi = fUMax(b);
295 if (std::isfinite(fDUMax(b)) && (k > 0 || !fLast.Empty()))
296 {
297 const double before = k > 0 ? fPlan[k - 1](b) : fLast(b);
298 lo = std::max(lo, before - fDUMax(b));
299 hi = std::min(hi, before + fDUMax(b));
300 }
301 fPlan[k](b) = std::min(hi, std::max(lo, U(k * l + b)));
302 }
303
304 // The states this plan leads through, by the model itself.
305 fPredicted.assign(N + 1, x0);
306 for (std::size_t k = 0; k < N; ++k)
307 fPredicted[k + 1] = fModel->f(fPredicted[k], fPlan[k], t + static_cast<double>(k) * dt);
308 }

310 Vector MPC::Policy(const Vector &xHat, const Matrix &, double t, const Vector *)
311 {
312 if (xHat.Size() != fModel->NState())
313 throw std::invalid_argument("Controllers::MPC: the estimate has the wrong size.");
314
315 // Start from the plan of the previous step, moved one step on.
316 if (!fLast.Empty())
317 {
318 for (std::size_t k = 0; k + 1 < fHorizon; ++k) fPlan[k] = fPlan[k + 1];
319 }
320 for (std::size_t pass = 0; pass < fNIterations; ++pass) Solve(xHat, t);
321
322 fLast = fPlan.front();
323 return fLast;
324 }
