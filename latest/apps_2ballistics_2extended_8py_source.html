147 # Process noise covariance generator (internal)
148 # ----------------------------------------------------------------------
149 def Q(self, t=None, x=None):
150 # Constant base covariance, unless air-friction (drag) UNCERTAINTY is enabled
151 # AND a state is supplied (the engine passes the predicted x). Then the
152 # process-noise acceleration std rides the local speed,
153 # σ_a(v) = ε · (c_drag/m) · |v|² ,
154 # added in quadrature to the constant base floor → Q is large at the muzzle
155 # (fast, drag poorly known) and ≈ base near apex (slow). Q scales as σ_a²,
156 # i.e. ∝ v⁴ in the drag term. The truth sampler still uses the constant base.
157 if self._drag_uncertainty is None or x is None:
158 return self._Q_Q_Q
159 xv = np.asarray(x, dtype=self.dtypedtype).reshape(-1)
160 if self.order < 2:
161 return self._Q_Q_Q
162 vi = [self.order * i + 1 for i in range(3)] # velocity slot of each axis
163 vmag = float(np.linalg.norm(xv[vi]))
164 sigma_drag = self._drag_uncertainty * (self["c_drag"] / self["m"]) * vmag**2
165 sigma2 = self._sigma_a_base**2 + sigma_drag**2
166 return sigma2 * self._Q_unit

168 def Qd(self, sigma: float, order: int, process_noise_model=BallisticNoise.ACCELERATION) -> np.ndarray:
169 """
170 Build the block-diagonal (per-axis) discrete process covariance Q for 3D
171 ballistic motion. Uses the same per-axis kinematic blocks as
172 BallisticLinearModel (kept consistent by convention, not shared code).
173
174 Parameters
175 ----------
176 sigma : float
177 Strength of the white disturbance (square root of its spectral
178 density); its UNIT is set by ``process_noise_model``
179 (BallisticNoise.ACCELERATION → m/s^1.5: the velocity wanders by
180 sigma·√t; BallisticNoise.JERK → m/s^2.5).
181 order : int
182 Model order (2 or 3).
183 process_noise_model : BallisticNoise
184 ACCELERATION (white-noise-acceleration) or JERK (white-noise-jerk) —
185 the same per-axis blocks BallisticLinearModel uses.
186
187 Returns
188 -------
189 Q : ndarray, shape (3*order, 3*order)
190 Block-diagonal process covariance matrix.
191
192 Note
193 ----
194 Default ``ACCELERATION`` drives the disturbance through velocity &
195 position, which is the right choice here: this EKF's ``f()`` recomputes the
196 acceleration from the dynamics every step (gravity + drag), so it is not a
197 free integrator state — a JERK disturbance placed on it would be dropped
198 from the propagated truth while still inflating the predicted covariance.
199 ``JERK`` is supported for completeness but is only physically meaningful for
200 a pure-kinematic constant-acceleration model (i.e. BallisticLinearModel).
201 """
202 process_noise_model = BallisticNoise.of(process_noise_model)
203 q = self.dtypedtype(sigma)**2
204 dt = self.dtypedtype(1.0 / self.fsfsfsfsfs)
205
206 if process_noise_model == BallisticNoise.ACCELERATION:
207 if order == 3:
208 Q1D = np.array([[dt**3/3.0, dt**2/2.0, 0.0],
209 [dt**2/2.0, dt, 0.0],
210 [0.0, 0.0, 0.0]], dtype=self.dtypedtype)
211 elif order == 2:
212 Q1D = np.array([[dt**3/3.0, dt**2/2.0],
213 [dt**2/2.0, dt]], dtype=self.dtypedtype)
214 else:
215 raise ValueError(f"order must be 2 or 3, got {order}.")
216 else: # BallisticNoise.JERK
217 if order != 3:
218 raise ValueError("JERK noise requires order==3 (acceleration must be a state).")
219 Q1D = np.array([[dt**5/20.0, dt**4/8.0, dt**3/6.0],
220 [dt**4/8.0, dt**3/3.0, dt**2/2.0],
221 [dt**3/6.0, dt**2/2.0, dt]], dtype=self.dtypedtype)
222
223 Q = np.zeros((3*order, 3*order), dtype=self.dtypedtype)
224 for i in range(3): # fixed XYZ
225 s = slice(order*i, order*(i+1))
226 Q[s, s] = q * Q1D
227
228 return Q

235 def f(self, x, u=None, w=None, t=None):
236 """
237 The motion as a derivative, per axis [p, v] or [p, v, a]:
238 dp/dt = v
239 dv/dt = a_grav(p) + a_drag(v) + a_ctrl(u)
240 da/dt = 0 (the acceleration slot is not integrated, see after_step)
241 The framework integrates it over one sample with the model's integrator.
242 """
243 x = np.asarray(x, dtype=self.dtypedtype).reshape(-1, 1)
244 # Axis-major layout: [px,vx,ax, py,vy,ay, pz,vz,az] at order 3, or
245 # [px,vx, py,vy, pz,vz] at order 2. The stride is `order`.
246 pi = [self.order * i for i in range(3)]
247 vi = [self.order * i + 1 for i in range(3)]
248 rate = np.zeros_like(x)
249 rate[pi, 0] = x[vi, 0]
250 rate[vi, 0] = self._acceleration(x[pi, 0], x[vi, 0], u)
251 return rate
252
253 def after_step(self, x, u=None, t=None):
254 """The acceleration slot reports the forces where the step lands."""
255 x = np.asarray(x, dtype=self.dtypedtype).reshape(-1, 1)
256 if self.order >= 3:
257 pi = [self.order * i for i in range(3)]
258 vi = [self.order * i + 1 for i in range(3)]
259 x[[self.order * i + 2 for i in range(3)], 0] = self._acceleration(x[pi, 0], x[vi, 0], u)
260 return x

262 def substeps(self, x=None):
263 if x is None:
264 return 1 if self._substeps == "auto" else self._substeps
265 p = np.asarray(x, dtype=float).reshape(-1)[[self.order * i for i in range(3)]]
266 return self._substeps_for(p, 1.0 / self.fsfsfsfsfs)

268 #: "auto" sub-steps: each turns an orbit by at most this angle (ω = sqrt(GM/r³)).
269 AUTO_SUBSTEP_RADIANS = 0.01
270
271 def _substeps_for(self, p, dt):
272 """Sub-steps for one sample: the fixed count, or in orbital mode with "auto" enough
273 that each covers at most AUTO_SUBSTEP_RADIANS of the local orbital motion."""
274 if self._substeps != "auto":
275 return self._substeps
276 if self["g"] is not None:
277 return 1
278 r = float(np.linalg.norm(p))
279 if r <= 0.0:
280 return 1
281 omega = np.sqrt(self["G"] * self["M"] / r**3)
282 return max(1, min(10_000, int(np.ceil(omega * dt / self.AUTO_SUBSTEP_RADIANSAUTO_SUBSTEP_RADIANS))))

284 def _acceleration(self, p, v, u=None):
285 """Gravity + drag + control at position p and velocity v (the forces f() integrates)."""
286 # Near-surface mode (g provided): constant downward acceleration in Z.
287 # Orbital mode (g=None): full geocentric G*M/r³ formula — requires
288 # positions in geocentric metres (origin = Earth's centre).
289 if self["g"] is not None:
290 a_grav = np.array([0.0, 0.0, -self["g"]], dtype=self.dtypedtype)
291 else:
292 r = np.linalg.norm(p)
293 a_grav = -self["G"] * self["M"] * p / r**3 if r > 0 else np.zeros(3, dtype=self.dtypedtype)
294
295 # Drag (depends on velocity, without control)
296 vmag = np.linalg.norm(v)
297 a_drag = -(self["c_drag"] / self["m"]) * vmag * v if vmag > 0 else np.zeros(3, dtype=self.dtypedtype)
298
299 # Control (optional controlled thrust force)
300 a_ctrl = np.zeros(3, dtype=self.dtypedtype)
301 if u is not None:
302 u_arr = np.asarray(u, dtype=self.dtypedtype).ravel()
303 if u_arr.size > 0:
304 a_ctrl = u_arr.reshape(3) / self["m"]
305
306 return a_grav + a_drag + a_ctrl

311 def h(self, x, v=None, t=None):
312 """
313 Measurement model: observes position only.
314 Extracts [px, py, pz] from axis-major state [px,vx,ax, py,vy,ay, pz,vz,az].
315 z = [px, py, pz] + v (additive noise, by default)
316 """
317 x = np.asarray(x, dtype=self.dtypedtype).reshape(-1, 1)
318 # Positions are the first slot of each axis in the axis-major layout, so they are
319 # `order` apart -- not always 0, 3, 6. Hardcoding that stride made the documented
320 # order=2 (position+velocity, no acceleration state) raise IndexError on a 6-state
321 # vector. Derived the same way self.velocity_indices already derives its own.
322 pi = [self.order * i for i in range(3)]
323 z = x[pi, :].copy()
324
325 if v is not None:
326 v = np.asarray(v, dtype=self.dtypedtype).reshape(3, 1)
327 z += v
328
329 return z
