1"""
2Signal-processing helpers — *is the Kalman filter just a low-pass filter?*
3
4These are **not** recursive Bayesian filters and they need no Kalman model, so they live
5in ``runtime`` next to the other utilities (``chrono``, ``timer``, …) rather than under
6``apps``. They make the classic identity computable: for the scalar **local-level /
7random-walk** model
8
9 x_{k+1} = x_k + w_k w_k ~ N(0, q²) (process / drift)
10 z_k = x_k + v_k v_k ~ N(0, r²) (measurement)
11
12the Kalman gain converges to a constant ``K_ss`` and the recursion collapses to
13
14 μ_k = (1 − K_ss)·μ_{k-1} + K_ss·z_k
15
16— *exactly* a first-order low-pass (exponential moving average) with pole ``1 − K_ss``.
17The complementary high-pass is ``z − lowpass(z)``, and in the stationary limit the whole
18thing coincides with the frequency-domain **Wiener filter** ``H = S_xx /(S_xx + S_nn)``.
19
20Build the baselines straight from the two noise standard deviations ``q``/``r`` with the
21``from_qr`` constructors (or directly from a known smoothing factor / variances).
22:func:`pull` and :func:`error_ratio` quantify the improvement (Gaussian pull distributions
23and error-std ratios). ``00_Kalman101.ipynb`` derives the same identity inline with a plain
24``LinearKalmanFilter``, underscoring that none of this needs a special model class.
25"""

33# Steady-state Riccati fixed point (constant-gain Kalman = Wiener = low-pass).
34SteadyState = namedtuple("SteadyState", ["gain", "prior_var", "posterior_var"])
35
36
37def steady_state_gain(process_uncertainty: float, measurement_uncertainty: float) -> SteadyState:
38 """Constant-gain fixed point of the scalar Riccati recursion, from the noise stds ``q``/``r``.
39
40 Solving ``P⁻ = P + Q``, ``K = P⁻/(P⁻+R)``, ``P = (1−K)P⁻`` for the steady state (with
41 ``Q = q²``, ``R = r²``) gives ``P⁻ = ½(q² + √(q⁴ + 4q²r²))`` and the closed-form gain
42
43 K_ss = (√(q⁴ + 4q²r²) − q²) / (2 r²).
44
45 Returns ``(gain, prior_var P⁻, posterior_var P)``.
46 """
47 q2 = float(process_uncertainty) ** 2
48 r2 = float(measurement_uncertainty) ** 2
49 if r2 <= 0.0: # perfect sensor → trust it fully
50 return SteadyState(1.0, q2, 0.0)
51 if q2 <= 0.0: # static signal → gain decays to 0
52 return SteadyState(0.0, 0.0, 0.0)
53 disc = np.sqrt(q2 * q2 + 4.0 * q2 * r2)
54 prior = 0.5 * (q2 + disc)
55 gain = prior / (prior + r2) # == (disc − q2) / (2 r2)
56 post = (1.0 - gain) * prior
57 return SteadyState(float(gain), float(prior), float(post))

63class LowPassFilter:
64 """First-order IIR low-pass / exponential moving average ``y_k=(1−α)y_{k-1}+α z_k``.
65
66 Seeded with ``alpha = K_ss`` (via :meth:`from_qr`) it reproduces the steady-state
67 Kalman estimate exactly once the gain has converged.
68 """
69
70 def __init__(self, alpha: float):
71 self.alpha = float(alpha)
72
73 @classmethod
74 def from_qr(cls, process_uncertainty: float, measurement_uncertainty: float) -> "LowPassFilter":
75 """Seed ``alpha`` with the steady-state Kalman gain ``K_ss(q, r)``."""
76 return cls(steady_state_gain(process_uncertainty, measurement_uncertainty).gain)
77
78 def apply(self, z) -> np.ndarray:
79 z = np.asarray(z, dtype=float).ravel()
80 y = np.empty_like(z)
81 acc = z[0] if z.size else 0.0
82 for k, zk in enumerate(z):
83 acc = (1.0 - self.alpha) * acc + self.alpha * zk
84 y[k] = acc
85 return y
86
87 __call__ = apply

107class WienerFilter:
108 """Non-causal frequency-domain Wiener filter ``H(ω)=S_xx/(S_xx+S_nn)``.
109
110 For the random-walk-plus-white-noise model ``S_xx(ω)=q²/(2−2cosω)`` (diverging at DC,
111 so ``H→1``) and ``S_nn=r²``. This is the stationary optimum the Kalman filter morphs
112 into; here it is applied as a zero-phase FFT smoother for comparison.
113 """
114
115 def __init__(self, process_variance: float, measurement_variance: float):
116 self.q2 = float(process_variance)
117 self.r2 = float(measurement_variance)
118
119 @classmethod
120 def from_qr(cls, process_uncertainty: float, measurement_uncertainty: float) -> "WienerFilter":
121 """Build from the noise standard deviations ``q``/``r`` (variances ``q²``/``r²``)."""
122 return cls(float(process_uncertainty) ** 2, float(measurement_uncertainty) ** 2)
123
124 def response(self, omega) -> np.ndarray:
125 omega = np.asarray(omega, dtype=float)
126 denom = 2.0 - 2.0 * np.cos(omega)
127 with np.errstate(divide="ignore", invalid="ignore"):
128 Sxx = np.where(denom > 0.0, self.q2 / denom, np.inf)
129 H = Sxx / (Sxx + self.r2)
130 return np.where(np.isinf(Sxx), 1.0, H) # DC passes fully
131
132 def apply(self, z) -> np.ndarray:
133 z = np.asarray(z, dtype=float).ravel()
134 N = z.size
135 if N == 0:
136 return z
137 Z = np.fft.rfft(z)
138 omega = 2.0 * np.pi * np.fft.rfftfreq(N) # rad/sample
139 return np.fft.irfft(Z * self.response(omega), n=N)
140
141 __call__ = apply

147def pull(estimate, truth, sigma) -> np.ndarray:
148 """Normalised residual ``(estimate − truth)/σ`` — should be ~N(0,1) when calibrated."""
149 estimate = np.asarray(estimate, dtype=float).ravel()
150 truth = np.asarray(truth, dtype=float).ravel()
151 sigma = np.asarray(sigma, dtype=float)
152 sigma = np.where(np.abs(sigma) < 1e-30, 1e-30, sigma)
153 return (estimate - truth) / sigma

156def error_ratio(reference_err, filtered_err) -> float:
157 """``std(reference_err)/std(filtered_err)`` — >1 means the filter reduced the error."""
158 ref = np.std(np.asarray(reference_err, dtype=float).ravel())
159 flt = np.std(np.asarray(filtered_err, dtype=float).ravel())
160 return float(ref / flt) if flt > 0 else float("inf")
