11def _recording(init):
12 """``init`` wrapped so that the instance remembers the arguments it was
13 built with (``_init_arguments``, the outermost constructor's, defaults
14 included). That is what lets a model given by code be written to a file as
15 a reference to its class: the file carries those arguments, and reading it
16 calls the constructor with them again."""
17 import functools
18 import inspect
19 if getattr(init, "_kalman_recording", False):
20 return init
21 try:
22 signature = inspect.signature(init)
23 except (TypeError, ValueError):
24 return init
25
26 @functools.wraps(init)
27 def wrapper(self, *args, **kwargs):
28 if "_init_arguments" not in self.__dict__:
29 try:
30 bound = signature.bind(self, *args, **kwargs)
31 bound.apply_defaults()
32 recorded, first = {}, None
33 for position, (name, value) in enumerate(bound.arguments.items()):
34 kind = signature.parameters[name].kind
35 if position == 0 or kind is inspect.Parameter.VAR_POSITIONAL:
36 continue
37 if position == 1 and name in ("sensor", "name"):
38 first = name # the sensor id: the reader gives its own
39 if kind is inspect.Parameter.VAR_KEYWORD:
40 recorded.update(value)
41 else:
42 recorded[name] = value
43 self.__dict__["_init_arguments"] = recorded
44 self.__dict__["_init_first"] = first
45 except TypeError:
46 pass # the call itself will say what is wrong
47 return init(self, *args, **kwargs)
48
49 wrapper._kalman_recording = True
50 return wrapper

53class KalmanModel(KalmanModelMetadata, KalmanModelMixin):
54 """
55 The single, flexible Kalman **model** contract (the *system*, WHAT).
56
57 A model defines ``f/h`` and the noise covariances ``Q/R``; the Jacobians
58 ``F/H/B/L/M`` are optional refinements with a numerical default. Construct it two
59 ways, **same contract**, runnable by **any** engine:
60
61 - **matrices** — pass ``F=,H=,B=,L=,M=`` (constant or callable ``M(t)``); ``f``/``h``
62 then default to ``f(x)=Fx+Bu+Lw`` / ``h(x)=Hx+Mv`` and ``F()/H()/…`` return them;
63 - **continuous** — pass ``A=`` (and, for process noise, ``Qc=`` with optional
64 ``Lc=``) instead of ``F=``/``Q=``; the model discretizes exactly via
65 ``expm``/Van Loan (needs ``fs=``/``sample_rate=``). ``Ac()``/``Qc()``/``Lc()``
66 return the continuous inputs back for inspection (e.g.
67 ``helpers.symbolic_discretize(model.Ac())``); mutually exclusive with
68 ``F=``/``Q=`` respectively;
69 - **functions** — override ``f``/``h`` (nonlinear); ``F/H`` come from
70 :func:`numerical_jacobian`.
71
72 Engines consume only what they need: UKF/PF use ``f/h/Q/R``; EKF additionally reads
73 ``F/H`` (numerical/analytic); KF reads ``F/H`` as constants.
74
75 Engine binding is **declared, not imported**: ``DEFAULT_ENGINE`` /
76 ``ALLOWED_ENGINES`` are :class:`KalmanEngineKind` values, so a model never imports an
77 engine; the :class:`~kalman.framework.filter.KalmanFilter` resolves/validates them.
78 """

215 def _read_sensors(self, sensor, kwargs):
216 """Read the ``sensor`` argument. A name alone is the common case and
217 returns ``None``. Otherwise the list of parts (names and sensors) is
218 returned for :meth:`_install_sensors`; when every part is a
219 :class:`KalmanSensor` they replace the model's own measurement, and the
220 ``H`` / ``R`` / ``m`` the model is built with are set from them here."""
221 from .sensor import KalmanSensor
222 from scipy.linalg import block_diag
223 parts = list(sensor) if isinstance(sensor, (list, tuple)) else [sensor]
224 if not parts:
225 raise ValueError(f"{type(self).__name__}: give a sensor (a name, a KalmanSensor, or a list of them).")
226 for part in parts:
227 if not isinstance(part, (str, int, KalmanSensor)):
228 raise TypeError(f"{type(self).__name__}: a sensor is a name or a KalmanSensor, got {type(part).__name__}.")
229 for part in parts:
230 if isinstance(part, KalmanSensor) and part.delay:
231 raise ValueError(f"{type(self).__name__}: the sensor {part.name!r} reports late (delay={part.delay}); "
232 "add it with model.add_sensor(sensor), then model.lag(L).")
233 names = [part.name if isinstance(part, KalmanSensor) else str(part) for part in parts]
234 if len(set(names)) != len(names):
235 raise ValueError(f"{type(self).__name__}: two sensors share a name in {names}.")
236 own = [part for part in parts if not isinstance(part, KalmanSensor)]
237 if len(own) > 1:
238 raise ValueError(f"{type(self).__name__}: one name only (the model's own sensor), got {own}.")
239 self._sensor_name_sensor_name = str(own[0]) if own else names[0]
240 self._labels_labels_labels = [self._sensor_name_sensor_name]
241 if len(parts) == 1 and own:
242 return None
243
244 if not own: # the sensors are the whole measurement
245 given = [part for part in parts]
246 counts = [g.channels for g in given]
247 if any(k is None for k in counts):
248 raise ValueError(f"{type(self).__name__}: a sensor that replaces the model's measurement must say "
249 "how many channels it has: give R as a matrix or a list, or labels.")
250 covs = []
251 for g, k in zip(given, counts):
252 R = np.asarray(g.R, dtype=np.float64)
253 covs.append(R * np.eye(k) if R.ndim == 0 else np.diag(R) if R.ndim == 1 else R)
254 kwargs["R"] = block_diag(*covs)
255 self.mmm = int(sum(counts))
256 if all(g.is_linear for g in given):
257 kwargs["H"] = np.vstack([g.G for g in given])
258 else:
259 kwargs.pop("H", None)
260 self.hhh = self._stacked_h(given)
261 return parts

263 def _stacked_h(self, sensors, first=None):
264 """``h`` made of ``first`` (a measurement function, or none) and then each sensor's readings."""
265 def h(x, v=None, t=None):
266 x = np.asarray(x, dtype=float).reshape(-1, 1)
267 rows = [] if first is None else [np.asarray(first(x, t=t), dtype=float).reshape(-1, 1)]
268 rows += [np.asarray(sensor.measure(x, t), dtype=float).reshape(-1, 1) for sensor in sensors]
269 z = np.vstack(rows)
270 return z if v is None else z + np.asarray(v, dtype=float).reshape(-1, 1)
271 return h

317 def _validate_kind(value, allowed, label):
318 """Normalise an ``integrator=``/``solver=`` override.
319
320 Accepts a Kind, a registered name, a concrete class, a ready instance,
321 or None. A **custom** scheme — one whose ``KIND`` is not a member of the
322 framework enum — is validated by its *base class* instead of by
323 ``ALLOWED_*``, since a closed enum cannot describe a kind it has never
324 heard of. Names are checked by resolving them (raising on a typo) but
325 returned unresolved, so ``default_solver(**kwargs)`` can still pass
326 constructor arguments.
327 """
328 if value is None:
329 return None
330
331 from .integrator import KalmanIntegrator
332 from .solver import KalmanSolver
333 if label == "integrator":
334 base, enum_cls = KalmanIntegrator, KalmanIntegratorKind
335 from .integrators import resolve_integrator as _resolve
336 else:
337 base, enum_cls = KalmanSolver, KalmanSolverKind
338 from .solvers import resolve_solver as _resolve
339
340 if isinstance(value, str):
341 _resolve(value) # raises ValueError on an unknown name
342 return value # resolved later, with kwargs
343
344 is_impl = isinstance(value, base) or (isinstance(value, type) and issubclass(value, base))
345 kind = getattr(value, "KIND", value) # instance/class → its KIND; else the Kind itself
346 if is_impl and not isinstance(kind, enum_cls):
347 return value # custom scheme: the base class is the contract
348 if allowed is not None and kind not in allowed:
349 raise TypeError(f"{label} {kind} not in ALLOWED_{label.upper()}S "
350 f"({sorted(k.name for k in allowed)}).")

378 # Continuous dynamics. A class with DYNAMICS = CONTINUOUS writes f as the
379 # derivative dx/dt. When the class is created, that f is kept as
380 # `derivative` and f becomes the decoded step below, so engines and

385 super().__init_subclass__(**kwargs)
386 for alias, symbol in (("step", "f"), ("sense", "h")):

424 def decode_step(self, x, u=None, t=None):
425 """One sample of a CONTINUOUS model: the derivative integrated over
426 ``1/fs`` by the model's integrator, then :meth:`after_step`. No noise."""
427 steps = self.substeps(x)
428 dt = ((1.0 / self.fsfsfs) if self.fsfsfs else 1.0) / steps
429 # The state is a column throughout, whatever shape the caller passed: a
430 # flat state plus a column derivative would broadcast to a matrix.
431 rate = lambda xx, uu, tt: np.asarray(self.derivativederivative(xx, u=uu, t=tt), dtype=self.dtype).reshape(-1, 1)
432 xk = self.default_integrator().integrate(rate, np.asarray(x, dtype=self.dtype).reshape(-1, 1), u=u,
433 t=(0.0 if t is None else t), dt=dt, n=steps)
434 return self.after_step(xk, u=u, t=t)

439 def f(self, x, u=None, w=None, t=None):
440 """State transition, the discrete step. Uses constant matrices if
441 provided; otherwise a subclass writes it (the step, or with
442 ``DYNAMICS = CONTINUOUS`` the derivative, which is decoded)."""
443 if self._F is None:
444 raise NotImplementedError(
445 "Model must define f() or be constructed with matrices (F=, H=, ...).")
446 x = self.FFF(t=t) @ x
447 if u is not None:
448 B = self.BBB(t=t)
449 x = x + (u if B is None else B @ u)
450 if w is not None:
451 L = self.LLL(t=t)
452 x = x + (w if L is None else L @ w)
453 return x

565 initial state as generator. Decoded once per (sample rate, initial state)."""
566 x0 = np.zeros((self.nnn, 1)) if getattr(self, "x0", None) is None else np.asarray(self.x0x0x0x0, dtype=float).reshape(-1, 1)

572 _, Q = van_loan_discretize(np.asarray(A, dtype=float), Lc, self._Qc, 1.0 / self.fsfsfs)
