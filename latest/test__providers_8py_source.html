164class TestProviderDiscovery(unittest.TestCase):
165 """A provider must advertise what it can supply, not only that it exists.
166
167 ``catalog()['providers']`` answers "which sources are there"; without the
168 measurement declaration a caller still has to read the source to learn that
169 ERA5 yields 500 hPa wind, which is exactly the question ``fetch`` needs
170 answered before it can be written.
171 """
172
173 def test_every_provider_declares_its_measurements(self):
174 for name, record in api.providers()["providers"].items():
175 self.assertTrue(record["measurements"],
176 f"provider {name!r} advertises no measurements")
177 for measurement in record["measurements"]:
178 self.assertIn("name", measurement)
179 self.assertIn("description", measurement)

213class TestCatalogSpecInitialState(unittest.TestCase):
214 """initial.x0/P0 must reach catalog models, not just the matrices path.
215
216 Regression for a SILENT failure: make_model filtered kw against the class's NAMED
217 signature, so a model taking **kwargs (as the app models do) never received x0/P0. The
218 filter then started from the origin with a default covariance and returned confident
219 nonsense instead of raising -- a satellite spec carrying its true initial state diverged
220 by 892 km before anyone noticed the state had been dropped.
221 """
222
223 def _spec(self, order, n, x0_first=6771e3):
224 MU, G = 3.986004418e14, 6.674e-11
225 x0 = [0.0] * n
226 x0[0] = x0_first
227 return {
228 "base": "BallisticModel", "fs": 1.0,
229 "dynamics": {"params": {"G": G, "M": MU / G, "m": 1.0, "c_drag": 0, "g": None,
230 "order": order, "fs": 1.0,
231 "process_uncertainty": 0.05,
232 "measurement_uncertainty": 500.0}},
233 "initial": {"x0": x0,
234 "P0": [[(1234.0 ** 2 if i == j else 0.0) for j in range(n)]
235 for i in range(n)]},
236 }
237
238 def test_x0_and_p0_reach_a_catalog_model(self):
239 m = api.make_model(self._spec(3, 9))
240 self.assertAlmostEqual(float(np.ravel(m.x0)[0]), 6771e3, places=3)
241 self.assertAlmostEqual(float(np.asarray(m.P0)[0, 0]), 1234.0 ** 2, places=3)
242
243 def test_ballistic_order_2_is_not_an_index_error(self):
244 """order=2 is documented and used to raise IndexError from a hardcoded [0,3,6]."""
245 for order, n in ((3, 9), (2, 6)):
246 with self.subTest(order=order):
247 m = api.make_model(self._spec(order, n))
248 z = np.ravel(m.h(m.x0))
249 self.assertEqual(z.shape, (3,))
250 self.assertAlmostEqual(float(z[0]), 6771e3, places=3)
251
252 def test_orbital_gravity_is_order_independent(self):
253 """g=None selects GM/r^3; acceleration must not depend on the state layout."""
254 for order, n in ((3, 9), (2, 6)):
255 with self.subTest(order=order):
256 m = api.make_model(self._spec(order, n))
257 x = np.zeros((n, 1))
258 x[0, 0] = 6771e3
