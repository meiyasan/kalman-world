13# Single rotating-frame mode — the building block AND the standalone model
14# ======================================================================
15class AnalyticOscillatorModel(LinearKalmanModel):
16 """
17 A single oscillator tracked as a **complex envelope** ``ψ = Re + i·Im`` in a frame
18 rotating at carrier ``fc`` [Hz]. On resonance (``fc = f0``) ψ is a constant (DC
19 in the rotating frame — the lock-in / heterodyne limit); off resonance it precesses
20 at the detuning ``Δf = f0 − fc``.
21
22 The state is two real numbers ``[Re, Im]`` — the slowly-varying amplitude and phase
23 of the mode.
24
25 **Parameters (preferred — Hz):**
26 - ``f0`` : resonance frequency [Hz]
27 - ``fc`` : carrier / demodulation frequency [Hz]
28
29 **Legacy parameters (rad/s) — still accepted for backward compat:**
30 - ``w0`` : resonance angular frequency [rad/s]
31 - ``wc`` : carrier angular frequency [rad/s]
32
33 **Frames:** This is the rotating-frame atom. The carrier rotation lives in the
34 **signal** (``preprocess`` multiplies by ``e^{−iω_c t}``); ``H = identity``; and
35 the frequency-offset precession sits in ``F = expm(Δω·dt)``. Its lab-frame dual
36 is :class:`PhasorOscillatorModel`, where the carrier lives in the time-varying
37 ``H(t)`` and ``F = I``.
38
39 This model is the atomic block the windowed :class:`OscillatorModel` stacks. Used
40 standalone it expects a 2-column analytic signal ``[real, imag]`` (e.g. from
41 ``scipy.signal.hilbert``); inside an :class:`OscillatorModel` window the
42 demodulation is done once at the window level.
43
44 **pre/in/postprocess symmetry:** ``preprocess`` rotates the analytic input into the
45 rotating frame; ``postprocess`` inverse-rotates back to the lab-frame analytic signal.
46 This means ``sample().zk`` is in the same space as the model's input → you can call
47 ``model.filter(sample.zk)`` directly.
48 """

61 def __init__(self, name, **kwargs):
62 measurement_uncertainty = kwargs.pop("measurement_uncertainty", 0.5)
63 process_uncertainty = kwargs.pop("process_uncertainty", 0.05)
64
65 self.fsfsfsfs = kwargs.pop("fs", None) or kwargs.pop("sample_rate", None) or self.fsfsfsfs
66 self.measurement_uncertainty = measurement_uncertainty
67 self.process_uncertainty = process_uncertainty
68
69 dt = 1.0 / self.fsfsfsfs
70
71 # Accept f0/fc [Hz] as primary parameters; fall through to w0/wc [rad/s]
72 # for backward compatibility with existing code that passes w0/wc.
73 f0_hz = kwargs.pop("f0", None)
74 fc_hz = kwargs.pop("fc", None)
75 w0_rad = kwargs.pop("w0", None)
76 wc_rad = kwargs.pop("wc", None)
77
78 if f0_hz is not None:
79 w0 = 2.0 * np.pi * float(f0_hz)
80 elif w0_rad is not None:
81 w0 = float(w0_rad)
82 else:
83 w0 = 2.0 * np.pi # default 1 Hz
84
85 if fc_hz is not None:
86 wc = 2.0 * np.pi * float(fc_hz)
87 elif wc_rad is not None:
88 wc = float(wc_rad)
89 else:
90 wc = w0 # default: carrier on resonance (lock-in)
91
92 self["f0"] = w0 / (2.0 * np.pi)
93 self["fc"] = wc / (2.0 * np.pi)
94 self["w0"] = w0 # kept for internal / legacy access
95 self["wc"] = wc
96
97 dw = w0 - wc # detuning Δω [rad/s]
98 Q = kwargs.pop("Q", None)
99
100 # F = expm(A·dt), A = [[−γ, −Δω], [Δω, −γ]], γ = w0/(2Q)
101 gamma = w0 / (2.0 * Q) if Q else 0.0
102 A = expm(np.array([[-gamma * dt, -dw * dt],
103 [ dw * dt, -gamma * dt]]))
104 C = np.identity(2)
105 # Both noise arguments are standard deviations, per component (Re and Im):
106 # of the measured envelope, and of the random change of the state per step.
107 R = measurement_uncertainty ** 2 * np.identity(2)
108 Qn = process_uncertainty ** 2 * np.identity(2)
109
110 x0, P0 = kwargs.pop("x0", None), kwargs.pop("P0", None)
111 super().__init__(name, F=A, H=C, Q=Qn, R=R, P0=P0, x0=x0, **kwargs)

113 def preprocess(self, zk, uk=None, **kwargs):
114 """Analytic signal [real, imag] → rotate into the rotating frame at wc."""
115 zr, zi = zk[:, 0], zk[:, 1]
116 times = self.t0t0 + np.arange(len(zk)) / self.fsfsfsfs
117 arg = self["wc"] * times
118 zk_rot = np.column_stack([zr * np.cos(arg) + zi * np.sin(arg),
119 zi * np.cos(arg) - zr * np.sin(arg)])
120 return zk_rot, uk
121
122 def inprocess(self, zk_hat, **kwargs):
123 return zk_hat
124
125 def postprocess(self, zk_hat, **kwargs):
126 """Inverse-rotate back to the lab-frame analytic signal."""
127 zr, zi = zk_hat[:, 0], zk_hat[:, 1]
128 times = self.t0t0 + np.arange(len(zk_hat)) / self.fsfsfsfs
129 arg = self["wc"] * times
130 return np.column_stack([zr * np.cos(arg) - zi * np.sin(arg),
131 zi * np.cos(arg) + zr * np.sin(arg)])
