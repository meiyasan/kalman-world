4Space science — travelling through space (the companion of :mod:`kalman.apps.rocket`,
5which handles takeoff and landing near the pad).
6
7Two regimes, both tracked with an Extended Kalman Filter:
8
9* **Fly to the Moon** — :class:`TransLunarModel`: a geocentric restricted four-body coast
10 (Earth + Moon + Sun), the Apollo free-return and its lunar **gravitational catapult**.
11 With ``use_ephemeris=True`` the Moon/Sun starting geometry is taken from the real
12 ephemeris (astropy) at a chosen ``epoch``.
13
14* **Interplanetary travel** — :class:`InterplanetaryModel`: a heliocentric **N-body** cruise
15 (Sun + planetary perturbations) general over any *departure → target* planet (Mars is the
16 worked example). Built up **gradually**:
17
18 1. :func:`hohmann_transfer` — analytic two-impulse estimate (Δv, time of flight, phasing,
19 synodic period, next ideal departure);
20 2. :func:`lambert` / :func:`porkchop` / :func:`best_departure` — Lambert targeting and a
21 porkchop scan over the live ephemeris to find the **minimum-Δv launch window**, refined
22 to a time-of-day at a real launch site (Earth rotation);
23 3. :class:`InterplanetaryModel` — the full N-body EKF cruise, seeded by the Lambert
24 solution, with impulsive mid-course Δv maneuvers as the control input.
25
26Positions are in scaled units (Mm), velocities in km/s; physical constants and the ephemeris

65def _kep_arc(r0, v0, tof_ks, n=200, mu=None):
66 """Two-body RK4 arc: returns (n+1, 3) position array in Mm."""
67 if mu is None:
68 mu = GM_SUN
69 def _deriv(s, t): return np.r_[s[3:], -mu * s[:3] / np.linalg.norm(s[:3]) ** 3]
70 s = np.concatenate([r0, v0]); dt = tof_ks / n
71 pts = [s[:3].copy()]
72 for _ in range(n):
73 s = _rk4(s, 0, dt, _deriv)
74 pts.append(s[:3].copy())
75 return np.array(pts)

78def _orbit_ellipse(body, t_ref=None, n=300):
79 """True heliocentric orbit path for *body* (one full period, two-body RK4).
80
81 Returns an (n+1, 2) array of (x, y) in AU, or ``None`` for hyperbolic orbits.
82 Using the actual ephemeris position+velocity avoids the circular-orbit approximation
83 and gives the correct elliptical shape (eccentricity, inclination projection).
84 """
85 r0, v0 = body_state(body, t_ref)
86 r0_mag = np.linalg.norm(r0)
87 v0_sq = float(np.dot(v0, v0))
88 denom = 2.0 * GM_SUN / r0_mag - v0_sq
89 if denom <= 0:
90 return None
91 a = GM_SUN / denom
92 T_ks = 2.0 * np.pi * np.sqrt(a ** 3 / GM_SUN)
93 pts = _kep_arc(r0, v0, T_ks, n=n)
94 return pts[:, :2] / D_SUN

100def _third_body(r, r_body, GM):
101 """Third-body perturbation in a frame centred on the primary: ``GM[(r_b-r)/|r_b-r|³ −
102 r_b/|r_b|³]``. The first term is the body's direct pull on the spacecraft; the second
103 (indirect) subtracts its pull on the central body, so a distant body contributes only its
104 true (tiny) tidal effect rather than a spurious bulk force."""
105 d = r_body - r
106 return GM * (d / np.linalg.norm(d) ** 3 - r_body / np.linalg.norm(r_body) ** 3)

109def gravity(r, t, moon_phase=0.0, sun_phase=0.0, bodies=("earth", "moon", "sun")):
110 """Geocentric acceleration on a spacecraft at ``r`` (scaled): Earth's central pull plus
111 the **third-body perturbations** of the Moon and Sun (analytic circular orbits)."""
112 r = np.asarray(r, dtype=float).reshape(3)
113 a = np.zeros(3)
114 if "earth" in bodies:
115 a += -GM_EARTH * r / np.linalg.norm(r) ** 3
116 if "moon" in bodies:
117 a += _third_body(r, moon_position(t, moon_phase), GM_MOON)
118 if "sun" in bodies:
119 a += _third_body(r, sun_position(t, sun_phase), GM_SUN)
120 return a

152class InterplanetaryModel(NativeTwin, ExtendedKalmanModel):
153 """
154 Heliocentric **N-body** interplanetary cruise, general over any *departure → target*
155 planet (Mars is the worked example).
156
157 State ``x = [rx, ry, rz, vx, vy, vz]`` (Sun-centred, position in Mm, velocity in km/s);
158 only the position is tracked (deep-space network). ``f()`` integrates the Sun's central
159 pull plus the **third-body perturbations of the planets** (from the live ephemeris,
160 interpolated over the mission span) with RK4. With ``control=True`` the input ``u`` is an
161 impulsive mid-course **Δv** [km/s] applied at the start of a step (``l = 3``). Seed it from
162 a Lambert solution, e.g. ``model.initial_state(best["x0"])`` where ``best`` comes from
163 :func:`best_departure`. If ``epoch`` is ``None`` the perturbers are dropped and the model
164 reduces to a fast two-body Sun cruise (handy for tests without the ephemeris).
165
166 **Spacecraft defaults** describe a generic small Mars probe at departure (illustrative
167 round numbers, not a particular mission):
168
169 * dry mass 600 kg (structure + instruments + avionics)
170 * wet mass 1200 kg (+ 600 kg of propellant)
171
172 Pass these to :class:`TrajectoryControlManeuver` to get realistic burn durations and
173 propellant accounting.
174 """

625# ===========================================================================
626# 4) Trajectory Correction Manoeuvre (TCM) controller
627# ===========================================================================
628class TrajectoryControlManeuver(KalmanController):
629 """Realistic finite-burn Trajectory Correction Manoeuvre (TCM) controller.
630
631 Inherits from :class:`~kalman.framework.KalmanController`. At each
632 scheduled epoch the controller:
633
634 1. Solves **Lambert's problem** from the current EKF position to the
635 target arrival position to find the velocity vector needed.
636 2. Computes **Δv** = v_needed − v_current.
637 3. Applies the **Tsiolkovsky rocket equation** to derive propellant
638 consumed and **burn duration** (seconds to minutes, not an instant).
639 4. Returns the velocity impulse; the EKF applies it as a state update.
640
641 **Early TCMs have the most leverage.** A 1 m/s Δv fired
642 175 days before Mars arrival shifts the arrival point by ≈ 15 000 km,
643 whereas the same burn fired 5 days out moves it only ≈ 430 km. The
644 default schedule (day 5, 30, 90) uses this: an early burn for the
645 launch-injection error, then two trims.
646
647 Lambert targeting assumes the Sun's gravity alone, so the planets' pull
648 after the last burn leaves a residual miss; a further burn close to
649 arrival removes it, at a higher Δv per kilometre.
650
651 Default spacecraft: a generic small probe (600 kg dry, 1200 kg wet,
652 one 22 N hydrazine thruster, Isp 220 s). Pass ``dry_mass_kg`` /
653 ``wet_mass_kg`` / ``Isp_s`` / ``thrust_N`` to override.
654
655 Parameters
656 ----------
657 model : InterplanetaryModel
658 Cruise model. ``control=True`` is enforced automatically.
659 r_target : (3,) array_like
660 Heliocentric arrival position [Mm] (e.g. ``best["x0"][:3]`` for
661 the Lambert best solution at the *target* date, or compute via
662 ``body_state(target, arrival_date)[0]``).
663 t_arrival_ks : float
664 Mission-elapsed arrival time [ks] (``tof_days * DAY``).
665 tcm_days : list of float, optional
666 Mission-elapsed days at which TCMs fire. Default: [5, 30, 90].
667 dry_mass_kg, wet_mass_kg : float
668 Spacecraft dry / wet mass [kg].
669 Isp_s : float
670 Specific impulse [s].
671 thrust_N : float
672 Total thruster thrust [N].
673 max_dv_kms : float
674 Per-TCM Δv cap [km/s] (default 0.05 = 50 m/s). Guards against
675 overburn when the Lambert solution diverges from the current state
676 by more than a realistic injection error.
677 mu : float, optional
678 Gravitational parameter for Lambert solve [Mm³/ks²]. Defaults to
679 ``GM_SUN`` (heliocentric cruise).
680 """
681
682 DRY_MASS_KG = 600.0 # kg — generic small probe, dry
683 WET_MASS_KG = 1200.0 # kg — including 600 kg of propellant
684 ISP_S = 220.0 # s — hydrazine monopropellant thruster
685 THRUST_N = 22.0 # N — one small thruster (real missions fire several)
686 MAX_DV_KMS = 0.050 # km/s — 50 m/s per-TCM cap
687 TCM_DAYS_DEFAULT = [5.0, 30.0, 90.0]
688
689 def __init__(self, model, r_target, t_arrival_ks, *,
690 tcm_days=None, dry_mass_kg=None, wet_mass_kg=None,
691 Isp_s=None, thrust_N=None, max_dv_kms=None, mu=None):
692 model.l = 3 # ensure model accepts control input
693 super().__init__(model, Rcost=np.eye(3))
694 self.r_target = np.asarray(r_target, float).reshape(3)
695 self.t_arrival = float(t_arrival_ks)
696 tcm_days = list(tcm_days or self.TCM_DAYS_DEFAULT)
697 self._tcm_times = sorted(float(d) * DAY for d in tcm_days) # [TU = ks]
698 self.dry_mass = float(dry_mass_kg if dry_mass_kg is not None else self.DRY_MASS_KG)
699 self._wet_mass_init = float(wet_mass_kg if wet_mass_kg is not None else self.WET_MASS_KG)
700 self.current_mass = self._wet_mass_init
701 self.Isp = float(Isp_s if Isp_s is not None else self.ISP_S)
702 self.thrust = float(thrust_N if thrust_N is not None else self.THRUST_N)
703 self.max_dv = float(max_dv_kms if max_dv_kms is not None else self.MAX_DV_KMS)
704 self.mu = float(mu if mu is not None else GM_SUN)
705 self.burns = [] # list of burn metadata dicts (populated at each TCM)
706 self._fired = set() # indices of TCMs already executed
707 self._dt_half = 0.5 * model._dt # match tolerance: half an EKF step
708
709 @property
710 def propellant_remaining_kg(self):
711 """Propellant left in the tanks [kg]."""
712 return max(self.current_mass - self.dry_mass, 0.0)
713
714 @property
715 def propellant_consumed_kg(self):
716 """Total propellant consumed across all TCMs [kg]."""
717 return sum(b["prop_kg"] for b in self.burns)
718
719 def _policy(self, x_hat, P, t, *, nu=None):
720 t = float(t)
721 for i, t_tcm in enumerate(self._tcm_times):
722 if i in self._fired:
723 continue
724 if abs(t - t_tcm) <= self._dt_half:
725 return self._execute_tcm(i, x_hat, t)
726 return np.zeros(3)
727
728 def _execute_tcm(self, idx, x_hat, t):
729 """Compute and log the burn for scheduled TCM *idx*."""
730 self._fired.add(idx)
731 r_now = x_hat[:3].flatten()
732 v_now = x_hat[3:6].flatten()
733 tof = self.t_arrival - t # remaining time-of-flight [ks]
734
735 if tof <= 0.0 or self.propellant_remaining_kgpropellant_remaining_kg <= 0.0:
736 return np.zeros(3)
737
738 # Lambert targeting: what velocity do we need NOW to reach r_target in tof?
739 try:
740 v_needed, _ = lambert(r_now, self.r_target, tof, mu=self.mu)
741 except Exception:
742 return np.zeros(3)
743
744 dv_vec = v_needed - v_now # required Δv [km/s]
745 dv_mag = float(np.linalg.norm(dv_vec))
746
747 # Cap: prevent overburn if injection error is unrealistically large
748 if dv_mag > self.max_dv:
749 dv_vec = dv_vec * (self.max_dv / dv_mag)
750 dv_mag = self.max_dv
751
752 # Tsiolkovsky rocket equation (SI: Isp in s, g0 in m/s², Δv in m/s)
753 dv_ms = dv_mag * 1e3 # km/s → m/s
754 max_dv_ms = self.Isp * G0 * np.log(self.current_mass / max(self.dry_mass, 1.0))
755 if dv_ms > max_dv_ms > 0.0: # out of propellant — burn what's left
756 dv_ms = max_dv_ms
757 dv_vec = dv_vec * (dv_ms / max(dv_mag * 1e3, 1e-10))
758 dv_mag = dv_ms * 1e-3
759
760 prop_kg = self.current_mass * (1.0 - np.exp(-dv_ms / (self.Isp * G0)))
761 prop_kg = min(prop_kg, self.propellant_remaining_kgpropellant_remaining_kg)
762
763 # Burn duration: Δm × Isp × g0 / F (constant-thrust, isentropic approximation)
764 burn_s = prop_kg * self.Isp * G0 / max(self.thrust, 1e-6)
765
766 m_before = self.current_mass
767 self.current_mass -= prop_kg
768
769 self.burns.append({
770 "tcm_idx": idx + 1,
771 "t_ks": t,
772 "t_days": t / DAY,
773 "dv_kms": float(dv_mag),
774 "dv_ms": float(dv_ms),
775 "dv_vec_kms": dv_vec.copy(),
776 "prop_kg": float(prop_kg),
777 "mass_before_kg": float(m_before),
778 "mass_after_kg": float(self.current_mass),
779 "burn_s": float(burn_s),
780 "tof_rem_days": float(tof / DAY),
781 })
782 return dv_vec.reshape(3)
783
784 def reset(self):
785 """Reset to the pre-launch state (full tanks, no burns fired).
786
787 Call this between a truth-propagation pass and the EKF filter pass so
788 the controller can re-fire the same TCM schedule on the filter estimate.
789 """
790 super().reset()
791 if not hasattr(self, "_wet_mass_init"):
792 return # called by KalmanController.__init__ before our attrs exist
793 self.current_mass = self._wet_mass_init
794 self.burns = []
795 self._fired = set()
796
797 def summary(self):
798 """Print a formatted table of all TCMs executed so far."""
799 if not self.burns:
800 print("No TCMs executed.")
801 return
802 print("TCM Summary")
803 print("─" * 76)
804 for b in self.burns:
805 print(" TCM-%-2d t=%6.1f d Δv=%8.2f m/s burn=%6.1f s "

807 % (b["tcm_idx"], b["t_days"], b["dv_ms"], b["burn_s"],
808 b["prop_kg"], b["mass_after_kg"], b["tof_rem_days"]))
809 print("─" * 76)
810 total_dv = sum(b["dv_ms"] for b in self.burns)
811 total_prop = sum(b["prop_kg"] for b in self.burns)
812 print(" TOTAL Δv=%8.2f m/s Δm=%6.2f kg propellant left: %.2f kg"

852def plot_porkchop(grid, *, ax=None, dv_max=None, figsize=(9.5, 5.8)):
853 """Render a :func:`porkchop` ``grid`` as a proper porkchop plot: filled Δv contours
854 focused on the launch-window basin (optimum + ``dv_max`` km/s), labelled contour lines, a
855 colour bar, and the minimum-Δv node starred. Returns the figure."""
856 import matplotlib.pyplot as plt

858 from astropy.time import Time
859 x = mdates.date2num(Time(grid["dep_mjd"], format="mjd").to_datetime()) # departure calendar dates
860 y, DV = grid["tof_days"], grid["dv_total"]
861 # Lambert can fail to converge for awkward transfer geometries and silently return NaN
862 # (sqrt of a negative number) rather than raising — contourf then renders those NaN cells
863 # as gaps (the "white diamond" artefacts). Treat them as "very large Δv", same as the
864 # excluded/failed nodes already marked np.inf, so they blend into the dark-red background.
865 DV = np.where(np.isfinite(DV), DV, np.inf)

870 # structure is visible; cap at vmin+4 for very wide distributions.
871 spread = float(np.percentile(DV_fin, 60)) - vmin
872 dv_max = vmin + max(spread * 2.5, 1.0)
873 dv_max = min(dv_max, vmin + 4.0)
874 else:
875 dv_max = float(dv_max)
876 # contourf uses clipped Z (clean colors); contour uses NaN-masked Z so line labels
877 # only appear inside the launch-window basin.
878 Z_fill = np.clip(DV, vmin, dv_max).T # (n_tof, n_dep)
879 Z_line = np.where(DV.T <= dv_max, DV.T, np.nan)
880 n_levels = 20
881 levels = np.linspace(vmin, dv_max, n_levels + 1)
882 fig = ax.figure if ax is not None else plt.subplots(figsize=figsize)[1].figure
883 if ax is None:
