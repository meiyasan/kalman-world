34 std::shared_ptr<Models::Linear> Cart(const std::string &name = "cart", double q = 1.)
35 {
36 auto cart = std::make_shared<Models::Linear>(name, 2, 1);
37 cart->SetSampleRate(FS);
38 const double dt = cart->DeltaT();
39
40 Matrix F = Matrix::Identity(2);
41 F(0, 1) = dt;
42 cart->SetF(F);
43
44 Matrix H(1, 2, 0.);
45 H(0, 0) = 1.;
46 cart->SetH(H);
47
48 Matrix Q(2, 2, 0.);
49 Q(0, 0) = q * dt * dt * dt * dt / 4.;
50 Q(0, 1) = Q(1, 0) = q * dt * dt * dt / 2.;
51 Q(1, 1) = q * dt * dt;
52 cart->SetQ(Q);
53
54 Matrix R(1, 1, 0.);
55 R(0, 0) = 0.25;
56 cart->SetR(R);
57
58 Vector x0(2);
59 x0(0) = 0.;
60 x0(1) = 1.;
61 cart->SetInitialState(x0);
62 cart->SetInitialCovariance(Matrix::Identity(2));
63 return cart;
64 }

81 // --- AddSensor, then Lag: a precise camera that reports 0.3 s late -----------------------
82 // A late sensor reads a past state, so the model has to carry it: Lag(model, 3)
83 // stacks the last three states under the present one.
84 {
85 Sensor camera = Position("camera", 0.01);
86 camera.delay = 0.3;
87 const std::shared_ptr<Models::Lag> late = Lag(AddSensor(Cart(), camera), 3);
88
89 Sampler sampler(late.get(), 1);
90 sampler.Generate(N, late->X0());
91 Filters::KF both(late.get());
92 const Record record = both.Forward(sampler.Measurements());
93
94 // the same readings without the camera: the first channel only, on the plain cart
95 auto cart = Cart();
96 std::vector<Vector> first;
97 for (const Vector &z : sampler.Measurements())
98 {
99 Vector one(1);
100 one(0) = z(0);
101 first.push_back(one);
102 }
103 Filters::KF alone(cart.get());
104 const Record without = alone.Forward(first);
105
106 std::printf("AddSensor + Lag: position error %.3f with the late camera, %.3f without\n",
107 record.RMSE(0, sampler.Truth()), without.RMSE(0, sampler.Truth()));
108 std::printf(" the position three steps ago is known to %.3f (fixed-lag smoothing)\n",
109 record.RMSE(6, sampler.Truth()));
110 }
111
112 // --- AddInstrument, then Joint: the offset of an uncalibrated gauge ----------------------------
113 // An instrument is sensors under one name. Calibrate() makes it carry a gain and
114 // an offset, which the model then declares as parameters; Joint estimates one.
115 {
116 const Sensor probe = Position("probe", 0.04);
117 const std::shared_ptr<Model> biased = AddInstrument(Cart(), Compose({probe}, "gauge").Calibrate(1., 0.8));
118 Sampler sampler(biased.get(), 2);
119 sampler.Generate(N, biased->X0());
120
121 const std::shared_ptr<Model> assumed = AddInstrument(Cart(), Compose({probe}, "gauge").Calibrate(1., 0.));
122 const std::shared_ptr<Models::Joint> model = Joint(assumed, {{"gauge.offset", 2., 0.}});
123 Filters::EKF ekf(model.get());
124 const Record record = ekf.Forward(sampler.Measurements());
125 std::printf("Joint: gauge.offset = %.3f +/- %.3f (truth 0.8)\n",
126 record.State(N - 1)(2), record.Sigma(2)[N - 1]);
127 }
128
129 // --- Compose: two carts in one filter ----------------------------------------------------------------
130 {
131 const std::shared_ptr<Models::Composite> pair = Compose({Cart("left"), Cart("right")});
132 Sampler sampler(pair.get(), 3);
133 sampler.Generate(N, pair->X0());
134 Filters::KF kf(pair.get());
135 const Record record = kf.Forward(sampler.Measurements());
136 std::printf("Compose: n = %zu, m = %zu, parts %s and %s; position errors %.3f and %.3f\n",
137 pair->NState(), pair->NMeasure(), pair->PartName(0).c_str(), pair->PartName(1).c_str(),
138 record.RMSE(0, sampler.Truth()), record.RMSE(2, sampler.Truth()));
139 }
140
141 // --- Route: a cart that is quiet, then shaken -----------------------------------------------------------
142 // The same state seen by two models; the interacting engine runs both and keeps
143 // the probability of each. Filter picks that engine from the model.
144 {
145 auto quiet = Cart("quiet", 0.01), shaken = Cart("shaken", 100.);
146 Sampler first(quiet.get(), 4);
147 first.Generate(N / 2, quiet->X0());
148 Sampler second(shaken.get(), 5);
149 second.Generate(N / 2, first.Truth().back());
150 std::vector<Vector> zk = first.Measurements();
151 zk.insert(zk.end(), second.Measurements().begin(), second.Measurements().end());
152
153 Matrix Pi(2, 2, 0.02);
154 Pi(0, 0) = Pi(1, 1) = 0.98;
155 const std::shared_ptr<Models::Router> modes = Route({quiet, shaken}, Pi);
156 Filter filter(modes.get());
157 filter.Forward(zk);
158
159 const Matrix mu = dynamic_cast<const Engines::Interacting &>(*filter.GetEngine()).ModeHistory();
160 double before = 0., after = 0.;
161 for (std::size_t k = 0; k < N / 2; ++k)
162 {
163 before += mu(k, 1);
164 after += mu(N / 2 + k, 1);
165 }
166 std::printf("Route: probability of the shaken regime: %.2f in the first half, %.2f in the second\n",
167 before / (N / 2), after / (N / 2));
168 }
169
170 return 0;
171}
