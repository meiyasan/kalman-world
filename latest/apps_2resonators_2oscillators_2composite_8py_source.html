14# One analysis window [fc ± dfc] — the atomic windowed model
15# ======================================================================
16class OscillatorModel(CompositeKalmanModel):
17 """
18 One analysis window ``[fc ± dfc]`` over a **real** signal — the generic rotating-frame
19 oscillator tracker.
20
21 ``frequencies`` is the full candidate mode list (already expanded, e.g. from
22 :func:`partial_frequencies`). The window acts as a **mask**: only modes with
23 ``|f − fc| ≤ dfc`` are kept and tracked; the rest are silently dropped from the
24 state. This is the key property: even if :class:`HarmonicOscillatorModel` or
25 :class:`InharmonicOscillatorModel` generates a full ladder ``[f0, 2f0, 3f0, …]``,
26 only those partials inside ``[fc ± dfc]`` are tracked by *this* window. To track
27 the whole ladder, stack one window per partial inside an
28 :class:`OscillatorBankModel`.
29
30 The in-band modes share the single demodulated band channel in **FUSED** mode
31 (``H = [I₂ … I₂]``, ``z = Σ_i ψ_i``) — the joint Kalman filter of Moisset et al.
32 (2025) / Mukherjee & Finn (1999). A single in-band mode is the trivial (degenerate)
33 case.
34
35 **Signal flow:**
36
37 * ``preprocess`` — real strain → Hilbert → demodulate at ``fc`` → ``[Re, Im]``
38 * Filter runs in rotating frame on the ``[Re, Im]`` envelope
39 * ``postprocess`` — ``[Re, Im]`` → remodulate at ``fc`` → **real band reconstruction**
40
41 The postprocess remodulation ensures that ``sample().zk`` is a real signal in the
42 same space as the model's input, so ``model.filter(sample.zk)`` works directly.
43
44 Parameters
45 ----------
46 sensor_id : str
47 frequencies : float | array-like
48 Candidate mode frequencies [Hz]; masked to those inside ``[fc ± dfc]``.
49 fc : float
50 Window center frequency [Hz] — the demodulation carrier.
51 dfc : float
52 Window half-width [Hz].
53 Q : float | list
54 Quality factor (scalar, or one per kept mode).
55 fs : float
56 measurement_uncertainty : float
57 Standard deviation of the white noise of the real input signal, per sample
58 at ``fs``. The model converts it to the noise of the demodulated,
59 decimated envelope (see ``__init__``).
60 process_uncertainty : float
61 Standard deviation of the random change of each envelope component per
62 filter step (the filter runs at the decimated rate ``4·dfc``).
63 p0 : float, optional
64 Initial variance of each envelope component. Default: ``(1000·σ)²``, a
65 prior loose enough that the first measurements set the amplitude.
66 """
67

124 # --- resample_poly helper (handles float sample rates) ---
125 @staticmethod
126 def _resample(y, fs_old, fs_new, tolerance=1e2, kaiser_beta=12.0):
127 """Rational resample with **edge-replicate padding** and a **sharp Kaiser
128 anti-alias/anti-image filter**.
129
130 Two boundary effects of a plain ``resample_poly`` are corrected here:
131
132 * **Edge transient** — ``resample_poly`` implicitly zero-pads beyond the
133 data, so its FIR runs into zeros at the edges; on **upsampling** the
134 trailing ~half-filter of output ramps linearly to zero (an asymmetric
135 tail decay — the start recovers, the end collapses). We pad both ends
136 with the edge sample, resample, then trim the padding back out, so the
137 FIR always sees real data across the original span. Output length
138 matches the unpadded ``resample_poly(y, up, down)`` exactly. The pad is
139 a **multiple of ``down``** so the trim offset ``pad·up/down`` is an exact
140 integer — otherwise the rounded trim leaves a fractional-sample group
141 delay (here ~2 ms) that rotates every off-carrier phasor by ``2π·Δf·δt``,
142 silently capping the subtraction of detuned modes at ~30 dB.
143
144 * **Upsampling images** — the decimated phasor's spectral copies sit at
145 ``±k·fs_dec`` and re-appear around ``fc ± k·fs_dec`` after remodulation.
146 The default Kaiser window (``β=5``) only suppresses the first image to
147 ≈ −73 dB. Because the carried signal lives in ``[fc ± dfc]`` while the
148 decimated Nyquist is ``2·dfc`` (a full octave of guard band), a steeper
149 stopband is essentially free: ``β=12`` pushes the first image below
150 −120 dB without touching in-band content.
151 """
152 from scipy.signal import resample_poly
153 from math import gcd
154 y = np.asarray(y)
155 fs_old_int = int(round(float(fs_old) * tolerance))
156 fs_new_int = int(round(float(fs_new) * tolerance))
157 g = gcd(fs_old_int, fs_new_int)
158 up = fs_new_int // g
159 down = fs_old_int // g
160 window = ("kaiser", float(kaiser_beta))
161
162 n_in = len(y)
163 if n_in == 0:
164 return resample_poly(y, up, down, window=window)
165 out_len = -(-n_in * up // down) # ceil(n_in*up/down), scipy's length
166 # Transient ≈ 2×(10·max(up,down)/up) input samples; round the pad up to a
167 # whole number of ``down``-blocks so the trim offset is an exact integer.
168 base = int(np.ceil(10.0 * max(up, down) / up)) * 2 + 8
169 reps = max(1, -(-base // down)) # ceil(base/down), ≥ 1 block
170 pad = reps * down
171 yp = np.concatenate([np.full(pad, y[0], dtype=y.dtype), y,
172 np.full(pad, y[-1], dtype=y.dtype)])
173 yr = resample_poly(yp, up, down, window=window)
174 off = reps * up # = pad·up/down, exact (zero delay)
175 return yr[off:off + out_len]

177 # --- real strain → heterodyne rotation → resample_poly downsample ---
178 def _demod_preprocess(self, zk, uk=None, **kwargs):
179 """Preprocess: real strain → complex baseband phasor at decimated rate.
180
181 Heterodyne rotation ``ψ = 2·x·e^{-jωc·t}`` followed by anti-aliased
182 downsampling to ``fs_dec = 4·dfc``. Returns ``(N_dec, 2)`` = ``[Re(ψ), Im(ψ)]``.
183 Symmetric counterpart: ``_reconstruct_postprocess``.
184 """
185 x = np.asarray(zk, dtype=float).reshape(np.shape(zk)[0], -1)[:, 0] # real strain (T,)
186 N = len(x)
187 self._preprocess_N = N
188 t = self.t0t0 + np.arange(N) / self._fs_original
189 psi = 2.0 * x * np.exp(-1j * self._wc * t)
190 psi_dec = self._resample(psi, self._fs_original, self._fs_dec)
191 return np.column_stack([psi_dec.real, psi_dec.imag]), uk # (N_dec, 2)
192
193 # --- optional band-limit: kill the residual upsampling images ---
194 def _band_limit_phasor(self, psi, cutoff_hz):
195 """Zero-phase low-pass a complex baseband phasor at ``cutoff_hz``.
196
197 Upsampling the decimated phasor leaves residual spectral images at
198 ``±fs_dec`` baseband (i.e. ``fc ± fs_dec`` after remodulation, well
199 outside the reconstruction band ``[fc ± 2·dfc]``). Low-passing the
200 upsampled phasor at ``cutoff_hz = 2·dfc`` removes them, confining the
201 reconstruction to exactly ``[fc ± cutoff_hz]``. In-band content is
202 untouched (the carried envelope lives within ``±dfc``).
203 """
204 from scipy.signal import firwin, filtfilt
205 psi = np.asarray(psi)
206 n = len(psi)
207 if n < 9 or cutoff_hz <= 0:
208 return psi
209 fnyq = self._fs_original / 2.0
210 # Transition band ~ the (2·dfc → fs_dec) guard; size taps to resolve it.
211 ntaps = int(round(3.3 * self._fs_original / cutoff_hz)) | 1
212 ntaps = min(ntaps, ((n - 1) // 3) | 1)
213 if ntaps < 3:
214 return psi
215 taps = firwin(ntaps, cutoff_hz / fnyq)
216 return filtfilt(taps, 1.0, psi, padlen=min(3 * ntaps, n - 1))
217
218 # --- postprocess: filtered phasor sum → real band signal at original rate ---
219 def _reconstruct_postprocess(self, zk_hat, band_limit=False, **kwargs):
220 """Postprocess: FUSED phasor sum (decimated) → real band signal at original rate.
221
222 ``zk_hat`` = ``h(xk)`` from the FUSED composite = ``Σψ̂_i``, shape ``(N_dec, 2)``.
223 Upsamples to ``fs_original`` then remodulates: ``Re[Σψ̂ · e^{jωc·t}]``.
224 Symmetric counterpart: ``_demod_preprocess``.
225 Returns ``(N, 1)`` real.
226
227 ``band_limit`` (default ``False``) — when ``True``, low-pass the upsampled
228 phasor at ``2·dfc`` to erase the residual ``fc ± fs_dec`` upsampling images.
229 """
230 z = np.asarray(zk_hat, dtype=float).reshape(len(zk_hat), -1)
231 psi = z[:, 0] + 1j * z[:, 1]
232 psi_up = self._resample(psi, self._fs_dec, self._fs_original)
233 N = self._preprocess_N
234 if len(psi_up) > N:
235 psi_up = psi_up[:N]
236 elif len(psi_up) < N:
237 psi_up = np.concatenate([psi_up, np.zeros(N - len(psi_up))])
238 if band_limit:
239 psi_up = self._band_limit_phasor(psi_up, 2.0 * self.dfc)
240 t = self.t0t0 + np.arange(N) / self._fs_original
241 arg = self._wc * t
242 return (psi_up.real * np.cos(arg) - psi_up.imag * np.sin(arg)).reshape(-1, 1)

251 """Upsample the decimated phasors in ``record.zk`` and remodulate to the

258 def amplitudes(self, record):
259 """Per-mode amplitude ``|ψ_i| = √(Re²+Im²)`` for the in-band modes — shape ``(T, n)``."""
260 xk = np.asarray(record.xk)
