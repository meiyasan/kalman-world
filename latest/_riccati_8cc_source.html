25 // Both return NaN as soon as an entry is not a number (std::max would skip it),
26 // so a diverging iteration cannot look converged.
27 double MaxAbs(const Matrix &M)
28 {
29 double out = 0.;
30 for (std::size_t i = 0; i < M.Rows() * M.Cols(); ++i)
31 {
32 const double v = std::abs(M.Data()[i]);
33 if (std::isnan(v)) return v;
34 out = std::max(out, v);
35 }
36 return out;
37 }

60 Matrix SolveStein(const Matrix &A, const Matrix &W, std::size_t maxDoublings)
61 {
62 const std::size_t n = A.Rows();
63 if (A.Cols() != n || W.Rows() != n || W.Cols() != n)
64 throw std::invalid_argument("Algebra::SolveStein: A and W must be (n, n).");
65
66 constexpr double eps = std::numeric_limits<double>::epsilon();
67 Matrix X = W, Ak = A;
68 for (std::size_t j = 0; j < maxDoublings; ++j)
69 {
70 const Matrix increment = Ak.Transpose() * X * Ak;
71 X += increment;
72 const double size = MaxAbs(X), step = MaxAbs(increment);
73 if (!std::isfinite(size)) break;
74 // The terms left are at most the last increment again, shrinking
75 // geometrically; below rounding the sum is done.
76 if (step <= eps * size || size == 0.) { Symmetrize(X); return X; }
77 Ak = Ak * Ak;
78 }
79 throw std::runtime_error("Algebra::SolveStein: the sum does not converge (is A Schur stable?).");
80 }

82 Matrix RiccatiGain(const Matrix &A, const Matrix &B, const Matrix &R, const Matrix &P)
83 {
84 const Matrix Bt = B.Transpose();

86 S += R;
87 return Inverse(S) * (Bt * P * A);

97 bool Newton(const Matrix &A, const Matrix &B, const Matrix &Q, const Matrix &R, Matrix &P)
98 {
99 constexpr double eps = std::numeric_limits<double>::epsilon();
100 Matrix current = P;
101 double previous = std::numeric_limits<double>::infinity();
102 for (std::size_t it = 0; it < 100; ++it)
103 {
104 const Matrix K = RiccatiGain(A, B, R, current);
105 Matrix closed = A;
106 closed -= B * K;
107 Matrix W = K.Transpose() * R * K;
108 W += Q;
109
110 Matrix next;
111 try { next = SolveStein(closed, W); }
112 catch (const std::runtime_error &) { return false; }
113
114 const double delta = MaxAbsDifference(next, current), scale = std::max(1., MaxAbs(next));
115 if (!std::isfinite(delta)) return false;
116 current = next;
117 // Rounding reached, or, close to it, no more progress.
118 if (delta <= 8. * eps * scale || (delta >= previous && delta <= 1e-10 * scale))
119 {
120 P = current;
121 return true;
122 }
123 previous = delta;
124 }
125 return false;
126 }
127 }

129 Matrix SolveDARE(const Matrix &A, const Matrix &B, const Matrix &Q, const Matrix &R,
130 double tolerance, std::size_t maxIterations)
131 {
132 CheckShapes(A, B, Q, R);
133
134 // The Riccati recursion from P = Q converges, monotonically, to the stabilising
135 // solution of a stabilisable pair, but linearly, at the square of the slowest
136 // closed-loop pole: near the unit circle it is slow and stops short of
137 // rounding (measured: 6e-8 from scipy for a pole at 0.999). Its gain, though,
138 // stabilises the pair long before P has converged. So Newton is tried at
139 // iterations 0, 1, 2, 4, 8 ... and takes over from the first gain that
140 // stabilises; the recursion alone answers only if Newton never settles.
141 const Matrix At = A.Transpose();
142 Matrix P = Q;
143 std::size_t nextTry = 0;
144 for (std::size_t it = 0; it < maxIterations; ++it)
145 {
146 if (it == nextTry)
147 {
148 Matrix polished = P;
149 if (Newton(A, B, Q, R, polished)) return polished;
150 nextTry = nextTry ? 2 * nextTry : 1;
151 }
152
153 const Matrix G = RiccatiGain(A, B, R, P);
154 Matrix next = At * P * A;
155 next -= (At * P * B) * G;
156 next += Q;
157 Symmetrize(next);
158
159 const double delta = MaxAbsDifference(next, P), scale = std::max(1., MaxAbs(next));
160 P = next;
161 if (!std::isfinite(delta)) break;
162 if (delta <= tolerance * scale)
163 {
164 Newton(A, B, Q, R, P); // polish if it can; else the recursion's P
165 return P;
166 }
167 }
168 throw std::runtime_error("Algebra::SolveDARE: the Riccati iteration did not converge "
169 "(is the pair (A, B) stabilisable?).");
170 }
