51 def test_a_nonlinear_continuous_model_states_an_intensity(self):
52 class Pendulum(KalmanModel):
53 DYNAMICS = KalmanDynamicsKind.CONTINUOUS
54 def __init__(self, **kw):
55 self.n = 2
56 super().__init__("p", H=np.array([[1.0, 0.0]]), R=np.eye(1) * 1e-2, **kw)
57 def f(self, x, u=None, w=None, t=None):
58 th, om = np.asarray(x, float).reshape(-1)
59 return np.array([[om], [-9.81 * np.sin(th) - 0.1 * om]])
60
61 q = 0.5
62 slow, fast = (Pendulum(fs=fs, Qc=np.diag([0.0, q]), x0=np.array([[0.3], [0.0]])) for fs in (50.0, 200.0))
63 for m in (slow, fast):
64 dt = 1.0 / m.fs
65 Q = m.Q()
66 self.assertAlmostEqual(Q[1, 1] / (q * dt), 1.0, places=2) # leading term: intensity x dt
67 self.assertAlmostEqual(Q[0, 0] / (q * dt**3 / 3), 1.0, places=1) # the angle's share, through dx/dt
68 np.testing.assert_allclose(Q, Q.T)
69 # the same physics at two rates: four times the rate, a quarter of the noise per step
70 self.assertAlmostEqual(slow.Q()[1, 1] / fast.Q()[1, 1], 4.0, places=1)
