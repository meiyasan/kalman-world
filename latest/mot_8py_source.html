1"""MOT-challenge dataset loading and evaluation helpers."""
2from __future__ import annotations
3"""
4Computer-vision object tracking with linear Kalman filters — the input is always a
5per-frame object **detector**'s output; this module owns the *filtering*, so a notebook
6never has to assemble ``F``/``H`` or a tracking loop by hand. Three levels, sharing one
7idea (detections in → smoothed, occlusion-robust state out):
8
9* :class:`DetectionModel` — **single object, centroid** ``x = [x, y, vx, vy]`` observing
10 ``(x, y)``. A constant-velocity filter that smooths a detector's centroid jitter and
11 **coasts through occlusions** (a ``NaN`` measurement row → predict-only, covariance grows
12 until re-acquisition).
13* :class:`BoxTrackModel` — **single object, bounding box** in the SORT parameterization
14 ``x = [x, y, s, r, vx, vy, vs]`` observing ``[x, y, s, r]`` (centre, area ``s = w·h``,
15 aspect ``r = w/h``). Now the *box size* is filtered/predicted too, so it coasts at its
16 last size rather than being faked from the covariance.
17* :class:`SortTracker` — **multiple objects** (SORT): per-frame detector **boxes** are
18 matched to existing tracks by IoU + Hungarian assignment, with track **birth/death** and
19 ``max_age`` occlusion coasting. Each track is one :class:`KalmanTrack` over a shared
20 :class:`BoxTrackModel`. :func:`synthetic_multi` generates a reproducible crossing-objects
21 scene (with occlusion) to drive it.
22
23The detector is swappable (a Haar cascade, an Otsu blob, a neural net…); this module only
24needs centroids (``DetectionModel``) or boxes (``BoxTrackModel``/``SortTracker``). See the
25``10_ComputerVision`` notebook. Distinct from ``apps.tracking`` — the combinatorial
26drift-chamber finder (HEP) — which shares the *combinatorial* idea but a different domain.
27"""

49def mot_detections(det_path, conf_thresh: float = 0.0) -> dict:
50 """MOTChallenge ``det.txt`` → ``{frame: (M,4) [x1,y1,x2,y2]}`` keeping rows with conf ≥ thresh."""
51 a = np.atleast_2d(np.loadtxt(det_path, delimiter=",", usecols=(0, 2, 3, 4, 5, 6)))
52 by = {}
53 for fr, x, y, w, h, c in a:
54 if c < conf_thresh:
55 continue
56 by.setdefault(int(fr), []).append([x, y, x + w, y + h])
57 return {f: np.asarray(v, float) for f, v in by.items()}

60def sort_mot_hypothesis(det_by_frame, **tracker_kwargs) -> np.ndarray:
61 """Run :class:`SortTracker` over ``{frame:(M,4)}`` → MOTChallenge rows
62 ``(N,10): frame,id,x,y,w,h,1,-1,-1,-1`` (detection frames only, not coasted predictions)."""
63 tracker_kwargs.setdefault("fs", 1.0) # dt = 1 frame (canonical SORT)
64 tr = SortTracker(**tracker_kwargs)
65 fmax = max(det_by_frame) if det_by_frame else 0
66 rows = []
67 for f in range(1, fmax + 1):
68 for tb in tr.step(det_by_frame.get(f, np.empty((0, 4)))):
69 if tb.coasting: # report detections, not pure predictions
70 continue
71 x1, y1, x2, y2 = tb.box
72 rows.append([f, tb.id + 1, x1, y1, x2 - x1, y2 - y1, 1, -1, -1, -1])
73 return np.asarray(rows, float).reshape(-1, 10)

76def _mot_active_gt_rows(gt_path) -> np.ndarray:
77 """MOT gt.txt → mot15 rows for evaluable targets. Auto-detects MOT16 (flag/class columns →
78 keep active pedestrians: flag==1 & class==1) vs MOT15 (no class → keep all)."""
79 a = np.atleast_2d(np.loadtxt(gt_path, delimiter=","))
80 col7 = a[:, 7] if a.shape[1] > 7 else np.full(len(a), -1.0)
81 is_mot16 = (a.shape[1] > 7
82 and np.all(np.isin(a[:, 6], (0.0, 1.0)))
83 and np.all((col7 >= 1) & (col7 <= 12)))
84 rows = []
85 for r in a:
86 if is_mot16 and not (r[6] == 1 and r[7] == 1):
87 continue
88 rows.append([int(r[0]), int(r[1]), r[2], r[3], r[4], r[5], 1, -1, -1, -1])
89 return np.asarray(rows, float)

92def mot_sequences_from_root(root) -> dict:
93 """Discover ``{name: (det_path, gt_path)}`` under a MOTChallenge ``train`` root (gt present)."""
94 import glob
95 seqs = {}
96 for det_path in sorted(glob.glob(os.path.join(root, "**", "det", "det.txt"), recursive=True)):
97 seqdir = os.path.dirname(os.path.dirname(det_path))
98 gt_path = os.path.join(seqdir, "gt", "gt.txt")
99 if os.path.exists(gt_path):
100 seqs[os.path.basename(seqdir)] = (det_path, gt_path)
101 return seqs

104def evaluate_mot(sequences, *, conf_thresh: float = 0.0, method_name: str | None = None,
105 metrics=None, **tracker_kwargs):
106 """CLEAR-MOT score :class:`SortTracker` on MOTChallenge sequences (needs ``motmetrics``).
107
108 ``sequences`` : ``{name: (det_path, gt_path)}`` (MOT15 or MOT16 format, auto-detected) — e.g.
109 from :func:`mot_sequences_from_root`. Returns the motmetrics summary DataFrame (per sequence
110 + ``OVERALL``); note motmetrics' ``motp`` is a distance (``1 − IoU``), so overlap-% =
111 ``100·(1 − motp)``. The method is labelled ``method_name`` (defaults to ``SortTracker.NAME``).
112 """
113 try:
114 import motmetrics as mm
115 except ImportError as exc: # keep motmetrics an optional dependency
116 raise ImportError("evaluate_mot needs `motmetrics` (pip install motmetrics).") from exc
117 if not hasattr(np, "asfarray"): # motmetrics ≤1.4 on NumPy ≥2.0
118 np.asfarray = lambda a, dtype=float: np.asarray(a, dtype=dtype)
119 import tempfile
120
121 method_name = method_name or SortTracker.NAME
122 metrics = metrics or ['num_frames', 'mota', 'motp', 'idf1', 'mostly_tracked', 'mostly_lost',
123 'num_false_positives', 'num_misses', 'num_switches', 'num_objects']
124 accs, names = [], []
125 with tempfile.TemporaryDirectory() as tmp:
126 for name, (det_path, gt_path) in sequences.items():
127 hyp = sort_mot_hypothesis(mot_detections(det_path, conf_thresh), **tracker_kwargs)
128 hp = os.path.join(tmp, "hyp.txt"); np.savetxt(hp, hyp, fmt="%d,%d,%.2f,%.2f,%.2f,%.2f,%d,%d,%d,%d")
129 gp = os.path.join(tmp, "gt.txt"); np.savetxt(gp, _mot_active_gt_rows(gt_path), fmt="%d,%d,%.2f,%.2f,%.2f,%.2f,%d,%d,%d,%d")
130 gt = mm.io.loadtxt(gp, fmt="mot15-2D")
131 hd = mm.io.loadtxt(hp, fmt="mot15-2D")
132 accs.append(mm.utils.compare_to_groundtruth(gt, hd, "iou", distth=0.5))
133 names.append(name)
134 summary = mm.metrics.create().compute_many(accs, metrics=metrics, names=names,
135 generate_overall=len(names) > 1)
136 summary.attrs["method_name"] = method_name
137 return summary
