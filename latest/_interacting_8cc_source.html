26 std::unique_ptr<Engine> EngineOf(const Model &mode)
27 {
28 switch (mode.DefaultEngine())
29 {
30 case EngineKind::Linear: return std::make_unique<Linear>();
31 case EngineKind::Extended: return std::make_unique<Extended>();
32 case EngineKind::Unscented: return std::make_unique<Unscented>();
33 case EngineKind::Combinatorial: return std::make_unique<Combinatorial>();
34 // Particle and Ensemble are deliberately absent: mixing hands each
35 // mode a Gaussian (x, P) to predict from, and a sample-set engine
36 // ignores that argument: the modes would silently decouple.
37 default: break;
38 }
39 throw std::invalid_argument(std::string("Engines::Interacting: the mode '") + mode.Name() + "' asks for the "
40 + ToString(mode.DefaultEngine()) + " engine, which cannot step a mode.");
41 }

62 void Interacting::Reset(const Model &model)
63 {
64 const auto *router = dynamic_cast<const Models::Router *>(&model);
65 if (!router)
66 throw std::invalid_argument("Engines::Interacting: the model must be a Models::Router (several modes "
67 "of one system); '" + model.Name() + "' is not.");
68 fModel = router;
69 const std::size_t M = router->NModes();
70 fEngines.clear();
71 fX.clear();
72 fP.clear();
73 for (std::size_t j = 0; j < M; ++j)
74 {
75 fEngines.push_back(EngineOf(router->Mode(j)));
76 fEngines.back()->Reset(router->Mode(j));
77 fX.push_back(router->Mode(j).X0());
78 fP.push_back(router->Mode(j).P0());

80 fMu = router->InitialProbabilities();

83 fModeHistory.clear();
84 std::size_t best = 0;
85 for (std::size_t j = 1; j < M; ++j)
86 if (fMu(j) > fMu(best)) best = j;

102 void Interacting::Combine(const Vector &weights, Vector &x, Matrix &P) const
103 {
104 const std::size_t M = fX.size(), n = fX.front().Size();
105 x = Vector(n, 0.);
106 for (std::size_t j = 0; j < M; ++j) x += weights(j) * fX[j];
107 P = Matrix(n, n, 0.);
108 for (std::size_t j = 0; j < M; ++j)
109 for (std::size_t a = 0; a < n; ++a)
110 for (std::size_t b = 0; b < n; ++b)
111 P(a, b) += weights(j) * (fP[j](a, b) + (fX[j](a) - x(a)) * (fX[j](b) - x(b)));
112 Utils::Symmetrize(P);
113 }

115 void Interacting::Predict(const Model &model, const Vector &, const Matrix &, const Vector *u, double t,
116 Vector &xk, Matrix &Pk, Matrix &F) const
117 {
118 const Models::Router &router = Bind(model);
119 const std::size_t M = fX.size(), n = router.NState();
120 const Matrix &Pi = router.Transition();
121
122 // --- the probabilities, one step on ---------------------------------------
123 Vector c(M, 0.);
124 for (std::size_t j = 0; j < M; ++j)
125 {
126 for (std::size_t i = 0; i < M; ++i) c(j) += Pi(i, j) * fMu(i);
127 if (!(c(j) > 1e-12)) c(j) = 1e-12;
128 }
129
130 // --- mixing: every mode starts from a blend of all of them ------------------
131 std::vector<Vector> xMix(M);
132 std::vector<Matrix> pMix(M);
133 for (std::size_t j = 0; j < M; ++j)
134 {
135 Vector x(n, 0.);
136 for (std::size_t i = 0; i < M; ++i) x += (Pi(i, j) * fMu(i) / c(j)) * fX[i];
137 Matrix P(n, n, 0.);
138 for (std::size_t i = 0; i < M; ++i)
139 {
140 const double w = Pi(i, j) * fMu(i) / c(j);
141 for (std::size_t a = 0; a < n; ++a)
142 for (std::size_t b = 0; b < n; ++b)
143 P(a, b) += w * (fP[i](a, b) + (fX[i](a) - x(a)) * (fX[i](b) - x(b)));
144 }
145 xMix[j] = x;
146 pMix[j] = P;
147 }
148
149 // --- each mode predicts from its mixed start ---------------------------------
150 std::vector<Matrix> Fs(M);
151 for (std::size_t j = 0; j < M; ++j)
152 {
153 Vector xPrior;
154 Matrix PPrior;
155 fEngines[j]->Predict(router.Mode(j), xMix[j], pMix[j], u, t, xPrior, PPrior, Fs[j]);
156 fX[j] = xPrior;
157 fP[j] = PPrior;
158 }
159 fC = c;
160
161 Vector weights = Routers::Router::Normalise(c);
162 Combine(weights, xk, Pk);
163 std::size_t best = 0;
164 for (std::size_t j = 1; j < M; ++j)
165 if (c(j) > c(best)) best = j;
166 F = Fs[best];
167 }

169 std::size_t Interacting::Update(const Model &model, const Vector &, const Matrix &, const Vector &z, double t,
170 Vector &xOut, Matrix &POut) const
171 {
172 const Models::Router &router = Bind(model);
173 const std::size_t M = fX.size(), m = router.NMeasure();
174
175 Vector likelihoods(M, 1.);
176 std::vector<double> nis(M, 0.);
177 std::size_t used = 0;
178 for (std::size_t j = 0; j < M; ++j)
179 {
180 Vector nu;
181 Matrix S;
182 fEngines[j]->Innovation(router.Mode(j), fX[j], fP[j], z, t, nu, S);
183
184 // Missing channels drop out of both the likelihood and the NIS, so a
185 // partly blind step carries less evidence, not none.
186 const std::vector<std::size_t> keep = Utils::ValidChannels(nu);
187 if (keep.empty())
188 {
189 likelihoods(j) = 1.;
190 nis[j] = static_cast<double>(m); // neutral: the expected value
191 }
192 else
193 {
194 const Vector nuc = Utils::Compress(nu, keep);
195 const Matrix Sc = Utils::CompressBoth(S, keep);
196 likelihoods(j) = Likelihood(nuc, Sc);
197 nis[j] = Utils::Mahalanobis(nuc, Utils::Inverse(Sc));
198 }
199
200 Vector xPost;
201 Matrix PPost;
202 used = fEngines[j]->Update(router.Mode(j), fX[j], fP[j], z, t, xPost, PPost);
203 fX[j] = xPost;
204 fP[j] = PPost;
205 }
206 fNis.push_back(nis);
207
208 // --- route: the only decision, and the only pluggable part ---------------------
209 fMu = Routers::Router::Normalise(router.Decision().Route(Table(fNis, M), fC, likelihoods));
210 std::vector<double> row(M);
211 std::size_t best = 0;
212 for (std::size_t j = 0; j < M; ++j)
213 {
214 row[j] = fMu(j);
215 if (fMu(j) > fMu(best)) best = j;
216 }
217 fModeHistory.push_back(row);
218 router.SetLeading(best);
219
220 Combine(fMu, xOut, POut);
221 return used;
222 }

224 double Interacting::Likelihood(const Vector &nu, const Matrix &S)
225 {
226 const std::size_t m = nu.Size();
227 if (m == 0) return 1.; // nothing measured: no evidence either way
228
229 const Matrix Sinv = Utils::Inverse(S);
230 const double quad = Utils::Mahalanobis(nu, Sinv);
231
232 // |S| from the Cholesky factor when it exists; the eigenvalues otherwise.
233 double logdet = 0.;
234 const Matrix L = Utils::CholeskyFactor(S, /* jitter */ true);
235 if (!L.Empty())
236 {
237 for (std::size_t i = 0; i < m; ++i) logdet += 2. * std::log(std::abs(L(i, i)));
238 }
239 else
240 {
241 Vector values;
242 Matrix vectors;
243 Utils::SymmetricEigen(S, values, vectors);
244 for (std::size_t i = 0; i < m; ++i)
245 if (std::abs(values(i)) > 1e-300) logdet += std::log(std::abs(values(i)));
246 }
247
248 const double logL = -0.5 * (quad + logdet + static_cast<double>(m) * std::log(2. * M_PI));
249 // Floored, not clamped to zero: a mode whose likelihood underflows must stay
250 // recoverable, and exp() of a very negative number is exactly what underflows.
251 return std::max(std::exp(logL), 1e-300);
252 }
