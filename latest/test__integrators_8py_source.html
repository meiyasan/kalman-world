28 def test_decay_accuracy_ordering(self):
29 """xdot=-x → x(dt)=x0 e^-dt; RK4 & DP far more accurate than Euler."""
30 f = lambda x, u, t: -x
31 x0 = np.array([1.0]); dt = 0.1
32 exact = np.exp(-dt)
33 e = abs(EulerIntegrator().step(f, x0, dt=dt)[0] - exact)
34 r = abs(RK4Integrator().step(f, x0, dt=dt)[0] - exact)
35 d = abs(DormandPrinceIntegrator().step(f, x0, dt=dt)[0] - exact)
36 self.assertLess(r, e / 100)
37 self.assertLess(d, r + 1e-12)
38 for I in (EulerIntegrator, RK4Integrator, DormandPrinceIntegrator):
39 self.assertLess(abs(I().step(f, x0, dt=1e-4)[0] - np.exp(-1e-4)), 1e-3)
40
41 def test_rk4_conserves_shm_energy(self):
42 """ẍ=-x (SHM): RK4 conserves energy far better than Euler over many steps."""
43 f = lambda s, u, t: np.array([s[1], -s[0]])
44 def energy(I, n=2000, dt=0.05):
45 s = np.array([1.0, 0.0])
46 for _ in range(n):
47 s = I.step(f, s, dt=dt)
48 return 0.5 * (s[0] ** 2 + s[1] ** 2)
49 self.assertLess(abs(energy(RK4Integrator()) - 0.5), 1e-3)
50 self.assertGreater(abs(energy(EulerIntegrator()) - 0.5), 0.1)
51
52 def test_integrate_n_steps(self):
53 f = lambda x, u, t: -x
54 one = RK4Integrator()
55 x = one.integrate(f, np.array([1.0]), dt=0.01, n=100)[0]
56 self.assertLess(abs(x - np.exp(-1.0)), 1e-6)
