26def cart():
27 dt, q = 1 / FS, 1.0
28 Q = q * np.array([[dt**4 / 4, dt**3 / 2], [dt**3 / 2, dt**2]])
29 return KalmanModel("cart", fs=FS, F=np.array([[1, dt], [0, 1]]), H=np.array([[1.0, 0.0]]), Q=Q,
30 R=np.array([[0.25]]), x0=np.array([[0.0], [1.0]]), P0=np.eye(2))

37def simulate(model, steps, seed):
38 """Truth and readings of a lagged model, from its own f and h."""
39 rng = np.random.default_rng(seed)
40 n0 = model.n_inner
41 Lq = np.linalg.cholesky(model.Q()[:n0, :n0] + 1e-15 * np.eye(n0))
42 Lr = np.linalg.cholesky(model.R())
43 x, xs, zs = np.asarray(model.x0, dtype=float).copy(), [], []
44 for _ in range(steps):
45 x = model.f(x)
46 x[:n0] += Lq @ rng.standard_normal((n0, 1))
47 xs.append(x.ravel().copy())
48 zs.append((model.h(x) + Lr @ rng.standard_normal((model.m, 1))).ravel())
49 return np.array(xs), np.array(zs)

63 def test_state_and_names(self):
64 model = cart().lag(2)
65 names = list(model.states.__members__)
66 self.assertEqual(len(names), 6)
67 self.assertEqual(names[2:], [names[0] + "@-1", names[1] + "@-1", names[0] + "@-2", names[1] + "@-2"])

82 def test_refusals(self):
83 for depth in (0, -1, 1.5, True):
84 with self.assertRaises(ValueError):
85 cart().lag(depth)

95 def test_fixed_lag_smoothing(self):
96 model = cart().lag(3)
97 truth, zk = simulate(model, 2000, 1)
98 with warnings.catch_warnings():
99 warnings.simplefilter("ignore")
100 rec = model.filter(zk, smooth=False, reset=True)
101 xk = np.asarray(rec.xk).reshape(len(zk), -1)
102 now, then = rmse(xk[:, 0], truth[:, 0]), rmse(xk[:, model.block(3)][:, 0], truth[:, 6])
103 self.assertLess(then, 0.9 * now)

127 def test_the_late_sensor_helps_and_ignoring_its_delay_hurts(self):
128 model = cart().add_sensor(late(0.3)).lag(3)
129 truth, zk = simulate(model, 2000, 2)
130 naive = cart().add_sensor(late(None))
131 with warnings.catch_warnings():
132 warnings.simplefilter("ignore")
133 good = np.asarray(model.filter(zk, smooth=False, reset=True).xk).reshape(len(zk), -1)
134 bad = np.asarray(naive.filter(zk, smooth=False, reset=True).xk).reshape(len(zk), -1)
135 alone = np.asarray(cart().filter(zk[:, :1], smooth=False, reset=True).xk).reshape(len(zk), -1)
136 e_lag, e_naive, e_alone = (rmse(x[:, 0], truth[:, 0]) for x in (good, bad, alone))
137 self.assertLess(e_lag, 0.8 * e_alone)
138 self.assertLess(e_lag, 0.8 * e_naive)
