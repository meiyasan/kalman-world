21 Fusion::Fusion(std::shared_ptr<Model> inner, Sensor sensor)
22 : fInner(std::move(inner)), fSensor(std::move(sensor))
23 {
24 if (!fInner) throw std::invalid_argument("Models::Fusion: the inner model is null.");
25 fSensor.Check(fInner->NState());
26 fNExtra = fSensor.Channels();
27 fNInnerMeasure = fInner->NMeasure();
28 SetName(fInner->Name());
29 Allocate(fInner->NState(), fNInnerMeasure + fNExtra, fInner->NControl());

31 SetSampleRate(fInner->SampleRate());
32 SetT0(fInner->T0());
33 SetTimeUnit(fInner->TimeUnit());
34 SetDefaultEngine(fInner->DefaultEngine());
35 SetBuiltInSensorName(fInner->BuiltInSensorName());
36 AdoptInitialState();
37
38 Settle();
39
40 DeclareState(fInner->StateComponents());
41 // An observable keeps its name unless the model already has one of that
42 // name: it then takes its sensor's name in front ("gyro2_RATE").
43 std::vector<Component> channels = fInner->MeasurementComponents();
44 std::vector<Component> added = fSensor.Observables();
45 for (auto &c : added)
46 {
47 for (const auto &known : channels)
48 if (known.name == c.name) { c.name = fSensor.name + "_" + c.name; break; }
49 channels.push_back(c);
50 }
51 fSensor.channels = added; // what Sensors() reports is what z carries
52 DeclareMeasurement(std::move(channels));
53 }

55 void Fusion::Settle()
56 {
57 const bool formulas = fSensor.IsLinear() || fSensor.jacobian;
58 if (fSensor.IsLinear() && !fSensor.calibrated)
59 {
60 // Still what the inner model is: its matrices with one more block, or code.
61 if (fInner->HasAnalyticJacobian()) SetAnalyticJacobian(true);
62 else SetDifferentiator(fInner->Differentiator(), fInner->Epsilon());
63 SetStructure(fInner->Structure());
64 return;
65 }
66 // A function on top of a linear model is no longer its matrices; nor is a
67 // calibrated reading, `gain * G x + offset`, which no matrix H carries. The
68 // rows of the Jacobian without a formula are differenced with the inner
69 // scheme (forward, at its default step, under an inner model that has formulas).
70 if (fInner->DefaultEngine() == EngineKind::Linear) SetDefaultEngine(EngineKind::Extended);
71 if (fInner->HasAnalyticJacobian() && formulas && fSensor.IsLinear()) SetAnalyticJacobian(true);
72 else if (fInner->HasAnalyticJacobian()) SetDifferentiator(DifferentiatorKind::Forward);
73 else SetDifferentiator(fInner->Differentiator(), fInner->Epsilon());
74 SetStructure(StructureKind::Maps);
75 }

93 {
94 const std::string &owner = fSensor.Owner();

173 Vector Fusion::MeasureWithNoise(const Vector &x, const Vector &v, double t) const
174 {
175 if (v.Size() != fNMeasure)
176 throw std::invalid_argument("Models::Fusion: the measurement noise has " + std::to_string(v.Size())
177 + " components, " + std::to_string(fNMeasure) + " expected.");
178 Sync();
179 Vector head(fNInnerMeasure);
180 for (std::size_t i = 0; i < fNInnerMeasure; ++i) head(i) = v(i);
181 Vector tail = Extra(x, t);
182 for (std::size_t i = 0; i < fNExtra; ++i) tail(i) += v(fNInnerMeasure + i);
183 return Stack(fInner->MeasureWithNoise(x, head, t), tail);
184 }

186 Matrix Fusion::JacobianH(const Vector &x, double t) const
187 {
188 Sync();
189 const Matrix Hs = fInner->JacobianH(x, t);
190 Matrix G;
191 const double gain = fSensor.calibrated ? fSensor.gain : 1.;
192 if (fSensor.IsLinear()) { G = fSensor.G; G *= gain; }
193 else if (fSensor.jacobian) { G = fSensor.jacobian(x, t); G *= gain; }
194 else
195 G = Differentiators::Jacobian(
196 fNumericalScheme, [this, t](const Vector &xx) { return Extra(xx, t); }, x, fEpsilon);
197 if (G.Rows() != fNExtra || G.Cols() != fNState)
198 throw std::runtime_error("Models::Fusion: the Jacobian of the extra channels is not k x n.");
199
200 Matrix out(fNMeasure, fNState, 0.);
201 for (std::size_t j = 0; j < fNState; ++j)
202 {
203 for (std::size_t i = 0; i < fNInnerMeasure; ++i) out(i, j) = Hs(i, j);
204 for (std::size_t i = 0; i < fNExtra; ++i) out(fNInnerMeasure + i, j) = G(i, j);
205 }
206 return out;
207 }

209 Matrix Fusion::H(double t) const
210 {
211 // The matrix a linear engine reads. With a function for the extra
212 // channels there is none: their rows are left at zero, and the engines
213 // that linearise ask JacobianH().
214 const Matrix Hs = fInner->H(t);
215 Matrix out(fNMeasure, fNState, 0.);
216 for (std::size_t j = 0; j < fNState; ++j)
217 {
218 for (std::size_t i = 0; i < fNInnerMeasure; ++i) out(i, j) = Hs(i, j);
219 if (fSensor.IsLinear())
220 for (std::size_t i = 0; i < fNExtra; ++i)
221 out(fNInnerMeasure + i, j) = (fSensor.calibrated ? fSensor.gain : 1.) * fSensor.G(i, j);
222 }
223 return out;
224 }

226 Matrix Fusion::R(double t) const
227 {
228 Sync();
229 const Matrix Rs = fInner->R(t);
230 Matrix out(fNMeasure, fNMeasure, 0.);
231 for (std::size_t i = 0; i < fNInnerMeasure; ++i)
232 for (std::size_t j = 0; j < fNInnerMeasure; ++j) out(i, j) = Rs(i, j);
233 for (std::size_t i = 0; i < fNExtra; ++i)
234 for (std::size_t j = 0; j < fNExtra; ++j) out(fNInnerMeasure + i, fNInnerMeasure + j) = fSensor.R(i, j);
235 return out;
