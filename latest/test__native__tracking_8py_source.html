32@needs_apps
33class TestDriftChamberTwin(unittest.TestCase):
34 # The search compares chi-squares: both sides must make the same choices, and
35 # then agree on each track. One propagation ends in a bisection onto the
36 # layer and both sides difference it with a step of 1e-6, which leaves about
37 # 1e-6 cm on a track of 100 cm (the callback path shows the same).
38 def check(self, model_kw, event_kw):
39 make = lambda: DriftChamberModel(**model_kw)
40 event = make().generate_event(**event_kw)
41 with native.python_reference():
42 python = find(make(), event)
43 model = make()
44 self.assertIsNotNone(model.native_handle())
45 twin = find(model, event)
46 self.assertEqual([t.hit_ids for t in python.tracks_raw], [t.hit_ids for t in twin.tracks_raw])
47 self.assertEqual([t.hit_ids for t in python.candidates], [t.hit_ids for t in twin.candidates])
48 for a, b in zip(python.tracks_raw, twin.tracks_raw):
49 self.assertLess(np.abs(np.asarray(a.xk) - np.asarray(b.xk)).max(), 1e-5)
50 self.assertLess(abs(a.chi2 - b.chi2), 1e-3 * max(1.0, a.chi2))
51 for a, b in zip(python.tracks, twin.tracks): # smoothed records
52 self.assertLess(np.abs(np.asarray(a.xk) - np.asarray(b.xk)).max(), 1e-5)
53 return python, twin

62 def test_one_propagation_agrees(self):
63 model = DriftChamberModel(Bz=1.2, q=-1.0, step=0.25)
64 handle = model.native_handle()
65 x = np.array([3.0, 1.0, 0.6, 0.5])
66 out = np.empty(4)
67 core.check(core.lib.kalman_model_f(handle, core.cdptr(x), core.ffi.NULL, 68.0, core.dptr(out)), "f")
68 self.assertLess(np.abs(out - model.f(x, t=68.0).reshape(-1)).max(), 1e-12)
69 self.assertAlmostEqual(float(np.hypot(out[0], out[1])), 68.0, places=9)
70 # a track that curls back before the layer
71 slow = np.array([3.0, 1.0, 0.02, 0.01])
72 core.check(core.lib.kalman_model_f(handle, core.cdptr(slow), core.ffi.NULL, 104.0, core.dptr(out)), "f")
73 self.assertTrue(np.all(np.isnan(out)) and np.all(np.isnan(model.f(slow, t=104.0))))
