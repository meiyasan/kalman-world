67 Vector PID::Policy(const Vector &xHat, const Matrix &, double, const Vector *)
68 {
69 Vector e = xHat;
70 e -= fXRef;
71 fIntegral += fDt * e;
72 Vector de = e;
73 de -= fPrevError;
74 de *= 1. / fDt;
75 fPrevError = e;
76
77 Vector u = fKp * e;
78 u += fKi * fIntegral;
79 u += fKd * de;
80 u *= -1.;
81 return u;
82 }

104 LQR &LQR::Fit()
105 {
106 if (fModel->NControl() == 0)
107 throw std::runtime_error("Controllers::LQR: the model has no control input (l = 0).");
108 Matrix B = fModel->B(fModel->T0());
109 if (B.Empty()) B = Matrix::Identity(fModel->NState()); // the control enters directly

113 LQR &LQR::Fit(const Matrix &A, const Matrix &B)
114 {
115 if (fModel->NControl() == 0)
116 throw std::runtime_error("Controllers::LQR: the model has no control input (l = 0).");
117 const std::size_t n = fModel->NState();
118 if (A.Rows() != n || A.Cols() != n || B.Rows() != n || B.Cols() != fRCost.Rows())
119 throw std::runtime_error("Controllers::LQR: A must be (n, n) and B (n, l).");
120 const Matrix P = SolveDARE(A, B, fQCost, fRCost, fTolerance, fMaxIterations);
121 fK = Algebra::RiccatiGain(A, B, fRCost, P);
122 return *this;
123 }

153 Vector MPPI::Policy(const Vector &xHat, const Matrix &, double t, const Vector *)
154 {
155 const std::size_t l = NControl(), n = fModel->NState();
156 const double dt = fModel->DeltaT();
157 std::normal_distribution<double> gauss(0., fNoiseSigma);
158
159 // The caller's perturbations, when it gave some, are used once.
160 const std::vector<double> draws = std::move(fDraws);
161 fDraws.clear();
162 std::size_t cursor = 0;
163
164 std::vector<Vector> nominal = fNominal;
165 std::vector<std::vector<Vector>> trial(fNRollouts, std::vector<Vector>(fHorizon, Vector(l, 0.)));
166 std::vector<double> costs(fNRollouts);
167
168 for (std::size_t iter = 0; iter < fNIters; ++iter)
169 {
170 for (std::size_t i = 0; i < fNRollouts; ++i)
171 {
172 Vector xi = xHat;
173 double cost = 0.;
174 for (std::size_t k = 0; k < fHorizon; ++k)
175 {
176 Vector uk = nominal[k];
177 for (std::size_t j = 0; j < l; ++j) uk(j) += draws.empty() ? gauss(fRandom) : draws[cursor++];
178 // Clip the candidate before rolling it out or costing it, so the
179 // averaged nominal stays inside what the actuator can deliver.
180 uk = fModel->LimitControl(uk);
181 trial[i][k] = uk;
182
183 xi = fModel->f(xi, fModel->NControl() > 0 ? uk : Vector(), t + static_cast<double>(k) * dt);
184 Vector e = xi;
185 e -= fXRef;
186 for (std::size_t a = 0; a < n; ++a)
187 for (std::size_t b = 0; b < n; ++b) cost += e(a) * fQCost(a, b) * e(b);
188 for (std::size_t a = 0; a < l; ++a)
189 for (std::size_t b = 0; b < l; ++b) cost += uk(a) * fRCost(a, b) * uk(b);
190 }
191 costs[i] = cost;
192 }
193
194 const double best = *std::min_element(costs.begin(), costs.end());
195 std::vector<double> weights(fNRollouts);
196 double total = 0.;
197 for (std::size_t i = 0; i < fNRollouts; ++i)
198 {
199 weights[i] = std::exp(-(costs[i] - best) / fLambda);
200 total += weights[i];
201 }
202 total += 1e-12;
203
204 for (std::size_t k = 0; k < fHorizon; ++k)
205 {
206 Vector acc(l, 0.);
207 for (std::size_t i = 0; i < fNRollouts; ++i) acc += (weights[i] / total) * trial[i][k];
208 nominal[k] = acc;
209 }
210 }
211
212 const Vector u0 = nominal[0];
213 // Shift the warm-started sequence for the next call's receding horizon.
214 for (std::size_t k = 0; k + 1 < fHorizon; ++k) fNominal[k] = nominal[k + 1];
215 fNominal[fHorizon - 1] = Vector(l, 0.);
216 return u0;
217 }
