56 // F and Q are derived from DeltaT() when asked for rather than fixed in
57 // the constructor, so a consumer that changes the sample rate after
58 // construction — the Python wrapper's `fs=` does exactly that — gets
59 // coefficients that agree with the new step.
60 Kalman::Matrix F(double) const override
61 {
62 Kalman::Matrix F = Kalman::Matrix::Identity(2);
63 F(0, 1) = DeltaT();
64 return F;
65 }
66
67 // Discrete white-noise acceleration (standard deviation sigma_a, constant
68 // over each step), off-diagonals included: the one acceleration moves both
69 // position and velocity, and a diagonal Q here would make the filter
70 // inconsistent.
71 Kalman::Matrix Q(double) const override
72 {
73 const double dt = DeltaT();
74 const double q = fSigmaAcc * fSigmaAcc;
75 Kalman::Matrix Q(2, 2, 0.);
76 Q(0, 0) = q * dt * dt * dt * dt / 4.;
77 Q(0, 1) = q * dt * dt * dt / 2.;
78 Q(1, 0) = Q(0, 1);
79 Q(1, 1) = q * dt * dt;
80 return Q;
81 }
82 };

112 class Pendulum : public Kalman::Models::Extended
113 {
114 private:
115 double fOmega2; // g / l
116 double fGamma; // damping
117
118 public:
119 Pendulum(double fs = 50., double length = 1., double gamma = 0.1)
120 : Kalman::Models::Extended("pendulum", 2, 1), fOmega2(9.81 / length), fGamma(gamma)
121 {
122 SetSampleRate(fs);
123 DeclareState({{"angle", "rad"}, {"rate", "rad/s"}});
124 DeclareMeasurement({{"angle", "rad"}});
125
126 Kalman::Matrix H(1, 2, 0.);
127 H(0, 0) = 1.;
128 SetH(H);
129
130 // Diagonal Q, unlike ConstantVelocity: these are two independent tuning
131 // values (angle and angular-velocity noise per step), not derived from one
132 // physical noise source. A random angular acceleration would instead give
133 // the correlated form used in ConstantVelocity::Q.
134 Kalman::Matrix Q(2, 2, 0.);
135 Q(0, 0) = 1e-6;
136 Q(1, 1) = 1e-4;
137 SetQ(Q);
138
139 Kalman::Matrix R(1, 1, 0.);
140 R(0, 0) = 1e-2;
141 SetR(R);
142
143 Kalman::Vector x0(2);
144 x0(0) = 0.5; // released from half a radian
145 SetInitialState(x0);
146 SetInitialCovariance(Kalman::Matrix::Identity(2));
147 }
148
149 const char *Title() const override { return "Damped Pendulum"; }
150 const char *Description() const override
151 { return "Nonlinear pendulum, one RK4 step per sample; angle observed."; }
152
153 Kalman::Vector f(const Kalman::Vector &x, const Kalman::Vector &, double) const override
154 {
155 const double dt = DeltaT();
156 auto deriv = [this](double th, double om, double &dth, double &dom)
157 {
158 dth = om;
159 dom = -fOmega2 * std::sin(th) - fGamma * om;
160 };
161
162 double k1t, k1o, k2t, k2o, k3t, k3o, k4t, k4o;
163 deriv(x(0), x(1), k1t, k1o);
164 deriv(x(0) + 0.5 * dt * k1t, x(1) + 0.5 * dt * k1o, k2t, k2o);
165 deriv(x(0) + 0.5 * dt * k2t, x(1) + 0.5 * dt * k2o, k3t, k3o);
166 deriv(x(0) + dt * k3t, x(1) + dt * k3o, k4t, k4o);
167
168 Kalman::Vector out(2);
169 out(0) = x(0) + dt / 6. * (k1t + 2. * k2t + 2. * k3t + k4t);
170 out(1) = x(1) + dt / 6. * (k1o + 2. * k2o + 2. * k3o + k4o);
171 return out;
172 }
173 };
