3"""GW interferometer noise subtraction via spectral-line Kalman tracking.
4
5Two-stage workflow
6------------------
7Stage 1 -- peak fitting
8 ``InterferometerModel.fit(sensor_id, strain, fs, bands)``
9 Estimates PSD via Welch, detects spectral peaks per band with
10 ``scipy.signal.find_peaks``, fits each peak to a Lorentzian profile to
11 extract (f0, Q, amplitude), and groups nearby peaks into analysis windows.
12
13Stage 2 -- Kalman tracking and subtraction
14 The returned ``InterferometerModel`` (a specialised ``OscillatorBankModel``)
15 tracks each window's phasor in the rotating frame -- the joint Kalman filter
16 of Moisset et al. 2025 (arXiv:2509.20658). After running the filter::
17
18 record = KalmanFilter(model).forward(strain, uk=model.make_controller())
19 clean = model.subtract(strain, record)
20
21 ``make_controller()`` returns a ``ViolinBankController`` that attaches one
22 ``AdaptiveQRKalmanController`` per window, self-adjusting each window's
23 process-noise level (Q) from empirical innovation statistics.
24
25Noise sources
26-------------
27``NOISE_SOURCES = ("violin",)`` -- violin-mode suspension noise only.
28Future additions:
29
30* ``"powergrid"`` -- use ``PowerGridStateModel`` (already implemented in
31 ``modules/apps/powergrids.py``) as a **separate parallel filter** since its
32 PhasorOscillatorModel architecture (lab-frame, time-varying H, scalar voltage)
33 is incompatible with ``OscillatorBankModel._broadcast_preprocess``.
34* ``"omc_pcal"`` -- fixed-frequency PCal calibration lines can be added as
35 additional ``OscillatorModel`` windows inside this bank (Q approx inf, known fc).
36
37Detector-specific subclasses
38-----------------------------
39Pre-loaded per-run violin-mode parameters are available via detector subclasses::
40
41 from kalman.apps.interferometer import InterferometerK1
42
43 # Compute band limits from the built-in catalog and fit from synthetic data:
44 bands = InterferometerK1.violin_bands("O3GK")
45 model = InterferometerK1.fit("K1", strain, fs=4096, bands=bands)
46
47 # Or download O3GK strain directly (requires gwpy):
48 model = InterferometerK1.from_gps(1271267328, 1271281728, run="O3GK")
49 ctrl = model.make_controller()
50 rec = model.filter(strain, uk=ctrl, smooth=True)
51 clean = model.subtract(strain, rec)
52
53Available: InterferometerK1, InterferometerH1, InterferometerL1, InterferometerV1.
54"""

68# Per-window leaf-level AdaptiveQR wrapper
69# ======================================================================
70
71class _LeafAdaptiveQR:
72 """Drives one ``AdaptiveQRKalmanController`` per leaf inside an ``OscillatorModel``.
73
74 ``OscillatorModel`` is a ``CompositeKalmanModel`` whose ``Q()`` is bound as
75 a closure over its leaf submodels — mutating ``window._Q`` has no effect.
76 This class targets each ``AnalyticOscillatorModel`` leaf directly so that the
77 ``model._Q`` mutations performed by ``AdaptiveQRKalmanController._update``
78 actually propagate through ``CompositeKalmanModel.Q()``.
79 """
80
81 def __init__(self, window: OscillatorModel, *, window_len: int = 30,
82 q_min: float = 0.1, q_max: float = 10.0):
83 self._leaves = window.models # list[AnalyticOscillatorModel]
84 self._ctrls = [
85 AdaptiveQRKalmanController(leaf, window=window_len,
86 q_min=q_min, q_max=q_max)
87 for leaf in self._leaves
88 ]
89
90 def update(self, x_hat_win, P_win, t, nu_win):
91 """Scale each leaf's Q from the window-level innovation ``nu_win`` (shape (2,1)).
92
93 All leaves in a FUSED window share the same 2-element [Re, Im] innovation
94 channel, so each leaf receives the same ``nu_win``. State/covariance are
95 sliced to match each leaf's own ``n=2`` state block.
96 """
97 offset = 0
98 for ctrl in self._ctrls:
99 n = ctrl.model.n
100 ctrl._update(
101 x_hat_win[offset:offset + n],
102 P_win[offset:offset + n, offset:offset + n],
103 t,
104 nu=nu_win,
105 )
106 offset += n
107
108 def reset(self):
109 for ctrl in self._ctrls:
110 ctrl.reset()

113# ======================================================================
114# Bank-level controller (one dispatcher for all windows)
115# ======================================================================
116
117class ViolinBankController(KalmanController):
118 """Bank-level controller: dispatches per-window ``AdaptiveQR`` from bank-level nu/P.
119
120 Intended to be passed as ``uk`` to ``KalmanFilter.forward()``::
121
122 ctrl = model.make_controller()
123 record = KalmanFilter(model).forward(strain, uk=ctrl)
124
125 Each call slices the bank-level state/covariance/innovation into per-window
126 sub-vectors and drives the corresponding ``_LeafAdaptiveQR``. No actuation
127 is produced (``_policy`` returns zeros).
128 """
129
130 def __init__(self, bank_model: "InterferometerModel",
131 window_aqrs: list[_LeafAdaptiveQR],
132 state_offsets: list[int]):
133 # Initialize before super().__init__() since that calls reset()
134 self._window_aqrs = window_aqrs
135 self._state_offsets = state_offsets
136 super().__init__(bank_model)
137
138 def _policy(self, x_hat, P, t, *, nu=None):
139 return np.zeros((max(self.model.l, 1), 1))
140
141 def _update(self, x_hat, P, t, *, nu=None):
142 if nu is None:
143 return None
144 for i, (win, aqr) in enumerate(zip(self.model.windows, self._window_aqrs)):
145 o = self._state_offsets[i]
146 x_i = x_hat[o:o + win.n]
147 P_i = P[o:o + win.n, o:o + win.n]
148 # OscillatorBankModel uses INDEPENDENT composition →
149 # bank nu = [nu_win0; nu_win1; ...], each window contributes 2 elements
150 # (FUSED internally → single 2-D [Re,Im] observation channel per window).
151 nu_i = nu[2 * i: 2 * i + 2]
152 aqr.update(x_i, P_i, t, nu_i)
153 return None
154
155 def reset(self):
156 super().reset()
157 for aqr in self._window_aqrs:
158 aqr.reset()
