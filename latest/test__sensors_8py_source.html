27def target():
28 F, Q, H = np.eye(6), np.zeros((6, 6)), np.zeros((3, 6))
29 for axis in range(3):
30 p, v = 2 * axis, 2 * axis + 1
31 F[p, v] = DT
32 H[axis, p] = 1.0
33 Q[p, p], Q[p, v], Q[v, p], Q[v, v] = DT**4 / 4, DT**3 / 2, DT**3 / 2, DT**2
34 return KalmanModel("gps", fs=1 / DT, F=F, H=H, Q=Q, R=25.0 * np.eye(3),
35 x0=np.array([[100.0], [30.0], [50.0], [-10.0], [200.0], [5.0]]), P0=25.0 * np.eye(6))

38def differences(sensor, x, step=1e-6):
39 """``dg/dx`` by central differences."""
40 columns = []
41 for j in range(x.size):
42 e = np.zeros(x.size)
43 e[j] = step
44 columns.append((np.asarray(sensor.g(x + e)) - np.asarray(sensor.g(x - e))) / (2 * step))
45 return np.array(columns).T

69 def test_doppler(self):
70 sensor = DopplerSensor("doppler", 6, P, V, sigma=0.5)
71 self.assertAlmostEqual(sensor.g(X)[0], 1.0 / 13.0, places=15) # (3 - 8 + 6) / 13
72 np.testing.assert_allclose(sensor.jacobian(X), differences(sensor, X), atol=1e-8)
73 below = DopplerSensor("below", 6, P, V, 0.5, at=[3.0, 4.0, 0.0])
74 self.assertEqual(below.settings, {"x": 3.0, "y": 4.0, "z": 0.0})
75 self.assertAlmostEqual(below.g(X)[0], 0.5, places=15)
76 on = X.copy()
77 on[4] = 0.0
78 self.assertEqual(below.g(on)[0], 0.0, "at the sensor itself the reading is 0")
79 with self.assertRaises(KeyError):
80 below.settings["focal"] = 1.0
81 self.assertFalse(below.jacobian(on).any())
82 below.settings.update(x=0.0, y=0.0)
83 self.assertAlmostEqual(below.g(X)[0], 1.0 / 13.0, places=15, msg="a setting changed moves the sensor")
84 flat = DopplerSensor("flat", 6, [0, 2], [1, 3], 0.5)
85 self.assertAlmostEqual(flat.g(X)[0], -1.0, places=15) # (3 - 8) / 5
86 self.assertEqual(list(flat.settings), ["x", "y"])

144 def test_nothing_is_called_back_during_a_pass(self):
145 doppler, camera = self.sensors()
146 calls = [0]
147 for sensor in (doppler, camera):
148 g = sensor.g
149
150 def counted(x, g=g):
151 calls[0] += 1
152 return g(x)
153 sensor.g = counted
154 model = target().add_sensor(doppler).add_sensor(camera)
155 data = model.sample(5.0, seed=3)
156 calls[0] = 0
157 model.filter(data.zk, reset=True)
158 self.assertEqual(calls[0], 0)
