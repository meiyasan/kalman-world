1"""Rocket liftoff — the first seconds off the pad."""
2from __future__ import annotations
3"""
4Rocket science — getting off the pad and back: **takeoff** and **landing**.
5
6This module is the *near-surface, powered-flight* half of the rocketry models (no third
7bodies). The space-travel half — flying to the Moon and interplanetary transfers — lives in
8:mod:`kalman.apps.space`.
9
10The "rocket equation" family (powered flight where what differs is the **phase**):
11
12* :class:`RocketAscentModel` — a planar **guided ascent**: gravity versus **thrust supplied
13 as the control input** ``u`` (constant mass). Demonstrates control channels (``l>0``).
14 *Scaled units.* Legacy alias: ``RocketLaunchModel``.
15* :class:`RocketLiftoffModel` — a 1-D **vertical liftoff** whose thrust is internal and whose
16 **mass burns down** (Tsiolkovsky): acceleration rises until **burnout**. *SI units.*
17 Legacy alias: ``RocketTakeoffModel``.
18* :class:`RocketLandingModel` — the same 1-D mass-burning dynamics run in reverse: a
19 SpaceX-style **hoverslam**, free-falling then firing a closed-loop landing burn to touch
20 down near zero velocity. *SI units.*
21
22All are :class:`ExtendedKalmanModel` subclasses whose ``f()`` advances the state by one
234th-order Runge–Kutta step; the EKF linearises ``f`` with a numerical Jacobian.
24
25The ascent model uses *scaled units* (``length = 1 Mm = 1000 km``, ``time = 1 ks = 1000 s``)
26to stay well-conditioned; the near-surface liftoff/landing models use plain **SI** (m, s, kg).
27Physical constants come from :mod:`kalman.runtime.astro` (astropy-sourced where available).
28"""

59class RocketLiftoffModel(NativeTwin, ExtendedKalmanModel):
60 """
61 Vertical **liftoff** from the launch pad with a finite propellant load (1-D).
62
63 The point a liftoff makes that a fixed-mass model cannot: the vehicle **burns mass**,
64 so even at constant thrust the acceleration *grows* as the rocket lightens, then turns
65 negative at **burnout** — Tsiolkovsky's rocket equation in flight. Mass is therefore a
66 state, not a parameter::
67
68 ḣ = v
69 v̇ = T(m)/m − g − ½ ρ(h) (C_d A) v|v| / m
70 ṁ = −ṁ_fuel (while propellant remains; 0 after burnout)
71
72 with thrust ``T = v_e · ṁ_fuel`` while ``m > m_dry`` (then the engine cuts off and the
73 rocket coasts). ``ρ(h) = ρ₀ e^{−h/H}`` is the exponential atmosphere. A ground radar /
74 altimeter measures **altitude only** (``z = h + noise``); the EKF reconstructs the
75 unmeasured climb rate and the remaining mass. ``n = 3``, ``m = 1``, ``l = 0``.
76
77 Contrast :class:`RocketAscentModel` (2-D guided ascent, thrust as control, constant
78 mass) and :class:`RocketLandingModel` (the same 1-D mass-burning dynamics run in
79 reverse to touch down). Importable as the legacy alias ``RocketTakeoffModel``.
80 """

89 def __init__(self, name="LIFTOFF", *,
90 m0=1000.0, # wet (fuelled) mass at liftoff [kg]
91 m_dry=400.0, # dry/structural mass [kg] (burnout mass)
92 ve=2500.0, # effective exhaust velocity [m/s] (Isp·g₀)
93 mdot=8.0, # propellant mass-flow rate [kg/s]
94 drag_area=0.25, # C_d·A [m²]
95 drag=True, # include atmospheric drag
96 g=G0, # gravity [m/s²] (near-surface constant)
97 length=40.0, width=3.7, # body size [m] (for the drawn rocket)
98 fs=10.0, # radar/altimeter sample rate [Hz]
99 track_m=20.0, # altitude measurement std [m]
100 q_alt=1e-3, q_vel=1e-2, q_mass=1e-4, # process-noise variances (model slack)
101 **kw):
102 self.dtypedtype = np.float64
103 self.m0 = float(m0)
104 self.m_dry = float(m_dry)
105 self.ve = float(ve)
106 self.mdot = float(mdot)
107 self.drag_area = float(drag_area)
108 self.drag = bool(drag)
109 self.g = float(g)
110 self.length, self.width = float(length), float(width)
111
112 self._dt = 1.0 / fs
113 self.nnnnn, self.mmmm, self.lll = 3, 1, 0
114 self.fsfsfsfsfs = fs
115
116 Q = np.diag([q_alt, q_vel, q_mass]).astype(float)
117 R = np.array([[track_m ** 2]], dtype=float)
118 # L = I: Q enters additively on every state. f() does not route an
119 # additive `w` through, so without it the engine would difference f with
120 # respect to w, find no sensitivity, and multiply Q by a zero matrix at
121 # every prediction: the covariance then collapses to zero.
122 # Start fuelled. The base default is a state of zeros, which for this
123 # model means no mass at all: the engine would cut off in the first step.
124 if kw.get("x0") is None:
125 x0 = np.zeros((3, 1))
126 x0[2, 0] = self.m0
127 kw["x0"] = x0
128 super().__init__(name, Q=Q, R=R, L=np.eye(self.nnnnn), fs=fs, **kw)

131 @property
132 def thrust(self):
133 """Engine thrust while burning [N] = exhaust velocity × mass-flow."""
134 return self.ve * self.mdot

136 @property
137 def burn_time(self):
138 """Propellant burn duration [s] = propellant mass / mass-flow."""
139 return (self.m0 - self.m_dry) / self.mdot

141 def twr(self, mass=None):
142 """Thrust-to-weight ratio at a given mass (defaults to liftoff mass)."""
143 mass = self.m0 if mass is None else mass
144 return self.thrust / (mass * self.g)

148 h, v, m = np.asarray(x, float).reshape(-1)[:3]

163 # --- dynamics ----------------------------------------------------------
164 def _deriv(self, s, t):
165 h, v, m = s
166 burning = m > self.m_dry
167 mdot = self.mdot if burning else 0.0
168 thrust = self.ve * mdot
169 m_eff = max(m, self.m_dry) # keep the division safe and physical
170 a = thrust / m_eff - self.g
171 if self.drag:
172 rho = RHO0 * np.exp(-max(h, 0.0) / H_ATM)
173 a -= 0.5 * rho * self.drag_area * v * abs(v) / m_eff
174 return np.array([v, a, -mdot])
175
176 DYNAMICS = KalmanDynamicsKind.CONTINUOUS # f is ds/dt; the framework integrates it (RK4)
177
178 def f(self, x, u=None, w=None, t=None):
179 return self._deriv(np.asarray(x, float).reshape(-1), float(t or 0.0)).reshape(np.shape(x))
180
181 def after_step(self, x, u=None, t=None):
182 s = np.asarray(x, float).reshape(-1)
183 s[2] = max(s[2], self.m_dry) # mass never drops below the dry mass
184 return s.reshape(-1, 1)
185
186 def h(self, x, v=None, t=None):
187 z = np.asarray(x, float).reshape(-1)[:1].reshape(1, 1)
188 return z if v is None else z + np.asarray(v, float).reshape(1, 1)

190 # --- flight metrics from a record -------------------------------------
191 def metrics(self, record):
192 """Headline flight figures from a sampled/filtered ``record`` (states
193 ``[h, v, m]``): burnout time & velocity, apogee, peak acceleration and peak
194 dynamic pressure (max-Q)."""
195 t = np.asarray(record.time, float).reshape(-1)
196 x = np.asarray(record.xk)
197 h, v, m = x[:, 0], x[:, 1], x[:, 2]
198 i_bo = int(np.argmax(m <= self.m_dry + 1e-6))
199 if not (m <= self.m_dry + 1e-6).any():
200 i_bo = len(t) - 1
201 rho = RHO0 * np.exp(-np.clip(h, 0, None) / H_ATM)
202 q = 0.5 * rho * v ** 2
203 a = np.array([self.accel(xi) for xi in x])
204 return {
205 "burnout time [s]": float(t[i_bo]),
206 "burnout speed [m/s]": float(v[i_bo]),
207 "apogee [km]": float(h.max() / 1e3),
208 "max accel [g]": float(np.nanmax(a) / G0),
209 "max-Q [kPa]": float(q.max() / 1e3),
210 "liftoff TWR": float(self.twr()),
211 }

213 def _fuel_frac(self, m):
214 return (float(m) - self.m_dry) / max(self.m0 - self.m_dry, 1e-9)
