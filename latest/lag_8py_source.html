1"""
2``lag``: a model that carries its last ``L`` states along with the present one.
3
4The Python face of ``Kalman::Models::Lag`` (``include/Kalman/Models/Lag.h``), which
5does the work. The state is the inner model's, ``L + 1`` times over,
6
7 X_k = [ x_k ; x_{k-1} ; ... ; x_{k-L} ]
8
9each step computing the new present state and shifting the others down. Two
10things come of it, with no change to any filter:
11
12- a sensor with a ``delay`` reads the state as it was (``KalmanSensor(...,
13 delay=0.3)``): the update corrects that past state and, through the
14 covariance between the blocks, the present one;
15- block ``j`` of the estimate is the state ``j`` steps ago given everything
16 measured since: fixed-lag smoothing, as the filter runs.
17
18Three ways to the same model::
19
20 model.lag(3)
21 lag(model, 3)
22 Lag(model, 3)
23
24The sensors are added first, then the model is lagged. Nothing is computed here.
25"""

41 def __init__(self, model: KalmanModel, depth: int, **kwargs):
42 if not isinstance(model, KalmanModel):
43 raise TypeError(f"lag: the model must be a KalmanModel, got {type(model).__name__}.")
44 if isinstance(depth, bool) or int(depth) != depth or int(depth) < 1:

49 handle = lib.kalman_model_create_lag(inner, self.depth)

54 def block(self, steps: int = 0) -> slice:
55 """Where the state ``steps`` ago sits in this model's state:
56 ``record.xk[:, model.block(3)]`` is the state three steps before each row,
57 given the measurements up to that row."""
58 if not 0 <= int(steps) <= self.depth:
59 raise IndexError(f"lag: the model carries {self.depth} past states, not {steps}.")
60 return slice(int(steps) * self.n_inner, (int(steps) + 1) * self.n_inner)
