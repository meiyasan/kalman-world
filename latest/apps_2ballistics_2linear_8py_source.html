40 # Continuous-time generator: ż=v, v̇=a, ȧ=0 (constant acceleration) —
41 # see the class docstring's Motion Model / A matrix. Passed as A= (not
42 # F=): the base KalmanModel discretizes exactly via F=expm(A/fs).
43 A = np.array([[0.0, 1.0, 0.0],
44 [0.0, 0.0, 1.0],
45 [0.0, 0.0, 0.0]], dtype=self.dtypedtype)
46
47 H = np.array([[1.0, 0.0, 0.0]], dtype=self.dtypedtype)
48
49 # -------------------------------------------
50 # Process noise — continuous white-noise-ACCELERATION disturbance,
51 # designed at the v̇ equation (Lc picks out that row); passed as Qc=/Lc=
52 # so the base model discretizes it exactly (Van Loan) alongside A=.
53 # This is a CONSTANT-acceleration model, so there is deliberately NO
54 # jerk option: the acceleration STATE itself gets no direct disturbance
55 # (Lc's 3rd entry is 0), so it stays flat (e.g. -9.81) and the resulting
56 # Q's acceleration row/col comes out identically zero — the disturbance
57 # only smears velocity (∝dt) and position (∝dt³); Qc=0 ⇒ truth == ideal.
58 #
59 # process_uncertainty is the square root of that disturbance's spectral
60 # density, in m/s^1.5: after a time t the velocity has wandered by q·√t and
61 # the position by q·√(t³/3). Order of magnitude after a 20 s flight:
62 # 0.05 → ~3 m 0.5 → ~26 m (default) 2 → ~100 m
63 # -------------------------------------------
64 Lc = np.array([[0.0], [1.0], [0.0]], dtype=self.dtypedtype)
65 Qc = np.array([[self.dtypedtype(process_uncertainty) ** 2]], dtype=self.dtypedtype)
66
67 # measurement_uncertainty is the position-sensor noise std σ [m]; the
68 # filter needs the variance, so R = σ²·I (consistent with the EKF model
69 # and with the P0 position prior below, both of which square it).
70 r = self.dtypedtype(measurement_uncertainty)
71 R = (r ** 2) * np.eye(1, dtype=self.dtypedtype)
72
73 # Initial covariance P0 — a positive-definite diagonal built from the two
74 # Kalman noises only (measurement R and process Q); there is no separate
75 # acceleration knob:
76 # position : R (position is what we measure)
77 # velocity : (measurement_uncertainty/10)² (unobserved → looser)
78 # acceleration : process_uncertainty² (from the process noise)
79 # Tying the acceleration prior to the PROCESS noise (not the sensor noise)
80 # keeps it sensible for quiet and noisy sensors alike, and means a smaller
81 # process noise automatically pins the acceleration closer to its initial value.
82 if "P0" not in kwargs:
83 p_var = self.dtypedtype(measurement_uncertainty) ** 2
84 v_var = (self.dtypedtype(measurement_uncertainty) / 10.0) ** 2
85 a_var = self.dtypedtype(process_uncertainty) ** 2
86 kwargs["P0"] = np.diag([p_var, v_var, a_var]).astype(self.dtypedtype)

94 # ------------------------------------------------------------------
95 # Trajectory estimators (closed-form for constant acceleration)
96 #
97 # Both take a [position, speed, acceleration] state — e.g. one row of a
98 # filtered/smoothed track, ``record.xk[k]`` — and assume the linear model's
99 # own constant-acceleration dynamics z(t) = z + ż·t + ½·z̈·t² hold from there
100 # on. Pass the matching covariance ``record.Pk[k]`` to also get 1-σ error bars
101 # by linear (first-order) propagation σ² = Jᵀ·P·J. Times are relative to the
102 # epoch of ``state``; add ``record.time[k]`` for an absolute time.
103 # ------------------------------------------------------------------
104 def _zva(self, state, cov):
105 """Unpack a state into scalars (z, ż, z̈) and a 3×3 covariance (or None)."""
106 x = self.x0x0x0 if state is None else state
107 x = np.asarray(x, dtype=self.dtypedtype).reshape(-1)
108 if x.shape[0] != 3:
109 raise ValueError(f"apex/impact_time need a [position, speed, acceleration] "
110 f"state (n=3); got length {x.shape[0]}.")
111 P = None if cov is None else np.asarray(cov, dtype=self.dtypedtype).reshape(3, 3)
112 return float(x[0]), float(x[1]), float(x[2]), P

114 def apex(self, state=None, cov=None):
115 """Maximum height of the constant-acceleration path and the time to reach it.
116
117 The height z(t) = z + ż·t + ½·z̈·t² peaks where the velocity vanishes, at
118 ``t* = −ż / z̈`` (needs z̈ ≠ 0; t* is negative if the body is already past its
119 peak), giving ``z* = z − ż² / (2·z̈)``. With ``cov`` given, returns 1-σ bars too.
120
121 Returns an :class:`ApexEstimate` (time, height, time_std, height_std);
122 ``state`` defaults to the model's initial state (the nominal launch).
123 """
124 z, v, a, P = self._zva(state, cov)
125 if a == 0.0:
126 raise ValueError("Acceleration is zero — a constant-velocity path has no apex.")
127 t_star = -v / a
128 height = z - v * v / (2.0 * a)
129 t_std = h_std = None
130 if P is not None:
131 J_t = np.array([0.0, -1.0 / a, v / (a * a)], dtype=self.dtypedtype)
132 J_h = np.array([1.0, -v / a, v * v / (2.0 * a * a)], dtype=self.dtypedtype)
133 t_std = float(np.sqrt(max(J_t @ P @ J_t, 0.0)))
134 h_std = float(np.sqrt(max(J_h @ P @ J_h, 0.0)))
135 return ApexEstimate(t_star, height, t_std, h_std)

137 def impact_time(self, state=None, cov=None, level=0.0):
138 """Time at which the trajectory crosses ``level`` (default 0) coming down.
139
140 Solves ½·z̈·t² + ż·t + (z − level) = 0 and returns the future **descending**
141 root (the "hitting time"). With ``cov`` given, the 1-σ time uncertainty is
142 propagated via the implicit derivative dt/dx = −[1, t, ½t²] / v_impact.
143
144 Returns an :class:`ImpactEstimate` (time, speed-at-impact, time_std). Raises if
145 the path never reaches ``level`` (negative discriminant) or only in the past.
146 """
147 z, v, a, P = self._zva(state, cov)
148 c = z - level
149 tol = 1e-9
150 if a == 0.0: # constant-velocity fallback
151 if abs(v) < tol:
152 raise ValueError("Static state never crosses the level.")
153 roots = [-c / v]
154 else:
155 disc = v * v - 2.0 * a * c
156 if disc < 0.0:
157 raise ValueError(f"Trajectory never reaches level={level} (no real crossing).")
158 sq = float(np.sqrt(disc))
159 roots = [(-v + sq) / a, (-v - sq) / a]
160 # prefer a future, descending (velocity < 0) crossing; else any future root
161 descending = [t for t in roots if t > tol and (v + a * t) < 0.0]
162 future = descending or [t for t in roots if t > tol]
163 if not future:
164 raise ValueError("No positive-time crossing of the level.")
165 t = min(future)
166 v_imp = v + a * t
167 t_std = None
168 if P is not None and abs(v_imp) > tol:
169 J = -np.array([1.0, t, 0.5 * t * t], dtype=self.dtypedtype) / v_imp
170 t_std = float(np.sqrt(max(J @ P @ J, 0.0)))
171 return ImpactEstimate(t, v_imp, t_std)
