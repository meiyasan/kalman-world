1"""TransformerNN — single-head self-attention router over the NIS window.
2
3The NIS history (window × M) is a natural sequence for a Transformer: each
4of the ``window`` timesteps is a token, each model's NIS is a feature.
5Self-attention lets every position attend to every other position in one
6step — no fixed receptive field (unlike the CNN) and no sequential
7bottleneck (unlike the LSTM).
8
9Architecture: per-token linear embedding → TransformerEncoderLayer (one block,
10single head) → global average pool over tokens → linear softmax head.
11
12Plug into NeuralRouter::
13
14 from kalman.framework.routers import NeuralRouter
15 from kalman.framework.routers.nn import TransformerNN
16
17 router = NeuralRouter(model=TransformerNN(d_model=16), window=10)
18 ikf.fit(training_zk_list, labels_list)
19"""

21import numpy as np
22import torch
23import torch.nn as nn

25_DEVICE = ("mps" if torch.backends.mps.is_available() else
26 "cuda" if torch.cuda.is_available() else "cpu")

29class _Block(nn.Module):
30 def __init__(self, M: int, d: int, d_ff: int, d_out: int):
31 super().__init__()
32 self.embed = nn.Linear(M, d)
33 self.encoder = nn.TransformerEncoderLayer(
34 d_model=d, nhead=1, dim_feedforward=d_ff,
35 dropout=0.0, batch_first=True,
36 )
37 self.head = nn.Linear(d, d_out)
38
39 def forward(self, x: torch.Tensor) -> torch.Tensor: # (B, W, M)
40 e = self.embed(x) # (B, W, d)
41 o = self.encoder(e) # (B, W, d)
42 return self.head(o.mean(1)) # (B, d_out)

45class TransformerNN:
46 """Single-head self-attention block + FFN + dense softmax head.
47
48 Parameters
49 ----------
50 d_model : int
51 Embedding / attention dimension.
52 d_ff : int, optional
53 Feed-forward hidden width. Defaults to ``2 * d_model``.
54 """
55
56 def __init__(self, d_model: int = 16, d_ff: int | None = None):
57 self.d_model = int(d_model)
58 self.d_ff = int(d_ff) if d_ff is not None else 2 * self.d_model
59 self._M: int | None = None
60 self._W: int | None = None
61 self._block: _Block | None = None
62 self._opt: torch.optim.Optimizer | None = None
63 self.loss_history: list[float] = []
64
65 def build(self, d_in: int, d_out: int, seed=None) -> "TransformerNN":
66 if seed is not None:
67 torch.manual_seed(seed)
68 M = d_out
69 W = d_in // M
70 self._M, self._W = M, W
71 self._block = _Block(M, self.d_model, self.d_ff, d_out).to(_DEVICE)
72 self._opt = torch.optim.Adam(self._block.parameters())
73 self.loss_history = []
74 return self
75
76 def _to_seq(self, X: np.ndarray) -> torch.Tensor:
77 """(N, M*W) → (N, W, M) float32 tensor on device."""
78 return torch.tensor(
79 X.reshape(-1, self._W, self._M), dtype=torch.float32, device=_DEVICE
80 )
81
82 def fit(self, X: np.ndarray, y: np.ndarray,
83 epochs: int = 300, lr: float = 2e-3,
84 batch_size: int = 64, **_) -> "TransformerNN":
85 self._opt.param_groups[0]["lr"] = lr
86 Xt = self._to_seq(X)
87 yt = torch.tensor(y, dtype=torch.long, device=_DEVICE)
88 loss_fn = nn.CrossEntropyLoss()
89 N = len(Xt)
90 for _ in range(epochs):
91 idx = torch.randperm(N, device=_DEVICE)
92 epoch_loss, n_batches = 0.0, 0
93 for s in range(0, N, batch_size):
94 b = idx[s:s + batch_size]
95 self._opt.zero_grad()
96 loss = loss_fn(self._block(Xt[b]), yt[b])
97 loss.backward()
98 self._opt.step()
99 epoch_loss += float(loss.item()); n_batches += 1
100 self.loss_history.append(epoch_loss / max(n_batches, 1))
101 return self
102
103 def __call__(self, x: np.ndarray) -> np.ndarray:
104 with torch.no_grad():
105 xt = self._to_seq(x[None])
106 logits = self._block(xt).squeeze(0).cpu().numpy()
107 e = np.exp(logits - logits.max())
108 return e / e.sum()
