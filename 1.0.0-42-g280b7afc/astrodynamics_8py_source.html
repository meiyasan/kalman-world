1"""
2Astrodynamics: constants, gravity, ephemerides and propagation shared by the
3aerospace and astrophysics applications (``kalman.apps.astrophysics.astrodynamics``).
4
5**Physical constants** are sourced from :mod:`astropy.constants` / :mod:`astropy.units`
6wherever astropy ships them (Earth, Sun, Jupiter, the astronomical unit, *G*, standard
7gravity ``g0``). The Moon and the remaining planets — which astropy does *not* expose as
8named constants — use cited IAU 2015 / JPL DE440 GM and mean-radius values. Everything is
9then expressed in the framework's **scaled units** (``length = 1 Mm = 1e6 m``,
10``time = 1 ks = 1e3 s``) that keep the orbital/ascent models well-conditioned.
11
12This module also provides the shared services both app modules build on:
13
14* :func:`rk4` — the 4th-order Runge–Kutta step used by every integrator;
15* :func:`body_state` / :func:`planet_positions` — **exact** ephemeris position & velocity of
16 any solar-system body at a given time (astropy's network-free ``builtin`` ephemeris), with
17 an analytic circular-orbit fallback when astropy is unavailable;
18* :func:`launch_site` / :func:`launch_site_state` — launch-pad coordinates resolved from a
19 **library** (``EarthLocation.of_address`` online geocoding) and their Earth-rotation state.
20"""

63# ---------------------------------------------------------------------------
64# Scaled units (1 Mm, 1 ks) and conversions
65# ---------------------------------------------------------------------------
66LU, TU = 1.0e6, 1.0e3 # 1 Mm (1e6 m), 1 ks (1e3 s)
67KM = 1.0e3 / LU # 1 km in LU (= 1e-3)
68DAY = 86400.0 / TU # 1 day in TU (= 86.4)
69VU = LU / TU # 1 scaled velocity unit = 1 km/s
70_GM_SCALE = TU**2 / LU**3 # SI GM (m^3 s^-2) -> scaled

121def _as_time(time):
122 """Coerce ``time`` (``astropy.time.Time``, ISO string, ``datetime`` or ``None``) to Time."""
123 from astropy.time import Time

131def moon_position(t, phase=0.0):
132 """Analytic Moon position (scaled, geocentric) on a circular orbit — fallback only."""
133 a = OMEGA_MOON * t + phase
134 return D_MOON * np.array([np.cos(a), np.sin(a), 0.0])

143def body_state(name, time=None, *, origin="heliocentric"):
144 """Exact position **and** velocity of a solar-system body in *scaled units*.
145
146 Returns ``(r, v)`` with position in Mm and velocity in km/s (= scaled ``LU/TU``).
147 ``origin`` is ``'heliocentric'`` (default, Sun-centred), ``'bary'`` (solar-system
148 barycentre) or ``'geocentric'`` (Earth-centred — used for the trans-lunar frame).
149 Uses astropy's network-free ``builtin`` ephemeris; if astropy is unavailable, falls
150 back to the analytic circular orbits for the Sun/Moon.
151 """
152 try:
153 from astropy.coordinates import get_body_barycentric_posvel, solar_system_ephemeris
154 import astropy.units as u
155 except Exception: # no astropy -> analytic fallback
156 r = (moon_position(0.0) if name.lower() == "moon" else
157 sun_position(0.0) if name.lower() == "sun" else np.zeros(3))
158 return r, np.zeros(3)
159
160 t = _as_time(time)
161 with solar_system_ephemeris.set(EPHEMERIS):
162 r_b, v_b = get_body_barycentric_posvel(name, t)
163 if origin in ("heliocentric", "sun"):
164 r0, v0 = get_body_barycentric_posvel("sun", t)
165 r_b, v_b = r_b - r0, v_b - v0
166 elif origin in ("geocentric", "earth"):
167 r0, v0 = get_body_barycentric_posvel("earth", t)
168 r_b, v_b = r_b - r0, v_b - v0
169 r = r_b.xyz.to_value(u.m) / LU
170 v = v_b.xyz.to_value(u.m / u.s) / VU
171 return r, v

182def ephemeris_samples(bodies, times, *, origin="heliocentric"):
183 """**Vectorised** ephemeris over an array of times.
184
185 Returns ``{body: (R, V)}`` where ``R``/``V`` are ``(len(times), 3)`` arrays of scaled
186 position [Mm] and velocity [km/s]. astropy evaluates the whole time array at once, so this
187 is the fast path for building interpolators that an N-body integrator can sample cheaply.
188 """
189 from astropy.coordinates import get_body_barycentric_posvel, solar_system_ephemeris
190 import astropy.units as u
191 out = {}
192 with solar_system_ephemeris.set(EPHEMERIS):
193 r0 = v0 = None
194 if origin in ("heliocentric", "sun"):
195 r0, v0 = get_body_barycentric_posvel("sun", times)
196 elif origin in ("geocentric", "earth"):
197 r0, v0 = get_body_barycentric_posvel("earth", times)
198 for b in bodies:
199 rb, vb = get_body_barycentric_posvel(b, times)
200 if r0 is not None:
201 rb, vb = rb - r0, vb - v0
202 R = (rb.xyz.to_value(u.m) / LU).T # (N, 3)
203 V = (vb.xyz.to_value(u.m / u.s) / VU).T # (N, 3)
204 out[b] = (np.atleast_2d(R), np.atleast_2d(V))
205 return out

208def times_from_epoch(epoch, t_scaled):
209 """astropy ``Time`` array at ``epoch + t_scaled`` (``t_scaled`` in scaled TU = ks)."""
210 from astropy.time import TimeDelta
211 epoch = _as_time(epoch)

246def launch_site(name):
247 """Return an ``astropy.coordinates.EarthLocation`` for a launch site.
248
249 Coordinates are resolved **from a library**: ``EarthLocation.of_address`` (online
250 OpenStreetMap geocoding); any address string works. Results are cached. When offline,
251 a small built-in coordinate table covers the most common pads.
252 """
253 key = name.strip().lower()
254 if key in _site_cache:
255 return _site_cache[key]
256 from astropy.coordinates import EarthLocation
257 import astropy.units as u
258 try:
259 loc = EarthLocation.of_address(name)
260 except Exception:
261 if key not in _SITE_FALLBACK:
262 raise
263 lat, lon, h = _SITE_FALLBACK[key]
264 loc = EarthLocation(lat=lat * u.deg, lon=lon * u.deg, height=h * u.m)
265 _site_cache[key] = loc
266 return loc

269def launch_site_state(name, time=None):
270 """Scaled GCRS position & velocity of a launch site, **including Earth's rotation**.
271
272 Returns ``(r, v)`` with position in Mm and velocity in km/s. The velocity is the
273 inertial (GCRS) velocity of the rotating pad — ~0.46 km/s eastward at the equator —
274 via ``EarthLocation.get_gcrs_posvel`` (astropy uses online IERS Earth-orientation data
275 when available, else the bundled IERS-B table).
276 """
277 import astropy.units as u
278 loc = launch_site(name)
279 t = _as_time(time)
280 r_gcrs, v_gcrs = loc.get_gcrs_posvel(t)
281 r = r_gcrs.xyz.to_value(u.m) / LU
282 v = v_gcrs.xyz.to_value(u.m / u.s) / VU
283 return r, v
