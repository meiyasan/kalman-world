80 void Sensor::Check(std::size_t n) const
81 {
82 if (name.empty()) throw std::invalid_argument("Kalman::Sensor: a sensor needs a name.");
83 const std::string who = "Kalman::Sensor: the sensor '" + name + "' ";
84 const std::size_t k = Channels();
85 if (k == 0) throw std::invalid_argument(who + "has no channel: give its noise covariance R (k x k).");
86 if (R.Cols() != k) throw std::invalid_argument(who + "needs a square R.");
87 if (static_cast<bool>(g) == !G.Empty())
88 throw std::invalid_argument(who + "needs a function g or a matrix G, one of them.");
89 if (!g && (G.Rows() != k || G.Cols() != n))
90 throw std::invalid_argument(who + "has a G of " + std::to_string(G.Rows()) + " x " + std::to_string(G.Cols())
91 + ", expected " + std::to_string(k) + " x " + std::to_string(n) + ".");
92 if (!channels.empty() && channels.size() != k)
93 throw std::invalid_argument(who + "has " + std::to_string(k) + " channels and "
94 + std::to_string(channels.size()) + " names.");
95 if (!(rate >= 0.)) throw std::invalid_argument(who + "has a negative rate.");
96 if (!(delay >= 0.)) throw std::invalid_argument(who + "has a negative delay.");
97 for (const auto &c : channels)
98 if (c.name.empty()) throw std::invalid_argument(who + "has a channel with no name.");
99 }
