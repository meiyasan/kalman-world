1"""Structural invariants of the framework/apps layout.
2
3These guard the architecture itself rather than any single model:
4
5- every abstract contract + base + registry lives in ``framework/``;
6- every app domain is a uniform *package* (variant submodules + an ``__init__``
7 that re-exports its public names and a ``VARIANTS`` registry);
8- the plug-in registries (engine / integrator / solver / provider) cover their
9 full kind enum and resolve by name;
10- the public ``kalman.apps`` surface stays importable after reorganisation;
11- ``apps/`` holds packages only (no flat domain modules).

36def _domain_packages():
37 """(import_path, name) for every leaf domain package, families expanded."""
38 out = []
39 for m in pkgutil.iter_modules([APPS_DIR]):
40 if not m.ispkg:
41 continue
42 if m.name in FAMILIES:
43 fam_dir = os.path.join(APPS_DIR, m.name)
44 for sub in pkgutil.iter_modules([fam_dir]):
45 if sub.ispkg:
46 out.append((f"{m.name}.{sub.name}", sub.name))
47 else:
48 out.append((m.name, m.name))
49 return sorted(out)

67class TestReservedNames(unittest.TestCase):
68 """The names of the model contract mean one thing each.
69
70 An operation gives back a model made from this one; an engine hook is a
71 method an engine looks up on the model. A name used for both, or redefined by
72 an application with another meaning, is found by the wrong caller without any
73 error (an ``observe`` operation once hijacked the ensemble engine's hook).
74 """
75
76 #: Methods of KalmanModel that return a model made from this one.
77 OPERATIONS = ("joint", "add_sensor", "add_instrument", "lag", "compose", "route")
78 #: Reserved for the operations still to come: nothing may take these names.
79 PLANNED = ()
80 #: Aliases of f and h, provided by the base class; a model never defines them.
81 ALIASES = ("step", "sense")
82 #: Looked up on a model by the ensemble engine only.
83 ENSEMBLE_HOOKS = ("covariance_mask",)
84 #: Ensemble-filter words that are not model methods: the engine's own steps.
85 NOT_ON_A_MODEL = ("forecast", "observe", "project", "analyze", "initial_ensemble", "localization")
86
87 @staticmethod
88 def _models():
89 for _, name, _ in pkgutil.walk_packages(apps.__path__, apps.__name__ + "."):
90 try:
91 importlib.import_module(name)
92 except Exception:
93 continue
94 seen, stack = [], [KalmanModel]
95 while stack:
96 for sub in stack.pop().__subclasses__():
97 if sub not in seen:
98 seen.append(sub)
99 stack.append(sub)
100 return seen

106 def test_f_and_h_take_one_state_or_many(self):

115 np.testing.assert_array_equal(model.sense(X), model.H() @ X)

286class TestCxxHeadersHaveOneLayout(unittest.TestCase):

299 def test_every_directory_has_its_umbrella(self):
300 import re
301 repo = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
302 where = {} # "Kalman/X/Name.h" -> file
303 for root in self.ROOTS:
304 base = os.path.join(repo, root)
305 for folder, _, files in os.walk(os.path.join(base, "Kalman")):
306 for name in files:
307 if name.endswith(".h"):
308 path = os.path.join(folder, name)
309 where[os.path.relpath(path, base)] = path
310 folders = {os.path.dirname(header) for header in where} - {"Kalman"}
311 self.assertGreater(len(folders), 15)
312 for folder in sorted(folders):
313 umbrella = folder + ".h"
314 self.assertIn(umbrella, where, f"{folder}/ needs the header {umbrella} beside it")
315 included = set(re.findall(r'#include "([^"]+)"', open(where[umbrella]).read()))
316 for header in sorted(where):
317 if os.path.dirname(header) == folder and header not in self.LEFT_OUT:
318 self.assertIn(header, included, f"{umbrella} must include {header}")
319 self.assertNotIn("Kalman/Apps/Apps.h", where)
