21 Lag::Lag(std::shared_ptr<Model> inner, std::size_t depth) : fInner(std::move(inner)), fDepth(depth)
22 {
23 if (!fInner) throw std::invalid_argument("Models::Lag: the inner model is null.");
24 if (fDepth == 0) throw std::invalid_argument("Models::Lag: carry at least one past state (depth > 0).");
25 fNInner = fInner->NState();
26 SetName(fInner->Name());
27 Allocate(fNInner * (fDepth + 1), fInner->NMeasure(), fInner->NControl());
28
29 SetSampleRate(fInner->SampleRate());
30 SetT0(fInner->T0());
31 SetTimeUnit(fInner->TimeUnit());
32 SetDefaultEngine(fInner->DefaultEngine());
33 SetBuiltInSensorName(fInner->BuiltInSensorName());
34 // Still what the inner model is: its matrices in a larger frame, or code.
35 if (fInner->HasAnalyticJacobian()) SetAnalyticJacobian(true);
36 else SetDifferentiator(fInner->Differentiator(), fInner->Epsilon());
37 SetStructure(fInner->Structure());
38
39 AdoptInitialState();
40
41 // The present state keeps its names; the state j steps ago carries "@-j".
42 const std::vector<Component> inner_components = fInner->StateComponents();
43 std::vector<Component> components;
44 for (std::size_t j = 0; j <= fDepth; ++j)
45 for (std::size_t i = 0; i < fNInner; ++i)
46 {
47 Component c = i < inner_components.size() ? inner_components[i] : Component{"x" + std::to_string(i), ""};
48 if (j > 0) c.name += "@-" + std::to_string(j);
49 components.push_back(std::move(c));
50 }
51 DeclareState(std::move(components));
52 DeclareMeasurement(fInner->MeasurementComponents());
53
54 Sync();
55 }

57 void Lag::AdoptInitialState()
58 {
59 const Vector &s0 = fInner->X0();
60 const Matrix &P0 = fInner->P0();
61 const std::size_t n = fNInner;
62 Vector x0(fNState, 0.);
63 Matrix Px(fNState, fNState, 0.);
64 for (std::size_t j = 0; j <= fDepth; ++j)
65 for (std::size_t i = 0; i < n; ++i)
66 {
67 x0(j * n + i) = s0.Size() == n ? s0(i) : 0.;
68 for (std::size_t c = 0; c < n; ++c)
69 Px(j * n + i, j * n + c) = (P0.Rows() == n && P0.Cols() == n) ? P0(i, c) : (i == c ? 1. : 0.);
70 }
71 SetInitialState(x0);
72 SetInitialCovariance(Px);
73 }

75 void Lag::Sync() const
76 {
77 if (fInner->SampleRate() != fSampleRate) fInner->SetSampleRate(fSampleRate);
78 if (fDelaysAt == fSampleRate) return;
79
80 // The block each row of z reads: its sensor's delay, in steps of the model.
81 std::vector<std::size_t> rows;
82 for (const Sensor &sensor : fInner->Sensors())
83 {
84 const double steps = sensor.delay * fSampleRate;
85 const std::size_t block = static_cast<std::size_t>(std::llround(steps));
86 if (!(sensor.delay >= 0.) || block > fDepth)
87 throw std::invalid_argument("Models::Lag: the sensor '" + sensor.name + "' is late by "
88 + std::to_string(steps) + " steps and the model carries "
89 + std::to_string(fDepth) + ".");
90 rows.insert(rows.end(), sensor.Channels(), block);
91 }
92 if (rows.size() != fNMeasure)
93 throw std::runtime_error("Models::Lag: the sensors of the inner model have " + std::to_string(rows.size())
94 + " channels and its measurement " + std::to_string(fNMeasure) + ".");
95 fRowBlock = std::move(rows);
96 fBlocksRead = fRowBlock;
97 std::sort(fBlocksRead.begin(), fBlocksRead.end());
98 fBlocksRead.erase(std::unique(fBlocksRead.begin(), fBlocksRead.end()), fBlocksRead.end());
99 fDelayed = fBlocksRead.back() != 0;
100 fDelaysAt = fSampleRate;
101 }

103 Vector Lag::Block(const Vector &x, std::size_t steps) const
104 {
105 if (x.Size() != fNState)
106 throw std::invalid_argument("Models::Lag: the state has " + std::to_string(x.Size())
107 + " components, " + std::to_string(fNState) + " expected.");
108 if (steps > fDepth)
109 throw std::out_of_range("Models::Lag: the model carries " + std::to_string(fDepth) + " past states, not "
110 + std::to_string(steps) + ".");
111 Vector s(fNInner);
112 for (std::size_t i = 0; i < fNInner; ++i) s(i) = x(steps * fNInner + i);
113 return s;
114 }

138 Vector Lag::f(const Vector &x, const Vector &u, const Vector &w, double t) const
139 {
140 if (w.Size() != fNState)
141 throw std::invalid_argument("Models::Lag: the process noise has " + std::to_string(w.Size())
142 + " components, " + std::to_string(fNState) + " expected.");
143 Sync();
144 Vector ws(fNInner);
145 for (std::size_t i = 0; i < fNInner; ++i) ws(i) = w(i);
146 Vector out = Shift(x, fInner->f(Block(x, 0), u, ws, t));

173 Vector Lag::h(const Vector &x, double t) const
174 {
175 Sync();
176 if (!fDelayed) return fInner->h(Block(x, 0), t);
177 // One reading of the inner model per block that a sensor reads; each row
178 // is taken from the reading of its own block, at its own time.
179 Vector z(fNMeasure);
180 for (const std::size_t block : fBlocksRead)
181 {
182 const Vector zb = fInner->h(Block(x, block), t - static_cast<double>(block) * DeltaT());
183 for (std::size_t i = 0; i < fNMeasure; ++i)
184 if (fRowBlock[i] == block) z(i) = zb(i);
185 }
186 return z;

262 std::vector<Sensor> Lag::Sensors() const
263 {
264 Sync();
265 std::vector<Sensor> out = fInner->Sensors();
266 std::size_t row = 0;
267 for (Sensor &sensor : out)
268 {
269 const std::size_t block = fRowBlock[row], n = fNInner, k = sensor.Channels();
270 const double late = static_cast<double>(block) * DeltaT();
271 row += k;
272 sensor.delay = 0.; // dealt with: on this model's state it reads on time
273 if (sensor.IsLinear())
274 {
275 Matrix G(k, fNState, 0.);
276 for (std::size_t i = 0; i < k; ++i)
277 for (std::size_t j = 0; j < n; ++j) G(i, block * n + j) = sensor.G(i, j);
278 sensor.G = G;
279 continue;
280 }
281 const Sensor::Measure g = sensor.g;
282 sensor.g = [this, g, block, late](const Vector &x, double t) { return g(Block(x, block), t - late); };
283 if (sensor.jacobian)
284 {
285 const Sensor::MeasureJacobian dg = sensor.jacobian;
286 sensor.jacobian = [this, dg, block, late, k, n](const Vector &x, double t) {
287 const Matrix Gb = dg(Block(x, block), t - late);
288 Matrix G(k, fNState, 0.);
289 for (std::size_t i = 0; i < k; ++i)
290 for (std::size_t j = 0; j < n; ++j) G(i, block * n + j) = Gb(i, j);
291 return G;
292 };
293 }
294 }
295 return out;
296 }
