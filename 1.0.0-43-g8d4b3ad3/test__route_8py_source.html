25def _two_models(q_slow=0.1, q_fast=2.0):
26 """Two 1-D random-walk models with very different process noise."""
27 def _m(name, q):
28 return KalmanModel(name,
29 F=np.eye(1), H=np.eye(1),
30 Q=np.array([[q ** 2]]),
31 R=np.array([[1.0 ** 2]]),
32 fs=1.0,
33 x0=np.array([[0.0]]))
34 return _m("slow", q_slow), _m("fast", q_fast)

37def _sim(regime_seq, q_slow=0.1, q_fast=2.0, obs_std=1.0, seed=0):
38 """Simulate a 1-D switching process and noisy observations."""
39 rng = np.random.default_rng(seed)
40 x = 0.0
41 zk, labels = [], []
42 for re in regime_seq:
43 x += rng.normal(0, q_slow if re == 0 else q_fast)
44 zk.append([x + rng.normal(0, obs_std)])
45 labels.append(re)
46 return np.array(zk), np.array(labels, dtype=int)
47

150 def test_mixing_starts_every_mode_from_the_beliefs_before_the_step(self):
151 # One step by hand, for two random walks: mode j starts from the blend
152 # of both beliefs as they were, with weights Pi_ij mu_i / c_j.
153 m0, m1 = _two_models()
154 z = np.array([[0.7]])
155 model = route([m0, m1], PI)
156 model.filter(z)
157 mu, c = np.array([0.5, 0.5]), PI.T @ np.array([0.5, 0.5])
158 P = [float(m0.P0[0, 0]), float(m1.P0[0, 0])]
159 post, like = [], []
160 for j, q in enumerate((0.1 ** 2, 2.0 ** 2)):
161 w = PI[:, j] * mu / c[j]
162 S = w @ P + q + 1.0
163 like.append(np.exp(-0.5 * 0.7 ** 2 / S) / np.sqrt(2 * np.pi * S))
164 post.append((w @ P + q) / S * 0.7)
165 want = c * np.array(like)
166 want /= want.sum()
167 np.testing.assert_allclose(model.mode_probabilities[0], want, rtol=1e-12)
168 np.testing.assert_allclose(float(model._filter.x[0, 0]), want @ np.array(post), rtol=1e-12)

229class TestRoutingAccuracy(unittest.TestCase):
230 """Routing accuracy on the 2-model Gaussian benchmark."""
231
232 def _run(self, router, epochs=200, n_train=3, seed=42):
233 m0, m1 = _two_models()
234 ikf = route([m0, m1], PI, router=router)
235
236 train_zk, train_lbl = [], []
237 for s in range(n_train):
238 zk, lbl = _sim(REGIME_200, seed=s)
239 train_zk.append(zk); train_lbl.append(lbl)
240
241 ikf.fit(train_zk, train_lbl, epochs=epochs, seed=seed)
242
243 zk_test, lbl_test = _sim(REGIME_200_TEST, seed=99)
244 ikf.filter(zk_test)
245 mu = ikf.mode_probabilities
246 acc = float((np.argmax(mu, axis=1) == lbl_test).mean())
247 return acc
248
249 def test_bayesian_accuracy_high(self):
