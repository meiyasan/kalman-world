1"""
2``joint``: a model whose state also carries some of another model's parameters.
3
4The Python face of ``Kalman::Models::Joint`` (``include/Kalman/Models/Joint.h``),
5which does the work: it appends the chosen parameters to the inner model's state,
6as constants that may drift a little,
7
8 x = [ s ; theta ] s: the inner state (n), theta: p parameters
9
10 s_{k+1} = f_inner(s_k, u_k, t ; theta_k)
11 theta_{k+1} = theta_k + w_theta, w_theta ~ N(0, diag(drift**2))
12 z_k = h_inner(s_k, t ; theta_k)
13
14sets the inner parameters before every evaluation and puts them back after, and
15assembles the Jacobian by blocks (the inner model's own for the motion, the
16parameter columns differenced). See the header for the details. Nothing is
17computed here.
18
19Three ways to the same model::
20
21 model.joint({"c_drag": (1e-3, 1e-6)})
22 joint(model, {"c_drag": (1e-3, 1e-6)})
23 Joint(model, {"c_drag": (1e-3, 1e-6)})
24"""

37def _spec(name, value):
38 """``(sigma, drift, tau)`` from a tuple or a mapping. The values are checked in C++."""
39 if isinstance(value, Mapping):
40 unknown = set(value) - {"sigma", "drift", "tau"}
41 if unknown:
42 raise ValueError(f"joint: {name!r} takes sigma=, drift= and tau=, not {sorted(unknown)}.")
43 if "sigma" not in value:
44 raise ValueError(f"joint: {name!r} needs sigma=.")
45 return float(value["sigma"]), float(value.get("drift", 0.0)), float(value.get("tau", 0.0))
46 try:
47 sigma, drift, *rest = value
48 (tau,) = rest or (0.0,)
49 except (TypeError, ValueError):
50 raise ValueError(f"joint: {name!r} needs (sigma, drift), (sigma, drift, tau) or "
51 f"{{'sigma': ..., 'drift': ..., 'tau': ...}}, got {value!r}.") from None
52 return float(sigma), float(drift), float(tau)

55class Joint(OperationModel):
56 """Estimate parameters of ``model`` along with its state.
57
58 ``Joint(model, {"c_drag": (1e-3, 1e-6)})`` appends the inner model's
59 ``c_drag`` to its state, with an initial standard deviation of ``1e-3`` and a
60 random walk of ``1e-6`` per step, starting from the value the inner model
61 holds. A parameter is one the model keeps as metadata (``model["c_drag"]``)
62 or one its C++ counterpart declares.
63
64 ``initial_state`` takes either the joint state (``n + p`` numbers) or one the
65 inner model takes; the latter is handed to the inner model, and the
66 parameters start again from the values it holds.
67 """

70 DESCRIPTION = "A model whose state also carries some of its parameters, estimated with the motion."
71 VERSION = "1.0"

115 def _warn_if_unobservable(self):
116 """Warn when the measurements cannot determine an estimated parameter from
117 this initial state: it would stay on its prior, or move only together with
118 the components named. Checked when a pass starts, where the initial state
119 is the real one (a system at rest can hide a parameter motion reveals)."""
120 import warnings
121 key = np.asarray(self.x0x0x0x0, dtype=np.float64).tobytes() + np.asarray(self.P0P0, dtype=np.float64).tobytes()
122 if self.__dict__.get("_observability_checked") == key:
123 return
124 self._observability_checked = key
125 try:

127 except Exception:

130 labels = list(weak)
131 for name in self.names:
132 if weak.get(name, 0.0) > self.OBSERVABILITY_SHARE:
133 others = [label for label in labels if label != name and weak[label] > self.OBSERVABILITY_SHARE]
134 together = f" apart from {', '.join(others)}" if others else ""
135 warnings.warn(f"joint: the measurements do not determine {name!r}{together} from this initial state "
