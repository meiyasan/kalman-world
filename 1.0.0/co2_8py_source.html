1"""CO2 in a room: a mass balance on a grid of cells.
2
3The room is a set of cells of a rectangular grid (an L-shaped room is the grid
4without one quarter). In each cell the CO2 concentration ``c`` [ppm] follows
5
6 dc/dt = D lap(c) - lambda (c - c_out) + S / V
7
8- ``D lap(c)``: mixing between neighbouring cells of the room (no flow through
9 the walls: a cell exchanges only with its neighbours inside the room);
10- ``lambda (c - c_out)``: ventilation, ``lambda`` the air changes per second,
11 bringing outdoor air at ``c_out``;
12- ``S / V``: the CO2 the occupants breathe out, spread over the seats.
13
14The equation is linear, so the model is a model of matrices: ``F`` and ``B``
15are its exact discretization over one step (matrix exponential), and it runs on
16the linear engine. The known inputs ``u = [occupants, 1]`` are the number of
17people in the room and a constant 1 that carries the outdoor air.
18
19With ``count_people=True`` the number of occupants is not given but estimated:
20it becomes the last component of the state, a number that drifts slowly
21(people come and go), and ``u = [1]``. The model stays linear.
22
23The model's own sensor reads one cell. Other sensors are added with the
24framework's operations; a sensor that drifts is an instrument with a
25calibration, whose offset ``joint`` estimates with the field.
26"""

36def l_shaped_room(rows: int = 12, cols: int = 12) -> np.ndarray:
37 """The cells of an L-shaped room: the grid without its upper-right quarter."""
38 room = np.ones((rows, cols), dtype=bool)
39 room[rows // 2:, cols // 2:] = False
40 return room

43class RoomCO2Model(LinearKalmanModel):
44 """CO2 concentration in every cell of a room.
45
46 Parameters
47 ----------
48 sensor : the name of the model's own CO2 sensor.
49 at : ``(row, column)`` of the cell that sensor reads.
50 room : boolean grid, True for the cells of the room; an L by default.
51 cell : the side of a cell [m]; height : the height of the room [m].
52 mixing : the mixing coefficient D [m^2/s].
53 ach : ventilation, in air changes per hour.
54 outdoor : the outdoor concentration [ppm].
55 seats : the cells where the occupants sit; their breath is spread over them.
56 emission : the CO2 one person breathes out [L/s].
57 sigma : the noise of the model's own sensor [ppm].
58 process_uncertainty : what the balance leaves out (a door opened, a person
59 moving), as a standard deviation per cell and per step [ppm].
60 count_people : estimate the number of occupants as the last state component
61 instead of taking it as an input.
62 people_drift : how much that number may change in a step, a standard
63 deviation [people]; used with ``count_people``.
64 fs : steps per second; one step a minute by default.
65 """

71 def __init__(self, sensor: str = "co2", at=(2, 2), *, room=None, cell: float = 0.5, height: float = 2.5,
72 mixing: float = 0.02, ach: float = 2.0, outdoor: float = 420.0, seats=None,
73 emission: float = 0.005, sigma: float = 15.0, process_uncertainty: float = 2.0,
74 count_people: bool = False, people_drift: float = 0.3, fs: float = 1 / 60, **kwargs):
75 self.room = l_shaped_room() if room is None else np.asarray(room, dtype=bool)
76 rows, cols = self.room.shape
77 self.cells = [(r, c) for r in range(rows) for c in range(cols) if self.room[r, c]]
78 index = {rc: i for i, rc in enumerate(self.cells)}
79 self.index = index
80 cells_n = len(self.cells)
81 self.count_people = bool(count_people)
82 n = cells_n + (1 if self.count_people else 0)
83 self.cell, self.height, self.outdoor = float(cell), float(height), float(outdoor)
84 seats = [(1, 1), (1, 2), (2, 1), (2, 2)] if seats is None else [tuple(s) for s in seats]
85 for s in [tuple(at), *seats]:
86 if s not in index:
87 raise ValueError(f"RoomCO2Model: the cell {s} is not in the room.")
88 self.seats = seats
89
90 # The continuous balance dc/dt = A c + Bc u, u = [occupants, 1].
91 lam = float(ach) / 3600.0
92 A = np.zeros((n, n))
93 A[:cells_n, :cells_n] = -lam * np.eye(cells_n)
94 k = float(mixing) / self.cell ** 2
95 for (r, c), i in index.items():
96 for nb in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
97 j = index.get(nb)
98 if j is not None: # a wall exchanges nothing
99 A[i, i] -= k
100 A[i, j] += k
101 volume = self.cell * self.cell * self.height # of one cell [m^3]
102 per_person = float(emission) * 1e-3 / volume * 1e6 # ppm/s in one cell
103 breath = np.zeros(n)
104 for s in seats:
105 breath[index[s]] += per_person / len(seats)
106 outdoor_air = np.zeros(n)
107 outdoor_air[:cells_n] = lam * self.outdoor
108 if self.count_people:
109 A[:, cells_n] = breath # the occupants are the last state
110 Bc = outdoor_air.reshape(n, 1)
111 else:
112 Bc = np.column_stack([breath, outdoor_air])
113
114 # Exact discretization over one step, the inputs held: exp([[A, Bc], [0, 0]] dt).
115 dt = 1.0 / float(fs)
116 l = Bc.shape[1]
117 block = np.zeros((n + l, n + l))
118 block[:n, :n], block[:n, n:] = A, Bc
119 E = expm(block * dt)
120 F, B = E[:n, :n], E[:n, n:]
121 H = np.zeros((1, n))
122 H[0, index[tuple(at)]] = 1.0
123 q = np.full(n, float(process_uncertainty) ** 2)
124 x0 = np.full((n, 1), self.outdoor)
125 P0 = np.full(n, 50.0 ** 2)
126 if self.count_people:
127 q[-1] = float(people_drift) ** 2
128 x0[-1], P0[-1] = 0.0, 10.0 ** 2
129 kwargs.setdefault("x0", x0)
130 kwargs.setdefault("P0", np.diag(P0))
131 super().__init__(sensor, F=F, B=B, H=H, Q=np.diag(q), R=np.array([[float(sigma) ** 2]]), fs=fs, **kwargs)
132 self.at = tuple(at)

135 ``rate`` readings per second (``None``: at every step of the model)."""
136 if tuple(at) not in self.index:
137 raise ValueError(f"RoomCO2Model.sensor: the cell {at} is not in the room.")
138 G = np.zeros((1, int(self.nnn)))
139 G[0, self.index[tuple(at)]] = 1.0
