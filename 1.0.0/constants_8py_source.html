81class KalmanEngineKind(Enum):
82 """
83 Algorithm family of an estimation engine.
84
85 Threads the shared term (Linear / Extended / Unscented / Particle) through every
86 layer so cross-layer compatibility is both readable and checkable:
87
88 - engines tag their ``KIND``;
89 - models declare ``DEFAULT_ENGINE`` + ``ALLOWED_ENGINES`` (these enum values,
90 so a model never imports an engine);
91 - the ``KalmanFilter`` resolves the default and validates
92 ``engine.KIND in model.ALLOWED_ENGINES``.
93 """
94 LINEAR = auto() # constant-matrix Kalman step (KF)
95 EXTENDED = auto() # Jacobian linearisation (EKF)
96 UNSCENTED = auto() # sigma-point transform (UKF)
97 PARTICLE = auto() # sequential importance resampling (PF)
98 COMBINATORIAL = auto() # branch-on-ambiguous-hit track finding (CKF)
99 ENSEMBLE = auto() # perturbed-observation ensemble Kalman filter (EnKF)
100 INTERACTING = auto() # several modes of one system, mixed and routed (IMM); see route()
101
102
103# All engine kinds — convenient default for ``ALLOWED_ENGINES``.
104ALL_ENGINES = frozenset(KalmanEngineKind)

110class KalmanIntegratorKind(Enum):
111 """
112 Time-stepping scheme of a :class:`~kalman.framework.integrator.KalmanIntegrator`
113 — advances ``x(t) → x(t+dt)`` given a continuous derivative ``ẋ = f(x,u,t)``.
114
115 Same declared-not-imported pattern as engines: integrators tag their ``KIND``;
116 models declare ``DEFAULT_INTEGRATOR`` + ``ALLOWED_INTEGRATORS``; the model
117 resolves/validates via ``resolve_integrator``.
118 """
119 EULER = auto() # explicit forward Euler (1st order)
120 RK4 = auto() # classic 4-stage Runge–Kutta (4th order)
121 DORMAND_PRINCE = auto() # RK45 embedded pair, adaptive substeps (Dormand–Prince)
122 IMPLICIT_EULER = auto() # backward Euler (stiff; needs a nonlinear solve — reserved)
123
124
125ALL_INTEGRATORS = frozenset(KalmanIntegratorKind)

167class KalmanStructureKind(Enum):
168 """
169 What a model is made of, which decides what a file can hold of it
170 (``model.export``). A model may declare it as ``STRUCTURE``; left at
171 ``None`` it is read from the model itself.
172
173 ``MAPS``: ``f`` and ``h`` are code. A file names the model and carries its
174 settings, or its Jacobians at one state.
175 ``MATRICES``: the model is its constant F, H, B, L, Q, R. A file holds all of it.
176 ``TIME_VARYING``: the model is its matrices and they change with time. A
177 file holds all of it over a stated number of samples, as tables.
178 """
179 MAPS = 0
180 MATRICES = 1
181 TIME_VARYING = 2

187class KalmanDynamicsKind(Enum):
188 """
189 What a model's ``f`` returns. A model declares it as ``DYNAMICS``.
190
191 ``DISCRETE``: ``f`` is the step, the state one sample later. Used as written.
192 ``CONTINUOUS``: ``f`` is the derivative dx/dt. The framework decodes it into
193 the step by integrating over one sample with the model's integrator, so
194 the filter always sees a discrete step.
195 """
196 DISCRETE = auto()
197 CONTINUOUS = auto()
