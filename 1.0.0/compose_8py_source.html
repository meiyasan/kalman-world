1"""
2``compose``: several models side by side, each with its own state.
3
4The Python face of ``Kalman::Models::Composite``
5(``include/Kalman/Models/Composite.h``), which does the work. The parts evolve
6independently, so the state is theirs one after the other and ``F`` and ``Q``
7are block-diagonal. What is measured depends on the readout:
8
9 STACKED: z = [ h1(x1) ; h2(x2) ; ... ] each part brings its sensors
10 SUMMED: z = h1(x1) + h2(x2) + ... the parts share one measurement
11
12Three ways to the same model::
13
14 a.compose(b)
15 compose(a, b, readout=KalmanReadout.SUMMED)
16 Composite(a, b)
17
18A parameter of a part is ``"<part>.<name>"``, the part going by its sensor's
19name (or by ``names=``). Models of matrices side by side are a model of
20matrices, on the linear engine. Nothing is computed here.
21
22This is not ``add_sensor``, where a second sensor reads the *same* state.
23"""

39 def __init__(self, *models: KalmanModel, readout: KalmanReadout = KalmanReadout.STACKED, names=None, **kwargs):
40 if len(models) == 1 and isinstance(models[0], (list, tuple)):
41 models = tuple(models[0])
42 if not models:
43 raise ValueError("compose: give at least one model.")
44 for model in models:
45 if not isinstance(model, KalmanModel):
46 raise TypeError(f"compose: a KalmanModel is expected, got {type(model).__name__}.")
47 if not isinstance(readout, KalmanReadout):
48 raise TypeError(f"compose: readout is a KalmanReadout, got {readout!r}.")
49 lib, ffi = core.require(), core.ffi
50 self.models, self.readout = list(models), readout
51 if names is None:
52 names, seen = [], {}
53 for model in models: # numbered from the second that shares a name
54 base = str(getattr(model, "_sensor_name", None) or type(model).__name__)
55 seen[base] = seen.get(base, 0) + 1
56 names.append(base if seen[base] == 1 else f"{base}{seen[base]}")
57 names = [str(name) for name in names]
58 if len(names) != len(models) or len(set(names)) != len(names) or not all(names):
59 raise ValueError(f"compose: every part needs a name of its own, got {names}.")
60 self.part_names = names
61
62 self._inner_cxxs = [self._inner(model) for model in models] # borrowed by the C++ model
63 encoded = [ffi.new("char[]", name.encode("utf-8", "replace")) for name in names]
64 handle = lib.kalman_model_create_composite(ffi.new("kalman_model_t *[]", self._inner_cxxs), len(models),
65 int(readout is KalmanReadout.SUMMED),
66 ffi.new("const char *[]", encoded))
67 if handle == ffi.NULL:
68 raise self._failed("compose")
69 self._inner_cxx_inner_cxx = self._inner_cxxs[0]
70 self._adopt(handle, list(models), **kwargs)

73 # time-varying matrices, and the filter runs it over their tables, as it does
74 # any model built from a function of time.

76 kinds = [model.structure for model in models]
77 self._time_varying = S.MAPS not in kinds and S.TIME_VARYING in kinds
78 if self._time_varying:

96 def part(self, which) -> slice:
97 """Where a part's state sits in this model's: ``record.xk[:, model.part("gps")]``.
98 ``which`` is a part's name, its index, or the model itself."""
99 if isinstance(which, KalmanModel):
100 index = next((i for i, model in enumerate(self.models) if model is which), None)
101 elif isinstance(which, str):
102 index = self.part_names.index(which) if which in self.part_names else None
103 else:
104 index = int(which) if -len(self.models) <= int(which) < len(self.models) else None
105 if index is None:
106 raise KeyError(f"compose: no part {which!r}; the parts are {self.part_names}.")
107 index %= len(self.models)
108 start = sum(int(model.n) for model in self.models[:index])
109 return slice(start, start + int(self.models[index].n))
