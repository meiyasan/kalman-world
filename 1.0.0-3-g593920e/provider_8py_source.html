2"""
3Abstract **provider** base — the data/knowledge-source role ("WHERE the data
4comes from"). Lives in ``framework/`` with every other base contract; concrete
5providers (synthetic/file/sensor/remote and the domain ones like weather's
6ERA5/turbulence) live outside it and register into ``kalman.providers``.
7
8A provider implements ``fetch`` (pull) and/or ``stream`` (live) and returns a
9plain dict payload — some of ``{"zk", "xk", "truth", "time", "meta"}`` — so any
10transport (facade/MCP/REST) can carry it as JSON. ``measurements``/``truth``
11are conveniences that pull the filter/OSSE-ready arrays out of the payload; the
12base offers a tiny query-keyed cache.
13"""

27 #: What this provider can actually supply, declared the way models declare
28 #: their params — each entry ``{"name":.., "units":.., "description":..}``.
29 #:
30 #: A provider that only advertises its *name* is half-discoverable: a caller
31 #: can see that ``era5`` exists but not that it yields 500 hPa wind, and has
32 #: to read the source to find out. Models are introspectable through
33 #: ``describe()``; measurements are the other half of the same question, so
34 #: they are declared here and harvested by ``kalman.api.describe_provider``.

39 #: before it tries (ERA5 needs ``cdsapi`` and CDS credentials, and finding
40 #: that out from an ImportError mid-download is nobody's idea of discovery).

46 # ------------------------------------------------------------------
47 @classmethod
48 def describe(cls) -> dict:
49 """Everything discoverable about this provider, as plain JSON-able data.
50
51 Harvests the declared metadata plus the real ``fetch`` signature, so the
52 query keys a caller must pass are read off the code rather than kept in
53 a docstring that can drift away from it."""
54 import inspect
55 try:
56 sig = inspect.signature(cls.fetchfetch)
57 query = [{"name": name,
58 "required": p.default is inspect.Parameter.empty
59 and p.kind is not inspect.Parameter.VAR_KEYWORD,
60 "default": None if p.default is inspect.Parameter.empty
61 else p.default}
62 for name, p in sig.parameters.items()
63 if name not in ("self", "kw", "kwargs")]
64 except (TypeError, ValueError):
65 query = []
66
67 kind = getattr(cls, "KIND", None)
68 return {
69 "name": getattr(cls, "NAME", cls.__name__),
70 "kind": getattr(kind, "name", str(kind)).lower() if kind else None,
71 "class": cls.__name__,
72 "description": getattr(cls, "DESCRIPTION", ""),
73 "version": getattr(cls, "VERSION", ""),
74 "measurements": [dict(m) for m in getattr(cls, "MEASUREMENTS", ())],
75 "requires": list(getattr(cls, "REQUIRES", ())),
76 "query": query,
77 "streams": cls.stream is not KalmanProvider.stream,
78 }

89 def _cached(self, key, produce):
90 if self._cache is None:
91 return produce()
92 if key not in self._cache:
93 self._cache[key] = produce()
94 return self._cache[key]

96 # ------------------------------------------------------------------
97 def measurements(self, **query):
98 """Filter-ready measurements ``zk`` from the payload."""
99 p = self.fetchfetch(**query)
100 z = p.get("zk", p.get("measurements"))
101 if z is None:
102 raise KeyError(f"{self.name()} payload has no measurements.")
103 return np.asarray(z, dtype=float)
104
105 def truth(self, **query):
106 """OSSE-ready truth trajectory from the payload."""
107 p = self.fetchfetch(**query)
108 x = p.get("truth", p.get("xk"))
109 if x is None:
110 raise KeyError(f"{self.name()} payload has no truth.")
111 return np.asarray(x, dtype=float)
