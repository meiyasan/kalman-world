28class path_taken:
29 """Which of the filter's C++ passes ran inside the block (``None``: the Python loop)."""
30
31 def __enter__(self):
32 self.taken, self._orig = [], {}
33 for name in PATHS:
34 self._orig[name] = getattr(BayesFilter, name)
35
36 def wrap(name=name):
37 def run(filt, *args, **kwargs):
38 self.taken.append(name)
39 return self._orig[name](filt, *args, **kwargs)
40 return run
41 setattr(BayesFilter, name, wrap())
42 return self
43
44 def __exit__(self, *exc):
45 for name, fn in self._orig.items():
46 setattr(BayesFilter, name, fn)
47
48 @property
49 def last(self):
50 return self.taken[-1] if self.taken else None

53def apart(a, b):
54 xa, xb = np.asarray(a.xk, float), np.asarray(b.xk, float)
55 Pa, Pb = np.asarray(a.Pk, float), np.asarray(b.Pk, float)
56 return (float(np.nanmax(np.abs(xa - xb)) / max(1.0, np.nanmax(np.abs(xa)))),
57 float(np.nanmax(np.abs(Pa - Pb)) / max(1.0, np.nanmax(np.abs(Pa)))))

60def both(make, z):
61 with native.python_reference():
62 reference = make().filter(z)
63 with path_taken() as path:
64 record = make().filter(z)
