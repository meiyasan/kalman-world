24class PhasorOscillatorModel(LinearKalmanModel):
25 r"""
26 Track a set of sinusoids as **lab-frame phasors**, straight from a **real** signal —
27 no demodulation, no Hilbert transform.
28
29 This is the *dual* of :class:`AnalyticOscillatorModel`:
30
31 * **Rotating frame** (`AnalyticOscillatorModel` / `OscillatorModel`): the carrier
32 rotation lives in the **signal** — we demodulate by multiplying ``z(t)`` by
33 ``e^{−iω_c t}`` so the mode appears as a slowly-varying envelope. ``F = expm(Δω)``,
34 ``H = identity``. Right for **weak, narrowband** lines (KAGRA violin modes).
35
36 * **Lab frame** (`PhasorOscillatorModel`): the carrier rotation lives in the
37 **observation map** instead. The phasors ``[Re_k, Im_k]`` are constant (``F = I``)
38 and the single real measurement is reconstructed from all of them at once through a
39 time-varying row ``H(t) = [cos ω₁t, −sin ω₁t, cos ω₂t, −sin ω₂t, …]``. No
40 Hilbert, causal/streaming. Because every mode must together explain the one
41 measurement, a strong mode cannot leak into an absent neighbour. Right for
42 **strong, wide-spread** lines (mains harmonics 50→250 Hz, or any wide-band set).
43
44 State layout (one phasor per frequency, scalar observation)::
45
46 x = [Re₁, Im₁, Re₂, Im₂, …, Re_N, Im_N] (2N states)
47 z = [v(t)] (1 observation)
48
49 ``F = I`` (phasors constant; ``process_uncertainty`` allows slow drift);
50 ``R = σ_meas²`` is a single channel.
51
52 Parameters
53 ----------
54 sensor_id : str
55 frequencies : float | array-like
56 Line frequencies [Hz] to track jointly (one ``[Re, Im]`` phasor each).
57 fs : float
58 measurement_uncertainty : float
59 Measurement-noise std (sets ``R = σ²``).
60 process_uncertainty : float
61 Per-phasor random-walk std (how fast the amplitudes may drift).
62 p0_std : float
63 Initial per-phasor prior std — large vs. the measurement noise for acquisition.
64 """
65
66 NAME = "Phasor Oscillator (lab-frame, real signal)"
67 DESCRIPTION = ("Joint lab-frame phasor tracker: real scalar in, time-varying H(t), "
68 "F = I. All modes coupled through one measurement (no demodulation).")
69 VERSION = "1.0"

79 def __init__(self, sensor_id, frequencies, *, fs=4096.0,
80 measurement_uncertainty=1.0, process_uncertainty=0.01,
81 p0_std=100.0, **kwargs):
82
83 freqs = np.atleast_1d(np.asarray(frequencies, dtype=float))
84 if freqs.size == 0:
85 raise ValueError("PhasorOscillatorModel needs at least one frequency.")
86 self.frequencies = freqs
87 self._omega = 2.0 * np.pi * freqs # (N,) carrier angular freqs
88 self.measurement_uncertainty = float(measurement_uncertainty)
89 self.process_uncertainty = float(process_uncertainty)
90 N = len(freqs)
91
92 F = np.eye(2 * N) # phasors constant in lab frame
93 Q = (process_uncertainty ** 2) * np.eye(2 * N) # slow phasor random-walk
94 R = np.array([[measurement_uncertainty ** 2]]) # scalar measurement noise
95
96 x0 = kwargs.pop("x0", None)
97 if x0 is None:
98 x0 = np.zeros((2 * N, 1))
99 P0 = kwargs.pop("P0", None)
100 if P0 is None:
101 P0 = (p0_std ** 2) * np.eye(2 * N) # generous prior for acquisition
102
103 super().__init__(sensor_id, F=F, H=self._observation_row, Q=Q, R=R,
104 fs=fs, x0=x0, P0=P0, **kwargs)

106 # ------------------------------------------------------------------
107 # Time-varying observation: the real scalar as the sum of all phasors.
108 # H(t) = [cos ω₁t, −sin ω₁t, cos ω₂t, −sin ω₂t, …] shape (1, 2N)
109 # ------------------------------------------------------------------
110 def _observation_row(self, t=None):
111 t = 0.0 if t is None else float(t)
112 row = np.zeros((1, 2 * len(self.frequencies)))
113 row[0, 0::2] = np.cos(self._omega * t)
114 row[0, 1::2] = -np.sin(self._omega * t)
115 return row

126 def amplitudes(self, record):
127 """Per-mode amplitude ``a_k = |ψ_k| = √(Re²+Im²)`` — shape ``(T, N)``."""
128 xk = np.asarray(record.xk)
129 return np.column_stack([np.hypot(xk[:, 2 * k], xk[:, 2 * k + 1])
130 for k in range(len(self.frequencies))])

132 def envelope(self, record):
133 """Per-mode amplitude and its ±1σ band from the phasor state ``record.xk`` and
134 covariance ``record.Pk``. Returns ``(amp, sigma)``, each shape ``(T, N)``."""
135 xk = np.asarray(record.xk)
136 Pk = np.asarray(record.Pk)
137 T, N = len(xk), len(self.frequencies)
138 amp = np.zeros((T, N))
139 sigma = np.zeros((T, N))
140 has_P = Pk.size > 0
141 for k in range(N):
142 re, im = xk[:, 2 * k], xk[:, 2 * k + 1]
143 a = np.hypot(re, im)
144 amp[:, k] = a
145 if has_P:
146 P_RR, P_II = Pk[:, 2 * k, 2 * k], Pk[:, 2 * k + 1, 2 * k + 1]
147 with np.errstate(divide="ignore", invalid="ignore"):
148 sigma[:, k] = np.where(
149 a > 1e-3,
150 np.sqrt((re ** 2 * P_RR + im ** 2 * P_II) / a ** 2), 0.0)
151 return amp, sigma
152
153 def reconstruct(self, record, t=None):
154 """Reconstruct the real signal from the phasor state — the model's own observation
155 map: ``z(t) = Σ_k Re_k·cos(ω_k t) − Im_k·sin(ω_k t)``."""
156 xk = np.asarray(record.xk)
157 t = np.asarray(record.time if t is None else t)
158 v = np.zeros(len(t))
159 for k in range(len(self.frequencies)):
160 w_k = self._omega[k]
161 v += xk[:, 2 * k] * np.cos(w_k * t) - xk[:, 2 * k + 1] * np.sin(w_k * t)
162 return v
