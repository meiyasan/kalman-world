3class EnumBis(Enum):
4 """Enhanced Enum base with class-safe introspection methods."""

6 def __eq__(self, other):
7 if isinstance(other, Enum): return self is other
8 if isinstance(other, str) : return super().name.lower() == other.lower()
9 if isinstance(other, int) : return self.value == other
10 return False
11 def __hash__(self):
12 return hash(self.name.lower())
13 def __repr__(self):
14 return f"{self.__class__.__name__}.{super().name}(index={int(self)}, value='{self.value}')"
15 def __len__(self):
16 self = type(self) if isinstance(self, Enum) else self
17 return len(self.__members__)
18 def __int__(self):
19 return list(self.__class__).index(self)
20 def __str__(self):

22 def matches(self, *args):

83class KalmanEngineKind(Enum):
84 """
85 Algorithm family of an estimation engine.
86
87 Threads the shared term (Linear / Extended / Unscented / Particle) through every
88 layer so cross-layer compatibility is both readable and checkable:
89
90 - engines tag their ``KIND``;
91 - models declare ``DEFAULT_ENGINE`` + ``ALLOWED_ENGINES`` (these enum values,
92 so a model never imports an engine);
93 - the ``KalmanFilter`` resolves the default and validates
94 ``engine.KIND in model.ALLOWED_ENGINES``.
95 """
96 LINEAR = auto() # constant-matrix Kalman step (KF)
97 EXTENDED = auto() # Jacobian linearisation (EKF)
98 UNSCENTED = auto() # sigma-point transform (UKF)
99 PARTICLE = auto() # sequential importance resampling (PF)
100 COMBINATORIAL = auto() # branch-on-ambiguous-hit track finding (CKF)
101 ENSEMBLE = auto() # perturbed-observation ensemble Kalman filter (EnKF)
102
103
104# All engine kinds — convenient default for ``ALLOWED_ENGINES``.
105ALL_ENGINES = frozenset(KalmanEngineKind)

111class KalmanIntegratorKind(Enum):
112 """
113 Time-stepping scheme of a :class:`~kalman.framework.integrator.KalmanIntegrator`
114 — advances ``x(t) → x(t+dt)`` given a continuous derivative ``ẋ = f(x,u,t)``.
115
116 Same declared-not-imported pattern as engines: integrators tag their ``KIND``;
117 models declare ``DEFAULT_INTEGRATOR`` + ``ALLOWED_INTEGRATORS``; the model
118 resolves/validates via ``resolve_integrator``.
119 """
120 EULER = auto() # explicit forward Euler (1st order)
121 RK4 = auto() # classic 4-stage Runge–Kutta (4th order)
122 DORMAND_PRINCE = auto() # RK45 embedded pair, adaptive substeps (Dormand–Prince)
123 IMPLICIT_EULER = auto() # backward Euler (stiff; needs a nonlinear solve — reserved)
124
125
126ALL_INTEGRATORS = frozenset(KalmanIntegratorKind)

168class KalmanStructureKind(Enum):
169 """
170 What a model is made of, which decides what a file can hold of it
171 (``model.export``). A model may declare it as ``STRUCTURE``; left at
172 ``None`` it is read from the model itself.
173
174 ``MAPS``: ``f`` and ``h`` are code. A file names the model and carries its
175 settings, or its Jacobians at one state.
176 ``MATRICES``: the model is its constant F, H, B, L, Q, R. A file holds all of it.
177 ``TIME_VARYING``: the model is its matrices and they change with time. A
178 file holds all of it over a stated number of samples, as tables.
179 """
180 MAPS = 0
181 MATRICES = 1
182 TIME_VARYING = 2

189 """
190 What a model's ``f`` returns. A model declares it as ``DYNAMICS``.
191
192 ``DISCRETE``: ``f`` is the step, the state one sample later. Used as written.
193 ``CONTINUOUS``: ``f`` is the derivative dx/dt. The framework decodes it into
194 the step by integrating over one sample with the model's integrator, so
195 the filter always sees a discrete step.
196 """
