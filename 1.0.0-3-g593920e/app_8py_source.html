1"""FastAPI app over the :mod:`kalman.api` facade — one thin adapter, no logic.
2
3Each route declares a required **scope**; a ``Authorization: Bearer <JWT>``
4header is verified by the same :func:`kalman.api.auth.authorize` the MCP server
5uses. FastAPI auto-generates the OpenAPI schema at ``/docs`` — a
6machine-discoverable contract for any non-MCP agent. ``fastapi`` is imported
7lazily so the core never depends on it.
8
9This is the registry's front door: models, plugins and measurement sources are
10registered here, everybody lists them here, and ``GET /events/stream``
11announces each change as it happens. The MCP server can run as a client of this
12API (``KALMAN_API_URL``), so agents and HTTP clients such as Unity share one
13registry and one set of runs.
14
15Errors map to status codes: 400 a bad request, 401 no or bad token (or a wrong
16password), 403 a token without the scope (or not the owner), 404 an unknown
17name or run, 409 a name taken by a different definition, 429 too many failed
18logins.
19"""

62 # ---- errors -> status codes, once for every route ----
63 def _err(status, headers=None):
64 async def handler(request, exc):
65 msg = exc.args[0] if isinstance(exc, KeyError) and exc.args else str(exc)
66 return JSONResponse(status_code=status, content={"detail": str(msg)},
67 headers=headers)
68 return handler

71 app.add_exception_handler(KeyError, _err(404))

73 app.add_exception_handler(AuthError, _err(403)) # a valid token, but not allowed

76 def _token(authorization):
77 if authorization and authorization.lower().startswith("bearer "):
78 return authorization.split(None, 1)[1].strip()
79 return None

81 def require(scope: str):
82 def dep(authorization: str = Header(default=None)):
83 token = _token(authorization)
84 try:
85 authorize(token, scope)
86 except auth.InvalidToken as e: # none, forged, expired, revoked
87 raise HTTPException(status_code=401, detail=str(e),
88 headers={"WWW-Authenticate": "Bearer"})
89 except AuthError as e: # a good token without the scope
90 raise HTTPException(status_code=403, detail=str(e))
91 return token
92 return dep

94 def need(body: dict, *keys):
95 missing = [k for k in keys if k not in body]
96 if missing:

113 @app.post("/token")
114 def token(body: dict, request: Request):
115 """``{user, password, scopes?, exp_hours?}`` -> ``{access_token, ...}``.
116 The scopes must be in the account's grant; see ``make api-user``."""
117 need(body, "user", "password")
118 try:
119 return auth.login(body["user"], body["password"], body.get("scopes"),
120 exp_hours=body.get("exp_hours"),
121 client=request.client.host if request.client else "")
122 except auth.RateLimited as e:
123 raise HTTPException(status_code=429, detail=str(e),
124 headers={"Retry-After": str(int(e.retry_after) + 1)})
125 except auth.BadCredentials as e:
126 raise HTTPException(status_code=401, detail=str(e),
127 headers={"WWW-Authenticate": "Bearer"})

129 @app.post("/token/revoke")

133 tok = _token(authorization)

138 headers={"WWW-Authenticate": "Bearer"})

202 @app.get("/events/stream")
203 async def events_stream(request: Request, since: int | None = None,
204 timeout: float | None = Query(default=None, ge=0),
205 last_event_id: str | None = Header(default=None),
206 _=Depends(require("read"))):
207 """Server-Sent Events: every registry change after ``since`` (or the
208 ``Last-Event-ID`` a reconnecting client sends; default: from now).
209 ``timeout`` closes the stream after that many seconds."""
210 if since is None:
211 since = int(last_event_id) if (last_event_id or "").isdigit() \
212 else await run_in_threadpool(store.last_event)
213 poll = float(os.environ.get("KALMAN_EVENTS_POLL", "0.5"))
214
215 async def gen():
216 cursor, start, quiet = since, time.monotonic(), time.monotonic()
217 yield "retry: 3000\n\n"
218 while True:
219 evs = await run_in_threadpool(store.events, cursor, 100)
220 for e in evs:
221 cursor = e["id"]
222 yield (f"id: {e['id']}\nevent: {e['type']}\n"
223 f"data: {json.dumps(e)}\n\n")
224 if evs:
225 quiet = time.monotonic()
226 elif time.monotonic() - quiet > 15: # keep proxies from closing it
227 yield ": keep-alive\n\n"
228 quiet = time.monotonic()
229 if timeout is not None and time.monotonic() - start >= timeout:
230 return
231 if await request.is_disconnected():
232 return
233 if not evs:
234 await asyncio.sleep(poll)
235
236 return StreamingResponse(gen(), media_type="text/event-stream",
237 headers={"Cache-Control": "no-cache",
238 "X-Accel-Buffering": "no"})
