27 std::unique_ptr<Engine> MakeEngine(EngineKind kind)
28 {
29 switch (kind)
30 {
31 case EngineKind::Linear: return std::make_unique<Engines::Linear>();
32 case EngineKind::Extended: return std::make_unique<Engines::Extended>();
33 case EngineKind::Unscented: return std::make_unique<Engines::Unscented>();
34 case EngineKind::Combinatorial: return std::make_unique<Engines::Combinatorial>();
35 // Particle and Ensemble are deliberately absent: the IKF mixing step
36 // hands each mode a Gaussian (x, P) to predict from, and a sample-set
37 // engine ignores that argument -- the modes would silently decouple.
38 default: break;
39 }
40 std::ostringstream os;
41 os << "Kalman::Filters::IKF: no C++ engine implements the '" << ToString(kind)
42 << "' kind requested by one of the modes.";
43 throw std::runtime_error(os.str());
44 }
45 }

47 IKF::IKF(const std::vector<Model *> &models, const Matrix &Pi, const Routers::Router *router)
48 : fModels(models), fPi(Pi)
49 {
50 const std::size_t M = fModels.size();
51 if (M < 2) throw std::runtime_error("Kalman::Filters::IKF: at least two models are required.");
52 if (Pi.Rows() != M || Pi.Cols() != M)
53 {
54 std::ostringstream os;
55 os << "Kalman::Filters::IKF: Pi has shape (" << Pi.Rows() << ", " << Pi.Cols()
56 << "), expected (" << M << ", " << M << ").";
57 throw std::runtime_error(os.str());
58 }
59
60 for (std::size_t i = 0; i < M; ++i)
61 {
62 if (!fModels[i]) throw std::runtime_error("Kalman::Filters::IKF: a model is null.");
63 fModels[i]->CheckShapes();
64
65 // Mixing averages states across modes, so it is only meaningful when every
66 // mode means the same thing by "the state".
67 if (fModels[i]->NState() != fModels[0]->NState() ||
68 fModels[i]->NMeasure() != fModels[0]->NMeasure())
69 {
70 std::ostringstream os;
71 os << "Kalman::Filters::IKF: model '" << fModels[i]->Name() << "' has (n, m) = ("
72 << fModels[i]->NState() << ", " << fModels[i]->NMeasure() << ") but '"
73 << fModels[0]->Name() << "' has (" << fModels[0]->NState() << ", "
74 << fModels[0]->NMeasure() << "). Every mode must share both.";
75 throw std::runtime_error(os.str());
76 }
77
78 // Each mode brings its own engine, so a linear and a nonlinear mode can
79 // coexist in one bank.
80 fEngines.push_back(MakeEngine(fModels[i]->DefaultEngine()));
81 }
82
83 for (std::size_t i = 0; i < M; ++i)
84 {
85 double row = 0.;
86 for (std::size_t j = 0; j < M; ++j)
87 {
88 if (Pi(i, j) < 0.)
89 throw std::runtime_error("Kalman::Filters::IKF: Pi has a negative entry.");
90 row += Pi(i, j);
91 }
92 if (std::abs(row - 1.) > 1e-9)
93 {
94 std::ostringstream os;
95 os << "Kalman::Filters::IKF: row " << i << " of Pi sums to " << row
96 << ", expected 1.";
97 throw std::runtime_error(os.str());
98 }
99 }
100
101 if (router) fRouter = router;
102 else
103 {
104 fOwnRouter = std::make_unique<Routers::Bayesian>();
105 fRouter = fOwnRouter.get();
106 }
107
108 Reset();
109 }

144 double IKF::Likelihood(const Vector &nu, const Matrix &S)
145 {
146 const std::size_t m = nu.Size();
147 if (m == 0) return 1.; // nothing measured: no evidence either way
148
149 const Matrix Sinv = Utils::Inverse(S);
150 const double quad = Utils::Mahalanobis(nu, Sinv);
151
152 // |S| from the Cholesky factor when it exists; the eigenvalues otherwise.
153 double logdet = 0.;
154 const Matrix L = Utils::CholeskyFactor(S, /* jitter */ true);
155 if (!L.Empty())
156 {
157 for (std::size_t i = 0; i < m; ++i) logdet += 2. * std::log(std::abs(L(i, i)));
158 }
159 else
160 {
161 Vector values;
162 Matrix vectors;
163 Utils::SymmetricEigen(S, values, vectors);
164 for (std::size_t i = 0; i < m; ++i)
165 if (std::abs(values(i)) > 1e-300) logdet += std::log(std::abs(values(i)));
166 }
167
168 const double logL = -0.5 * (quad + logdet + static_cast<double>(m) * std::log(2. * M_PI));
169 // Floored, not clamped to zero: a mode whose likelihood underflows must stay
170 // recoverable, and exp() of a very negative number is exactly what underflows.
171 return std::max(std::exp(logL), 1e-300);
172 }

174 Record IKF::Forward(const std::vector<Vector> &zk)
175 {
176 if (zk.empty()) throw std::runtime_error("Kalman::Filters::IKF::Forward: no measurements given.");
177
178 const std::size_t M = fModels.size();
179 const std::size_t m = fModels[0]->NMeasure();
180 const std::size_t n = fModels[0]->NState();
181 const std::size_t N = zk.size();
182
183 Reset();
184
185 fNis = Matrix(N, M, 0.);
186 fModeHistory = Matrix(N, M, 0.);
187 fCombined.reserve(N);
188 fCombinedP.reserve(N);
189 fTime.reserve(N);
190
191 const double dt = fModels[0]->DeltaT();
192 double t = fModels[0]->T0();
193
194 for (std::size_t k = 0; k < N; ++k)
195 {
196 if (zk[k].Size() != m)
197 {
198 std::ostringstream os;
199 os << "Kalman::Filters::IKF::Forward: measurement " << k << " has "
200 << zk[k].Size() << " channels, expected " << m << ".";
201 throw std::runtime_error(os.str());
202 }
203
204 // --- 1. Markov prediction ------------------------------------
205 Vector c(M, 0.);
206 for (std::size_t j = 0; j < M; ++j)
207 for (std::size_t i = 0; i < M; ++i) c(j) += fPi(i, j) * fMu(i);
208
209 // --- 2. Mixing: every mode starts from a blend of all modes ----
210 // This is what separates an IKF from M independent filters: a mode that
211 // becomes relevant inherits the state the others built up.
212 std::vector<Vector> xMix(M);
213 std::vector<Matrix> pMix(M);
214 for (std::size_t j = 0; j < M; ++j)
215 {
216 Vector x(n, 0.);
217 for (std::size_t i = 0; i < M; ++i)
218 {
219 const double w = (c(j) > 1e-300) ? fPi(i, j) * fMu(i) / c(j) : 1. / M;
220 x += w * fX[i];
221 }
222
223 Matrix P(n, n, 0.);
224 for (std::size_t i = 0; i < M; ++i)
225 {
226 const double w = (c(j) > 1e-300) ? fPi(i, j) * fMu(i) / c(j) : 1. / M;
227 Vector d(fX[i]);
228 d -= x;
229
230 Matrix spread(n, n, 0.);
231 for (std::size_t a = 0; a < n; ++a)
232 for (std::size_t b = 0; b < n; ++b) spread(a, b) = d(a) * d(b);
233
234 Matrix contrib(fP[i]);
235 contrib += spread;
236 contrib *= w;
237 P += contrib;
238 }
239 Utils::Symmetrize(P);
240
241 xMix[j] = x;
242 pMix[j] = P;
243 }
244
245 // --- 3. Predict + update each mode from its mixed start --------
246 Vector likelihoods(M, 1.);
247 for (std::size_t j = 0; j < M; ++j)
248 {
249 Vector xPrior;
250 Matrix PPrior, F;
251 fEngines[j]->Predict(*fModels[j], xMix[j], pMix[j], nullptr, t, xPrior, PPrior, F);
252
253 Vector nu;
254 Matrix S;
255 fEngines[j]->Innovation(*fModels[j], xPrior, PPrior, zk[k], t, nu, S);
256
257 // Missing channels drop out of both the likelihood and the NIS, so a
258 // partly-blind step simply carries less evidence rather than none.
259 const std::vector<std::size_t> keep = Utils::ValidChannels(nu);
260 if (keep.empty())
261 {
262 likelihoods(j) = 1.;
263 fNis(k, j) = static_cast<double>(m); // neutral: the expected value
264 }
265 else
266 {
267 const Vector nuc = Utils::Compress(nu, keep);
268 const Matrix Sc = Utils::CompressBoth(S, keep);
269 likelihoods(j) = Likelihood(nuc, Sc);
270 fNis(k, j) = Utils::Mahalanobis(nuc, Utils::Inverse(Sc));
271 }
272
273 Vector xPost;
274 Matrix PPost;
275 fEngines[j]->Update(*fModels[j], xPrior, PPrior, zk[k], t, xPost, PPost);
276 fX[j] = xPost;
277 fP[j] = PPost;
278 }
279
280 // --- 4. Route: the only decision, and the only pluggable part --
281 Matrix nisSoFar(k + 1, M, 0.);
282 for (std::size_t r = 0; r <= k; ++r)
283 for (std::size_t j = 0; j < M; ++j) nisSoFar(r, j) = fNis(r, j);
284
285 fMu = Routers::Router::Normalise(fRouter->Route(nisSoFar, c, likelihoods));
286 for (std::size_t j = 0; j < M; ++j) fModeHistory(k, j) = fMu(j);
287
288 // --- 5. Combination -------------------------------------------
289 Vector x(n, 0.);
290 for (std::size_t j = 0; j < M; ++j) x += fMu(j) * fX[j];
291
292 Matrix P(n, n, 0.);
293 for (std::size_t j = 0; j < M; ++j)
294 {
295 Vector d(fX[j]);
296 d -= x;
297
298 Matrix spread(n, n, 0.);
299 for (std::size_t a = 0; a < n; ++a)
300 for (std::size_t b = 0; b < n; ++b) spread(a, b) = d(a) * d(b);
301
302 Matrix contrib(fP[j]);
303 contrib += spread;
304 contrib *= fMu(j);
305 P += contrib;
306 }
307 Utils::Symmetrize(P);
308
309 fCombined.push_back(x);
310 fCombinedP.push_back(P);
311 fTime.push_back(t);
312 t += dt;
313 }
314
315 Record record("ikf", StateFlavor::Posterior);
316 record.Reserve(N);
317 for (std::size_t k = 0; k < N; ++k)
318 record.Add(fTime[k], fCombined[k], fCombinedP[k], zk[k]);
319 return record;
320 }

322 std::size_t IKF::DominantMode(std::size_t k) const
323 {
324 if (k >= fModeHistory.Rows())
325 throw std::out_of_range("Kalman::Filters::IKF::DominantMode: step out of range.");
326
327 std::size_t best = 0;
328 for (std::size_t j = 1; j < fModeHistory.Cols(); ++j)
329 if (fModeHistory(k, j) > fModeHistory(k, best)) best = j;
330 return best;
331 }
