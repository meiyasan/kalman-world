9class ParticleEngine(KalmanEngine):
10 """
11 Bootstrap / SIR particle filter.
12
13 The **one stateful engine**: it holds the particle cloud for the duration of a
14 forward pass (the filter calls ``reset(model)`` to (re)seed it from ``x0/P0``).
15 ``_predict`` propagates particles through ``model.f`` with ``model.Q`` draws;
16 ``_update`` reweights by the Gaussian innovation likelihood (``model.h``,
17 ``model.R``), normalises, and systematic-resamples when the effective sample size
18 drops. Both return the weighted mean/cov as ``(x, P)`` so records/logs are
19 unchanged. No RTS smoother (forward-only).
20 """
21
22 NAME = "Particle Engine"
23 DESCRIPTION = "Sequential importance-resampling particle filter (forward-only)."
24 VERSION = "1.0"
25 KIND = KalmanEngineKind.PARTICLE

27 def __init__(self, n_particles: int = 500, resample_threshold: float = 0.5):
28 self.n_particles = int(n_particles)
29 self.resample_threshold = float(resample_threshold)
30 self._particles = None # (N, n)
31 self._weights = None # (N,)
32
33 # ------------------------------------------------------------------
34 # Per-run state
35 # ------------------------------------------------------------------
36 def reset(self, model):
37 n = model.n
38 x0 = np.asarray(model.x0, dtype=np.float64).reshape(-1)
39 P0 = np.asarray(model.P0, dtype=np.float64)
40 self._particles = np.random.multivariate_normal(x0, P0, size=self.n_particles)
41 self._weights = np.full(self.n_particles, 1.0 / self.n_particles)
42 return self

44 def _estimate(self):
45 w = self._weights
46 mean = np.einsum("i,ij->j", w, self._particles)
47 d = self._particles - mean
48 cov = (d * w[:, None]).T @ d
49 cov = 0.5 * (cov + cov.T) + 1e-9 * np.eye(cov.shape[0])
50 return mean.reshape(-1, 1), cov

52 def _resample(self):
53 N = self.n_particles
54 from ..native import HAS_C_ACCEL
55 if HAS_C_ACCEL:
56 from ..native import lib, as_contiguous, dptr, iptr
57 weights = as_contiguous(self._weights)
58 idx = np.empty(N, dtype=np.intc)
59 lib.pf_systematic_resample(N, dptr(weights), float(np.random.uniform()), iptr(idx))
60 else:
61 positions = (np.arange(N) + np.random.uniform()) / N
62 cumsum = np.cumsum(self._weights)
63 cumsum[-1] = 1.0
64 idx = np.searchsorted(cumsum, positions)
65 self._particles = self._particles[idx]
66 self._weights = np.full(N, 1.0 / N)

71 def _predict(self, model, x, P, u=None, t=None, **kw):
72 if self._particles is None:
73 self.resetreset(model)
74
75 n = model.n
76 u = (np.asarray(u).reshape(-1, 1) if (u is not None and model.l > 0) else None)
77 Q = np.asarray(model.Q(t=t), dtype=np.float64)
78 accepts_w = self._noise_through_f(model)
79 if not accepts_w:
80 # Added after f, through the model's coupling: w ~ N(0, L Q L^T).
81 L = self._safe_L(model, np.asarray(model.x0, dtype=np.float64).reshape(-1, 1), u, t)
82 if L is not None:
83 Q = L @ Q @ L.T
84 noise = np.random.multivariate_normal(np.zeros(n), Q, size=self.n_particles)
85
86 prop = np.empty_like(self._particles)
87 for i, xp in enumerate(self._particles):
88 xi = xp.reshape(-1, 1)
89 if accepts_w:
90 xi2 = model.f(xi, u=u, w=noise[i].reshape(-1, 1), t=t)
91 else:
92 xi2 = np.asarray(self._f_plain(model, xi, u, t)).reshape(-1, 1) + noise[i].reshape(-1, 1)
93 prop[i] = np.asarray(xi2, dtype=np.float64).reshape(-1)
94 self._particles = prop
95
96 x_mean, P_cov = self._estimate()

106 def _update(self, model, x, P, z, t=None, **kw):
107 z = np.asarray(z, dtype=np.float64).reshape(-1)
108 keep = ~np.isnan(z)
109 if not keep.any():
110 return self._estimate()
111
112 idx = np.where(keep)[0]
113 mv = len(idx)
114 R = np.asarray(model.R(t=t), dtype=np.float64)[np.ix_(idx, idx)]
115 Rinv = np.linalg.pinv(R)
116 zc = z[idx]
117
118 # model.h() is unavoidable Python (per-particle nonlinear map); only the
119 # quadratic-form reduction below is pushed into C.
120 zp = np.empty((self.n_particles, mv), dtype=np.float64)
121 for i, xp in enumerate(self._particles):
122 zp[i] = np.asarray(model.h(xp.reshape(-1, 1), t=t), dtype=np.float64).reshape(-1)[idx]
123
124 logw = np.log(self._weights + 1e-300)
125 from ..native import HAS_C_ACCEL
126 if HAS_C_ACCEL:
127 from ..native import lib, as_contiguous, dptr
128 lib.pf_weight_update(
129 self.n_particles, mv,
130 dptr(as_contiguous(zc)), dptr(as_contiguous(zp)), dptr(as_contiguous(Rinv)), dptr(logw),
131 )
132 else:
133 for i in range(self.n_particles):
134 dz = zc - zp[i]
135 logw[i] += -0.5 * dz @ Rinv @ dz
136 logw -= logw.max()
137 wts = np.exp(logw)
138 wts /= wts.sum()
139 self._weights = wts
140
141 ess = 1.0 / np.sum(wts ** 2)
142 if ess < self.resample_threshold * self.n_particles:
143 self._resample()
144
145 return self._estimate()
