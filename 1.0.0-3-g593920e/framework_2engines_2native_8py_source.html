1"""
2The C++ estimation step, driven by the Python filter.
3
4:class:`NativeEngine` is a :class:`~kalman.framework.engine.KalmanEngine` whose
5``_predict`` / ``_update`` / ``_smooth`` run in ``libKalman`` — the same
6``Engines::Linear`` / ``Extended`` / ``Unscented`` / ``Particle`` / ``Ensemble`` a
7C++ or ROOT+ program uses — through the C API. The Python
8:class:`~kalman.framework.filter.KalmanFilter` keeps the workflow, the records,
9the diagnostics and the plots; only the arithmetic crosses.
10
11It needs a :class:`~kalman.framework.models.native.NativeModel`: the C++ engine
12evaluates the model's ``f``/``h``/``F``/``Q`` on the C++ side, and there is no
13route from there back into a Python-defined model — by design, since a
14per-step callback into Python is what the measurements in ``docs/15`` rule out.
15A Python model keeps its Python engines.
16
17Not registered in ``ENGINE_BY_KIND``: it is a *backend* for an existing kind,
18not a kind. ``KalmanFilter(model, engine=NativeEngine("extended"))`` is the
19per-step form; :meth:`NativeModel.filter_native` is the fused one.
20"""

38 def __init__(self, kind="extended", **params):
39 """``kind`` is an engine name (``"linear"``, ``"extended"``, ``"unscented"``,
40 ``"particle"``, ``"ensemble"``) or a :class:`KalmanEngineKind`; ``params`` are the
41 C++ constructor's (``alpha/beta/kappa``, ``n_particles/threshold/seed``,
42 ``n_members/inflation/seed``)."""
43 name = kind.name.lower() if isinstance(kind, KalmanEngineKind) else str(kind).lower()
44 if name not in core.ENGINE_KINDS:
45 raise ValueError(f"unknown native engine kind {kind!r}; choose from {sorted(core.ENGINE_KINDS)}.")
46 self.kind_name = name
47 self.KINDKINDKIND = KIND_BY_NAME[name]
48 self.params = dict(params)

83 def _predict(self, model, x, P, u=None, t=None, **kw):
84 mh = self._native(model)
85 self._ensure_reset(model)
86 n = model.n
87 xf = np.ascontiguousarray(np.asarray(x, dtype=np.float64).reshape(-1))
88 Pf = np.ascontiguousarray(np.asarray(P, dtype=np.float64).reshape(n, n))
89 uf = None
90 if u is not None and model.l > 0:
91 uf = np.ascontiguousarray(np.asarray(u, dtype=np.float64).reshape(-1))
92 xo, Po, Fo = np.empty(n), np.empty((n, n)), np.empty((n, n))
93 core.check(core.lib.kalman_engine_predict(
94 self._handle, mh, core.cdptr(xf), core.cdptr(Pf),
95 core.cdptr(uf) if uf is not None else core.ffi.NULL, self._t(t),
96 core.dptr(xo), core.dptr(Po), core.dptr(Fo)), "kalman_engine_predict")
97 return xo.reshape(-1, 1), Po, Fo

99 def _update(self, model, x, P, z, t=None, **kw):
100 mh = self._native(model)
101 self._ensure_reset(model)
102 n, m = model.n, model.m
103 xf = np.ascontiguousarray(np.asarray(x, dtype=np.float64).reshape(-1))
104 Pf = np.ascontiguousarray(np.asarray(P, dtype=np.float64).reshape(n, n))
105 zf = np.ascontiguousarray(np.asarray(z, dtype=np.float64).reshape(-1))
106 if zf.size != m:
107 raise ValueError(f"z has {zf.size} channels, expected {m}.")
108 xo, Po = np.empty(n), np.empty((n, n))
109 core.check(core.lib.kalman_engine_update(
110 self._handle, mh, core.cdptr(xf), core.cdptr(Pf), core.cdptr(zf), self._t(t),
111 core.dptr(xo), core.dptr(Po)), "kalman_engine_update")
112 return xo.reshape(-1, 1), Po

114 def _smooth(self, model, post_k, prior_next, smooth_next, F_k):
115 n = model.n
116 c = lambda a, shape: np.ascontiguousarray(np.asarray(a, dtype=np.float64).reshape(shape))
117 x, P = post_k
118 xp, Pp = prior_next
119 xs, Ps = smooth_next
120 xo, Po = np.empty(n), np.empty((n, n))
121 core.check(core.lib.kalman_engine_smooth(
122 self._handle, n, core.cdptr(c(x, -1)), core.cdptr(c(P, (n, n))),
123 core.cdptr(c(xp, -1)), core.cdptr(c(Pp, (n, n))),
124 core.cdptr(c(xs, -1)), core.cdptr(c(Ps, (n, n))), core.cdptr(c(F_k, (n, n))),
125 core.dptr(xo), core.dptr(Po)), "kalman_engine_smooth")
126 return xo.reshape(-1, 1), Po
