8class BoostKalmanController(KalmanController):
9 """When the filter is surprised (NIS >> m), shrink R so K grows.
10
11 Monitors Normalised Innovation Squared over a sliding window.
12 When NIS exceeds ``threshold × m``, R is scaled down by
13 ``boost_factor``, causing K to grow (trust measurements more).
14 When NIS is nominal, R returns to its original value.
15
16 Integrates with ``KalmanFilter.forward(uk=boost)`` — S is computed
17 internally from model.H and model.R so no manual step-by-step loop
18 is needed.
19
20 Note: ``model._R`` mutation is effective only for leaf
21 ``LinearKalmanModel`` instances. For composite window models (e.g.
22 ``OscillatorModel``) R is bound as a closure and ``_R`` is ignored;
23 use ``AdaptiveQRKalmanController`` for composite models.
24
25 Usage::
26
27 boost = BoostKalmanController(model, boost_factor=3.0, threshold=2.0)
28 record = filter.forward(zk, uk=boost)
29
30 With libKalman installed the NIS window and the decision are computed by
31 ``Kalman::Controllers::Boost`` (reading the model's ``H`` and its current ``R``
32 through callbacks); the factor it decides is applied here, to ``model._R``,
33 exactly as the Python reference (:meth:`_reference_update`) does. One
34 difference: an innovation with missing (NaN) channels is scored on the
35 channels present in C++, where the reference's NIS is NaN (and never boosts).
36 """

70 def _reference_update(self, x_hat, P, t, *, nu=None):
71 if nu is None:
72 return None
73 if self._R_nominal is None:
74 self._R_nominal = self.model.R(t=t).copy()
75
76 H = self.model.H(x=x_hat, t=t)
77 S = H @ P @ H.T + self.model.R(t=t)
78 nis = float((nu.T @ np.linalg.pinv(S) @ nu).item())
79 self._nis_history.append(nis)
80
81 recent = self._nis_history[-self.window:]
82 nis_avg = np.mean(recent)
83 expected = int(nu.shape[0])
84
85 scale = 1.0 / self.boost_factor if nis_avg > self.threshold * expected else 1.0
86 self.model._R = scale * self._R_nominal
87 self.scale_log.append((t, scale))
88 return None
