21 Matrix Cholesky(const Matrix &S)
22 {
23 const std::size_t n = S.Rows();
24 Matrix L(n, n, 0.);
25 for (std::size_t i = 0; i < n; ++i)
26 for (std::size_t j = 0; j <= i; ++j)
27 {
28 double sum = S(i, j);
29 for (std::size_t k = 0; k < j; ++k) sum -= L(i, k) * L(j, k);
30 if (i == j)
31 {
32 if (!(sum > 0.)) return Matrix();
33 L(i, j) = std::sqrt(sum);
34 }
35 else
36 {
37 L(i, j) = sum / L(j, j);
38 }
39 }
40 return L;
41 }

45 Matrix InverseFromCholesky(const Matrix &L)
46 {
47 const std::size_t n = L.Rows();
48 Matrix inv(n, n, 0.);
49 for (std::size_t c = 0; c < n; ++c)
50 {
51 Vector y(n, 0.), x(n, 0.);
52 for (std::size_t i = 0; i < n; ++i) // forward substitution
53 {
54 double sum = (i == c) ? 1. : 0.;
55 for (std::size_t k = 0; k < i; ++k) sum -= L(i, k) * y(k);
56 y(i) = sum / L(i, i);
57 }
58 for (std::size_t ii = n; ii-- > 0;) // back substitution
59 {
60 double sum = y(ii);
61 for (std::size_t k = ii + 1; k < n; ++k) sum -= L(k, ii) * x(k);
62 x(ii) = sum / L(ii, ii);
63 }
64 for (std::size_t i = 0; i < n; ++i) inv(i, c) = x(i);
65 }
66 return inv;
67 }

70 Matrix Inverse(const Matrix &S, bool *ok)
71 {
72 const std::size_t n = S.Rows();
73 const Matrix sym = SymmetricPart(S);
74
75 // Cholesky is both the cheapest route and the one that preserves the symmetry
76 // we expect of a covariance, so it is tried first and unmodified.
77 if (const Matrix L = Cholesky(sym); !L.Empty())
78 {
79 if (ok) *ok = true;
80 return InverseFromCholesky(L);
81 }
82
83 // Not positive definite: build the Moore-Penrose pseudo-inverse from the
84 // eigendecomposition, dropping the null directions instead of dividing by
85 // (nearly) zero. This is what numpy.linalg.pinv does on the Python side.
86 //
87 // Note there is deliberately NO Tikhonov ladder here, unlike
88 // CholeskyFactor(). Ridging a singular innovation covariance would let the
89 // Cholesky succeed on `S + 1e-12 I` and hand back an inverse of order 1e12
90 // -- a colossal Kalman gain conjured out of a direction the measurement
91 // says nothing about. Dropping the direction is the honest answer.
92 if (ok) *ok = false;
93
94 Vector values;
95 Matrix vectors;
96 SymmetricEigen(sym, values, vectors);
97
98 double vmax = 0.;
99 for (std::size_t i = 0; i < n; ++i) vmax = std::max(vmax, std::abs(values(i)));
100 const double tol = vmax * static_cast<double>(n) * 1e-15;
101
102 Matrix inv(n, n, 0.);
103 for (std::size_t k = 0; k < n; ++k)
104 {
105 if (std::abs(values(k)) <= tol) continue;
106 const double w = 1. / values(k);
107 for (std::size_t i = 0; i < n; ++i)
108 for (std::size_t j = 0; j < n; ++j)
109 inv(i, j) += w * vectors(i, k) * vectors(j, k);
110 }
111 return inv;
112 }

114 Vector Solve(const Matrix &S, const Vector &b)
115 {
116 // L L^T x = b by two substitutions; the inverse is never formed. A matrix
117 // that is not positive-definite goes to Inverse(), which degrades gracefully.
118 const std::size_t n = S.Rows();
119 const Matrix L = Cholesky(SymmetricPart(S));
120 if (L.Empty()) return Inverse(S) * b;
121
122 Vector y(n, 0.), x(n, 0.);
123 for (std::size_t i = 0; i < n; ++i)
124 {
125 double sum = b(i);
126 for (std::size_t k = 0; k < i; ++k) sum -= L(i, k) * y(k);
127 y(i) = sum / L(i, i);
128 }
129 for (std::size_t ii = n; ii-- > 0;)
130 {
131 double sum = y(ii);
132 for (std::size_t k = ii + 1; k < n; ++k) sum -= L(k, ii) * x(k);
133 x(ii) = sum / L(ii, ii);
134 }
135 return x;
136 }

138 Matrix CholeskyFactor(const Matrix &S, bool jitter)
139 {
140 const std::size_t n = S.Rows();
141 const Matrix sym = SymmetricPart(S);
142
143 const int attempts = jitter ? 12 : 1;
144 for (int attempt = 0; attempt < attempts; ++attempt)
145 {
146 Matrix reg = sym;
147 if (attempt > 0)
148 {
149 const double ridge = std::pow(10., -18. + attempt);
150 for (std::size_t i = 0; i < n; ++i) reg(i, i) += ridge;
151 }
152 const Matrix L = Cholesky(reg);
153 if (!L.Empty()) return L;
154 }
155 return Matrix();
156 }
