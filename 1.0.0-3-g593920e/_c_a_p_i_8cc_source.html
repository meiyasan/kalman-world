73 // One error slot and a small ring of string slots per thread. The ring is
74 // what lets kalman_factory_describe() hand back five strings at once without
75 // any of them overwriting another before the caller has read them.
76 thread_local std::string tError;
77 thread_local std::string tStrings[8];
78 thread_local std::size_t tNext = 0;

80 const char *Keep(const std::string &s)
81 {
82 std::string &slot = tStrings[tNext++ % 8];
83 slot = s;
84 return slot.c_str();
85 }

161 {
162 if (!out) throw std::invalid_argument(std::string("null output buffer for ") + what);
163 if (M.Rows() != rows || M.Cols() != cols)
164 throw std::runtime_error(std::string(what) + " has shape (" + std::to_string(M.Rows()) + ", "
165 + std::to_string(M.Cols()) + "), expected (" + std::to_string(rows)
166 + ", " + std::to_string(cols) + ")");
167 std::memcpy(out, M.Data(), rows * cols * sizeof(double));

557{
558 return Guard([&] {

561 if (steps <= 0 || !data) throw std::invalid_argument("a table needs at least one matrix");

563 const std::size_t cols = (w == "R") ? M.NMeasure() : M.NState();
564 std::vector<Kalman::Matrix> table;
565 for (int k = 0; k < steps; ++k)
566 table.push_back(Mat(data + static_cast<std::size_t>(k) * rows * cols, rows, cols, w.c_str()));
567 M.SetTable(w, std::move(table));
568 return 0;
569 });

1099 class FunctionController : public Kalman::Controller
1100 {
1101 kalman_control_fn fPolicy;
1102 void *fUser;
1103 std::size_t fM;
1104
1105 public:
1106 FunctionController(Kalman::Model *model, kalman_control_fn policy, void *user)
1107 : Kalman::Controller(model), fPolicy(policy), fUser(user), fM(model->NMeasure()) {}
1108
1109 const char *Title() const override { return "Function controller"; }
1110
1111 Kalman::Vector Policy(const Kalman::Vector &x, const Kalman::Matrix &P, double t,
1112 const Kalman::Vector *nu) override
1113 {
1114 const std::size_t n = x.Size(), l = NControl();
1115 std::vector<double> xs(n), Ps(n * n), nus(fM, 0.), us(l, 0.);
1116 for (std::size_t i = 0; i < n; ++i) xs[i] = x(i);
1117 for (std::size_t i = 0; i < n; ++i)
1118 for (std::size_t j = 0; j < n; ++j) Ps[i * n + j] = P(i, j);
1119 if (nu)
1120 for (std::size_t i = 0; i < fM && i < nu->Size(); ++i) nus[i] = (*nu)(i);
1121 if (fPolicy(fUser, xs.data(), Ps.data(), t, nus.data(), us.data()) != 0)
1122 throw std::runtime_error("the control function failed");
1123 Kalman::Vector u(l, 0.);
1124 for (std::size_t i = 0; i < l; ++i) u(i) = us[i];
1125 return u;
1126 }
1127 };

1129 int RunFilter(const kalman_model_t *model, kalman_engine_t *engine, int N,
1130 const double *zk, const double *uk, kalman_control_fn control, void *user, int smooth,
1131 double *x_prior, double *P_prior, double *x_post, double *P_post,
1132 double *x_smooth, double *P_smooth, double *innovations, double *transitions,
1133 double *controls)
1134 {
1135 if (N <= 0) throw std::invalid_argument("N must be positive");
1136 if (!zk) throw std::invalid_argument("null zk");
1137 // The filter mutates nothing in the model, but its API takes a pointer.
1138 Kalman::Model &M = const_cast<Kalman::Model &>(Deref(model));
1139 Kalman::Engine &E = Deref(engine);
1140 const std::size_t n = M.NState(), m = M.NMeasure(), l = M.NControl();
1141 const std::size_t steps = static_cast<std::size_t>(N);
1142 // A model with no control input (l = 0) may still run a closed loop: its
1143 // controller commands nothing and only retunes the model (adaptive Q and R).
1144
1145 std::vector<Kalman::Vector> z(steps), u;
1146 for (std::size_t k = 0; k < steps; ++k) z[k] = Vec(zk + k * m, m, "zk");
1147 if (!control && uk && l > 0)
1148 {
1149 u.resize(steps);
1150 for (std::size_t k = 0; k < steps; ++k) u[k] = Vec(uk + k * l, l, "uk");
1151 }
1152
1153 Kalman::Filter filter(&M, &E);
1154 if (control)
1155 {
1156 FunctionController controller(&M, control, user);
1157 filter.Forward(z, controller);
1158 }
1159 else filter.Forward(z, u.empty() ? nullptr : &u);
1160 if (smooth) filter.Backward();
1161
1162 for (std::size_t k = 0; k < steps; ++k)
1163 {
1164 if (x_prior) Copy(filter.PriorStates()[k], x_prior + k * n, n, "x_prior");
1165 if (P_prior) Copy(filter.PriorCovariances()[k], P_prior + k * n * n, n, n, "P_prior");
1166 if (x_post) Copy(filter.PosteriorStates()[k], x_post + k * n, n, "x_post");
1167 if (P_post) Copy(filter.PosteriorCovariances()[k], P_post + k * n * n, n, n, "P_post");
1168 if (innovations) Copy(filter.Innovations()[k], innovations + k * m, m, "innovations");
1169 if (transitions) Copy(filter.Transitions()[k], transitions + k * n * n, n, n, "transitions");
1170 if (smooth && x_smooth) Copy(filter.SmoothedStates()[k], x_smooth + k * n, n, "x_smooth");
1171 if (smooth && P_smooth) Copy(filter.SmoothedCovariances()[k], P_smooth + k * n * n, n, n, "P_smooth");
1172 if (controls && control && l > 0) Copy(filter.Controls()[k], controls + k * l, l, "controls");
1173 }
1174 return 0;
1175 }
1176}

1262 double nu, double beta, const double *forcing_re, const double *forcing_im,
1263 int integrator, const double *w, int batch, int nsteps, double *out)
1264{

1310int kalman_propagate(const kalman_model_t *model, kalman_engine_t *engine, int N,
1311 const double *x0, const double *P0, double t0, double fs,
1312 const double *uk, const double *wk, double *xk, double *Pk)
1313{
1314 return Guard([&] {
1315 if (N <= 0 || !xk) throw std::invalid_argument("bad arguments");
1316 if (!(fs > 0.)) throw std::invalid_argument("the sample rate must be positive");
1317 const Kalman::Model &M = Deref(model);
1318 Kalman::Engine &E = Deref(engine);
1319 const std::size_t n = M.NState(), l = M.NControl();
1320
1321 Kalman::Vector x = Vec(x0, n, "x0");
1322 Kalman::Matrix P = Mat(P0, n, n, "P0");
1323 for (int k = 0; k < N; ++k)
1324 {
1325 const double t = t0 + static_cast<double>(k) / fs;
1326 Kalman::Vector u;
1327 if (uk && l > 0) u = Vec(uk + static_cast<std::size_t>(k) * l, l, "uk");
1328
1329 Kalman::Vector xNext;
1330 Kalman::Matrix PNext, F;
1331 E.Predict(M, x, P, (uk && l > 0) ? &u : nullptr, t, xNext, PNext, F);
1332 if (wk)
1333 for (std::size_t i = 0; i < n; ++i) xNext(i) += wk[static_cast<std::size_t>(k) * n + i];
1334
1335 x = xNext;
1336 P = PNext;
1337 Copy(x, xk + static_cast<std::size_t>(k) * n, n, "xk");
1338 if (Pk) Copy(P, Pk + static_cast<std::size_t>(k) * n * n, n, n, "Pk");
1339 }
1340 return 0;
1341 });
1342}

1344int kalman_jacobian(int kind, double step, kalman_map_fn fn, void *user, int n, int m,
1345 const double *x, const double *y0, double *J)
1346{
1347 return Guard([&] {
1348 if (!fn) throw std::invalid_argument("null map");
1349 if (n <= 0 || m <= 0 || !J) throw std::invalid_argument("bad arguments");
1350 const std::size_t nn = static_cast<std::size_t>(n), mm = static_cast<std::size_t>(m);
1351 std::vector<double> in(nn), out(mm);
1352 const Kalman::Map map = [&](const Kalman::Vector &point) {
1353 for (std::size_t i = 0; i < nn; ++i) in[i] = point(i);
1354 if (fn(user, in.data(), out.data()) != 0) throw std::runtime_error("the map failed");
1355 Kalman::Vector y(mm);
1356 for (std::size_t i = 0; i < mm; ++i) y(i) = out[i];
1357 return y;
1358 };
1359 const Kalman::Vector base = y0 ? Vec(y0, mm, "y0") : Kalman::Vector();
1360 const Kalman::Matrix jac = Kalman::Differentiators::Jacobian(
1361 static_cast<Kalman::DifferentiatorKind>(kind), map, Vec(x, nn, "x"), step, y0 ? &base : nullptr);
1362 Copy(jac, J, mm, nn, "J");
1363 return 0;
1364 });
1365}

1380int kalman_integrate(int kind, double rtol, double atol, int max_substeps,
1381 kalman_derivative_fn f, void *user, int n,
1382 const double *x, double t, double dt, int nsteps, double *out)
1383{
1384 return Guard([&] {
1385 if (!f) throw std::invalid_argument("null derivative function");
1386 if (n <= 0 || nsteps < 0 || !x || !out) throw std::invalid_argument("bad arguments");
1387 const std::size_t size = static_cast<std::size_t>(n);
1388
1389 std::unique_ptr<Kalman::Integrator> integrator;
1390 if (kind == static_cast<int>(Kalman::IntegratorKind::DormandPrince))
1391 integrator = std::make_unique<Kalman::Integrators::DormandPrince>(
1392 rtol > 0. ? rtol : 1e-6, atol > 0. ? atol : 1e-9,
1393 max_substeps > 0 ? static_cast<std::size_t>(max_substeps) : 10000);
1394 else
1395 integrator = Kalman::Integrators::Make(static_cast<Kalman::IntegratorKind>(kind));
1396
1397 std::vector<double> in(size), dx(size);
1398 const Kalman::Derivative derivative = [&](const Kalman::Vector &state, const Kalman::Vector &, double time) {
1399 for (std::size_t i = 0; i < size; ++i) in[i] = state(i);
1400 if (f(user, in.data(), time, dx.data()) != 0)
1401 throw std::runtime_error("the derivative function failed");
1402 Kalman::Vector d(size);
1403 for (std::size_t i = 0; i < size; ++i) d(i) = dx[i];
1404 return d;
1405 };
1406
1407 const Kalman::Vector result = integrator->Integrate(derivative, Vec(x, size, "x"), Kalman::Vector(), t, dt,
1408 static_cast<std::size_t>(nsteps));
1409 Copy(result, out, size, "out");
1410 return 0;
1411 });
1412}

1791kalman_tracks_t *kalman_ckf_find(const kalman_model_t *model, int n_layers, const double *radii,
1792 const int *hits_per_layer, const double *hits,
1793 int n_seeds, const int *seed_layer, const int *seed_index,
1794 const double *seed_x, const double *seed_P,
1795 double chi2_cut, double hole_penalty, int max_candidates,
1796 int max_holes, int min_hits, double shared_hit_frac)
1797{
1798 try
1799 {
1800 Ok();
1801 const Kalman::Model &M = Deref(model);
1802 const std::size_t n = M.NState(), m = M.NMeasure();
1803 if (n_layers <= 0 || !radii || !hits_per_layer) throw std::invalid_argument("no layers given");
1804 if (n_seeds < 0 || (n_seeds > 0 && (!seed_layer || !seed_index || !seed_x || !seed_P)))
1805 throw std::invalid_argument("incomplete seeds");
1806
1807 Kalman::Tracking::Event event(static_cast<std::size_t>(n_layers));
1808 std::vector<double> rr(radii, radii + n_layers);
1809 const double *cursor = hits;
1810 for (int l = 0; l < n_layers; ++l)
1811 for (int j = 0; j < hits_per_layer[l]; ++j, cursor += m)
1812 event[static_cast<std::size_t>(l)].push_back(Vec(cursor, m, "hits"));
1813
1814 const Kalman::Matrix I = Kalman::Matrix::Identity(n);
1815 std::vector<Kalman::Tracking::Candidate> seeds(static_cast<std::size_t>(n_seeds));
1816 for (int s = 0; s < n_seeds; ++s)
1817 {
1818 const int l = seed_layer[s], j = seed_index[s];
1819 if (l < 0 || l >= n_layers || j < 0 || j >= hits_per_layer[l])
1820 throw std::out_of_range("a seed points at a hit that does not exist");
1821 Kalman::Tracking::Candidate &c = seeds[static_cast<std::size_t>(s)];
1822 c.x = Vec(seed_x + static_cast<std::size_t>(s) * n, n, "seed_x");
1823 c.P = Mat(seed_P + static_cast<std::size_t>(s) * n * n, n, n, "seed_P");
1824 c.hitIds.push_back({static_cast<std::size_t>(l), static_cast<std::size_t>(j)});
1825 c.radii.push_back(rr[static_cast<std::size_t>(l)]);
1826 c.xk.push_back(c.x);
1827 c.Pk.push_back(c.P);
1828 c.zk.push_back(event[static_cast<std::size_t>(l)][static_cast<std::size_t>(j)]);
1829 c.priorX.push_back(c.x);
1830 c.priorP.push_back(c.P);
1831 c.Fc.push_back(I);
1832 c.Faccum = I;
1833 }
1834
1835 const Kalman::Engines::Combinatorial engine(
1836 chi2_cut, static_cast<std::size_t>(max_candidates), static_cast<std::size_t>(max_holes),
1837 static_cast<std::size_t>(min_hits), shared_hit_frac, hole_penalty);
1838 auto out = std::make_unique<kalman_tracks>();
1839 out->n = n;
1840 out->m = m;
1841 out->tracks = engine.Find(M, event, rr, &seeds);
1842 return out.release();
1843 }
1844 catch (const std::exception &e) { Fail(e.what()); return nullptr; }
1845 catch (...) { Fail("unknown C++ exception"); return nullptr; }
1846}
