22def wire_inharmonicity(E: float, d: float, L: float, T: float) -> float:
23 """
24 Inharmonicity coefficient B for a stiff wire under tension (Fletcher 1964).
25
26 Predicts partial frequencies: ``f_n = n·f0·√(1 + B·n²)``.
27
28 Parameters
29 ----------
30 E : Young's modulus [Pa] (steel ≈ 2e11 Pa, tungsten ≈ 4e11 Pa)
31 d : wire diameter [m]
32 L : wire length [m]
33 T : wire tension [N]
34
35 Returns
36 -------
37 B : inharmonicity coefficient (dimensionless)
38 """
39 return np.pi**3 * E * d**4 / (64.0 * T * L**2)

42def wire_fundamental(d: float, L: float, T: float, rho: float) -> float:
43 """
44 Fundamental transverse vibration frequency of a wire under tension.
45
46 ``f0 = (1/2L) · √(T/μ)``, ``μ = ρ·π·d²/4`` (linear mass density).
47
48 Parameters
49 ----------
50 d : wire diameter [m]
51 L : wire length [m]
52 T : wire tension [N]
53 rho : material density [kg/m³] (steel ≈ 7800, tungsten ≈ 19300)
54
55 Returns
56 -------
57 f0 : fundamental frequency [Hz]
58 """
59 mu = rho * np.pi * d**2 / 4.0 # linear mass density [kg/m]
60 return (1.0 / (2.0 * L)) * np.sqrt(T / mu)

66def partial_frequencies(f0: float, B: float, n_harmonics: int) -> np.ndarray:
67 """
68 Fletcher (in)harmonic partial series: ``f_n = n·f0·√(1 + B·n²)``.
69
70 ``B = 0`` gives the perfectly harmonic series ``f_n = n·f0``. For a stiff wire
71 under tension, B can be derived from physical parameters via
72 :func:`wire_inharmonicity`. A typical KAGRA suspension wire has B ≈ 0.1.
73
74 Parameters
75 ----------
76 f0 : fundamental frequency [Hz]
77 B : inharmonicity coefficient (dimensionless, ≥ 0)
78 n_harmonics: number of partials to compute
79
80 Returns
81 -------
82 freqs : (n_harmonics,) array of partial frequencies [Hz]
83 """
84 n = np.arange(1, n_harmonics + 1, dtype=float)
85 return n * f0 * np.sqrt(1.0 + B * n ** 2)
