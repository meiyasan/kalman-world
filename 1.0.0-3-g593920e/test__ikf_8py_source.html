16def _two_models(q_slow=0.1, q_fast=2.0):
17 """Two 1-D random-walk models with very different process noise."""
18 def _m(name, q):
19 return KalmanModel(name,
20 F=np.eye(1), H=np.eye(1),
21 Q=np.array([[q ** 2]]),
22 R=np.array([[1.0 ** 2]]),
23 fs=1.0,
24 x0=np.array([[0.0]]))
25 return _m("slow", q_slow), _m("fast", q_fast)

28def _sim(regime_seq, q_slow=0.1, q_fast=2.0, obs_std=1.0, seed=0):
29 """Simulate a 1-D switching process and noisy observations."""
30 rng = np.random.default_rng(seed)
31 x = 0.0
32 zk, labels = [], []
33 for re in regime_seq:
34 x += rng.normal(0, q_slow if re == 0 else q_fast)
35 zk.append([x + rng.normal(0, obs_std)])
36 labels.append(re)
37 return np.array(zk), np.array(labels, dtype=int)

157class TestRoutingAccuracy(unittest.TestCase):
158 """Routing accuracy on the 2-model Gaussian benchmark."""
159
160 def _run(self, router, epochs=200, n_train=3, seed=42):
161 m0, m1 = _two_models()
162 ikf = InteractiveKalmanFilter([m0, m1], PI, router=router)
163
164 train_zk, train_lbl = [], []
165 for s in range(n_train):
166 zk, lbl = _sim(REGIME_200, seed=s)
167 train_zk.append(zk); train_lbl.append(lbl)
168
169 ikf.fit(train_zk, train_lbl, epochs=epochs, seed=seed)
170
171 zk_test, lbl_test = _sim(REGIME_200_TEST, seed=99)
172 ikf.forward(zk_test)
173 mu = ikf.mode_probabilities
174 acc = float((np.argmax(mu, axis=1) == lbl_test).mean())
175 return acc
176
177 def test_bayesian_accuracy_high(self):
178 # Classical IKF should reach ≥85% on clean Gaussian data
179 acc = self._run(BayesianRouter())
180 self.assertGreater(acc, 0.85, f"Bayesian accuracy {acc:.1%} too low")
181
182 def test_threshold_accuracy_reasonable(self):
183 acc = self._run(ThresholdRouter(m=1, window=5, soft=True))
184 self.assertGreater(acc, 0.70, f"Threshold accuracy {acc:.1%} too low")
185
186 def test_neural_accuracy_reasonable(self):
187 # After 3 training sequences, MLP should reach ≥70%
188 acc = self._run(NeuralRouter(MLP(hidden=16), window=5), epochs=300)
189 self.assertGreater(acc, 0.70, f"Neural accuracy {acc:.1%} too low")
190
191 def test_hybrid_close_to_bayesian(self):
192 r_bay = BayesianRouter()
193 r_mlp = NeuralRouter(MLP(hidden=8), window=5)
194 acc = self._run(HybridRouter(r_bay, r_mlp, alpha=0.3), epochs=200)
