25class TestJacobianAndShapes(unittest.TestCase):
26
27 def test_numerical_jacobian_linear_map(self):
28 """For a linear map f(x)=Ax the numerical Jacobian must recover A."""
29 A = np.array([[2.0, 1.0], [0.0, 3.0]])
30 J = numerical_jacobian(lambda x: A @ x, np.array([1.0, 2.0]))
31 np.testing.assert_allclose(J, A, atol=1e-5)
32
33 def test_numerical_jacobian_nonlinear(self):
34 """d/dx [x0^2, sin(x1)] = diag(2 x0, cos x1)."""
35 x = np.array([1.5, 0.3])
36 J = numerical_jacobian(lambda v: np.array([[v[0, 0] ** 2], [np.sin(v[1, 0])]]), x)
37 np.testing.assert_allclose(J, np.diag([2 * x[0], np.cos(x[1])]), atol=1e-5)
38
39 def test_as_col_promotes_1d(self):

123 def test_bandpass_preserves_length_and_passband(self):
124 fs, N = 256, 1024
125 t = np.arange(N) / fs
126 x = np.sin(2 * np.pi * 40 * t) + np.sin(2 * np.pi * 5 * t)
127 y = bandpass_filter(x, fs, fc=40, bandwidth=10)
128 self.assertEqual(y.shape, (N,))
129 # the 40 Hz component survives; the 5 Hz component is strongly attenuated
130 burn = slice(N // 4, 3 * N // 4)
131 self.assertGreater(np.std(y[burn]), 0.3)
132

138 def test_make_multirate_input_nan_pads_slow_sensors(self):
139 slow = np.arange(10, dtype=float).reshape(10, 1) # 10 Hz
140 fast = np.random.RandomState(1).randn(50, 2) # 50 Hz
141 zk = make_multirate_input([(slow, 10), (fast, 50)], base_fs=50)
142 self.assertEqual(zk.shape, (50, 3))
143 # slow sensor fires 10/50 of the time -> ~80% NaN in its column
144 self.assertAlmostEqual(np.isnan(zk[:, 0]).mean(), 0.8, delta=0.05)
145 # fast sensor fires every step -> no NaN
146 self.assertFalse(np.any(np.isnan(zk[:, 1:])))
