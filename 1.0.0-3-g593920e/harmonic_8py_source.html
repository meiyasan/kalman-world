16class InharmonicOscillatorModel(OscillatorModel):
17 """
18 One :class:`OscillatorModel` window whose candidate frequency list is an **inharmonic
19 ladder**: ``f_n = n·f0·√(1+B·n²)`` for ``n = 1 … n_harmonics``, generated from
20 ``(f0, B)``.
21
22 As any :class:`OscillatorModel` window it **masks** to ``[fc ± dfc]``: only partials
23 inside the window are tracked, even though the full ladder is generated. To track
24 the whole ladder, place one :class:`InharmonicOscillatorModel` per partial and
25 compose them in an :class:`OscillatorBankModel` (INDEPENDENT).
26
27 Physical parameters (Young's modulus, wire diameter, length, tension) can be
28 converted to ``(f0, B)`` via :func:`wire_fundamental` and :func:`wire_inharmonicity`.
29
30 Parameters
31 ----------
32 sensor_id : str
33 f0 : float
34 Fundamental frequency [Hz].
35 fc : float
36 Window center frequency [Hz] — which partial to track.
37 dfc : float
38 Window half-width [Hz].
39 n_harmonics : int
40 Number of partials to include in the candidate ladder.
41 B : float
42 Inharmonicity coefficient (``B = 0`` → harmonic, use :class:`HarmonicOscillatorModel`).
43 """
44
45 NAME = "Inharmonic Oscillator (windowed)"
46 DESCRIPTION = "Windowed oscillator over an inharmonic ladder f_n = n·f0·√(1+B·n²)."
47 VERSION = "3.1"

49 def __init__(self, sensor_id, f0, fc, dfc, *, n_harmonics=3, B=0.0,
