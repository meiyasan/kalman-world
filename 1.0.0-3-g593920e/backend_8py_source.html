1"""Where the MCP tools send their work: this process, or the REST API.
2
3 - :class:`LocalBackend` calls :mod:`kalman.api` in-process and checks each
4 scope here. Models and plugins are still shared through the registry file
5 (:mod:`kalman.api.store`), but runs stay in this process.
6 - :class:`RemoteBackend` (``KALMAN_API_URL`` set) forwards every call to the
7 REST API with the caller's bearer token. The REST API is then the one
8 authority: it checks the token, owns the registry and keeps the runs, so a
9 ``run_id`` from an agent works in Unity and the other way round. This
10 process needs no JWT secret and never loads native code.
11
12Both expose the same verbs, so :mod:`.server` does not care which one it has.
13"""

110class RemoteBackend:
111 kind = "remote"

113 def __init__(self, base_url: str, *, timeout: float | None = None, client=None):
114 if client is None:
115 try:
116 import httpx
117 except ImportError as e:
118 raise ImportError("KALMAN_API_URL needs httpx (pip install 'kalman[remote]').") from e
119 client = httpx.Client(base_url=base_url.rstrip("/"),
120 timeout=timeout or float(os.environ.get("KALMAN_API_TIMEOUT", "300")))
121 self.base_url = base_url.rstrip("/")
122 self._http = client
123
124 def _call(self, method, path, token, *, json=None, params=None):
125 headers = {"Authorization": f"Bearer {token}"} if token else {}
126 r = self._http.request(method, path, json=json, params=params, headers=headers)
127 if r.status_code >= 400:
128 try:
129 detail = r.json().get("detail", r.text)
130 except Exception:
131 detail = r.text
132 if r.status_code in (401, 403):
133 from ..api.auth import AuthError, InvalidToken
134 raise (InvalidToken if r.status_code == 401 else AuthError)(detail)
135 raise RemoteError(r.status_code, str(detail))
136 return r.json()
