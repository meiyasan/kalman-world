54 double Combinatorial::Chi2Quantile99(std::size_t dof)
55 {
56 // scipy.stats.chi2.ppf(0.99, dof) for dof = 1..10; beyond that the
57 // Wilson-Hilferty approximation, which is within 0.1% at these sizes.
58 static const double table[] = {6.6349, 9.2103, 11.3449, 13.2767, 15.0863,
59 16.8119, 18.4753, 20.0902, 21.6660, 23.2093};
60 if (dof == 0) return 9.2103;
61 if (dof <= 10) return table[dof - 1];
62 const double k = static_cast<double>(dof), z = 2.3263;
63 return k * std::pow(1. - 2. / (9. * k) + z * std::sqrt(2. / (9. * k)), 3);
64 }

88 GateContext Combinatorial::Context(const Model &model, const Matrix &Pk, const Vector &zpred,
89 const Matrix &H, const Matrix &R) const
90 {
91 GateContext ctx;
92 ctx.zpred = zpred;
93 ctx.H = H;
94 ctx.R = R;
95 Matrix S = Utils::Similarity(H, Pk);
96 S += R;
97 ctx.Sinv = Utils::Inverse(S);
98 ctx.K = Pk * H.Transpose() * ctx.Sinv;
99 ctx.P = Utils::JosephForm(ctx.K, H, Pk, R);
100 (void)model;
101 return ctx;
102 }

113 double Combinatorial::Gate(const Model &model, const Vector &xk, const Matrix &Pk,
114 const Tracking::Hit &z, double t, Vector &xOut, Matrix &POut) const
115 {
116 const std::vector<std::size_t> keep = Utils::ValidChannels(z);
117 if (keep.empty())
118 {
119 xOut = xk;
120 POut = Pk;
121 return std::numeric_limits<double>::infinity();
122 }
123
124 Vector zpred;
125 Matrix H;
126 Measure(model, xk, t, zpred, H);
127
128 const GateContext ctx = Context(model, Pk, Utils::Compress(zpred, keep),
129 Utils::CompressRows(H, keep), Utils::CompressBoth(model.R(t), keep));
130 const double chi2 = ctx.Score(xk, Utils::Compress(z, keep), xOut);
131 POut = ctx.P;
132 return chi2;
133 }

137 std::vector<Candidate> Combinatorial::Seed(const Model &model, const Event &hits,
138 const std::vector<double> &radii) const
139 {
140 const Tracking::Detector &detector = DetectorOf(model);
141 const Matrix I = Matrix::Identity(model.NState());
142
143 std::vector<Candidate> seeds;
144 for (std::size_t l = 0; l < hits.size(); ++l)
145 {
146 if (hits[l].empty()) continue;
147 const double r = radii.at(l);
148 for (std::size_t j = 0; j < hits[l].size(); ++j)
149 {
150 Candidate c;
151 detector.SeedState(hits[l][j], c.x, c.P);
152 c.hitIds.push_back(HitId{l, j});
153 c.radii.push_back(r);
154 c.xk.push_back(c.x);
155 c.Pk.push_back(c.P);
156 c.zk.push_back(hits[l][j]);
157 c.priorX.push_back(c.x);
158 c.priorP.push_back(c.P);
159 c.Fc.push_back(I);
160 c.Faccum = I;
161 seeds.push_back(std::move(c));
162 }
163 break; // seed only from the first populated layer
164 }
165 return seeds;
166 }

168 std::vector<Candidate> Combinatorial::Find(const Model &model, const Event &hits,
169 const std::vector<double> &radiiIn,
170 const std::vector<Candidate> *seedsIn) const
171 {
172 const std::vector<double> radii = radiiIn.empty() ? DetectorOf(model).Radii() : radiiIn;
173 if (radii.size() < hits.size())
174 throw std::runtime_error("Engines::Combinatorial: fewer radii than layers.");
175 for (std::size_t l = 0; l < hits.size(); ++l)
176 for (const Vector &z : hits[l])
177 if (z.Size() != model.NMeasure())
178 throw std::runtime_error("Engines::Combinatorial: a hit on layer " + std::to_string(l)
179 + " has " + std::to_string(z.Size()) + " channels, expected "
180 + std::to_string(model.NMeasure()) + ".");
181
182 const std::size_t L = hits.size();
183 const Matrix I = Matrix::Identity(model.NState());
184 const double cut = GateThreshold(model);
185 const double penalty = HoleCost(model);
186 const std::vector<Candidate> seeds = seedsIn ? *seedsIn : Seed(model, hits, radii);
187
188 std::vector<Candidate> seedTracks;
189 for (const Candidate &seed : seeds)
190 {
191 std::vector<Candidate> cands{seed};
192
193 // Follow outward from the seed's own layer, whichever that is — Seed()
194 // picks the innermost *populated* layer, and a caller may pass seeds of
195 // its own. Starting at a fixed layer 1 would re-scan the seeding layer.
196 for (std::size_t l = static_cast<std::size_t>(seed.SeedLayer() + 1); l < L; ++l)
197 {
198 const double r = radii[l];
199 const Tracking::Layer &layer = hits[l];
200 // NaN channels need the per-hit compression in Gate(); without them the
201 // whole layer shares one prepared gate.
202 const bool masked = !layer.empty() && HasNaN(layer);
203
204 std::vector<Candidate> grown;
205 for (const Candidate &c : cands)
206 {
207 Vector x;
208 Matrix P, F;
209 Predict(model, c.x, c.P, nullptr, r, x, P, F);
210 if (!AllFinite(x)) continue; // track left the detector
211
212 const Matrix Faccum = F * c.Faccum;
213 GateContext ctx;
214 if (!masked && !layer.empty()) ctx = MakeGateContext(model, x, P, r);
215
216 for (std::size_t j = 0; j < layer.size(); ++j)
217 {
218 Vector xk;
219 Matrix Pk;
220 double chi2;
221 if (masked) chi2 = Gate(model, x, P, layer[j], r, xk, Pk);
222 else { chi2 = ctx.Score(x, layer[j], xk); Pk = ctx.P; }
223 if (chi2 < cut)
224 {
225 Candidate nc = c;
226 nc.x = xk; nc.P = Pk;
227 nc.chi2 += chi2; nc.score += chi2;
228 nc.hitIds.push_back(HitId{l, j});
229 nc.radii.push_back(r);
230 nc.xk.push_back(xk); nc.Pk.push_back(Pk);
231 nc.zk.push_back(layer[j]);
232 nc.priorX.push_back(x); nc.priorP.push_back(P);
233 nc.Fc.push_back(Faccum);
234 nc.Faccum = I;
235 grown.push_back(std::move(nc));
236 }
237 }
238 // The hole branch: the track survives a missed layer.
239 if (c.holes < fMaxHoles)
240 {
241 Candidate nc = c;
242 nc.x = x; nc.P = P;
243 ++nc.holes; nc.score += penalty;
244 nc.Faccum = Faccum;
245 grown.push_back(std::move(nc));
246 }
247 }
248 if (grown.empty()) break;
249 std::stable_sort(grown.begin(), grown.end(),
250 [](const Candidate &a, const Candidate &b) { return a.score < b.score; });
251 if (grown.size() > fMaxCandidates) grown.resize(fMaxCandidates);
252 cands.swap(grown);
253 }
254
255 const Candidate *best = nullptr;
256 for (const Candidate &c : cands)
257 if (c.NHits() >= fMinHits && (!best || c.Quality() < best->Quality())) best = &c;
258 if (best) seedTracks.push_back(*best);
259 }
260
261 return Arbitrate(std::move(seedTracks));
262 }

264 std::vector<Candidate> Combinatorial::Arbitrate(std::vector<Candidate> tracks) const
265 {
266 std::stable_sort(tracks.begin(), tracks.end(),
267 [](const Candidate &a, const Candidate &b) { return a.Quality() < b.Quality(); });
268 std::vector<Candidate> accepted;
269 std::set<HitId> used;
270 for (Candidate &c : tracks)
271 {
272 std::set<HitId> ids(c.hitIds.begin(), c.hitIds.end());
273 std::size_t shared = 0;
274 for (const HitId &id : ids) shared += used.count(id);
275 if (static_cast<double>(shared) <= fSharedHitFrac * static_cast<double>(ids.size()))
276 {
277 used.insert(ids.begin(), ids.end());
278 accepted.push_back(std::move(c));
279 }
280 }
281 return accepted;
282 }

284 std::vector<Candidate> Combinatorial::SmoothTracks(const Model &, const std::vector<Candidate> &tracks) const
285 {
286 std::vector<Candidate> smoothed;
287 smoothed.reserve(tracks.size());
288 for (const Candidate &c : tracks)
289 {
290 const std::size_t N = c.NHits();
291 Candidate out = c;
292 if (N == 0) { smoothed.push_back(out); continue; }
293 for (std::size_t k = N - 1; k-- > 0;)
294 {
295 Vector xs;
296 Matrix Ps;
297 Smooth(c.xk[k], c.Pk[k], c.priorX[k + 1], c.priorP[k + 1],
298 out.xk[k + 1], out.Pk[k + 1], c.Fc[k + 1], xs, Ps);
299 out.xk[k] = xs;
300 out.Pk[k] = Ps;
301 }
302 smoothed.push_back(std::move(out));
303 }
304 return smoothed;
305 }
