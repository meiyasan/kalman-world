10class KalmanSampler:
11 """
12 Generate synthetic trajectories and measurements from a **model**:
13 - propagate the state with the model's dynamics (via an engine's predict step),
14 - draw process/measurement noise from model.Q / model.R,
15 - emit measurements through model.h.
16
17 Sampling is a model capability (it needs only the system + an engine's predict);
18 by default it uses ``model.default_engine()``. A filter may pass its own engine
19 (e.g. the UKF engine) for consistency.
20 """
21
22 def __init__(self, model, engine=None):
23 self.model = model
24 self._engine = engine if engine is not None else model.default_engine()
25 self.reset() # initialise the running cursor (time + state)

27 def reset(self):
28 """
29 Rewind the sampler stream to the model's initial time/state.
30
31 A ``KalmanSampler`` is *stateful*: successive ``measurement()`` calls on the
32 same instance **resume** from the last sample (continuous time and state), so
33 you can build a long trajectory in chunks. Call ``reset()`` (alias ``clear()``)
34 to start a fresh, independent stream from ``t0`` / ``x0`` / ``P0``.
35 (``model.sample()`` always uses a fresh sampler, so it is unaffected.)
36 """
37 self._t = float(self.model.t0) if self.model.t0 is not None else 0.0
38 x0 = self.model.get_initial_state_vector()
39 P0 = self.model.get_initial_state_covariance()
40 self._x = None if x0 is None else np.array(x0, dtype=float, copy=True)
41 self._P = None if P0 is None else np.array(P0, dtype=float, copy=True)
42 return self
43
44 clear = reset

68 def measurement(self, duration: float, uk=None, **kwargs):
69 '''
70 Generate synthetic measurements for a given duration.
71 Returns a KalmanRecord with the sampled truth (xk) and measurements (zk).
72
73 Each step routes through ``engine._predict`` (so ``Pk`` carries the
74 analytic prior covariance band, ``F P Fᵀ + Q``) and then injects an
75 explicit process-noise draw into the mean, so the returned ``xk`` is a
76 genuine stochastic realisation of the system (it actually drifts under
77 ``Q``) while ``Pk`` still reflects the model's analytic uncertainty growth.
78 '''
79
80 # seed=: the same run every time. It was accepted and ignored before.
81 seed = kwargs.pop("seed", None)
82 if seed is not None:
83 np.random.seed(int(seed))
84
85 inprocess = kwargs.pop("inprocess", True)
86 postprocess = kwargs.pop("postprocess", True)
87 indexing = kwargs.pop("indexing", "xy")
88
89 measurement_noise = kwargs.pop("measurement_noise", True)
90 process_noise = kwargs.pop("process_noise", True)
91
92 n_samples = int(duration * self.fsfs)
93 zk = np.zeros((n_samples, self.m, 1))
94 xk = np.zeros((n_samples, self.n, 1))
95 Pk = np.zeros((n_samples, self.n, self.n))
96
97 # Resume from the running cursor unless an explicit x0/P0 is supplied.
98 x = kwargs.pop("x0", None); x = self._x if x is None else x
99 P = kwargs.pop("P0", None); P = self._P if P is None else P
100 if x is None or x.shape != (self.n, 1):
101 raise ValueError(f"Initial state vector has incorrect dimensions. Expected {self.n}x1, got {format_shape(x)}.")
102 if P is None or P.shape != (self.n, self.n):
103 raise ValueError(f"Initial state covariance has incorrect dimensions. Expected {self.n}x{self.n}, got {format_shape(P)}.")
104
105 uk = uk if uk is not None else [self.model.u0] * n_samples
106 wk = self.process_noise(duration, **kwargs) if process_noise else [self.model.w0] * n_samples
107 vk = self.measurement_noise(duration, **kwargs) if measurement_noise else [self.model.v0] * n_samples
108 t_start = self._t # resume time from the previous call
109 # Step first, then record: sample k is the state *after* the k-th step
110 # (driven by u[k] and w[k]) and z[k] is a measurement of that state.
111 # This is the order the filter assumes (predict with u[k], then update
112 # with z[k], starting from x0) and the order Kalman::Sampler uses.
113 # Recording before the step made every measurement, and every control,
114 # one step late for the filter.
115 chain = self._native_chain(n_samples, x, P, t_start, uk, wk, kwargs)
116 if chain is not None: # the loop ran in C++
117 xk[:, :, 0], Pk[:] = chain
118 x, P = xk[-1].copy(), Pk[-1].copy()
119 else:
120 for i in range(n_samples):
121 t_i = t_start + i / self.fsfs # propagate time (for time-varying f/h)
122 u_i = np.asarray(uk[i]).reshape(-1, 1) if self.model.l > 0 else None
123
124 x, P, _ = self.engineengine._predict(self.model, x, P, u=u_i, t=t_i, **kwargs)
125 x = x + wk[i]
126 xk[i] = x
127 Pk[i] = P
128
129 # advance the cursor so the next measurement() resumes at t+1 with the latest state
130 self._t = t_start + n_samples / self.fsfs
131 self._x, self._P = x, P
132
133 if inprocess:
134 xk = self.model.inprocess(xk, **kwargs)
135
136 # Squeeze trailing unit dim: (N, n, 1) → (N, n) so record.xk indexing works correctly
137 xk = KalmanRecord._normalize_out(xk)
138
139 zk = self._native_measurements(xk, vk, t_start, chain is not None and not inprocess)
140 if zk is None:
141 zk = [self.model.h(xx.reshape(-1, 1), v, t_start + k / self.fsfs) for xx, v, k in zip(xk, vk, range(n_samples))]
142 zk = KalmanRecord._normalize_out(zk, indexing=indexing, dtype=self.model.dtype)
143 if postprocess:
144 zk = self.model.postprocess(zk, **kwargs)
145
146 times = t_start + np.arange(n_samples) / self.fsfs
147 return KalmanRecord(times, zk=zk, xk=xk, Pk=Pk, flavor=KalmanStateFlavor.SAMPLED, model=self.model)

149 def _native_measurements(self, xk, vk, t_start, eligible):
150 """``h`` over the whole run in one C++ call, for a model that is C++ (a twin,
151 a ``NativeModel``, a model of matrices); the noise drawn by numpy is added
152 after it, which is what those models' ``h`` does with its ``v``. ``None``
153 when the model's own Python ``h`` must be asked sample by sample."""
154 if not eligible:
155 return None
156 model = self.model
157 handle = self._matrices_handle()
158 if handle is None:
159 twin = getattr(model, "native_handle", None)
160 handle = twin() if twin is not None else None
161 if handle is None or self.engineengine._noise_through_h(model) is None:
162 return None
163 from .native import core
164 n, m, N = int(model.n), int(model.m), len(xk)
165 try:
166 states = np.ascontiguousarray(np.asarray(xk, dtype=np.float64).reshape(N, n))
167 noise = np.asarray(vk, dtype=np.float64).reshape(N, m)
168 except (TypeError, ValueError):
169 return None
170 zk = np.empty((N, m), dtype=np.float64)
171 core.check(core.lib.kalman_model_measure(handle, N, core.cdptr(states), float(t_start), float(self.fsfs),
172 core.dptr(zk)), "kalman_model_measure")
173 return zk + noise

175 def _native_chain(self, n_samples, x, P, t_start, uk, wk, kwargs):
176 """The state chain computed by the C++ loop (kalman_propagate), with the
177 noise drawn here by numpy so a seeded run does not change. ``None`` when
178 the Python loop must run: another engine than the extended one (or the
179 linear one on a model of constant matrices), extra arguments for the
180 prediction, a model that is not double precision, or no library."""
181 from .models import native
182 from .engines.extended import ExtendedKalmanEngine
183 if kwargs or not native.native_loop_enabled():
184 return None
185 model = self.model
186 matrices = self._matrices_handle()
187 if matrices is None and type(self.engineengine) is not ExtendedKalmanEngine:
188 return None
189 if n_samples <= 0 or not native.callback_eligible(model):
190 return None
191 from .native import core
192 n, l = int(model.n), int(model.l)
193 try:
194 draws = np.asarray(wk, dtype=np.float64).reshape(n_samples, n)
195 controls = np.asarray(uk, dtype=np.float64).reshape(n_samples, l) if l > 0 else None
196 except (TypeError, ValueError):
197 return None
198 if matrices is not None:
199 # A model of constant matrices on the linear engine: the chain runs
200 # on those matrices in C++ (a Kalman::Models::Tabulated holding them).
201 return core.propagate(matrices, core.create_engine("linear"), n, l, n_samples, x, P,
202 t_start, self.fsfs, uk=controls, wk=draws)
203 twin = getattr(model, "native_handle", None)
204 handle, keep = (twin() if twin is not None else None), None
205 if handle is None:
206 handle, keep = native.callback_handle(model, "extended", live_noise=True)
207 try:
208 return core.propagate(handle, core.create_engine("extended"), n, l, n_samples, x, P,
209 t_start, self.fsfs, uk=controls, wk=draws)
210 except RuntimeError:
211 if keep and keep[-1]:
212 raise keep[-1][0] # the Python exception, not its C++ shadow
213 raise

232 def process_noise(self, duration: float, **_):
233 """
234 Generate process noise for the given duration.
235 If Q is a block diagonal covariance matrix, ensure proper sampling.
236 """
237
238 n_samples = int(duration * self.fsfs)
239 if self.model.Q() is None: process_noise = np.zeros((n_samples, self.n))
240 else: # Check if Q is block diagonal
241 if np.allclose(self.model.Q(), np.diag(np.diag(self.model.Q()))): # Diagonal matrix
242 process_noise = np.random.normal(0, np.sqrt(np.diag(self.model.Q())), size=(n_samples, self.n))
243 else:
244 process_noise = np.random.multivariate_normal(np.zeros(self.n), self.model.Q(), size=n_samples)
245
246 process_noise = process_noise.reshape((n_samples, self.n, 1))
247 return process_noise
