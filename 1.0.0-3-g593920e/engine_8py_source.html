12class KalmanEngine(KalmanEngineMetadata, KalmanEngineMixin):
13 """
14 Abstract Kalman **engine** base — the inference algorithm (HOW), one step at a time.
15
16 **Core quantities** (passed to every method as arguments, never stored here):
17
18 - ``x`` (ψ in math notation) — state vector, shape ``(n, 1)``. The hidden
19 quantities the filter estimates (e.g. position, velocity, acceleration).
20 - ``P`` — state-error **covariance matrix**, shape ``(n, n)``. Encodes uncertainty
21 in the current estimate: ``P = E[(ψ - ψ̂)(ψ - ψ̂)ᵀ]``. Diagonal entries are
22 per-state variances; off-diagonals capture cross-correlations.
23 - ``F`` — **state-transition matrix**, shape ``(n, n)``. The discrete-time linear
24 map ``ψ_{k+1} = F ψ_k`` (no noise); equals ``exp(A Δt)`` for a continuous system
25 with dynamics matrix ``A``. For nonlinear models ``F = ∂f/∂ψ`` (Jacobian).
26 - ``Q`` — process-noise covariance, ``(n, n)``; ``R`` — measurement-noise
27 covariance, ``(m, m)``. Larger ``Q`` → trust model less; larger ``R`` → trust
28 sensor less.
29
30 **Two layers**:
31
32 - ``_propagate`` / ``_measure`` — pure model evaluation for one step: the
33 nonlinear map and its Jacobian (``f``/``F`` for the state transition,
34 ``h``/``H`` for the observation). This is the *only* part that differs
35 between algorithms, so it's the hook each engine overrides
36 (:class:`LinearKalmanEngine` / :class:`ExtendedKalmanEngine`).
37 - ``_predict`` / ``_update`` — the **shared** covariance machinery: propagate/
38 update ``P``, inject process noise ``Q``, compute the gain ``K``, apply the
39 masked Joseph-form correction. Identical for every Jacobian-based engine;
40 it calls ``_propagate``/``_measure`` to get the algorithm-specific piece.
41
42 UKF/PF have no single Jacobian (sigma points / particles instead), so they
43 skip the hooks and override ``_predict``/``_update`` wholesale.
44
45 The engine *knows the model* (it calls ``model.F/H`` or ``model.f/h``) but owns no
46 run state and nothing about workflow — that is the ``KalmanFilter``'s job.
47
48 **Whole-pass engines.** For some algorithms one predict/update step is not the
49 whole story: a combinatorial track finder branches, prunes and arbitrates across a
50 whole event, and no per-step loop over single measurements describes it. Such an
51 engine sets ``OWNS_PASS = True`` and implements :meth:`forward` / :meth:`backward`;
52 ``KalmanFilter`` then hands it the data and turns whatever it returns into records,
53 instead of running its own loop. The algorithm still lives in the engine and the
54 workflow still lives in the filter — the split just falls in a different place.
55 Objects returned by such a ``forward`` must expose ``to_record(model, flavor)``.
56 """

103 @staticmethod
104 def _noise_through_f(model) -> bool:
105 """Whether the process noise goes *through* ``model.f`` (passed as ``w``)
106 instead of being added after it.
107
108 A model that declares its noise coupling ``L`` (a matrix given to the
109 constructor, or an ``L`` method of its own) has said the noise is added,
110 as ``L w``; that is what the extended engine does, and every engine must
111 agree with it. Many models take ``w`` in ``f``'s signature and ignore
112 it: reading the signature alone made the sigma-point and particle
113 engines drop their process noise altogether. So the noise goes through
114 ``f`` only when ``f`` takes ``w``, the model declares no coupling, *and*
115 ``f`` responds to ``w`` when tried (:meth:`_responds_to_noise`).
116 """
117 from .model import KalmanModel
118 if not KalmanEngine._accepts_param(model.f, "w"):
119 return False
120 declared = getattr(model, "_L", None) is not None or type(model).L is not KalmanModel.L
121 return not declared and KalmanEngine._responds_to_noise(model, "f")
122
123 @staticmethod
124 def _noise_through_h(model) -> bool:
125 """The same for the measurement noise, ``h(..., v)`` and the coupling ``M``."""
126 from .model import KalmanModel

128 return False

132 @staticmethod
133 def _responds_to_noise(model, which: str) -> bool:
134 """Whether ``model.f`` (``which="f"``) or ``model.h`` actually changes when
135 handed a noise sample.
136
137 ``f(x, u, w, t)`` and ``h(x, v, t)`` are the contract's signatures, so a
138 model has ``w`` and ``v`` in them whether or not it uses them. One that
139 ignores the argument and declares no coupling would lose its ``Q`` (or
140 ``R``) in the sigma-point and particle engines, which hand the noise to
141 the map: the unscented filter then trusted its sensor completely. So the
142 map is tried once, at the initial state, with and without a sample; a map
143 that does not move takes its noise added (``L = I``, ``M = I``), as the
144 extended engine does. The answer is kept on the model.
145 """
146 cache = model.__dict__.setdefault("_noise_response", {})
147 if which in cache:
148 return cache[which]
149 answer = True # if the map cannot be tried, believe its signature
150 try:
151 n, m = int(model.n), int(model.m)
152 x0 = getattr(model, "x0", None)
153 x = np.zeros((n, 1)) if x0 is None else np.asarray(x0, dtype=float).reshape(n, 1)
154 t = getattr(model, "t0", None)
155 t = 0.0 if t is None else float(t)
156 if which == "f":
157 q = np.asarray(model.Q(t=t)).shape[0]
158 u = np.zeros((int(model.l), 1)) if int(model.l) > 0 else None
159 sample = np.linspace(0.5, 1.0, q).reshape(q, 1)
160 quiet = np.asarray(model.f(x, u=u, w=np.zeros((q, 1)), t=t), dtype=float)
161 moved = np.asarray(model.f(x, u=u, w=sample, t=t), dtype=float)
162 else:
163 r = np.asarray(model.R(t=t)).shape[0]
164 sample = np.linspace(0.5, 1.0, r).reshape(r, 1)
165 quiet = np.asarray(model.h(x, v=np.zeros((r, 1)), t=t), dtype=float)
166 moved = np.asarray(model.h(x, v=sample, t=t), dtype=float)
167 if np.all(np.isfinite(quiet)) and np.all(np.isfinite(moved)):
168 answer = bool(np.any(quiet != moved))
169 except Exception:
170 pass
171 cache[which] = answer
172 return answer

191 def _predict(self, model, x, P, u=None, t=None, **kw):
192 """Prediction step → ``(xk, Pk, F)``.
193
194 Propagates the prior belief one step forward using the motion model:
195
196 .. code-block:: text
197
198 xk = F · x + B · u (mean prediction)
199 Pk = F · P · Fᵀ + Q (covariance prediction)
200
201 where ``F`` is the state-transition matrix (``∂f/∂x`` for nonlinear models)
202 and ``Q`` is the process-noise covariance. Returns the predicted mean
203 ``xk``, the predicted covariance ``Pk``, and the transition matrix
204 ``F`` used (needed later by the RTS smoother).
205 """
206 if u is not None:
207 u = np.asarray(u).reshape(-1, 1)
208 u = u if model.l > 0 else None
209
210 xk, F = self._propagate(model, x, u, t)
211
212 # Pass the current state to Q only for models whose Q accepts it (e.g. a
213 # speed-dependent air-friction Q). Constant-Q models keep Q(t=t) untouched.
214 Q = model.Q(t=t, x=x) if self._accepts_param(model.Q, "x") else model.Q(t=t)
215 L = self._safe_L(model, x, u, t)
216 if L is not None:
217 Q = L @ Q @ L.T
218
219 return xk, F @ P @ F.T + Q, F

221 def _update(self, model, x, P, z, t=None, **kw):
222 """Update step → ``(x, P)``. Missing channels (NaN in ``z``) are skipped.
223
224 Fuses the predicted belief with the measurement using the standard equations:
225
226 .. code-block:: text
227
228 S = H · P · Hᵀ + R (innovation covariance)
229 K = P · Hᵀ · S⁻¹ (Kalman gain)
230 x = x + K · (z − H · x) (corrected mean, overwrites prior)
231 P = (I − KH) · P · (I − KH)ᵀ + K · R · Kᵀ (Joseph form, overwrites prior)
232
233 The Joseph form is used for ``P`` to preserve symmetry and positive-definiteness
234 even under numerical round-off.
235 """
236 z = np.asarray(z, dtype=float).reshape(-1, 1)
237 keep = ~np.isnan(z).ravel()
238 if not keep.any(): # all measurements missing → predict-only
239 return x, P
240
241 zk, H = self._measure(model, x, t)
242 H = np.asarray(H, dtype=float)
243 R = np.asarray(model.R(t=t), dtype=float)
244 # Residual against the true (possibly nonlinear) h(x), NOT H @ x -- NaN
245 # propagates automatically wherever z is missing, matching `keep`.
246 yk = z.reshape(-1) - np.asarray(zk, dtype=float).reshape(-1)
247
248 from .native import HAS_C_ACCEL
249 if HAS_C_ACCEL:
250 from .native import lib, as_contiguous, dptr
251 n = model.n
252 xk = np.empty(n, dtype=np.float64)
253 Pk = np.empty((n, n), dtype=np.float64)
254 lib.kf_update_residual(
255 n, H.shape[0],
256 dptr(as_contiguous(H)), dptr(as_contiguous(np.asarray(x, dtype=float).reshape(-1))),
257 dptr(as_contiguous(P)),
258 dptr(as_contiguous(yk)), dptr(as_contiguous(R)),
259 dptr(xk), dptr(Pk),
260 )
261 return xk.reshape(-1, 1), Pk
262
263 H = H[keep, :]
264 R = R[np.ix_(keep, keep)]
265 y = yk[keep].reshape(-1, 1)
266
267 S = R + H @ P @ H.T
268 try:
269 K = P @ H.T @ np.linalg.pinv(S)
270 except np.linalg.LinAlgError:
271 raise RuntimeError("Kalman update failed; check model stability / noise covariances.")
272
273 I = np.eye(model.n)
274 x = x + K @ y
275 IKH = I - K @ H
276 P = IKH @ P @ IKH.T + K @ R @ K.T # Joseph form (numerically stable)
277 return x, P

279 def _smooth(self, model, post_k, prior_next, smooth_next, F_k):
280 """One reverse RTS (Rauch–Tung–Striebel) step → ``(x_s, P_s)``.
281
282 Given the forward-pass posterior at step ``k`` and the already-smoothed estimate
283 at step ``k+1``, computes the smoothed estimate at step ``k``:
284
285 .. code-block:: text
286
287 C = P_k · Fᵀ · P_{k+1|k}⁻¹ (smoother gain)
288 x_s = x_k + C · (x_s_{k+1} − x_{k+1|k})
289 P_s = P_k + C · (P_s_{k+1} − P_{k+1|k}) · Cᵀ
290
291 where ``F_k`` is the transition matrix stored during the forward pass at step ``k``.
292 """
293 x, P = post_k
294 xp, Pp = prior_next
295 xs, Ps = smooth_next
296 if np.linalg.cond(Pp) < 1e12:
297 C = P @ F_k.T @ np.linalg.inv(Pp)
298 else:
299 C = P @ F_k.T @ np.linalg.pinv(Pp)
300 return x + C @ (xs - xp), P + C @ (Ps - Pp) @ C.T
