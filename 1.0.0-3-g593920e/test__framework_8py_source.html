91class _Pendulum(ExtendedKalmanModel):
92 """Damped pendulum: x = [theta, omega]; only theta is observed."""
93
94 def __init__(self):
95 self.n, self.m, self.l = 2, 1, 0
96 super().__init__("pendulum", Q=np.eye(2) * 1e-5, R=np.array([[0.04]]), fs=20)
97
98 def f(self, x, u=None, w=None, t=None):
99 th, om = float(x[0, 0]), float(x[1, 0])
100 dt = 0.05
101 return np.array([[th + om * dt], [om - 9.81 * np.sin(th) * dt]])
102
103 def h(self, x, v=None, t=None):
104 # additive measurement noise (the sampler passes v from R; matrix models add it for free)
105 return x[:1] if v is None else x[:1] + v
106

116 def test_numerical_jacobian_matches_default(self):
117 """The model's default F()/H() are the numerical Jacobians of f()/h()."""
118 m = _Pendulum()
119 x = np.array([[0.4], [0.2]])
120 np.testing.assert_allclose(m.F(x), numerical_jacobian(lambda xx: m.f(xx), x), atol=1e-5)
121 np.testing.assert_allclose(m.H(x), numerical_jacobian(lambda xx: m.h(xx), x), atol=1e-6)
122
123 def test_sample_then_filter_tracks(self):
124 np.random.seed(0)
125 m = _Pendulum()
126 m.initial_state([0.5, 0.0])
127 truth = m.sample(3.0)
128 rec = m.filter(truth.zk)
129 self.assertEqual(rec.xk.shape, (60, 2))
130 _no_nan_inf(rec.xk, "pendulum xk")
131 # after the initial transient the filtered angle beats the raw measurement
132 tail = slice(20, None)
133 rmse_filt = np.sqrt(np.mean((rec.xk[tail, 0] - truth.xk[tail, 0]) ** 2))
134 rmse_meas = np.sqrt(np.mean((truth.zk[tail, 0] - truth.xk[tail, 0]) ** 2))
135 self.assertLess(rmse_filt, rmse_meas)

186class TestSampler(unittest.TestCase):
187
188 def test_sample_tracks_analytic_pk(self):
189 """sample() always routes through engine._predict, so Pk carries the
190 analytic prior covariance band; with process_noise=False the mean is
191 also deterministic (no random draw injected)."""
192 m = _linear_model()
193 m.initial_state([0.0, 1.0])
194 a = m.sample(1.0, process_noise=False)
195 b = m.sample(1.0, process_noise=False)
196 np.testing.assert_allclose(a.xk, b.xk) # deterministic mean
197 self.assertGreater(float(np.sum(np.abs(a.Pk))), 0.0) # analytic Pk present
198
199 def test_process_noise_toggle(self):
200 """process_noise=False removes the stochastic drift from the truth path."""
201 np.random.seed(2)
202 m = LinearKalmanModel("s", F=F_CV, H=H_POS, Q=np.eye(2) * 0.5, R=R_POS, fs=50)
203 m.initial_state([0.0, 0.0])
204 noisy = m.sample(2.0, process_noise=True)
205 clean = m.sample(2.0, process_noise=False)
206 self.assertGreater(np.std(np.diff(noisy.xk[:, 1])),
207 np.std(np.diff(clean.xk[:, 1])))
