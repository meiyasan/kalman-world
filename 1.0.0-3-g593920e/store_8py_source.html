1"""
2The shared registry behind :mod:`kalman.api` — one SQLite file that every
3process reading the same state directory sees.
4
5Before this, authored models and loaded plugins lived in a dict inside each
6process. The REST server and the MCP server are two processes, so a model
7registered through one was invisible to the other, and a restart lost it. Now
8the registry is on disk:
9
10 - ``models`` authored KalmanSpecs (name -> spec, dimensions, owner)
11 - ``plugins`` compiled model libraries loaded through the API, so every
12 process (and every restart) loads them again
13 - ``sources`` named measurement sources an admin or author set up once
14 and everybody fetches by name
15 - ``measurements`` rows pushed into a ``push`` source by a device, a
16 simulation or a game engine
17 - ``events`` an append-only log of all of the above; ``GET /events``
18 streams it, so clients subscribe instead of polling the
19 catalog
20 - ``revoked_*`` the token revocation list: single tokens by ``jti``, and
21 "every token of this user issued until now"
22 - ``login_failures`` failed logins per user and client, so a lockout holds
23 across workers, processes and restarts
24
25The file is ``$KALMAN_STATE_DIR/registry.sqlite`` (default ``~/.kalman/state``).
26WAL mode lets one writer and many readers work at the same time, across
27processes and threads. Values are JSON text; nothing here executes code.
28"""

118@contextlib.contextmanager
119def _db():
120 """One connection per call: sqlite3 connections must not cross threads, and
121 FastAPI runs sync routes on a thread pool. Commits on success."""
122 p = path()
123 with _ready_lock:
124 if p not in _ready:
125 os.makedirs(os.path.dirname(p), exist_ok=True)
126 con = sqlite3.connect(p, timeout=30)
127 try:
128 con.execute("PRAGMA journal_mode=WAL")
129 con.executescript(_SCHEMA)
130 con.commit()
131 finally:
132 con.close()
133 try:
134 os.chmod(p, 0o600) # owner-only, like the credentials file
135 except OSError:
136 pass
137 _ready.add(p)
138 con = sqlite3.connect(p, timeout=30)
139 con.row_factory = sqlite3.Row
140 try:
141 yield con
142 con.commit()
143 finally:
144 con.close()

151def _emit(con, type_: str, payload: dict) -> int:
152 cur = con.execute("INSERT INTO events(type, payload, ts) VALUES (?,?,?)",
153 (type_, json.dumps(payload), time.time()))
154 eid = cur.lastrowid
155 if MAX_EVENTS > 0:
156 con.execute("DELETE FROM events WHERE id <= ?", (eid - MAX_EVENTS,))
157 return eid

189def put_model(name: str, spec: dict, info: dict, owner: str | None, *,
190 replace: bool = False, may_replace=lambda rec: True) -> tuple[dict, str]:
191 """Insert or update a model. Returns ``(record, action)`` where action is
192 ``created``, ``replaced`` or ``unchanged``.
193
194 The same spec under the same name is a no-op, so a client can register on
195 every start. A different spec needs ``replace=True`` and ``may_replace(old)``
196 true; otherwise :class:`ConflictError` — a shared registry must not let one
197 author silently overwrite another's model."""
198 now = time.time()
199 with _db() as con:
200 row = con.execute("SELECT * FROM models WHERE name=?", (name,)).fetchone()
201 if row is not None:
202 old = _model(row)
203 if _same(old["spec"], spec):
204 return old, "unchanged"
205 if not replace:
206 raise ConflictError(f"model {name!r} already exists with a different spec "
207 f"(owner {old['owner'] or 'unknown'}); pass replace=true "
208 "to update it, or choose another name.")
209 may_replace(old) # raises AuthError when not allowed
210 con.execute("UPDATE models SET spec=?, info=?, updated=? WHERE name=?",
211 (json.dumps(spec), json.dumps(info), now, name))
212 action = "replaced"
213 else:
214 con.execute("INSERT INTO models VALUES (?,?,?,?,?,?)",
215 (name, json.dumps(spec), json.dumps(info), owner, now, now))
216 action = "created"
217 _emit(con, f"model.{action}", {"name": name, "owner": owner, **info})
218 row = con.execute("SELECT * FROM models WHERE name=?", (name,)).fetchone()
219 return _model(row), action

237def put_plugin(name: str, plugin_path: str, models: list, owner: str | None) -> bool:
238 """Record a loaded plugin. Returns True when it was not recorded before."""
239 with _db() as con:
240 row = con.execute("SELECT models FROM plugins WHERE name=?", (name,)).fetchone()
241 if row is not None and sorted(json.loads(row[0])) == sorted(models):
242 return False
243 con.execute("INSERT OR REPLACE INTO plugins VALUES (?,?,?,?,?)",
244 (name, plugin_path, json.dumps(sorted(models)), owner, time.time()))
245 _emit(con, "plugin.loaded", {"name": name, "models": sorted(models), "owner": owner})
246 return True

321def append_measurements(source: str, rows: list, times: list | None,
322 owner: str | None) -> dict:
323 now = time.time()
324 with _db() as con:
325 first = last = None
326 for i, row in enumerate(rows):
327 t = None if times is None else times[i]
328 cur = con.execute("INSERT INTO measurements(source, row, t, owner, received) "
329 "VALUES (?,?,?,?,?)", (source, json.dumps(row), t, owner, now))
330 first = first or cur.lastrowid
331 last = cur.lastrowid
332 if MAX_ROWS > 0:
333 con.execute("DELETE FROM measurements WHERE source=? AND seq NOT IN "
334 "(SELECT seq FROM measurements WHERE source=? "
335 " ORDER BY seq DESC LIMIT ?)", (source, source, MAX_ROWS))
336 out = {"source": source, "first": first, "last": last, "count": len(rows)}
337 _emit(con, "measurement.received", {**out, "owner": owner})
338 return out

369def revoke_token(jti: str, sub: str | None, exp: float | None, by: str | None):
370 """Refuse the token with this ``jti`` from now on. Entries whose token has
371 expired anyway are dropped here, so the list stays as short as the set of
372 live revoked tokens."""
373 now = time.time()
374 with _db() as con:
375 con.execute("DELETE FROM revoked_tokens WHERE exp IS NOT NULL AND exp < ?", (now,))
376 con.execute("INSERT OR REPLACE INTO revoked_tokens VALUES (?,?,?,?,?)",
377 (jti, sub, exp, by, now))

380def revoke_user(sub: str, by: str | None) -> float:

383 with _db() as con:
384 con.execute("INSERT OR REPLACE INTO revoked_users VALUES (?,?,?,?)", (sub, now, by, now))

388def is_revoked(jti: str | None, sub: str | None, iat: float | None) -> bool:
389 with _db() as con:
390 if jti and con.execute("SELECT 1 FROM revoked_tokens WHERE jti=?", (jti,)).fetchone():
391 return True
392 if sub is not None:
393 row = con.execute("SELECT before FROM revoked_users WHERE sub=?", (sub,)).fetchone()
394 # A token with no iat cannot be shown to be newer than the cutoff.
395 if row is not None and (iat is None or float(iat) <= row[0]):
396 return True
397 return False
