8class MPPIKalmanController(KalmanController):
9 """Direct (online) MPPI: replans a sampled receding-horizon control sequence at *every* step.
10
11 Unlike :class:`LQRKalmanController` — a single fixed gain computed once from one
12 linearisation of ``model.f`` — MPPI resolves
13 ``u* = argmin E[ sum_k eᵀ Qcost e + uᵀ Rcost u ]`` from scratch at every control tick by
14 Monte-Carlo rollout of the **true nonlinear** ``model.f()``, including whatever actuator
15 limits the model enforces internally (e.g. ``CarLaneModel``'s ``steer_limit`` clip).
16
17 That is the structural advantage over LQR under hard constraints: LQR computes its gain
18 once, offline, against the *unconstrained* linear-quadratic optimum, and the clip is
19 applied after the fact, blind to the fact that saturation is coming. MPPI's rollouts hit
20 the same clip during planning, so the chosen action already accounts for it — no
21 look-ahead is wasted commanding steering the actuator can never deliver.
22
23 The price is cost: MPPI resolves an ``n_rollouts x horizon`` Monte-Carlo optimisation on
24 **every** call — orders of magnitude more expensive per step than reading a stored gain
25 matrix, and generally too slow for a tight real-time loop. That gap is exactly what
26 :class:`NeuralKalmanController` is for: train an MLP offline to imitate MPPI's own
27 decisions (``fit_mppi()``), then deploy the single cheap forward pass instead of
28 resolving MPPI online at every tick.
29
30 Warm-started receding horizon: the nominal sequence found at step *k* is shifted by one
31 and reused as the starting point at step *k+1* (with the vacated last slot zeroed), so
32 each step only has to refine an already-good guess rather than search from scratch.
33
34 With libKalman installed the rollouts run in ``Kalman::Controllers::MPPI``, which then
35 holds the warm-started sequence: ``model.f`` and ``model.clip_u`` are called back from
36 C++ for a model written in Python (and not at all for a model written in C++). The
37 perturbations are still drawn here, from the same seeded numpy generator in the same
38 order, so a seeded controller makes the same decisions on either path.
39 :meth:`_reference_policy` is the Python reference.
40 """

69 def _reference_policy(self, x_hat, P, t, *, nu=None):
70 model = self.model
71 dt = 1.0 / model.fs if model.fs else 1.0
72 u_nom = self._u_nom
73
74 for _ in range(self.n_iters):
75 du = self._rng.normal(0, self.noise_sigma, (self.n_rollouts, self.horizon, self._l))
76 u_try = u_nom[None, :, :] + du
77 costs = np.zeros(self.n_rollouts)
78 for i in range(self.n_rollouts):
79 xi = x_hat.copy()
80 cost = 0.0
81 for k in range(self.horizon):
82 # Clip the *candidate* action before rolling it out or costing it, not
83 # after -- otherwise dynamics/cost go flat past the limit (any raw value
84 # out there looks identical), and the cost-weighted average of many
85 # different "however far past the limit" samples can land anywhere out
86 # there with no pull back toward what's deliverable. Clipping here keeps
87 # every candidate, and so the weighted-average nominal, inside the box
88 # the actuator can actually reach -- matching real MPC practice, and
89 # keeping u_try consistent with what u_nom will be averaged from.
90 u_k = np.asarray(model.clip_u(u_try[i, k].reshape(-1, 1)), dtype=float).reshape(-1)
91 u_try[i, k] = u_k
92 u_k = u_k.reshape(-1, 1)
93 xi = np.asarray(model.f(xi, u=u_k, t=t + k * dt), dtype=float).reshape(-1, 1)
94 e = xi - self.x_ref
95 cost += float((e.T @ self.Qcost @ e + u_k.T @ self.Rcost @ u_k).item())
96 costs[i] = cost
97 weights = np.exp(-(costs - costs.min()) / self.lam)
98 weights /= weights.sum() + 1e-12
99 u_nom = np.einsum("i,ihl->hl", weights, u_try)
100
101 u0 = u_nom[0].reshape(-1, 1)
102 # shift the warm-started nominal sequence for the next call's receding horizon
103 self._u_nom = np.vstack([u_nom[1:], np.zeros((1, self._l))])
104 return u0
