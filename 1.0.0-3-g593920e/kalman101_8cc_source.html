23class ConstantVelocity : public Kalman::Models::Linear
24{
25 public:
26 ConstantVelocity(double fs, double sigmaAcc, double sigmaMeas)
27 : Kalman::Models::Linear("constant-velocity", 2, 1)
28 {
29 SetSampleRate(fs);
30
31 const double dt = DeltaT();
32 const double q = sigmaAcc * sigmaAcc;
33
34 Kalman::Matrix F = Kalman::Matrix::Identity(2);
35 F(0, 1) = dt;
36 SetF(F);
37
38 Kalman::Matrix H(1, 2, 0.);
39 H(0, 0) = 1.; // only the position is measured
40 SetH(H);
41
42 // Discrete white-noise acceleration: an unmodelled acceleration of standard
43 // deviation sigma_a [m/s^2], held constant over each step of length dt.
44 // (The continuous-time version has dt^3/3, dt^2/2, dt and a spectral
45 // density instead.) The off-diagonal terms matter: one acceleration moves
46 // both the position and the velocity, so their errors are correlated, and
47 // dropping these terms makes the filter inconsistent.
48 Kalman::Matrix Q(2, 2, 0.);
49 Q(0, 0) = q * dt * dt * dt * dt / 4.;
50 Q(0, 1) = q * dt * dt * dt / 2.;
51 Q(1, 0) = Q(0, 1);
52 Q(1, 1) = q * dt * dt;
53 SetQ(Q);
54
55 Kalman::Matrix R(1, 1, 0.);
56 R(0, 0) = sigmaMeas * sigmaMeas;
57 SetR(R);
58 }
59};

98 // and whether that becomes a TGraph, a matplotlib axis or a gnuplot file is

101 std::printf("# t\ttruth\tmeasured\tfiltered\tsigma\tsmoothed\n");

103 std::printf("%.4f\t%.6f\t%.6f\t%.6f\t%.6f\t%.6f\n",
104 forward.Time(k), sampler.Truth()[k](0), forward.Measurement(k)(0),
105 forward.State(k)(0), sigma[k], smoothed.State(k)(0));
106
