45def _random_walk(n, q, r, seed=0):
46 """Synthetic local-level random walk: truth x_{k+1}=x_k+w (w~N(0,q²)) and z=x+v (v~N(0,r²))."""
47 rng = np.random.default_rng(seed)
48 x = np.cumsum(rng.normal(0.0, q, n))
49 z = x + rng.normal(0.0, r, n)
50 return x, z

53def _scalar_kf(z, q, r):
54 """Plain scalar constant-model Kalman filter; returns (estimate, posterior_var) per step."""
55 q2, r2 = q * q, r * r
56 x, P = float(z[0]), r2
57 xs, Ps = np.empty_like(z), np.empty_like(z)
58 for k, zk in enumerate(z):
59 P = P + q2 # predict
60 K = P / (P + r2) # gain
61 x = x + K * (zk - x) # update
62 P = (1.0 - K) * P
63 xs[k], Ps[k] = x, P
64 return xs, Ps

69 def test_steady_state_gain_matches_converged_kalman(self):
70 """The closed-form K_ss must match the gain the recursion converges to."""
71 q, r = 0.05, 1.0
72 ss = steady_state_gain(q, r)
73 # Iterate the scalar Riccati recursion to convergence.
74 q2, r2 = q ** 2, r ** 2
75 P = r2
76 for _ in range(5000):
77 P = P + q2 # predict
78 K = P / (P + r2) # gain
79 P = (1.0 - K) * P # update
80 self.assertAlmostEqual(K, ss.gain, places=6)
81 self.assertTrue(0.0 < ss.gain < 1.0)

85 self.assertEqual(steady_state_gain(1.0, 0.0).gain, 1.0) # perfect sensor → follow it fully

87 def test_lowpass_from_qr_reproduces_steady_state_kalman(self):
88 """A low-pass seeded with K_ss matches the steady-state scalar Kalman estimate."""
89 q, r = 0.1, 1.0

191 def test_covariance_inflates_through_occlusion(self):
192 """During the occlusion (predict-only) the position variance must grow."""
193 d = DetectionModel("cam", fs=30, measurement_uncertainty=8.0, process_uncertainty=40.0)
194 truth = d.synthetic_track(6.0, motion="linear", seed=2)

196 rec = d.filter(det)

269class TestSamplerAndFilterAgreeOnTiming(unittest.TestCase):
270 """sample() and filter() use one convention: sample k is the state after the
271 k-th step, driven by u[k], and z[k] measures it. A noise-free run filtered
272 from the same x0 must then leave no innovation at all, however far the state
273 moves in a step. (Recording the state *before* each step left every fix, and
274 every control, one step late.)"""
275
276 def run_clean(self, control):
277 import numpy as np
278 from modules.framework.model import KalmanModel
279 F = np.array([[1.0, 1.0], [0.0, 1.0]])
280 kw = dict(F=F, H=np.array([[1.0, 0.0]]), Q=1e-9 * np.eye(2), R=np.array([[1e-6]]),
281 fs=1.0, x0=np.array([[0.0], [50.0]]), P0=1e-9 * np.eye(2))
282 if control:
283 kw["B"] = np.array([[0.0], [1.0]])
284 make = lambda: KalmanModel("m", **kw)
285 N = 12
286 uk = (np.arange(N, dtype=float) * 3.0).reshape(N, 1) if control else None
287 sample = make().sample(N, uk=uk, process_noise=False, measurement_noise=False) if control \
288 else make().sample(N, process_noise=False, measurement_noise=False)
289 record = make().filter(np.asarray(sample.zk), uk) if control else make().filter(np.asarray(sample.zk))
290 return np.asarray(sample.xk), np.asarray(sample.zk), np.asarray(record.xk)
291
292 def test_first_sample_is_one_step_after_x0(self):
293 xk, zk, _ = self.run_clean(control=False)
294 self.assertAlmostEqual(float(xk[0, 0]), 50.0) # x0 = 0 moved by one step at 50 per step
295 self.assertAlmostEqual(float(zk[0, 0]), 50.0)
296
297 def test_noise_free_run_filters_back_exactly(self):
298 import numpy as np
299 for control in (False, True):
300 xk, _, est = self.run_clean(control)
301 self.assertLess(float(np.max(np.abs(est - xk))), 1e-3,
302 f"filter and sampler are a step apart (control={control})")
