105class KalmanEngineABC(ABC):
106
107 """
108 Abstract base class for Kalman engines — the inference algorithm (HOW).
109
110 An engine implements one estimation *step*; it is handed the model and the
111 current belief ``(x, P)`` and returns the new belief. It owns no run state and
112 knows nothing about workflow (that is the Filter's job).
113 """
114 @abstractmethod
115 def _predict(self, model, x, P, u=None, t=None, **kwargs):
116 """Prediction step: return ``(xk, Pk, F)``."""
117 ...
118 @abstractmethod
119 def _update(self, model, x, P, z, t=None, **kwargs):
120 """Update step: return ``(x, P)`` (missing-measurement aware)."""
121 ...
122 @abstractmethod
123 def _smooth(self, model, post_k, prior_next, smooth_next, F_k):
124 """RTS smoothing equations for one reverse step: return ``(x_s, P_s)``."""
125 ...
