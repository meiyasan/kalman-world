149 Matrix JosephForm(const Matrix &K, const Matrix &H, const Matrix &P, const Matrix &R)
150 {
151 Matrix IKH = Matrix::Identity(P.Rows());
152 IKH -= K * H;
153
154 Matrix out = Similarity(IKH, P);
155 out += Similarity(K, R);

160 void Correct(const Matrix &Hc, const Matrix &Rc, const Vector &yc,
161 const Vector &x, const Matrix &P, Vector &xOut, Matrix &POut)
162 {
163 Matrix S = Similarity(Hc, P); // H P H^T
164 S += Rc;
165 const Matrix K = P * Hc.Transpose() * Inverse(S);
166 xOut = x;
167 xOut += K * yc;
168 POut = JosephForm(K, Hc, P, Rc);
169 }

171 void WeightedOuterSum(std::size_t nsigma, std::size_t na, std::size_t nb,
172 const double *w, const double *A, const double *a,
173 const double *B, const double *b, double *out)
174 {
175 for (std::size_t i = 0; i < na * nb; ++i) out[i] = 0.;
176 for (std::size_t s = 0; s < nsigma; ++s)
177 for (std::size_t i = 0; i < na; ++i)
178 {
179 const double da = w[s] * (A[s * na + i] - a[i]);
180 if (da == 0.) continue;
181 for (std::size_t j = 0; j < nb; ++j) out[i * nb + j] += da * (B[s * nb + j] - b[j]);
182 }
183 }

185 void SystematicResample(std::size_t n, const double *weights, double u0, int *indices)
186 {
187 // Walk the cumulative weight once; the last entry is pinned to 1 so
188 // round-off in the sum cannot leave the final positions unmatched.
189 double cumulative = weights[0];
190 std::size_t j = 0;
191 for (std::size_t i = 0; i < n; ++i)
192 {
193 const double position = (static_cast<double>(i) + u0) / static_cast<double>(n);
194 while (j + 1 < n && cumulative < position) cumulative += weights[++j];
195 indices[i] = static_cast<int>(j);
196 }
197 }
