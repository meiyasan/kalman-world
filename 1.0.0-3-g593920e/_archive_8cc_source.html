29 std::uint32_t Crc32(const std::string &bytes)
30 {
31 static const std::array<std::uint32_t, 256> table = [] {
32 std::array<std::uint32_t, 256> t{};
33 for (std::uint32_t i = 0; i < 256; ++i)
34 {
35 std::uint32_t c = i;
36 for (int k = 0; k < 8; ++k) c = (c & 1U) ? (0xEDB88320U ^ (c >> 1)) : (c >> 1);
37 t[i] = c;
38 }
39 return t;
40 }();
41 std::uint32_t c = 0xFFFFFFFFU;
42 for (const unsigned char b : bytes) c = table[(c ^ b) & 0xFFU] ^ (c >> 8);
43 return c ^ 0xFFFFFFFFU;
44 }

59 std::uint32_t Get16(const std::string &in, std::size_t at)
60 {
61 if (at + 2 > in.size()) throw std::runtime_error("Kalman::IO::Archive: the archive is cut short.");
62 return static_cast<std::uint32_t>(static_cast<unsigned char>(in[at]))
63 | (static_cast<std::uint32_t>(static_cast<unsigned char>(in[at + 1])) << 8);

69 // Raw deflate, as a zip entry holds it. "" when it would not be smaller.
70 std::string Deflate(const std::string &bytes)
71 {
72 z_stream stream{};
73 if (deflateInit2(&stream, Z_BEST_COMPRESSION, Z_DEFLATED, -15, 8, Z_DEFAULT_STRATEGY) != Z_OK) return "";
74 std::string out(deflateBound(&stream, static_cast<uLong>(bytes.size())), '\0');
75 stream.next_in = reinterpret_cast<Bytef *>(const_cast<char *>(bytes.data()));
76 stream.avail_in = static_cast<uInt>(bytes.size());
77 stream.next_out = reinterpret_cast<Bytef *>(out.data());
78 stream.avail_out = static_cast<uInt>(out.size());
79 const int status = deflate(&stream, Z_FINISH);
80 const std::size_t size = stream.total_out;
81 deflateEnd(&stream);
82 if (status != Z_STREAM_END || size >= bytes.size()) return "";
83 out.resize(size);
84 return out;
85 }

93 stream.next_in = reinterpret_cast<Bytef *>(const_cast<char *>(packed.data()));
94 stream.avail_in = static_cast<uInt>(packed.size());
95 stream.next_out = reinterpret_cast<Bytef *>(out.data());
96 stream.avail_out = static_cast<uInt>(out.size());

99 inflateEnd(&stream);

137 std::string Archive::Bytes() const
138 {
139 if (fEntries.size() > 0xFFFEU) throw std::runtime_error("Kalman::IO::Archive: too many entries.");
140 std::string out, central;
141 for (const auto &e : fEntries)
142 {
143 if (e.second.size() > 0xFFFFFFFEULL || out.size() > 0xFFFFFFFEULL)
144 throw std::runtime_error("Kalman::IO::Archive: '" + e.first + "' passes the 4 GB of a plain zip archive.");
145 const std::uint32_t crc = Crc32(e.second), size = static_cast<std::uint32_t>(e.second.size());
146 const std::uint32_t offset = static_cast<std::uint32_t>(out.size());
147
148 // The readable part stays as it is; the rest is compressed when that helps.
149 std::string deflated;
150#if KALMAN_WITH_ZLIB
151 if (!Readable(e.first)) deflated = Deflate(e.second);
152#endif
153 const std::uint32_t method = deflated.empty() ? 0U : 8U;
154 const std::string &data = deflated.empty() ? e.second : deflated;
155 const std::uint32_t packed = static_cast<std::uint32_t>(data.size());
156
157 Put32(out, kLocal);
158 Put16(out, 20); Put16(out, 0); Put16(out, method); // version needed, flags, 0 stored / 8 deflated
159 Put16(out, 0); Put16(out, kDosDate); // time, date
160 Put32(out, crc); Put32(out, packed); Put32(out, size);
161 Put16(out, static_cast<std::uint32_t>(e.first.size())); Put16(out, 0);
162 out += e.first;
163 out += data;
164
165 Put32(central, kCentral);
166 Put16(central, 20); Put16(central, 20); Put16(central, 0); Put16(central, method);
167 Put16(central, 0); Put16(central, kDosDate);
168 Put32(central, crc); Put32(central, packed); Put32(central, size);
169 Put16(central, static_cast<std::uint32_t>(e.first.size()));
170 Put16(central, 0); Put16(central, 0); Put16(central, 0); Put16(central, 0);
171 Put32(central, 0); Put32(central, offset);
172 central += e.first;
173 }
174 const std::uint32_t start = static_cast<std::uint32_t>(out.size());
175 out += central;
176 Put32(out, kEnd);
177 Put16(out, 0); Put16(out, 0);
178 Put16(out, static_cast<std::uint32_t>(fEntries.size())); Put16(out, static_cast<std::uint32_t>(fEntries.size()));
179 Put32(out, static_cast<std::uint32_t>(central.size())); Put32(out, start);
180 Put16(out, 0);
181 return out;
182 }

189 Archive Archive::FromBytes(const std::string &bytes)
190 {
191 if (!Looks(bytes)) throw std::runtime_error("Kalman::IO::Archive: not a zip archive.");
192 // The end record is the last thing in the file, before an optional comment.
193 std::size_t end = std::string::npos;
194 for (std::size_t back = 22; back <= bytes.size() && back <= 22 + 0xFFFF; ++back)
195 if (Get32(bytes, bytes.size() - back) == kEnd) { end = bytes.size() - back; break; }
196 if (end == std::string::npos) throw std::runtime_error("Kalman::IO::Archive: no end record; the archive is cut short.");
197
198 const std::uint32_t count = Get16(bytes, end + 10);
199 std::size_t at = Get32(bytes, end + 16);
200 Archive archive;
201 for (std::uint32_t i = 0; i < count; ++i)
202 {
203 if (Get32(bytes, at) != kCentral) throw std::runtime_error("Kalman::IO::Archive: damaged directory.");
204 const std::uint32_t method = Get16(bytes, at + 10), crc = Get32(bytes, at + 16);
205 const std::uint32_t packed = Get32(bytes, at + 20), size = Get32(bytes, at + 24);
206 const std::uint32_t nameLength = Get16(bytes, at + 28), extra = Get16(bytes, at + 30),
207 comment = Get16(bytes, at + 32);
208 const std::size_t local = Get32(bytes, at + 42);
209 if (at + 46 + nameLength > bytes.size()) throw std::runtime_error("Kalman::IO::Archive: damaged directory.");
210 const std::string name = bytes.substr(at + 46, nameLength);
211 at += 46 + nameLength + extra + comment;
212
213 if (!name.empty() && name.back() == '/') continue; // a folder
214 if (method != 0 && (method != 8 || !Compresses()))
215 throw std::runtime_error("Kalman::IO::Archive: entry '" + name + "' is compressed"
216 + (method == 8 ? ", and this build of libKalman has no zlib (KALMAN_USE_ZLIB). "
217 "Repack the archive without compression (zip -0), or use a "
218 "build with zlib."
219 : " with a method other than deflate."));
220 if (Get32(bytes, local) != kLocal) throw std::runtime_error("Kalman::IO::Archive: damaged entry '" + name + "'.");
221 const std::size_t data = local + 30 + Get16(bytes, local + 26) + Get16(bytes, local + 28);
222 if ((method == 0 && packed != size) || data + packed > bytes.size())
223 throw std::runtime_error("Kalman::IO::Archive: entry '" + name + "' is cut short.");
224 std::string content = bytes.substr(data, packed);
225#if KALMAN_WITH_ZLIB
226 if (method == 8) content = Inflate(content, size, name);
227#endif
228 if (Crc32(content) != crc)
229 throw std::runtime_error("Kalman::IO::Archive: entry '" + name + "' fails its checksum; the file is damaged.");
230 archive.Set(name, std::move(content));
231 }
232 return archive;
233 }
