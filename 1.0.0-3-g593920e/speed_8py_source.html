3"""
4Vehicle dynamics models for the Kalman framework.
5
6* :class:`CarSpeedModel` — a **straight-line car speed test** (the "0–100 km/h /
7 standing quarter-mile" run). A single car accelerates in a straight line under a
8 realistic longitudinal force balance — **traction-limited then power-limited** drive
9 force, aerodynamic drag and rolling resistance — while a noisy GPS/radar measures
10 position only. The EKF reconstructs the clean **speed** and **acceleration** the test
11 is after, far better than naively differencing the noisy position fixes.
12
13* :class:`CarLaneModel` — **lateral lane-keeping** with steering control.
14 State ``x = [y, ψ, v]`` (lateral offset, heading, speed); control ``u = [δ]``
15 (steering angle). Bicycle kinematics; noisy lateral GPS.
16"""

37class CarSpeedModel(NativeTwin, ExtendedKalmanModel):
38 """
39 Straight-line acceleration ("speed") test of a single car.
40
41 Dynamics (longitudinal force balance, divided by mass ``m``)::
42
43 ṡ = v
44 v̇ = [ F_drive(v) − F_drag(v) − F_roll − F_grade ] / m
45
46 with
47
48 * ``F_drive(v) = min( μ·m·g , P_max / v )`` — the wheels can put down at most the
49 grip-limited force ``μ·m·g`` off the line; once the engine's **power** limit
50 ``P_max / v`` falls below that (above ~30–40 km/h) the run becomes power-limited.
51 That hand-off is exactly the kink you see in a real acceleration trace.
52 * ``F_drag = ½ ρ (C_d A) v |v|`` — aerodynamic drag (caps the top speed),
53 * ``F_roll = C_r m g`` — rolling resistance,
54 * ``F_grade = m g sin θ`` — road grade (0 on a flat straight).
55
56 A GPS/radar measures **position only** (``z = s + noise``); the EKF recovers the
57 smooth speed and acceleration curves. ``n = 2``, ``m = 1``, ``l = 0``.
58 """
59

61 DESCRIPTION = "Longitudinal car dynamics (traction/power-limited drive + drag); GPS position measured."

69 def __init__(self, name="CAR", *,
70 mass=1500.0, # vehicle mass [kg]
71 power=150_000.0, # peak engine power at the wheels [W] (~200 hp)
72 mu=1.0, # tyre–road grip coefficient (launch traction limit)
73 drag_area=0.66, # C_d·A [m²] (≈ 0.30 × 2.2 m²)
74 rho=1.225, # air density [kg/m³]
75 c_roll=0.015, # rolling-resistance coefficient
76 grade=0.0, # road grade [rad] (0 = flat)
77 g=9.81, # gravity [m/s²]
78 fs=20.0, # GPS/radar sample rate [Hz]
79 gps_uncertainty=0.5, # position measurement std [m]
80 q_pos=1e-3, q_vel=5e-2, # process-noise variances (model slack)
81 v_eps=0.5, # speed floor [m/s] so P/v stays finite at launch
82 **kw):
83 self.dtypedtype = np.float64
84 self.mass = float(mass)
85 self.power = float(power)
86 self.mu = float(mu)
87 self.drag_area = float(drag_area)
88 self.rho = float(rho)
89 self.c_roll = float(c_roll)
90 self.grade = float(grade)
91 self.g = float(g)
92 self.v_eps = float(v_eps)
93
94 self._dt = 1.0 / fs
95 self.nnnnn, self.mmmm, self.lll = 2, 1, 0
96 self.fsfsfsfsfs = fs
97
98 Q = np.diag([q_pos, q_vel]).astype(float)
99 R = np.array([[gps_uncertainty ** 2]], dtype=float)
100 # f() doesn't route an additive-noise `w` through (it's accepted for API
101 # compatibility only), so the engine's generic "numerically differentiate f
102 # w.r.t. w" fallback for L would silently see zero sensitivity and multiply Q
103 # by a zero matrix every predict step. L=I says explicitly: Q enters
104 # additively on every state, which is what q_pos/q_vel are actually meant to do.
105 super().__init__(name, Q=Q, R=R, L=np.eye(self.nnnnn), fs=fs, **kw)

124 # ----------------------------------------------------------------------
125 # Longitudinal dynamics
126 # ----------------------------------------------------------------------
127 def drive_force(self, v):
128 """Wheel drive force [N]: grip-limited off the line, power-limited at speed."""
129 grip = self.mu * self.mass * self.g
130 power = self.power / max(abs(v), self.v_eps)
131 return min(grip, power)
132
133 def accel(self, v):
134 """Longitudinal acceleration [m/s²] at speed ``v`` (the force balance / mass)."""
135 f_drive = self.drive_force(v)
136 f_drag = 0.5 * self.rho * self.drag_area * v * abs(v)
137 f_roll = self.c_roll * self.mass * self.g
138 f_grade = self.mass * self.g * np.sin(self.grade)
139 return (f_drive - f_drag - f_roll - f_grade) / self.mass
140
141 DYNAMICS = KalmanDynamicsKind.CONTINUOUS
142
143 def f(self, x, u=None, w=None, t=None):
144 """Longitudinal derivative dx/dt = [v, accel(v)] (the force balance)."""
145 s = np.asarray(x, float).reshape(-1)
146 return np.array([[s[1]], [self.accel(s[1])]])
147
148 def after_step(self, x, u=None, t=None):
149 # The car does not roll backwards in a speed run.
150 s = np.asarray(x, float).reshape(-1)
151 s[1] = max(s[1], 0.0)
152 return s.reshape(-1, 1)
153
154 def h(self, x, v=None, t=None):
155 z = np.asarray(x, float).reshape(-1)[:1].reshape(1, 1)
156 return z if v is None else z + np.asarray(v, float).reshape(1, 1)

158 # ----------------------------------------------------------------------
159 # Speed-test metrics extracted from a filtered/sampled record
160 # ----------------------------------------------------------------------
161 def metrics(self, record):
162 """Headline figures of a run: 0–100 km/h, 0–60 mph, standing 400 m
163 (≈ quarter-mile) time and trap speed, and the terminal speed reached.
164
165 ``record`` is any record exposing ``time`` (s), ``xk[:, 0]`` (distance) and
166 ``xk[:, 1]`` (speed). Times are linearly interpolated between samples; entries
167 are ``None`` when the run is too short to reach a threshold.
168 """
169 t = np.asarray(record.time, float).reshape(-1)
170 x = np.asarray(record.xk)
171 dist, spd = x[:, 0], x[:, 1]
172
173 def t_at_speed(v_target):
174 i = np.argmax(spd >= v_target)
175 if not (spd >= v_target).any() or i == 0:
176 return None
177 t0, t1, v0, v1 = t[i - 1], t[i], spd[i - 1], spd[i]
178 return float(t0 + (t1 - t0) * (v_target - v0) / (v1 - v0)) if v1 > v0 else float(t1)
179
180 def t_at_dist(d_target):
181 i = np.argmax(dist >= d_target)
182 if not (dist >= d_target).any() or i == 0:
183 return None, None
184 t0, t1, d0, d1 = t[i - 1], t[i], dist[i - 1], dist[i]
185 frac = (d_target - d0) / (d1 - d0) if d1 > d0 else 1.0
186 return float(t0 + (t1 - t0) * frac), float(spd[i - 1] + (spd[i] - spd[i - 1]) * frac)
187
188 t_400, v_400 = t_at_dist(402.336) # standing quarter-mile (402.336 m)
189 return {
190 "0-100 km/h [s]": t_at_speed(100 / 3.6),
191 "0-60 mph [s]": t_at_speed(26.8224),
192 "400 m time [s]": t_400,
193 "400 m trap [km/h]": None if v_400 is None else v_400 * 3.6,
194 "top speed [km/h]": float(spd.max() * 3.6),
195 }
