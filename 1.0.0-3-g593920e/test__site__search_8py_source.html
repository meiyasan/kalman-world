50 def test_sections_carry_their_heading_anchor_and_text(self):
51 entries = [e for e in site_search.build_index(self.site) if e["p"] == "md_docs_guide.html"]
52 by_heading = {e["h"]: e for e in entries}
53 self.assertEqual(set(by_heading), {"", "The physics", "The model"})
54 self.assertEqual(by_heading["The physics"]["a"], "physics") # anchor inside the heading
55 self.assertEqual(by_heading["The model"]["a"], "code") # anchor just before it
56 self.assertIn("cooling rate from the thermometer", by_heading["The physics"]["x"])
57 self.assertIn("stray closer", by_heading[""]["x"], "a stray </p> does not end the page")
58 text = " ".join(e["x"] for e in entries)
59 self.assertNotIn("in a script", text)
60 self.assertNotIn("navigation text", text)
61 self.assertNotIn("12", by_heading["The model"]["x"], "line numbers are not text")
