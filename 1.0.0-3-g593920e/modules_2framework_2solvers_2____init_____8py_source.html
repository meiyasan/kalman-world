1"""
2Concrete spatial-discretization solvers (the *how in space*).
3
4Each shipped solver tags its
5:class:`~kalman.framework.constants.KalmanSolverKind` and composes a
6:class:`~kalman.framework.integrator.KalmanIntegrator` for the time march;
7``SOLVER_BY_KIND`` lets a model resolve its declared ``DEFAULT_SOLVER`` without
8importing a solver. FD/FV/FEM entries are reserved (raise on use).
9
10As with the integrators, the enum is the authority for **shipped** solvers only:
11:func:`register_solver` binds a custom discretization to a name that
12:func:`resolve_solver` accepts, and an unknown kind raises rather than silently
13degrading to the spectral default.
14"""

43# Custom discretizations registered at runtime: lowercase name -> class. Kept
44# apart from SOLVER_BY_KIND so the "registry covers its whole enum" invariant holds.
45SOLVER_BY_NAME: dict[str, type] = {}

50def _builtin_names() -> set:
51 return {k.name.lower() for k in SOLVER_BY_KIND}

54def available_solvers() -> list:
55 """Every resolvable solver name — shipped kinds plus registered ones."""
56 return sorted(_builtin_names() | set(SOLVER_BY_NAME))

59def register_solver(name, cls):
60 """Register a custom :class:`KalmanSolver` subclass under ``name``.
61
62 ``name`` may be a string (a new kind, resolvable everywhere a kind name is
63 accepted — including a ``KalmanSpec``'s ``bindings.solver``) or a
64 :class:`KalmanSolverKind` member, which **rebinds** that shipped kind.
65 Returns ``cls``, so it also works as a decorator.
66 """
67 if not (isinstance(cls, type) and issubclass(cls, KalmanSolver)):
68 raise TypeError(f"{cls!r} is not a KalmanSolver subclass.")
69 if isinstance(name, KalmanSolverKind):
70 SOLVER_BY_KIND[name] = cls
71 return cls
72 key = str(name).strip().lower()
73 if not key:
74 raise ValueError("a solver name must be a non-empty string.")
75 if key in _builtin_names():
76 raise ValueError(f"{key!r} is a shipped kind; pass KalmanSolverKind.{key.upper()} "
77 f"to rebind it deliberately.")
78 SOLVER_BY_NAME[key] = cls
79 return cls
80
81
82def unregister_solver(name) -> bool:
83 """Drop a custom registration (no-op for shipped kinds). Returns whether it existed."""
84 return SOLVER_BY_NAME.pop(str(name).strip().lower(), None) is not None

87def resolve_solver(kind=None, **kwargs) -> KalmanSolver:
88 """Instantiate the solver for ``kind``; ``kwargs`` forward to its constructor.
89
90 Accepts a :class:`KalmanSolverKind`, a registered name, a ``KalmanSolver``
91 subclass, or a ready instance (returned as-is, ``kwargs`` ignored). ``None``
92 gives the default (spectral). An unrecognised kind raises
93 :class:`ValueError` instead of silently falling back. Enum lookup keeps the
94 ``name`` fallback that makes it robust to ``%autoreload``.
95 """
96 if isinstance(kind, KalmanSolver):
97 return kind
98 if isinstance(kind, type) and issubclass(kind, KalmanSolver):
99 return kind(**kwargs)
100 if kind is None:
101 return DEFAULT_SOLVER(**kwargs)
102
103 if isinstance(kind, str):
104 key = kind.strip().lower()
105 cls = SOLVER_BY_NAME.get(key) or next(
106 (c for k, c in SOLVER_BY_KIND.items() if k.name.lower() == key), None)
107 else:
108 cls = SOLVER_BY_KIND.get(kind)
109 if cls is None: # %autoreload identity fallback
110 name = getattr(kind, "name", None)
111 cls = next((c for k, c in SOLVER_BY_KIND.items()
112 if getattr(k, "name", None) == name), None)
113
114 if cls is None:
115 raise ValueError(f"unknown solver kind {kind!r}; "
116 f"choose from {available_solvers()} or register_solver(...).")
117 return cls(**kwargs)
