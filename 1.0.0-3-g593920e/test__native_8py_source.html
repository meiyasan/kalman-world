71 def _joseph_ref(self, n, H, x, P, z, R):
72 keep = ~np.isnan(z)
73 if not keep.any():
74 return x.copy(), P.copy()
75 Hk, Rk, zk = H[keep, :], R[np.ix_(keep, keep)], z[keep]
76 y = zk - Hk @ x
77 S = Hk @ P @ Hk.T + Rk
78 K = P @ Hk.T @ np.linalg.pinv(S)
79 I = np.eye(n)
80 IKH = I - K @ Hk
81 x_out = x + K @ y
82 P_out = IKH @ P @ IKH.T + K @ Rk @ K.T
83 return x_out, P_out

185 def _python_forward(self, n, m, l, N, F, H, Q, R, B, L, zk, uk, x0, P0):
186 x, P = x0.copy(), P0.copy()
187 x_prior = np.empty((N, n))
188 P_prior = np.empty((N, n, n))
189 x_post = np.empty((N, n))
190 P_post = np.empty((N, n, n))
191 inno = np.empty((N, m))
192 for k in range(N):
193 xp = F @ x
194 if B is not None and l > 0:
195 xp = xp + B @ uk[k]
196 LQ = L @ Q @ L.T if L is not None else Q
197 Pp = F @ P @ F.T + LQ
198 x_prior[k], P_prior[k] = xp, Pp
199
200 z = zk[k]
201 keep = ~np.isnan(z)
202 Hk = H[keep, :]
203 Rk = R[np.ix_(keep, keep)]
204 y = z[keep] - Hk @ xp
205 inno[k] = z - H @ xp
206
207 S = Hk @ Pp @ Hk.T + Rk
208 K = Pp @ Hk.T @ np.linalg.pinv(S)
209 IKH = np.eye(n) - K @ Hk
210 x = xp + K @ y
211 P = IKH @ Pp @ IKH.T + K @ Rk @ K.T
212 x_post[k], P_post[k] = x, P
213 return x_prior, P_prior, x_post, P_post, inno

281 def _python_rts(self, n, N, x_post, P_post, x_prior, P_prior, F):
282 xs = np.empty((N, n)); Ps = np.empty((N, n, n))
283 xs[-1], Ps[-1] = x_post[-1], P_post[-1]
284 for k in reversed(range(N - 1)):
285 Pp = P_prior[k + 1]
286 C = P_post[k] @ F.T @ np.linalg.pinv(Pp)
287 xs[k] = x_post[k] + C @ (xs[k + 1] - x_prior[k + 1])
288 Ps[k] = P_post[k] + C @ (Ps[k + 1] - Pp) @ C.T
289 return xs, Ps
