46 Bayesian router;

48 Vector c{0.5, 0.5};

62 // --- Threshold ------------------------------------------------------------
63 {
64 const double m = 2.;
65 const Matrix nis = MakeNis(10, 3, /* good */ 1, m);
66 Vector c(3, 1. / 3.);
67 Vector L(3, 1.);
68
69 Threshold hard(2, 5, /* soft */ false);
70 const Vector mh = hard.Route(nis, c, L);
71 CHECK_CLOSE(mh(1), 1., 1e-12, "Threshold(hard): all weight on the model whose NIS is m");
72 CHECK_CLOSE(mh(0), 0., 1e-12, "Threshold(hard): nothing on the others");
73
74 Threshold soft(2, 5, /* soft */ true);
75 const Vector ms = soft.Route(nis, c, L);
76 CHECK_CLOSE(ms(0) + ms(1) + ms(2), 1., 1e-12, "Threshold(soft): normalised");
77 CHECK(ms(1) > ms(0) && ms(1) > ms(2), "Threshold(soft): still prefers the best match");
78 CHECK(ms(0) > 0., "Threshold(soft): keeps the others alive");
79
80 // A short history is averaged as it is (as routers/threshold.py does), not
81 // padded: padding with the chi2(1) mean would drag every model toward 1.
82 const Matrix shortNis = MakeNis(2, 3, 1, m);
83 const Vector me = hard.Route(shortNis, c, L);
84 CHECK_CLOSE(me(1) + me(0) + me(2), 1., 1e-12, "Threshold: normalised on a short history");
85 CHECK_CLOSE(me(1), 1., 0., "Threshold: a short history still picks the model at m");
86 Matrix two(2, 2, 0.);
87 two(0, 0) = 3.; two(1, 0) = 3.; // model 0 sits at 3 = m, model 1 at 0
88 Threshold three(3, 5, true);
89 const Vector ms3 = three.Route(two, Vector(2, 0.5), Vector(2, 1.));
90 CHECK_CLOSE(ms3(0), (1. / 1e-6) / (1. / 1e-6 + 1. / (3. + 1e-6)), 1e-12,
91 "Threshold(soft): the mean of the rows there are, with m = 3");
92 // Window 0 is the whole history (numpy's nis[-0:]): model 0 averages 3 = m over
93 // all four rows, though its last row says 0.
94 Matrix grow(4, 2, 1.);
95 grow(0, 0) = 12.; grow(1, 0) = 0.; grow(2, 0) = 0.; grow(3, 0) = 0.;
96 Threshold all(3, 0, false), last(3, 1, false);
97 CHECK_CLOSE(all.Route(grow, Vector(2, 0.5), Vector(2, 1.))(0), 1., 0., "Threshold: window 0 averages every row");
98 CHECK_CLOSE(last.Route(grow, Vector(2, 0.5), Vector(2, 1.))(1), 1., 0., "Threshold: window 1 reads the last row");
99
100 const Matrix empty(0, 3, 0.);
101 const Vector mu0 = hard.Route(empty, c, L);
102 CHECK_CLOSE(mu0(0), 1. / 3., 1e-12, "Threshold: uniform before anything is observed");
103 }

105 // --- Hybrid ---------------------------------------------------------------
106 {
107 Bayesian a;
108 Threshold b(2, 5, true);
109 const Matrix nis = MakeNis(10, 2, /* good */ 0, 2.);
110 Vector c{0.5, 0.5};
111 Vector L{1.0, 9.0}; // the likelihood favours model 1 ...
112 // ... while the NIS favours model 0
113
114 Hybrid pureA(&a, &b, 0.0);
115 Hybrid pureB(&a, &b, 1.0);
116 Hybrid mixed(&a, &b, 0.5);
117
118 const Vector ma = pureA.Route(nis, c, L);
119 const Vector mb = pureB.Route(nis, c, L);
120 const Vector mm = mixed.Route(nis, c, L);
121
122 CHECK_CLOSE(ma(0), a.Route(nis, c, L)(0), 1e-9, "Hybrid: alpha=0 is exactly router a");
123 CHECK_CLOSE(mb(0), Router::Normalise(b.Route(nis, c, L))(0), 1e-9,
124 "Hybrid: alpha=1 is exactly router b");
125 CHECK(mm(0) > ma(0) && mm(0) < mb(0),
126 "Hybrid: the blend sits between its two children");
127 CHECK_CLOSE(mm(0) + mm(1), 1., 1e-12, "Hybrid: normalised");
128
129 bool threw = false;
130 try { Hybrid bad(&a, &b, 1.5); }
131 catch (const std::runtime_error &) { threw = true; }
132 CHECK(threw, "Hybrid: an out-of-range alpha throws");
133
134 threw = false;

136 catch (const std::runtime_error &) { threw = true; }
137 CHECK(threw, "Hybrid: a null child throws");

140 // --- the learned path: router owns features, model owns the forward pass --

142 // No model attached: degrades to Bayesian rather than failing. A missing

175 // --- NN::MLP — the backend that needs no optional dependency at all -------
176 {
177 // Two modes, ten features. Layer picks whichever of the two feature groups
178 // is larger, so the expected answer is known without training anything.
179 NN::MLP mlp;
180 Matrix W(2, 10, 0.);
181 for (std::size_t i = 0; i < 10; ++i) W(i % 2, i) = 1.0;
182 Vector b(2, 0.);
183 mlp.AddLayer(W, b);
184
185 CHECK(mlp.IsReady(), "NN::MLP: ready once a layer is added");
186 CHECK(mlp.NInputs() == 10 && mlp.NOutputs() == 2, "NN::MLP: reports its shape");
187
188 Vector feat(10, 0.);
189 for (std::size_t i = 0; i < 10; i += 2) feat(i) = 3.0; // favours mode 0
190 const Vector probs = mlp(feat);
191 CHECK_CLOSE(probs(0) + probs(1), 1., 1e-12, "NN::MLP: outputs a probability vector");
192 CHECK(probs(0) > probs(1), "NN::MLP: the excited output wins");
193
194 // Large logits must not overflow to NaN — the softmax shift earns its keep.
195 Vector big(10, 0.);
196 for (std::size_t i = 0; i < 10; i += 2) big(i) = 400.0;
197 const Vector stable = mlp(big);
198 CHECK(std::isfinite(stable(0)) && std::isfinite(stable(1)),
199 "NN::MLP: softmax is shift-stabilised against overflow");
200 CHECK_CLOSE(stable(0), 1., 1e-9, "NN::MLP: saturates to a confident answer");
201
202 // Shape mistakes are caught when the layer is added, not at inference.
203 bool threw = false;
204 try { NN::MLP bad; bad.AddLayer(Matrix(2, 3, 0.), Vector(5, 0.)); }
205 catch (const std::runtime_error &) { threw = true; }
206 CHECK(threw, "NN::MLP: a weight/bias mismatch throws on AddLayer");
207
208 threw = false;
209 try { NN::MLP bad; bad.AddLayer(Matrix(2, 3, 0.), Vector(2, 0.));
210 bad.AddLayer(Matrix(2, 9, 0.), Vector(2, 0.)); }
211 catch (const std::runtime_error &) { threw = true; }
212 CHECK(threw, "NN::MLP: a layer that does not fit the previous one throws");
213
214 // And it drives the router end to end, with no optional dependency present.
215 Neural router(&mlp, 5);
216 CHECK(router.IsReady(), "Neural: ready once a model is attached");
217 Matrix nis(5, 2, 0.);
218 for (std::size_t r = 0; r < 5; ++r) { nis(r, 0) = 3.0; nis(r, 1) = 0.0; }
219 Vector c{0.5, 0.5}, L{1.0, 1.0};
220 const Vector mu = router.Route(nis, c, L);
221 CHECK_CLOSE(mu(0) + mu(1), 1., 1e-12, "Neural + NN::MLP: normalised");
222 CHECK(mu(0) > mu(1), "Neural + NN::MLP: routes on the learned policy");
223
224 // A window that disagrees with the policy is a shape error, not a guess.
225 Neural wrong(&mlp, 7);
226 threw = false;
227 try { wrong.Route(nis, c, L); }
228 catch (const std::runtime_error &) { threw = true; }
229 CHECK(threw, "Neural: a window mismatching the policy throws");
230 }
