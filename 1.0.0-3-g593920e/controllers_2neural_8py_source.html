8class NeuralKalmanController(KalmanController):
9 """Neural controller trained offline to imitate :class:`MPPIKalmanController`.
10
11 ``fit_mppi()`` runs the same warm-started, receding-horizon MPPI planner online at a
12 grid of simulated states, records every ``(state, u*)`` decision, and fits a small MLP
13 to reproduce it — turning an expensive online Monte-Carlo optimisation into a single
14 cheap forward pass at deployment time.
15
16 Parameters
17 ----------
18 features : tuple of str
19 Which inputs the MLP sees. Any subset of ``("error", "P", "nu")``:
20
21 * ``"error"`` — state error ``x̂ − x_ref`` (shape n)
22 * ``"P"`` — navigation uncertainty ``√diag(P)`` (shape n)
23 * ``"nu"`` — latest innovation ``z − H x̂⁻`` (shape m)
24
25 Default is ``("error",)``. MPPI's own decision ``u*`` never depends on ``P`` or
26 ``ν`` — it plans from a state, not a belief — so for a *full-state-feedback* MPPI
27 teacher (the common case) those two channels carry no signal, only noise, unless the
28 training rollouts are given genuinely representative ``P``/``ν`` statistics to match
29 what the controller will see at deployment (``fit_mppi`` does this — see below).
30 Feeding the network a channel that is constant or always zero during training but
31 varies at test time is a real train/test mismatch, not a free extra feature: adding
32 ``"P"`` or ``"nu"`` without matched statistics measurably *hurts* closed-loop cost.
33 Only opt in if your controller's optimal action genuinely depends on uncertainty or
34 innovation (e.g. an uncertainty-aware or predictive policy you designed on purpose).
35 """
36
37 def __init__(self, model, hidden=64, features=("error",), **kw):
38 super().__init__(model, **kw)
39 self._hidden = hidden
40 self._features = tuple(features)
41 n, m = model.n, model.m
42 self._input_dim = (
43 (n if "error" in self._features else 0) +
44 (n if "P" in self._features else 0) +
45 (m if "nu" in self._features else 0)
46 )
47 self._x_mean = np.zeros(self._input_dim)
48 self._x_std = np.ones(self._input_dim)

50 # ------------------------------------------------------------------
51 def _build_input(self, x_hat, P, nu):
52 parts = []
53 if "error" in self._features:
54 parts.append((x_hat - self.x_refx_ref).ravel())
55 if "P" in self._features:
56 parts.append(np.sqrt(np.abs(np.diag(P))))
57 if "nu" in self._features:
58 parts.append(np.zeros(self.model.m) if nu is None
59 else np.asarray(nu).ravel())
60 return np.concatenate(parts) if parts else np.zeros(1)

62 def _policy(self, x_hat, P, t, *, nu=None):
63 if self._weights is None:
64 raise RuntimeError("Call fit_mppi() or load() before using NeuralKalmanController.")
65 inp = self._build_input(x_hat, P, nu)
66 return self._mlp_forward((inp - self._x_mean) / self._x_std)
67
68 def _mlp_forward(self, x):
69 W1, b1 = self._weights["W1"], self._weights["b1"]
70 W2, b2 = self._weights["W2"], self._weights["b2"]
71 W3, b3 = self._weights["W3"], self._weights["b3"]
72 h = np.tanh(W1 @ x + b1)
73 h = np.tanh(W2 @ h + b2)
74 return W3 @ h + b3

76 def _init_weights(self, seed=None):
77 rng = np.random.default_rng(seed)
78 d_in, d_h, d_out = self._input_dim, self._hidden, max(self.model.l, 1)
79 scale1 = np.sqrt(2.0 / d_in)
80 scale2 = np.sqrt(2.0 / d_h)
81 self._weights = {
82 "W1": rng.normal(0, scale1, (d_h, d_in)),
83 "b1": np.zeros(d_h),
84 "W2": rng.normal(0, scale2, (d_h, d_h)),
85 "b2": np.zeros(d_h),
86 "W3": rng.normal(0, scale2, (d_out, d_h)),
87 "b3": np.zeros(d_out),
88 }

90 def fit_mppi(self, n_rollouts=100, horizon=15, noise_sigma=0.1, lam=1.0,
91 epochs=200, lr=1e-3, n_episodes=8, rollout_steps=30, x0_std=1.0, seed=None,
92 n_iters=3):
93 """Train by **imitating** a warm-started, receding-horizon MPPI planner on ``model.f()``.
94
95 This runs the same planner as :class:`MPPIKalmanController` — see its docstring for
96 why warm-starting matters — as the *teacher*, and distils its decisions into the MLP:
97
98 For every step of ``n_episodes`` simulated episodes:
99
100 1. Starting from the nominal sequence carried over from the previous step (zero on
101 the first step of an episode), refine it over ``n_iters`` rounds: sample
102 ``n_rollouts`` perturbations around the nominal, roll each through
103 ``model.f(x, u+δu, t)`` (full nonlinear physics, so any actuator clip the model
104 enforces internally is respected during planning), weight by
105 ``w_i = exp(−J_i / λ)`` on cost ``J = Σ eᵀ Qcost e + uᵀ Rcost u``, and replace the
106 nominal with the weighted average.
107 2. Record ``(state, u*_0)`` — the *first* action of the refined nominal sequence.
108 3. Apply that action, advance the true state, shift the nominal by one step
109 (receding horizon), and repeat.
110
111 Refining a warm-started nominal (rather than resampling raw noise from zero every
112 step, as one-shot random shooting does) is what lets MPPI actually converge toward
113 its cost-optimal sequence within a practical rollout budget — without it, the
114 collected ``u*`` targets are far noisier and the imitating MLP inherits that noise
115 as a training-time bias.
116
117 Replanning from the actual current state at every step — rather than committing to
118 one open-loop sequence per episode — is what makes the collected dataset genuinely
119 state-conditioned, so the trained MLP generalises as a feedback policy instead of
120 just memorising a single nominal trajectory.
121
122 Episodes start near ``model.x0`` (the system's typical operating point — not
123 ``x_ref``, which may sit at an unreachable corner of state space, e.g. zero speed)
124 with spread ``x0_std``.
125
126 Rollouts are advanced with the model's own process noise ``Q`` injected at every
127 step, matching how :func:`closed_loop_run`-style deployment actually behaves. A
128 clean, noise-free rollout only ever visits *one* smooth corrective arc per episode
129 — it converges toward ``x_ref`` and stays there, so the dataset ends up almost
130 entirely one-sided (states still short of the target, actions that push toward it)
131 and starves the network of examples on the *other* side of the target (small
132 negative error, having overshot) that real sensor/process noise puts it in
133 constantly once deployed. Without noise here, the MLP has no data to learn a
134 symmetric policy from and inherits whatever asymmetric bias the unseen side
135 defaults to.
136 """
137 from .mppi import MPPIKalmanController
138
139 rng = np.random.default_rng(seed)
140 model = self.model
141 q_std = np.sqrt(np.diag(model.Q()))
142 n, l, m = model.n, max(model.l, 1), model.m
143 dt = 1.0 / model.fs if model.fs else 1.0
144
145 # Realistic P(t)/S(t) schedule for the "P"/"nu" features, if requested — the
146 # covariance recursion (unlike the state trajectory) doesn't depend on which control
147 # is applied, so it can be precomputed once from the model's own F/H/Q/R rather than
148 # left as the constant placeholder that would silently mismatch what a real filter
149 # produces at deployment.
150 need_P = "P" in self._features or "nu" in self._features
151 if need_P:
152 F0, H0, Q0, R0 = model.F(), model.H(), model.Q(), model.R()
153 P0 = model.P0.copy() if hasattr(model, "P0") else np.eye(n)
154 P_sched, S_sched = [], []
155 Pc = P0
156 for _ in range(rollout_steps + 1):
157 Pm = F0 @ Pc @ F0.T + Q0
158 S = H0 @ Pm @ H0.T + R0
159 K = Pm @ H0.T @ np.linalg.inv(S)
160 Pc = (np.eye(n) - K @ H0) @ Pm
161 P_sched.append(Pm)
162 S_sched.append(S)
163 else:
164 P_dummy = np.eye(n)
165
166 planner = MPPIKalmanController(
167 model, n_rollouts=n_rollouts, horizon=horizon, noise_sigma=noise_sigma,
168 lam=lam, n_iters=n_iters, seed=seed, Qcost=self.Qcost, Rcost=self.Rcost,
169 x_ref=self.x_refx_ref,
170 )
171
172 dataset_x, dataset_u = [], []
173
174 for _ in range(n_episodes):
175 x = (model.x0.ravel() + rng.normal(0, x0_std, n)).reshape(-1, 1)
176 planner.reset()
177
178 for step in range(rollout_steps):
179 t0 = step * dt
180 u0 = planner._policy(x, None, t0)
181
182 if need_P:
183 P_here, S_here = P_sched[step], S_sched[step]
184 nu_here = (rng.multivariate_normal(np.zeros(m), S_here).reshape(-1, 1)
185 if "nu" in self._features else None)
186 else:
187 P_here, nu_here = P_dummy, None
188
189 inp = self._build_input(x, P_here, nu_here)
190 dataset_x.append(inp)
191 dataset_u.append(u0.ravel())
192
193 x = np.asarray(model.f(x, u=u0, t=t0), dtype=float).reshape(-1, 1)
194 x = x + rng.normal(0, q_std, n).reshape(-1, 1)
195
196 X = np.array(dataset_x)
197 Y = np.array(dataset_u)
198
199 # Raw input features span wildly different scales (e.g. a lateral offset ~O(1)
200 # next to a cruise speed ~O(15)) — normalising is what lets gradient descent see
201 # the small-scale features at all, instead of the largest feature dominating.
202 self._x_mean = X.mean(axis=0)
203 self._x_std = X.std(axis=0)
204 self._x_std[self._x_std < 1e-8] = 1.0
205 Xn = (X - self._x_mean) / self._x_std
206
207 self._init_weights(seed=seed)
208 self._train_mlp(Xn, Y, epochs=epochs, lr=lr)
209 return self

211 def _train_mlp(self, X, Y, epochs=100, lr=1e-3):
212 """Adam-optimised gradient descent — plain SGD converges too unevenly across
213 parameters of very different gradient scale (first layer vs output layer)."""
214 params = self._weights
215 m_t = {k: np.zeros_like(v) for k, v in params.items()}
216 v_t = {k: np.zeros_like(v) for k, v in params.items()}
217 beta1, beta2, eps = 0.9, 0.999, 1e-8
218
219 N = len(X)
220 for step in range(1, epochs + 1):
221 W1, b1 = params["W1"], params["b1"]
222 W2, b2 = params["W2"], params["b2"]
223 W3, b3 = params["W3"], params["b3"]
224
225 h1 = np.tanh(X @ W1.T + b1)
226 h2 = np.tanh(h1 @ W2.T + b2)
227 y_pred = h2 @ W3.T + b3
228
229 err = y_pred - Y
230
231 grads = {}
232 grads["W3"] = (2.0 / N) * err.T @ h2
233 grads["b3"] = (2.0 / N) * err.sum(axis=0)
234 dh2 = (2.0 / N) * err @ W3 * (1 - h2 ** 2)
235 grads["W2"] = dh2.T @ h1
236 grads["b2"] = dh2.sum(axis=0)
237 dh1 = dh2 @ W2 * (1 - h1 ** 2)
238 grads["W1"] = dh1.T @ X
239 grads["b1"] = dh1.sum(axis=0)
240
241 for k in params:
242 m_t[k] = beta1 * m_t[k] + (1 - beta1) * grads[k]
243 v_t[k] = beta2 * v_t[k] + (1 - beta2) * grads[k] ** 2
244 m_hat = m_t[k] / (1 - beta1 ** step)
245 v_hat = v_t[k] / (1 - beta2 ** step)
246 params[k] = params[k] - lr * m_hat / (np.sqrt(v_hat) + eps)
247
248 self._weights = params
