28 bool Factory::Register(const std::string &name, Creator create)
29 {
30 if (name.empty()) throw std::invalid_argument("Factory::Register: the model name is empty.");
31 if (!create) throw std::invalid_argument("Factory::Register: no creator for '" + name + "'.");
32
33 std::lock_guard<std::mutex> lock(fMutex);
34 const auto it = fModels.find(name);
35 if (it != fModels.end())
36 {
37 const std::string what = "Factory::Register: '" + name + "' is already registered"
38 + (it->second.plugin.empty() ? "" : " (by " + it->second.plugin + ")")
39 + "; model names are unique.";
40 // Inside a dlopen the registrar is a static initialiser, and an exception
41 // escaping one of those terminates the process. Park the message instead
42 // and let LoadPlugin() throw it once the loader has returned.
43 if (!fLoading.empty()) { fLoadError = what; return false; }
44 throw std::runtime_error(what);
45 }
46 fModels[name] = Registration{std::move(create), fLoading};
47 return true;
48 }

77 std::unique_ptr<Model> Factory::Create(const std::string &name) const
78 {
79 Creator create;
80 {
81 std::lock_guard<std::mutex> lock(fMutex);
82 const auto it = fModels.find(name);
83 if (it == fModels.end())
84 {
85 std::ostringstream os;
86 os << "Factory::Create: unknown model '" << name << "'; registered:";
87 if (fModels.empty()) os << " (none)";
88 for (const auto &kv : fModels) os << " '" << kv.first << "'";
89 os << ".";
90 throw std::out_of_range(os.str());
91 }
92 create = it->second.create;
93 }
94 // Run the creator outside the lock: a constructor may itself consult the factory.
95 std::unique_ptr<Model> model = create();
96 if (!model) throw std::runtime_error("Factory::Create: the creator for '" + name + "' returned null.");
97 return model;
98 }

122 std::vector<std::string> Factory::LoadPlugin(const std::string &path)
123 {
124 if (path.empty()) throw std::invalid_argument("Factory::LoadPlugin: the path is empty.");
125
126 {
127 std::lock_guard<std::mutex> lock(fMutex);
128 if (std::find(fPlugins.begin(), fPlugins.end(), path) != fPlugins.end())
129 {
130 std::vector<std::string> names;
131 for (const auto &kv : fModels)
132 if (kv.second.plugin == path) names.push_back(kv.first);
133 return names;
134 }
135 if (!fLoading.empty())
136 throw std::runtime_error("Factory::LoadPlugin: '" + path + "' requested while '"
137 + fLoading + "' is still loading.");
138 fLoading = path;
139 fLoadError.clear();
140 }
141
142 // RTLD_NOW so an unresolved symbol fails here, with a message, rather than
143 // at the first call into the model. RTLD_LOCAL keeps two plugins that
144 // both define an internal helper from clashing with each other.
145 void *handle = ::dlopen(path.c_str(), RTLD_NOW | RTLD_LOCAL);
146
147 std::lock_guard<std::mutex> lock(fMutex);
148 fLoading.clear();
149
150 if (!handle)
151 {
152 const char *why = ::dlerror();
153 throw std::runtime_error("Factory::LoadPlugin: cannot load '" + path + "': "
154 + (why ? why : "unknown loader error") + ".");
155 }
156
157 std::vector<std::string> names;
158 for (const auto &kv : fModels)
159 if (kv.second.plugin == path) names.push_back(kv.first);
160
161 if (!fLoadError.empty())
162 {
163 // The library is in memory (dlopen has no undo for its initialisers), but
164 // it is not recorded as loaded and whatever it did register is withdrawn,
165 // so the failure is complete rather than half-applied.
166 for (const auto &name : names) fModels.erase(name);
167 const std::string what = fLoadError;
168 fLoadError.clear();
169 throw std::runtime_error("Factory::LoadPlugin: '" + path + "' rejected — " + what);
170 }
171
172 if (names.empty())
173 throw std::runtime_error("Factory::LoadPlugin: '" + path + "' loaded but registered no model; "
174 "did it forget KALMAN_REGISTER_MODEL, or was it linked against "
175 "a different libKalman?");
176
177 fPlugins.push_back(path); // the handle is kept open for the life of the process
178 return names;
179 }
