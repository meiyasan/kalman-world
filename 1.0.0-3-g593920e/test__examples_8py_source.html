38 def test_cxx_and_python_filter_the_same_readings_alike(self):
39 with tempfile.TemporaryDirectory() as out:
40 cxx = subprocess.run([PROGRAM, out, toy_plugin()], capture_output=True, text=True, timeout=120)
41 self.assertEqual(cxx.returncode, 0, cxx.stderr)
42 self.assertTrue(os.path.exists(os.path.join(out, "toy_sensor.csv")))
43 self.assertTrue(os.path.exists(os.path.join(out, "toy_estimate.csv")))
44
45 env = dict(os.environ, KALMAN_PLUGIN_PATH=os.path.dirname(toy_plugin()))
46 py = subprocess.run([sys.executable, SCRIPT, out], capture_output=True, text=True,
47 timeout=300, cwd=REPO, env=env)
48 self.assertEqual(py.returncode, 0, py.stderr)
49
50 # Both report what the filter learned; the Python lines also give the
51 # distance to the C++ program's estimate.
52 learned = re.search(r"learned : k = ([\d.]+) \+/- ([\d.]+) /s\s+\‍(truth ([\d.]+)", cxx.stdout)
53 self.assertIsNotNone(learned, cxx.stdout)
54 k, sigma, truth = map(float, learned.groups())
55 self.assertLess(abs(k - truth), 3 * sigma, "the cooling rate is learned within 3 sigma")
56 self.assertLess(abs(k - truth), 0.1 * truth, "and to better than 10%")
57 self.assertIn(learned.group(0).split("(")[0].strip(), py.stdout, "Python reports the same estimate")
58
59 differences = [float(x) for x in re.findall(r"[Tk] ([\d.]+e[-+]\d+)", py.stdout)]
60 self.assertEqual(len(differences), 6, py.stdout)
61 self.assertEqual(max(differences[:2]), 0.0, "the model written in C++: the C++ program's own computation")
62 # The two models take their Jacobians by differences of f, whose last
63 # digit differs between numpy and the C math library.
64 self.assertLess(max(differences[2:4]), 1e-6, "the model written in Python, run by the C++ filter")
65 self.assertLess(max(differences[4:]), 1e-6, "the model written in Python, run by the Python loop")
