1"""
2Astrodynamics runtime services for the Kalman framework.
3
4**Physical constants** are sourced from :mod:`astropy.constants` / :mod:`astropy.units`
5wherever astropy ships them (Earth, Sun, Jupiter, the astronomical unit, *G*, standard
6gravity ``g0``). The Moon and the remaining planets — which astropy does *not* expose as
7named constants — use cited IAU 2015 / JPL DE440 GM and mean-radius values. Everything is
8then expressed in the framework's **scaled units** (``length = 1 Mm = 1e6 m``,
9``time = 1 ks = 1e3 s``) that keep the orbital/ascent models well-conditioned.
10
11This module also provides the shared services both app modules build on:
12
13* :func:`rk4` — the 4th-order Runge–Kutta step used by every integrator;
14* :func:`body_state` / :func:`planet_positions` — **exact** ephemeris position & velocity of
15 any solar-system body at a given time (astropy's network-free ``builtin`` ephemeris), with
16 an analytic circular-orbit fallback when astropy is unavailable;
17* :func:`launch_site` / :func:`launch_site_state` — launch-pad coordinates resolved from a
18 **library** (``EarthLocation.of_address`` online geocoding) and their Earth-rotation state.
19"""

63# Scaled units (1 Mm, 1 ks) and conversions
64# ---------------------------------------------------------------------------
65LU, TU = 1.0e6, 1.0e3 # 1 Mm (1e6 m), 1 ks (1e3 s)
66KM = 1.0e3 / LU # 1 km in LU (= 1e-3)
67DAY = 86400.0 / TU # 1 day in TU (= 86.4)
68VU = LU / TU # 1 scaled velocity unit = 1 km/s
69_GM_SCALE = TU**2 / LU**3 # SI GM (m^3 s^-2) -> scaled

130def moon_position(t, phase=0.0):
131 """Analytic Moon position (scaled, geocentric) on a circular orbit — fallback only."""
132 a = OMEGA_MOON * t + phase
133 return D_MOON * np.array([np.cos(a), np.sin(a), 0.0])

142def body_state(name, time=None, *, origin="heliocentric"):
143 """Exact position **and** velocity of a solar-system body in *scaled units*.
144
145 Returns ``(r, v)`` with position in Mm and velocity in km/s (= scaled ``LU/TU``).
146 ``origin`` is ``'heliocentric'`` (default, Sun-centred), ``'bary'`` (solar-system
147 barycentre) or ``'geocentric'`` (Earth-centred — used for the trans-lunar frame).
148 Uses astropy's network-free ``builtin`` ephemeris; if astropy is unavailable, falls
149 back to the analytic circular orbits for the Sun/Moon.
150 """
151 try:
152 from astropy.coordinates import get_body_barycentric_posvel, solar_system_ephemeris
153 import astropy.units as u
154 except Exception: # no astropy -> analytic fallback
155 r = (moon_position(0.0) if name.lower() == "moon" else
156 sun_position(0.0) if name.lower() == "sun" else np.zeros(3))
157 return r, np.zeros(3)
158
159 t = _as_time(time)
160 with solar_system_ephemeris.set(EPHEMERIS):
161 r_b, v_b = get_body_barycentric_posvel(name, t)
162 if origin in ("heliocentric", "sun"):
163 r0, v0 = get_body_barycentric_posvel("sun", t)
164 r_b, v_b = r_b - r0, v_b - v0
165 elif origin in ("geocentric", "earth"):
166 r0, v0 = get_body_barycentric_posvel("earth", t)
167 r_b, v_b = r_b - r0, v_b - v0
168 r = r_b.xyz.to_value(u.m) / LU
169 v = v_b.xyz.to_value(u.m / u.s) / VU
170 return r, v

181def ephemeris_samples(bodies, times, *, origin="heliocentric"):
182 """**Vectorised** ephemeris over an array of times.
183
184 Returns ``{body: (R, V)}`` where ``R``/``V`` are ``(len(times), 3)`` arrays of scaled
185 position [Mm] and velocity [km/s]. astropy evaluates the whole time array at once, so this
186 is the fast path for building interpolators that an N-body integrator can sample cheaply.
187 """
188 from astropy.coordinates import get_body_barycentric_posvel, solar_system_ephemeris
189 import astropy.units as u
190 out = {}
191 with solar_system_ephemeris.set(EPHEMERIS):
192 r0 = v0 = None
193 if origin in ("heliocentric", "sun"):
194 r0, v0 = get_body_barycentric_posvel("sun", times)
195 elif origin in ("geocentric", "earth"):
196 r0, v0 = get_body_barycentric_posvel("earth", times)
197 for b in bodies:
198 rb, vb = get_body_barycentric_posvel(b, times)
199 if r0 is not None:
200 rb, vb = rb - r0, vb - v0
201 R = (rb.xyz.to_value(u.m) / LU).T # (N, 3)
202 V = (vb.xyz.to_value(u.m / u.s) / VU).T # (N, 3)
203 out[b] = (np.atleast_2d(R), np.atleast_2d(V))
204 return out

207def times_from_epoch(epoch, t_scaled):

245def launch_site(name):
246 """Return an ``astropy.coordinates.EarthLocation`` for a launch site.
247
248 Coordinates are resolved **from a library**: ``EarthLocation.of_address`` (online
249 OpenStreetMap geocoding); any address string works. Results are cached. When offline,
250 a small built-in coordinate table covers the most common pads.
251 """
252 key = name.strip().lower()
253 if key in _site_cache:
254 return _site_cache[key]
255 from astropy.coordinates import EarthLocation
256 import astropy.units as u
257 try:
258 loc = EarthLocation.of_address(name)
259 except Exception:
260 if key not in _SITE_FALLBACK:
261 raise
262 lat, lon, h = _SITE_FALLBACK[key]
263 loc = EarthLocation(lat=lat * u.deg, lon=lon * u.deg, height=h * u.m)
264 _site_cache[key] = loc
265 return loc

268def launch_site_state(name, time=None):
269 """Scaled GCRS position & velocity of a launch site, **including Earth's rotation**.
270
271 Returns ``(r, v)`` with position in Mm and velocity in km/s. The velocity is the
272 inertial (GCRS) velocity of the rotating pad — ~0.46 km/s eastward at the equator —
273 via ``EarthLocation.get_gcrs_posvel`` (astropy uses online IERS Earth-orientation data
274 when available, else the bundled IERS-B table).
275 """
276 import astropy.units as u
277 loc = launch_site(name)
278 t = _as_time(time)
279 r_gcrs, v_gcrs = loc.get_gcrs_posvel(t)
280 r = r_gcrs.xyz.to_value(u.m) / LU
281 v = v_gcrs.xyz.to_value(u.m / u.s) / VU
282 return r, v
