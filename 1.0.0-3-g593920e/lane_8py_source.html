28class CarLaneModel(NativeTwin, ExtendedKalmanModel):
29 """Lateral lane-keeping with steering control (bicycle kinematics).
30
31 State ``x = [y, ψ, v]``:
32 - ``y`` — lateral offset from lane centre [m]
33 - ``ψ`` — heading angle relative to lane [rad]
34 - ``v`` — longitudinal speed [m/s] (driven externally or constant)
35
36 Control ``u = [δ]`` — steering angle [rad].
37
38 Dynamics (discrete, Euler)::
39
40 y_{k+1} = y_k + v_k · sin(ψ_k) · dt
41 ψ_{k+1} = ψ_k + v_k · tan(δ_k) / L · dt
42 v_{k+1} = v_k (constant speed)
43
44 Measurement: the lateral offset seen by a lane camera, ``z = y + noise``.
45
46 ``n = 3, m = 1, l = 1``.
47
48 Parameters
49 ----------
50 wheelbase : float
51 Distance between the axles ``L`` [m].
52 v0 : float
53 Speed [m/s].
54 lateral_uncertainty : float
55 Standard deviation of the measured lateral offset [m] (a lane camera: a few cm).
56 q_lat, q_head, q_vel : float
57 Process-noise variances per step, for ``y`` [m²], ``ψ`` [rad²] and ``v`` [(m/s)²].
58 The defaults are 1 cm, 1 mrad and 1 cm/s per 20 ms step (standard deviations).
59 steer_limit : float
60 Largest steering angle [rad]. The lateral acceleration is ``v² tan δ / L``; at
61 15 m/s the default 0.1 rad gives 8.4 m/s², about what tyres hold on a dry road.
62 """
63
64 NAME = "Car Lane Keeping"
65 DESCRIPTION = "Bicycle-kinematic lateral lane-keeping with steering control."
66 VERSION = "1.0"

112 def clip_u(self, u):
113 """Steering-angle actuator limit: ``|delta| <= steer_limit``."""
114 u_arr = np.asarray(u, float).reshape(-1).copy()
115 if len(u_arr) > 0:
116 u_arr[0] = np.clip(u_arr[0], -self.steer_limit, self.steer_limit)
117 return u_arr.reshape(-1, 1)

123 def f(self, x, u=None, w=None, t=None):
124 """Bicycle-kinematic derivative ẋ = [v·sin ψ, v·tan δ / L, 0]."""
125 x = np.asarray(x, float).reshape(-1)
126 y, psi, v = x[0], x[1], x[2]
127 delta = 0.0
128 if u is not None:
129 u_arr = self.clip_uclip_u(u).reshape(-1)
130 if len(u_arr) > 0:
131 delta = u_arr[0]
132 return np.array([[v * np.sin(psi)], [v * np.tan(delta) / self.wheelbase], [0.0]])

135 x = np.asarray(x, float).reshape(-1)

139 def control_cost(self):
140 """Quadratic cost weights ``(Qcost, Rcost, x_ref)``: 1 per m² of lateral offset
141 and 100 per rad² of steering, so 1 m of offset costs as much as 0.1 rad. The
142 reference is the lane centre, heading along the lane, at the cruise speed."""
143 Qcost = np.diag([1.0, 0.0, 0.0])
144 Rcost = np.array([[100.0]])
145 x_ref = np.array([[0.0], [0.0], [self.speed]])
146 return Qcost, Rcost, x_ref
