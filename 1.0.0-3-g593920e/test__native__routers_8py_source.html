23def inputs(rows, M, seed=0):
24 rng = np.random.default_rng(seed)
25 nis = rng.chisquare(2, size=(rows, M)) * rng.uniform(0.5, 3.0, size=M)
26 c = rng.dirichlet(np.ones(M))
27 likelihoods = rng.uniform(0.0, 2.0, size=M)
28 likelihoods[rng.integers(M)] = 0.0 # one mode written off
29 return nis, c, likelihoods

37class _Softmax(KalmanRouter):
38 """A router written in Python only (stands in for a NeuralRouter)."""
39
40 def route(self, nis_arr, c, likelihoods):
41 z = -nis_arr[-1]
42 return np.exp(z - z.max()) / np.exp(z - z.max()).sum()

48 def agree(self, router, atol=1e-15, shapes=((1, 2), (3, 2), (12, 3), (40, 5))):
49 for seed, (rows, M) in enumerate(shapes):
50 nis, c, L = inputs(rows, M, seed)
51 mu = router(nis, c, L)
52 ref = reference_route(router, nis, c, L)
53 np.testing.assert_allclose(mu, ref, rtol=0, atol=atol, err_msg=f"{rows} x {M}")
54 self.assertAlmostEqual(mu.sum(), 1.0, places=12)
55 self.assertIsNotNone(router.__dict__.get("_native_state"), "the C++ router ran")

110 def test_mode_probabilities_agree(self):
