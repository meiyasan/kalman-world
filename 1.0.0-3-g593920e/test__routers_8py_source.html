231class _ArchitectureTrainingMixin:
232 """Shared functional checks: build → call → fit reduces loss → reproducible."""
233
234 ARCH = None
235 BUILD_KW = {}
236 EPOCHS = 60
237
238 def _data(self, seed=0, N=60, window=4, M=2):
239 rng = np.random.default_rng(seed)
240 d_in = window * M
241 X = rng.normal(size=(N, d_in))
242 y = (X[:, 0] > X[:, 1]).astype(int)
243 return X, y, d_in, M
244
245 def test_build_succeeds(self):
246 model = self.ARCH(**self.BUILD_KW)
247 result = model.build(8, 2, seed=0)
248 self.assertIs(result, model) # build() returns self
249
250 def test_call_returns_prob_vector(self):
251 X, y, d_in, M = self._data()
252 model = self.ARCH(**self.BUILD_KW)
253 model.build(d_in, M, seed=0)
254 out = model(X[0])
255 self.assertEqual(out.shape, (M,))
256 self.assertAlmostEqual(float(out.sum()), 1.0, places=6)
257 self.assertTrue(np.all(out >= 0))
258
259 def test_fit_reduces_loss(self):
260 X, y, d_in, M = self._data()
261 model = self.ARCH(**self.BUILD_KW)
262 model.build(d_in, M, seed=0)
263
264 def ce():
265 return np.mean([-np.log(float(model(x)[int(yi)]) + 1e-12)
266 for x, yi in zip(X, y)])
267
268 loss_before = ce()
269 model.fit(X, y, epochs=self.EPOCHS, lr=5e-3)
270 loss_after = ce()
271 self.assertLess(loss_after, loss_before)
272
273 def test_reproducible_with_seed(self):
274 X, y, d_in, M = self._data()
275 m1 = self.ARCH(**self.BUILD_KW); m1.build(d_in, M, seed=3)
276 m2 = self.ARCH(**self.BUILD_KW); m2.build(d_in, M, seed=3)
277 np.testing.assert_allclose(m1(X[0]), m2(X[0]), atol=1e-6)

337 def _data(self, seed=0):

343 def test_save_load_round_trip(self):
344 import tempfile, os
