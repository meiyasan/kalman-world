8def _window_feats(nu_window: np.ndarray) -> np.ndarray:
9 """[std, lag-1 autocorr, lag-3 autocorr] of a window of scalar innovations.
10
11 Std alone is what ``AdaptiveQRKalmanController``'s trace-ratio test uses -- it
12 reacts strongly to noise entering *directly* at the measured state, but is nearly
13 blind to noise on an *unmeasured* state that only reaches the measurement
14 indirectly through the dynamics coupling (e.g. heading noise reaching a
15 lateral-position measurement), because that shows up as added smoothness /
16 autocorrelation in the innovation, not added instantaneous variance.
17 """
18 nu = np.asarray(nu_window, dtype=float).ravel()
19 a = nu - nu.mean()
20 std = float(np.std(nu))
21
22 def _autocorr(lag):
23 if len(a) <= lag:
24 return 0.0
25 x, y = a[:-lag], a[lag:]
26 denom = np.std(x) * np.std(y)
27 return float(np.mean(x * y) / denom) if denom > 1e-12 else 0.0
28
29 return np.array([std, _autocorr(1), _autocorr(3)])

32class NeuralAdaptiveQRKalmanController(KalmanController):
33 """Classify *which* state's process noise is structurally elevated, from a window
34 of innovations, and apply a **per-state** Q correction — something the single
35 scalar in :class:`AdaptiveQRKalmanController` cannot do (it must scale every state's
36 Q by the same factor, whether that state's noise actually changed or not).
37
38 Usage::
39
40 nqr = NeuralAdaptiveQRKalmanController(model, window=20)
41 nqr.fit(regime_scales={
42 "nominal": np.array([1.0, 1.0, 1.0]),
43 "lat": np.array([25.0, 1.0, 1.0]),
44 "head": np.array([1.0, 25.0, 1.0]),
45 })
46 record = filter.forward(zk, uk=nqr)
47
48 ``fit()`` simulates labelled synthetic episodes for each named regime (using the
49 model's own ``f()``/``h()``, exactly like ``NeuralKalmanController.fit_mppi()``
50 simulates its own training data), trains a small classifier on windowed innovation
51 features, and stores ``regime_scales`` so a predicted label can be looked up as a
52 concrete per-state multiplier at inference time.
53
54 This is a *classifier*, not a continuous regressor: it identifies which of a small
55 number of known noise "shapes" best matches the recent innovation window and
56 applies that shape's multiplier vector. It does not invent a shape it wasn't
57 trained on.
58 """

76 def _policy(self, x_hat, P, t, *, nu=None):
77 return np.zeros((max(self.model.l, 1), 1))

79 def _update(self, x_hat, P, t, *, nu=None):
80 if nu is None or self._model_nn is None:
81 return None
82 if self._Q_nominal is None:
83 self._Q_nominal = self.model.Q(t=t).copy()
84
85 self._nu_history.append(float(np.asarray(nu).ravel()[0]))
86 if len(self._nu_history) < self.window:
87 return None
88
89 window = np.array(self._nu_history[-self.window:])
90 feat = (_window_feats(window) - self._x_mean) / self._x_std
91 probs = np.asarray(self._model_nn(feat), dtype=float)
92 label = self._labels[int(np.argmax(probs))]
93 scale = np.clip(self.regime_scales[label], self.q_min, self.q_max)
94
95 self.model._Q = np.diag(scale) @ self._Q_nominal
96 self.scale_log.append((t, scale.copy(), label))
97 return None

99 def reset(self):
100 super().reset()
101 if self._Q_nominal is not None:
102 self.model._Q = self._Q_nominal.copy()

107 def fit(self, regime_scales: dict, n_episodes=300, episode_len=60,
108 epochs=150, lr=2e-3, batch_size=32, seed=None, model_arch=None):
109 """Simulate labelled episodes for each named regime and train a classifier.
110
111 Parameters
112 ----------
113 regime_scales : dict[str, array-like of shape (n,)]
114 Named noise "shapes" — e.g. ``{"nominal": [1,1,1], "lat": [25,1,1]}``.
115 Training simulates ``model.f()`` with process noise scaled by each vector
116 in turn; at inference the predicted regime's vector is applied to Q.
117 model_arch : object, optional
118 Any callable ``model_arch(feat) -> probs`` with the ``build``/``fit``
119 convention used by ``NeuralRouter`` (e.g. ``routers.nn.MLP``). Defaults to
120 a small ``MLP``.
121 """
122 from ..routers.nn import MLP
123 from ..models.extended import ExtendedKalmanModel
124 from ..engines.extended import ExtendedKalmanEngine
125
126 rng = np.random.default_rng(seed)
127 model = self.model
128 n = model.n
129 dt = 1.0 / model.fs if model.fs else 1.0
130 Q0 = model.Q().copy()
131 q_std0 = np.sqrt(np.diag(Q0))
132 r_std = np.sqrt(np.diag(model.R()))
133
134 self.regime_scales = {k: np.asarray(v, dtype=float) for k, v in regime_scales.items()}
135 self._labels = list(self.regime_scales.keys())
136 label_to_idx = {lbl: i for i, lbl in enumerate(self._labels)}
137
138 engine = ExtendedKalmanEngine()
139
140 def simulate(scale_vec, seed_i):
141 rng_i = np.random.default_rng(seed_i)
142 x_true = model.x0.copy()
143 x_est, P_est = model.x0.copy(), (model.P0.copy() if hasattr(model, "P0") else np.eye(n))
144 nu_hist = []
145 t_ = 0.0
146 for _ in range(episode_len):
147 x_true = np.asarray(model.f(x_true, t=t_), dtype=float).reshape(-1, 1)
148 # scale_vec multiplies Q (variance), like AdaptiveQRKalmanController's
149 # lambda and the diag(scale) @ Q_nominal applied at inference below --
150 # so the std used to draw noise is sqrt(scale_vec), not scale_vec itself.
151 w = rng_i.normal(0, q_std0 * np.sqrt(scale_vec)).reshape(-1, 1)
152 x_true = x_true + w
153 z = np.asarray(model.h(x_true, t=t_), dtype=float).reshape(-1, 1)
154 z = z + rng_i.normal(0, r_std).reshape(-1, 1)
155
156 x_est, P_est, _ = engine._predict(model, x_est, P_est, t=t_)
157 h_pred = model.h(x_est, t=t_)
158 nu = float((z - np.asarray(h_pred, dtype=float).reshape(-1, 1)).ravel()[0])
159 nu_hist.append(nu)
160 x_est, P_est = engine._update(model, x_est, P_est, z, t=t_)
161 t_ += dt
162 return np.array(nu_hist)
163
164 X_all, y_all = [], []
165 for ep in range(n_episodes):
166 label = self._labels[ep % len(self._labels)]
167 nu_hist = simulate(self.regime_scales[label], seed_i=(seed or 0) * 100000 + ep)
168 X_all.append(_window_feats(nu_hist[-self.window:]))
169 y_all.append(label_to_idx[label])
170
171 X = np.array(X_all, dtype=np.float32)
172 y = np.array(y_all)
173 self._x_mean = X.mean(0)
174 self._x_std = X.std(0)
175 self._x_std[self._x_std < 1e-8] = 1.0
176 Xn = (X - self._x_mean) / self._x_std
177
178 self._model_nn = model_arch if model_arch is not None else MLP(hidden=self.hidden)
179 self._model_nn.build(X.shape[1], len(self._labels), seed=seed)
180 self._model_nn.fit(Xn, y, epochs=epochs, lr=lr, batch_size=batch_size)
181 return self
