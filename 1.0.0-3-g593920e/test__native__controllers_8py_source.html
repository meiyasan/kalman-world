48def beliefs(n_steps, seed=0):
49 rng = np.random.default_rng(seed)
50 for k in range(n_steps):
51 x = np.array([[0.8], [0.05], [15.0]]) + 0.1 * rng.standard_normal((3, 1))
52 A = 0.05 * rng.standard_normal((3, 3))
53 yield x, 0.01 * np.eye(3) + A @ A.T, 0.02 * k, 0.3 * rng.standard_normal((1, 1))

60def closed_loop_run(controller, N, seed=0):
61 """Notebook 12's closed loop (cell 5), Gaussian noise: the true car and an EKF steered by
62 ``controller``. Returns the true states."""
63 from modules.framework.filters import ExtendedKalmanFilter
64 rng = np.random.default_rng(seed)
65 model = controller.model
66 ekf = ExtendedKalmanFilter(model)
67 ekf.reset()
68 x_true, nu, t_now = model.x0.copy(), np.zeros((model.m, 1)), 0.0
69 q_std, r_std = np.sqrt(np.diag(model.Q())), np.sqrt(np.diag(model.R()))
70 xs = np.zeros((N, model.n))
71 for k in range(N):
72 u = controller(ekf.x, ekf.P, t_now, nu=nu)
73 x_true = np.asarray(model.f(x_true, u=u, t=t_now), dtype=float).reshape(-1, 1)
74 x_true = x_true + rng.normal(0, q_std, model.n).reshape(-1, 1)
75 z = np.asarray(model.h(x_true, t=t_now), dtype=float).reshape(-1, 1) + rng.normal(0, r_std).reshape(-1, 1)
76 ekf.x, ekf.P, _ = ekf.engine._predict(model, ekf.x, ekf.P, u=u, t=t_now)
77 nu = z - np.asarray(model.h(ekf.x, t=t_now), dtype=float).reshape(-1, 1)
78 ekf.x, ekf.P = ekf.engine._update(model, ekf.x, ekf.P, z, t=t_now)
79 xs[k] = x_true.ravel()
80 t_now += 1.0 / model.fs
81 return xs

229class _Unstable(ExtendedKalmanModel):
230 """A driven model whose f raises after a few calls, and whose clip can raise."""
231
232 def __init__(self, fail_f=False, fail_clip=False):
233 self.nnnn, self.mmmm, self.lll = 2, 1, 1
234 self.dtypedtype = np.float64
235 self.calls, self.fail_f, self.fail_clip = 0, fail_f, fail_clip
236 super().__init__("unstable", Q=np.eye(2) * 1e-4, R=np.eye(1) * 1e-2, L=np.eye(2), fs=10.0,
237 x0=np.zeros((2, 1)), P0=np.eye(2))
238
239 def f(self, x, u=None, w=None, t=None):
240 self.calls += 1
241 if self.fail_f and self.calls > 3:
242 raise ZeroDivisionError("f gave up")
243 x = np.asarray(x, float).reshape(-1)
244 torque = 0.0 if u is None else float(np.ravel(u)[0])
245 return np.array([[x[0] + 0.1 * x[1]], [x[1] + 0.1 * torque]])
246
247 def h(self, x, v=None, t=None):
248 return np.asarray(x, float).reshape(-1)[:1].reshape(1, 1)
249
250 def clip_u(self, u):
251 if self.fail_clip:
252 raise KeyError("no actuator")
253 return np.clip(u, -0.5, 0.5)

329@unittest.skipUnless(core.HAS_CXX_CORE, "libKalman not built")
330class TestRiccati(unittest.TestCase):
331 """Kalman::Algebra::SolveDARE against scipy.linalg.solve_discrete_are."""
332
333 def cases(self):
334 rng = np.random.default_rng(3)
335 out = [(np.array([[1, 0.02], [0, 1]]), np.array([[2e-4], [0.02]]), np.eye(2), np.eye(1)),
336 (np.array([[1, 0.1], [0, 1]]), np.array([[0.005], [0.1]]), np.diag([1.0, 0.0]), np.array([[100.0]])),
337 (np.array([[0.999, 0.01], [0, 0.999]]), np.array([[0.0], [0.01]]), np.eye(2), np.array([[1e3]]))]
338 m = lane()
339 out.append((m.F(), m.B(), np.diag([1.0, 0.0, 0.0]), np.array([[100.0]])))
340 for n, l in ((4, 2), (6, 1), (8, 3)):
341 A = rng.normal(size=(n, n))
342 A /= max(abs(np.linalg.eigvals(A))) / 1.05
343 out.append((A, rng.normal(size=(n, l)), np.eye(n), np.eye(l)))
344 return out
345
346 def test_agrees_with_scipy(self):
347 from scipy.linalg import solve_discrete_are
348 for A, B, Q, R in self.cases():
349 P_ref = solve_discrete_are(A, B, Q, R)
350 K_ref = np.linalg.solve(R + B.T @ P_ref @ B, B.T @ P_ref @ A)
351 P, K = core.solve_dare(A, B, Q, R)
352 # Measured: 1.4e-11 for the pole at 0.999 (where scipy's own residual is the
353 # larger one, 2.4e-14 against 1.9e-16), 4e-14 or better elsewhere.
354 self.assertLess(np.abs(K - K_ref).max() / np.abs(K_ref).max(), 1e-10)
355 self.assertLess(np.abs(P - P_ref).max() / np.abs(P_ref).max(), 1e-10)
356 G = np.linalg.solve(R + B.T @ P @ B, B.T @ P @ A)
357 residual = np.abs(A.T @ P @ A - A.T @ P @ B @ G + Q - P).max() / np.abs(P).max()
358 self.assertLess(residual, 1e-13)
