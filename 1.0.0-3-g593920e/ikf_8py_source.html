28class InteractiveKalmanFilter:
29 """Multiple models interact to handle regime switching (IKF filter).
30
31 Runs M ``KalmanFilter`` instances in parallel — one per operating regime.
32 At every step:
33
34 1. **Interaction / mixing** — states mixed by conditional probs μ_{i|j}.
35 2. **Predict + update** — each model runs its own Kalman step.
36 3. **NIS** — NIS_j = ν_j^T S_j^{-1} ν_j logged per model.
37 4. **Routing** — ``router.route(nis_arr, c, likelihoods)`` → updated μ.
38 5. **Combination** — x̂ = Σ μ_j x̂_j.
39
40 Parameters
41 ----------
42 models : list of KalmanModel
43 transition : ndarray (M, M)
44 Markov transition matrix. Rows sum to 1.
45 initial_probs : ndarray (M,), optional
46 Prior mode probabilities. Default: uniform.
47 router : KalmanRouter or callable, optional
48 Any callable ``(nis_arr, c, likelihoods) → mu`` or ``KalmanRouter``
49 instance. Default: ``BayesianRouter()`` (classical IKF).
50 """
51
52 NAME = "Interactive Kalman Filter"
53 DESCRIPTION = "IKF — runs several models and switches between regimes (the IMM of the literature)."

86 def fit(self, training_zk_list, true_regime_list,
87 epochs: int = 500, lr: float = 2e-3, seed=None,
88 **kwargs) -> "InteractiveKalmanFilter":
89 """Train the router using NIS histories from labelled data.
90
91 Temporarily switches to ``BayesianRouter`` to collect neutral NIS
92 histories, then calls ``router.fit(nis_arrays, labels, **kwargs)``.
93
94 For stateless routers (``BayesianRouter``, ``ThresholdRouter``) this
95 is a no-op — the call completes without training.
96
97 Parameters
98 ----------
99 training_zk_list : list of ndarray (T_i, m)
100 true_regime_list : list of int ndarray (T_i,)
101 epochs, lr, seed
102 Forwarded to ``router.fit``.
103 """
104 saved, self.router = self.router, BayesianRouter()
105 nis_arrays, label_arrays = [], []
106 for zk, labels in zip(training_zk_list, true_regime_list):
107 self.forward(zk, record=False)
108 nis_arrays.append(np.array(self.nis_log))
109 label_arrays.append(np.asarray(labels, dtype=int))
110
111 self.router = saved
112 self.router.fit(nis_arrays, label_arrays,
113 epochs=epochs, lr=lr, seed=seed, **kwargs)
114 return self

118 def forward(self, gk, uk=None, *, record=True, **kwargs):
119 indexing = kwargs.pop("indexing", "xy")
120 self.reset()
121
122 gk_arr = gk.zk if isinstance(gk, KalmanRecord) else gk
123 gk_arr = KalmanRecord._normalize_in(gk_arr, indexing=indexing,
124 dtype=self.models[0].dtype)
125
126 uk_fn = uk if callable(uk) else None
127 t = float(self.models[0].t0) if self.models[0].t0 is not None else 0.0
128 dt = 1.0 / self.sample_rate() if self.sample_rate() else 1.0
129 nu_c = np.zeros((self.models[0].m, 1))
130
131 for z in gk_arr:
132 # 1. Markov prediction: c_j = Σ_i Π[i,j] μ_i
133 c = np.maximum(self.Pi.T @ self.mu, 1e-12)
134
135 # 2. State mixing
136 mu_ij = (self.Pi * self.mu[:, None]) / c # (M, M)
137 for j in range(self.M_count):
138 x_mix = sum(mu_ij[i, j] * self.filters[i].x
139 for i in range(self.M_count))
140 P_mix = sum(mu_ij[i, j] * (
141 self.filters[i].P
142 + (self.filters[i].x - x_mix) @ (self.filters[i].x - x_mix).T)
143 for i in range(self.M_count))
144 self.filters[j].x = x_mix
145 self.filters[j].P = P_mix
146
147 # 3. Per-model predict + update; compute NIS and Gaussian likelihood
148 if uk_fn is not None:
149 x_c = sum(self.mu[j] * self.filters[j].x
150 for j in range(self.M_count))
151 P_c = sum(self.mu[j] * (
152 self.filters[j].P
153 + (self.filters[j].x - x_c) @ (self.filters[j].x - x_c).T)
154 for j in range(self.M_count))
155 u = np.asarray(uk_fn(x_c, P_c, t, nu=nu_c),
156 dtype=float).reshape(-1, 1)
157 else:
158 u = None
159
160 nis_step = np.zeros(self.M_count)
161 likelihoods = np.zeros(self.M_count)
162
163 for j in range(self.M_count):
164 fj = self.filters[j]
165 mj = self.models[j]
166 uj = u if u is not None else np.zeros((max(mj.l, 1), 1))
167
168 fj.x, fj.P, F = fj.engine._predict(mj, fj.x, fj.P, u=uj, t=t)
169 fj.prior.append((fj.x, fj.P))
170 fj.transition.append(F)
171
172 zk, H_j = fj.engine._measure(mj, fj.x, t)
173 nu_j = (np.asarray(z, dtype=float).reshape(-1, 1)
174 - np.asarray(zk, dtype=float).reshape(-1, 1))
175 S_j = H_j @ fj.P @ H_j.T + mj.R(t=t)
176 fj.innovation.append(nu_j)
177
178 try:
179 nis_step[j] = float(
180 (nu_j.T @ np.linalg.solve(S_j, nu_j)).item())
181 likelihoods[j] = multivariate_normal.pdf(
182 nu_j.ravel(),
183 mean=np.zeros(nu_j.shape[0]),
184 cov=S_j)
185 except np.linalg.LinAlgError:
186 nis_step[j] = float(nu_j.shape[0])
187 likelihoods[j] = 1e-12
188
189 fj.x, fj.P = fj.engine._update(mj, fj.x, fj.P, z, t=t)
190 fj.posterior.append((fj.x, fj.P))
191 fj.measurement.append(z)
192
193 self.nis_log.append(nis_step)
194
195 # 4. Route
196 nis_arr = np.array(self.nis_log)
197 self.mu = np.asarray(self.router(nis_arr, c, likelihoods),
198 dtype=float)
199 self.mu = np.maximum(self.mu, 1e-12)
200 self.mu /= self.mu.sum()
201 self.mu_log.append(self.mu.copy())
202
203 # 5. Combined estimate
204 x_c = sum(self.mu[j] * self.filters[j].x
205 for j in range(self.M_count))
206 P_c = sum(self.mu[j] * (
207 self.filters[j].P
208 + (self.filters[j].x - x_c) @ (self.filters[j].x - x_c).T)
209 for j in range(self.M_count))
210 self.combined_posterior.append((x_c, P_c))
211
212 h_c = self.models[0].h(x_c, t=t)
213 nu_c = (np.asarray(z, dtype=float).reshape(-1, 1)
214 - np.asarray(h_c, dtype=float).reshape(-1, 1))
215 t += dt
216
217 return self._build_record(gk_arr, indexing=indexing) if record else self

221 f0.posterior = self.combined_posterior
222 f0.measurement = f0.measurement if f0.measurement else list(zk)
223 return f0.get_record(indexing=indexing)
