2kalman.framework.native.learn — libKalmanLearn from Python.

4``libKalmanLearn`` holds what needs an AI runtime. From Python it serves the
5libTorch field solver, ``Kalman::Learn::Solvers::Vorticity``: the batched
6pseudo-spectral step behind ``SpectralSolver(backend="torch")``.
7
8The library is looked for like the core (see :mod:`.core`), under the name
9``libKalmanLearn``; ``KALMAN_LEARN_LIBRARY`` names the file itself. It is loaded
10on first use, not at import: it brings libTorch with it.
11
12libTorch here is the C++ library; Python's torch package is not used. A process

14the same installation (in a conda environment, ``libtorch`` and ``pytorch`` are).

69def load():
70 """dlopen libKalmanLearn once. Returns ``lib``, or ``None`` (see ``LOAD_ERROR``)."""
71 global ffi, lib, LIBRARY_PATH, LOAD_ERROR, _tried
72 if lib is not None or _tried:
73 return lib
74 _tried = True
75 try:
76 from cffi import FFI
77 path = next((p for p in candidate_paths() if os.path.isfile(p)), None)
78 if path is None:
79 raise OSError("libKalmanLearn not found; build it (make build install) or set "
80 "KALMAN_LEARN_LIBRARY.")
81 _ffi = FFI()
82 _ffi.cdef(_CDEF)
83 lib = _ffi.dlopen(path)
84 ffi, LIBRARY_PATH = _ffi, path
85 except Exception as e: # missing file, a second libTorch, no cffi ...
86 LOAD_ERROR = f"{type(e).__name__}: {e}"
87 return lib

115class Vorticity:
116 """``Kalman::Learn::Solvers::Vorticity``: batched 2-D vorticity steps on libTorch.
117
118 ``dw/dt = -u.grad(w) - beta v + nu lap(w) + F`` on a periodic grid; ``forcing_hat`` is
119 the 2-D transform of ``F`` (complex, ``(ny, nx)``) or ``None``.
120 """
121
122 def __init__(self, kx, ky, k2, k2inv, dealias, *, nu, beta=0.0, forcing_hat=None, device="cpu"):
123 if not has_torch():
124 raise RuntimeError("libKalmanLearn with libTorch is not available — "
125 + (LOAD_ERROR or "it was built without libTorch"))
126 ny, nx = np.shape(kx)
127 self.shape = (int(ny), int(nx))
128 keep = [_d(a) for a in (kx, ky, k2, k2inv, np.asarray(dealias, dtype=float))]
129 fre = fim = ffi.NULL
130 if forcing_hat is not None:
131 f = np.asarray(forcing_hat, dtype=complex)
132 (_r, fre), (_i, fim) = _d(f.real), _d(f.imag)
133 handle = lib.kalman_vorticity_create(ny, nx, *(p for _, p in keep), float(nu), float(beta),
134 fre, fim, str(device or "cpu").encode())
135 if handle == ffi.NULL:
136 raise RuntimeError("kalman_vorticity_create: " + last_error())
137 self._handle = ffi.gc(handle, lib.kalman_vorticity_destroy)
138 self.device = ffi.string(lib.kalman_vorticity_device(self._handle)).decode()

140 def advance(self, w, nsteps, dt, tableau):
141 """Advance ``w`` of shape ``(batch, ny, nx)`` by ``nsteps`` steps of ``dt``.
142 ``tableau`` is ``(c, A, b)`` of an explicit Runge-Kutta scheme."""
143 ny, nx = self.shape
144 x, px = _d(w)
145 if x.ndim != 3 or x.shape[1:] != (ny, nx):
146 raise ValueError(f"expected (batch, {ny}, {nx}), got {x.shape}")
147 c, A, b = tableau
148 s = len(b)
149 full = np.zeros((s, s))
150 for i, row in enumerate(A): # rows may be ragged (lower triangle)
151 full[i, :len(row)] = row
152 (_c, pc), (_A, pA), (_b, pb) = _d(np.asarray(c, dtype=float)), _d(full), _d(np.asarray(b, dtype=float))
153 out = np.empty_like(x)
154 rc = lib.kalman_vorticity_advance(self._handle, px, x.shape[0], int(nsteps), float(dt), s,
155 pc, pA, pb, ffi.cast("double *", ffi.from_buffer(out)))
156 if rc != 0:
157 raise RuntimeError("kalman_vorticity_advance: " + last_error())
158 return out
