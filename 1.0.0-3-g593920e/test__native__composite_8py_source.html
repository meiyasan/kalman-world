1"""Composite models run as models of matrices, in C++.
2
3A :class:`CompositeKalmanModel` stacks its parts: block-diagonal F and Q, and an
4H that stacks (INDEPENDENT) or sums (FUSED) the parts' readouts. When every part
5is constant matrices the composite is too, and says so (``structure`` is
6MATRICES), so its filtering pass and its smoother run in C++ on those matrices
7(``kf_forward_linear`` / ``kf_backward_rts``) instead of the C++ loop calling the
8Python closures back at every step. These tests pin that the pass gives the
9Python answer, that it really is the C++ pass that runs, and that a composite
10which is not made of matrices keeps the path it had.
11
12The reference is the Python loop (``native.python_reference()`` with the fused
13linear pass switched off as well).
14"""

36# As in test_native_twins: the state to 1e-7 and the covariance to 1e-6 of the
37# largest entry. The two passes are the same arithmetic in another order (the
38# C++ update is not the Python one term for term), so they agree far closer,
39# to about 1e-12; the shared bound is kept.

46def close(test, a, b, rtol, what):
47 a, b = np.asarray(a, float), np.asarray(b, float)
48 scale = max(1.0, float(np.nanmax(np.abs(a))))
49 test.assertLess(float(np.nanmax(np.abs(a - b))) / scale, rtol, what)

52def strain(freqs, amps, Qs, noise, T=4.0, seed=0):
53 rng = np.random.default_rng(seed)
54 t = np.arange(int(T * FS)) / FS
55 x = noise * rng.standard_normal(t.size)
56 for a, f, q in zip(amps, freqs, Qs):
57 x += a * np.exp(-t / (q / (np.pi * f))) * np.cos(2 * np.pi * f * t)
58 return x

68def close_smoothed(test, py, cx, model, what):
69 """A smoothed covariance, against the scale of the forward pass it comes from.
70
71 The windows start from a loose prior ((1000 sigma)^2) and the RTS sweep takes
72 differences of the forward covariances on its way back, down to a smoothed
73 covariance thousands of times smaller: its rounding is that of the forward
74 covariance it was subtracted from. Two Python RTS sweeps over forward passes
75 that agree to 1e-14 differ there by 2e-4 of the smoothed value; measured
76 against the forward scale both stay below 1e-7."""
77 scale = max(1.0, max(float(np.max(np.abs(P))) for _, P in model._filter.prior))
78 a, b = np.asarray(py.Pk, float), np.asarray(cx.Pk, float)
79 test.assertLess(float(np.max(np.abs(a - b))) / scale, COVARIANCE_RTOL, what)

82def on_cpp(test, make, z, smooth=False):
83 """The default pass, asserting it is the fused C++ one."""
84 with mock.patch.object(KalmanFilter, "_forward_native", autospec=True,
85 side_effect=KalmanFilter._forward_native) as fused, \
86 mock.patch.object(KalmanFilter, "_forward_callback", autospec=True,
87 side_effect=KalmanFilter._forward_callback) as callback:
88 model = make()
89 record = model.filter(z, smooth=smooth)
90 test.assertEqual(fused.call_count, 1, "the pass ran in kf_forward_linear")
91 test.assertEqual(callback.call_count, 0, "no Python callbacks")
92 test.assertIsInstance(model._filter.engine, LinearKalmanEngine)
93 test.last_model = model
94 return record

247 """A part whose H is a function of time (the lab-frame phasor) makes the
