24 std::vector<double> flat;

30 // sum_i w_i (A_i - a)(B_i - b)^T — the one accumulation every moment here
31 // needs, and the same code the Python engine reaches through the C kernels.
32 Matrix WeightedOuter(const std::vector<double> &w,
33 const std::vector<Vector> &A, const Vector &a,
34 const std::vector<Vector> &B, const Vector &b)
35 {
36 Matrix C(a.Size(), b.Size(), 0.);
37 Utils::WeightedOuterSum(A.size(), a.Size(), b.Size(), w.data(),
38 Pack(A).data(), a.Data(), Pack(B).data(), b.Data(), C.Data());
39 return C;
40 }

42 Vector WeightedMean(const std::vector<double> &w, const std::vector<Vector> &Y)
43 {
44 Vector mean(Y.front().Size(), 0.);
45 for (std::size_t s = 0; s < Y.size(); ++s)
46 for (std::size_t i = 0; i < mean.Size(); ++i)
47 mean(i) += w[s] * Y[s](i);
48 return mean;
49 }

67 Matrix Augmented(const Matrix &P, const Matrix &N)
68 {
69 const std::size_t n = P.Rows(), q = N.Rows();
70 Matrix out(n + q, n + q, 0.);
71 for (std::size_t i = 0; i < n; ++i)
72 for (std::size_t j = 0; j < n; ++j) out(i, j) = P(i, j);
73 for (std::size_t i = 0; i < q; ++i)
74 for (std::size_t j = 0; j < q; ++j) out(n + i, n + j) = N(i, j);

106 void Unscented::Weights(std::size_t n, double &gamma, std::vector<double> &wm,
107 std::vector<double> &wc) const
108 {
109 const double nn = static_cast<double>(n);
110 const double lambda = fAlpha * fAlpha * (nn + fKappa) - nn;
111 gamma = std::sqrt(nn + lambda);
112
113 wm.assign(2 * n + 1, 0.5 / (nn + lambda));
114 wc = wm;
115 wm[0] = lambda / (nn + lambda);
116 wc[0] = lambda / (nn + lambda) + (1. - fAlpha * fAlpha + fBeta);
117 }

119 std::vector<Vector> Unscented::SigmaPoints(const Vector &mean, const Matrix &P, double gamma)
120 {
121 const std::size_t n = mean.Size();
122 const Matrix L = Utils::CholeskyFactor(P, true);
123 if (L.Empty())
124 throw std::runtime_error("Engines::Unscented: the covariance has no Cholesky factor "
125 "(not positive semi-definite even after jitter).");
126
127 std::vector<Vector> pts(2 * n + 1, mean);
128 for (std::size_t i = 0; i < n; ++i)
129 for (std::size_t r = 0; r < n; ++r)
130 {
131 const double d = gamma * L(r, i);
132 pts[i + 1](r) += d;
133 pts[n + i + 1](r) -= d;
134 }
135 return pts;
136 }

152 void Unscented::Predict(const Model &model, const Vector &x, const Matrix &P,
153 const Vector *u, double t,
154 Vector &xk, Matrix &Pk, Matrix &F) const
155 {
156 const std::size_t n = model.NState();
157 const Vector uu = ControlOf(model, u);
158
159 double gamma;
160 std::vector<double> wm, wc;
161 std::vector<Vector> Y;
162
163 if (model.HasProcessNoiseInput())
164 {
165 // The noise goes through f: sigma points over the state and the noise
166 // together, and nothing to add afterwards.
167 const Matrix Q = model.Q(t);
168 const std::size_t q = Q.Rows();
169 Weights(n + q, gamma, wm, wc);
170 const std::vector<Vector> sigma = SigmaPoints(Augmented(x, q), Augmented(P, Q), gamma);
171 Y.reserve(sigma.size());
172 for (const Vector &s : sigma) Y.push_back(model.f(Head(s, n), uu, Tail(s, n), t));
173
174 xk = WeightedMean(wm, Y);
175 Pk = WeightedOuter(wc, Y, xk, Y, xk);
176 Regularise(Pk);
177 }
178 else
179 {
180 Weights(n, gamma, wm, wc);
181 const std::vector<Vector> sigma = SigmaPoints(x, P, gamma);
182 Y.reserve(sigma.size());
183 for (const Vector &s : sigma) Y.push_back(model.f(s, uu, t));
184
185 xk = WeightedMean(wm, Y);
186 Pk = WeightedOuter(wc, Y, xk, Y, xk);
187
188 Matrix Q = model.Q(t);
189 const Matrix L = model.L(t);
190 if (!L.Empty()) Q = Utils::Similarity(L, Q);
191 Pk += Q;
192 Regularise(Pk);
193 }
194
195 F = Matrix::Identity(n);
196 }

198 std::size_t Unscented::Update(const Model &model, const Vector &x, const Matrix &P,
199 const Vector &z, double t,
200 Vector &xOut, Matrix &POut) const
201 {
202 xOut = x;
203 POut = P;
204
205 const std::vector<std::size_t> keep = Utils::ValidChannels(z);
206 if (keep.empty()) return 0;
207
208 const std::size_t n = model.NState();
209 double gamma;
210 std::vector<double> wm, wc;
211 std::vector<Vector> Z, X;
212 Vector zpred;
213 Matrix S, Pxz;
214
215 if (model.HasMeasurementNoiseInput())
216 {
217 // The noise goes through h: augment the state with it.
218 const Matrix R = model.R(t);
219 const std::size_t r = R.Rows();
220 Weights(n + r, gamma, wm, wc);
221 const std::vector<Vector> sigma = SigmaPoints(Augmented(x, r), Augmented(P, R), gamma);
222 Z.reserve(sigma.size());
223 X.reserve(sigma.size());
224 for (const Vector &s : sigma)
225 {
226 X.push_back(Head(s, n));
227 Z.push_back(model.MeasureWithNoise(X.back(), Tail(s, n), t));
228 }
229 zpred = WeightedMean(wm, Z);
230 S = WeightedOuter(wc, Z, zpred, Z, zpred);
231 Pxz = WeightedOuter(wc, X, x, Z, zpred);
232 }
233 else
234 {
235 Weights(n, gamma, wm, wc);
236 X = SigmaPoints(x, P, gamma);
237 Z.reserve(X.size());
238 for (const Vector &s : X) Z.push_back(model.h(s, t));
239 zpred = WeightedMean(wm, Z);
240 S = WeightedOuter(wc, Z, zpred, Z, zpred);
241 S += model.R(t);
242 Pxz = WeightedOuter(wc, X, x, Z, zpred);
243 }
244
245 // Restrict to the channels that arrived.
246 Matrix Sc = Utils::CompressBoth(S, keep);
247 Regularise(Sc);
248 Matrix Pxzc(model.NState(), keep.size());
249 for (std::size_t i = 0; i < model.NState(); ++i)
250 for (std::size_t j = 0; j < keep.size(); ++j)
251 Pxzc(i, j) = Pxz(i, keep[j]);
252
253 Vector y = Utils::Compress(z, keep);
254 y -= Utils::Compress(zpred, keep);
255
256 const Matrix K = Pxzc * Utils::Inverse(Sc);
257
258 xOut += K * y;
259 POut -= Utils::Similarity(K, Sc); // P - K S K^T
260 Regularise(POut);
261
262 return keep.size();
263 }
