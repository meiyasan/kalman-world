1"""Structural invariants of the framework/apps layout.
2
3These guard the architecture itself rather than any single model:
4
5- every abstract contract + base + registry lives in ``framework/``;
6- every app domain is a uniform *package* (variant submodules + an ``__init__``
7 that re-exports its public names and a ``VARIANTS`` registry);
8- the plug-in registries (engine / integrator / solver / provider) cover their
9 full kind enum and resolve by name;
10- the public ``kalman.apps`` surface stays importable after reorganisation;
11- ``apps/`` holds packages only (no flat domain modules).
12"""

36def _domain_packages():
37 """(import_path, name) for every leaf domain package, families expanded."""
38 out = []
39 for m in pkgutil.iter_modules([APPS_DIR]):
40 if not m.ispkg:
41 continue
42 if m.name in FAMILIES:
43 fam_dir = os.path.join(APPS_DIR, m.name)
44 for sub in pkgutil.iter_modules([fam_dir]):
45 if sub.ispkg:
46 out.append((f"{m.name}.{sub.name}", sub.name))
47 else:
48 out.append((m.name, m.name))
49 return sorted(out)

99 def test_every_domain_has_variants_and_all(self):
100 for path, name in _domain_packages():
101 mod = importlib.import_module(f"modules.apps.{path}")
102 self.assertTrue(hasattr(mod, "__all__"), f"{name} package needs __all__")
103 self.assertTrue(hasattr(mod, "VARIANTS"), f"{name} package needs VARIANTS")
104 self.assertGreater(len(mod.VARIANTS), 0, f"{name}.VARIANTS is empty")
105 for key, cls in mod.VARIANTS.items():
106 self.assertTrue(inspect.isclass(cls),
107 f"{name}.VARIANTS[{key!r}] must be a class")

109 def test_variants_are_kalman_models(self):

117 def test_public_names_resolve_from_apps(self):
118 """Every name a domain exports is reachable and identical via kalman.apps
119 when re-exported there (the stable public surface)."""
120 for path, name in _domain_packages():
121 mod = importlib.import_module(f"modules.apps.{path}")
122 for public in mod.__all__:
123 self.assertTrue(hasattr(mod, public), f"{name}.{public} missing")
124 if hasattr(apps, public):
125 self.assertIs(getattr(apps, public), getattr(mod, public),
126 f"kalman.apps.{public} differs from {name}.{public}")

128 def test_families_expose_their_domains(self):
129 """A family package groups domains and exposes them via DOMAINS."""

132 self.assertTrue(hasattr(mod, "DOMAINS"), f"{fam} family needs DOMAINS")
