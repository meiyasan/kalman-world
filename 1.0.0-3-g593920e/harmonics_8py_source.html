7def plot_phasor_diagram(f0, phasors, *, colors=None, frames=30, interval=60, figsize=(14, 4)):
8 """Animated phasor diagram for power-grid odd harmonics.
9
10 Three synchronized panels: lab-frame phasors rotating at each harmonic's own
11 angular frequency, the individual harmonic waveforms, and their composite vs the
12 ideal (fundamental-only) waveform — tied together by a moving time cursor.
13
14 Parameters
15 ----------
16 f0 : float
17 Fundamental frequency [Hz].
18 phasors : list of (V, phi)
19 Amplitude [V] and phase [rad] for H1, H3, H5, … (odd harmonics, in order) —
20 half-wave-symmetric grid harmonics. ``phasors[0]`` is the fundamental H1.
21 colors : list of str, optional
22 Per-harmonic colors. Default: matplotlib tab10 cycle.
23 frames, interval : int
24 Animation frame count and interval [ms] over one fundamental period.
25
26 Returns
27 -------
28 Jupyter HTML player (or raw ``FuncAnimation`` outside Jupyter) — see ``helpers.animate``.
29 """
30 import matplotlib.pyplot as plt
31 from matplotlib.patches import FancyArrowPatch
32 from ....helpers import animate
33
34 n = len(phasors)
35 if colors is None:
36 cmap = plt.cm.tab10
37 colors = [cmap(k) for k in range(n)]
38 labels_short = [f"H{2 * k + 1}" for k in range(n)]
39 labels = [f"{lbl} ({'f₀' if k == 0 else f'{2 * k + 1}f₀'})" for k, lbl in enumerate(labels_short)]
40 omegas = [2 * np.pi * (2 * k + 1) * f0 for k in range(n)]
41
42 T = 1.0 / f0
43 t_cyc = np.linspace(0, T, 500)
44 harmonic_waves = [V * np.cos(omegas[k] * t_cyc + phi) for k, (V, phi) in enumerate(phasors)]
45 v_clean = sum(harmonic_waves)
46 v_ideal = phasors[0][0] * np.cos(omegas[0] * t_cyc)
47
48 frame_t = np.linspace(0, T, frames, endpoint=False)
49
50 fig, axes = plt.subplots(1, 3, figsize=figsize)
51 fig.suptitle("Power grid phasors — amplitude, phase, and time-domain reconstruction", fontweight="bold")
52
53 # Phasor diagram — arrows rotate at each harmonic's own angular frequency. Axis is
54 # square and sized to the largest amplitude in BOTH dimensions: a rotating phasor
55 # swings through the full +/-amplitude range on y too, not just on x.
56 ax0 = axes[0]
57 ax0.set_aspect("equal")
58 Vmax = max(V for V, _ in phasors)
59 ax0.set_xlim(-Vmax * 1.15, Vmax * 1.15); ax0.set_ylim(-Vmax * 1.15, Vmax * 1.15)
60 ax0.axhline(0, color="gray", lw=0.5); ax0.axvline(0, color="gray", lw=0.5)
61 ax0.set_xlabel("Re [V]"); ax0.set_ylabel("Im [V]")
62 ax0.set_title("Phasor diagram (rotating, lab frame)")
63 arrows = []
64 for (V, phi), col in zip(phasors, colors):
65 arrow = FancyArrowPatch((0, 0), (V * np.cos(phi), V * np.sin(phi)),
66 color=col, arrowstyle="-|>", mutation_scale=14, lw=2.2)
67 ax0.add_patch(arrow)
68 arrows.append(arrow)
69 # Fixed-position readout instead of labels at each arrow tip: harmonic amplitudes
70 # vary by orders of magnitude, so tip-anchored labels collide near the origin.
71 info_text = ax0.text(0.02, 0.98, "", transform=ax0.transAxes, va="top", ha="left",
72 fontsize=7.5, family="monospace",
73 bbox=dict(facecolor="white", alpha=0.85, edgecolor="gray", pad=3))
74
75 # Individual harmonic waveforms — static curves + moving cursor dots
76 ax1 = axes[1]
77 for wave, lbl, col in zip(harmonic_waves, labels, colors):
78 ax1.plot(t_cyc * 1e3, wave, color=col, lw=1.5, label=lbl)
79 ax1.axhline(0, color="gray", lw=0.5)
80 ax1.set_xlabel("t [ms]"); ax1.set_ylabel("voltage [V]")
81 ax1.set_title("Individual harmonics")
82 ax1.legend(fontsize=8)
83 cursor1 = ax1.axvline(0, color="black", lw=0.8, ls=":")
84 dots1 = [ax1.plot([], [], "o", color=col, ms=5)[0] for col in colors]
85
86 # Composite waveform — static curves + moving cursor dot
87 ax2 = axes[2]
88 ax2.plot(t_cyc * 1e3, v_clean, color="purple", lw=2, label=" + ".join(labels_short))
89 ax2.plot(t_cyc * 1e3, v_ideal, "--", color="steelblue", lw=1, label="H1 only (ideal)")
90 ax2.axhline(0, color="gray", lw=0.5)
91 ax2.set_xlabel("t [ms]"); ax2.set_ylabel("voltage [V]")
92 ax2.set_title("Composite: distorted vs ideal")
93 ax2.legend(fontsize=8)
94 cursor2 = ax2.axvline(0, color="black", lw=0.8, ls=":")
95 dot2 = ax2.plot([], [], "o", color="purple", ms=5)[0]
96
97 def _draw(i):
98 t = frame_t[i]
99 tm = t * 1e3
100 lines = []
101 for k, ((V, phi), arrow, col) in enumerate(zip(phasors, arrows, colors)):
102 ang = omegas[k] * t + phi
103 x, y = V * np.cos(ang), V * np.sin(ang)
104 arrow.set_positions((0, 0), (x, y))
105 lines.append(f"{labels[k]}: V={V:>5.1f}V φ={np.degrees(ang) % 360:>5.1f}°")
106 dots1[k].set_data([tm], [x])
107 info_text.set_text("\n".join(lines))
108 cursor1.set_xdata([tm, tm])
109 cursor2.set_xdata([tm, tm])
110 v_now = sum(V * np.cos(omegas[k] * t + phi) for k, (V, phi) in enumerate(phasors))
111 dot2.set_data([tm], [v_now])
112 return tuple(arrows) + (info_text,) + tuple(dots1) + (cursor1, cursor2, dot2)
113
114 plt.tight_layout()
115 return animate(fig, _draw, frames=frames, interval=interval)

122class PowerGridStateModel(PhasorOscillatorModel):
123 """
124 Harmonic voltage tracker for power grid signals (50/60 Hz).
125
126 **Mains specialization of** :class:`PhasorOscillatorModel`. It *is* the **lab-frame**
127 joint phasor estimator — the base class supplies all the machinery; this subclass only
128 fixes the **mains harmonic structure**.
129
130 **Lab-frame formulation** (contrast with notebook 02's rotating-frame
131 :class:`~kalman.apps.oscillators.OscillatorModel`):
132
133 * ``F = I`` — phasors are constant in the lab frame (for a steady sinusoid the phasor
134 does not rotate; it is the signal that oscillates).
135 * ``H(t) = [cos ω₁t, −sin ω₁t, cos ω₂t, −sin ω₂t, …]`` — the carrier rotation lives
136 entirely in the **time-varying observation map**, not in the state transition.
137 * Single scalar measurement ``z(t) = v(t)`` — no Hilbert transform, no demodulation;
138 the raw voltage is ingested directly.
139
140 Because ``H(t)`` couples all harmonics to the one measured voltage, the strong
141 fundamental is fully claimed by H1 and leaves nothing for the even harmonics —
142 so **H2/H4 correctly read ≈ 0**. An independent-demodulator bank cannot do this:
143 the 230 V fundamental survives demodulation in the H2 frame as a rotating phasor and
144 shows up as a spurious amplitude.
145
146 **Why mains lines are (ideally) odd-only.** A clean AC waveform has half-wave
147 symmetry ``v(t ± T/2) = −v(t)``, which forces a Fourier series of **odd harmonics
148 only** (f0, 3·f0, 5·f0, …). Even harmonics vanish unless something breaks that
149 symmetry — DC bias, half-wave rectifier loads, transformer saturation, geomagnetically
150 induced currents. Tracking all harmonics and *seeing* the even ones collapse to ≈ 0
151 is therefore a **physics verification**; a non-zero even harmonic is a diagnostic
152 of load asymmetry. ``B > 0`` covers inharmonic partials
153 (``f_k = k·f0·√(1+B·k²)``) for completeness.
154
155 State layout (state = phasor per harmonic, observation = scalar voltage):
156 x = [Re_1, Im_1, Re_2, Im_2, ..., Re_N, Im_N] (2N states)
157 z = [v] (1 observation)
158
159 Parameters
160 ----------
161 sensor_id : str
162 Channel / sensor name (e.g. "L1").
163 f0 : float
164 Fundamental frequency [Hz], default 50.
165 harmonics : int
166 Number of harmonics to track (k = 1 … harmonics), default 3.
167 fs : float
168 Sample rate [Hz], default 1024.
169 B : float
170 Inharmonicity coefficient (default 0 → ideal harmonic grid).
171 measurement_uncertainty : float
172 Measurement noise std [V], default 1.0.
173 process_uncertainty : float
174 Phasor random-walk std [V] (how fast amplitudes may drift), default 0.01.
175 p0_std : float
176 Initial per-phasor prior std [V] for acquisition, default 100.
177
178 Usage
179 -----
180 Feed the raw measured voltage directly (no Hilbert transform required)::
181
182 record = grid_model.filter(v, smooth=True)
183
184 A 2-column analytic signal ``[real, imag]`` is still accepted — only the real
185 (measured) column is used.
186 """
187
188 NAME = "Power Grid Harmonic Tracker"
189 DESCRIPTION = (
190 "Mains specialization of PhasorOscillatorModel: joint lab-frame phasor "
191 "estimator tracking N harmonics of a 50/60 Hz grid from the scalar voltage — "
192 "all harmonics coupled through one measurement, so absent (even) harmonics read ≈ 0."
193 )
194 VERSION = "4.0"

230 def harmonic_envelope(self, record):
231 """Per-harmonic amplitude ``a_k = √(Re²+Im²)`` and its ±1σ band — shape ``(T, n_harmonics)``."""

234 def reconstruct_voltage(self, record, t=None):
235 """Reconstruct the time-domain voltage from the phasor state:
236 ``v(t) = Σ_k Re_k·cos(ω_k t) − Im_k·sin(ω_k t)``."""
237 return self.reconstruct(record, t)

239 def subtract_voltage(self, measured, record, t=None) -> np.ndarray:
240 """Return ``measured − reconstruct_voltage(record)``: the harmonic-suppressed
241 residual — e.g. mains hum/THD removed from a measured voltage, analogous to
242 :meth:`InterferometerModel.subtract` for spectral-line noise."""
243 return np.asarray(measured, dtype=float).ravel() - self.reconstruct_voltage(record, t)
