1"""Interferometer violin-mode bank model."""
2from __future__ import annotations
3"""GW interferometer noise subtraction via spectral-line Kalman tracking.
4
5Two-stage workflow
6------------------
7Stage 1 -- peak fitting
8 ``InterferometerModel.fit(sensor_id, strain, fs, bands)``
9 Estimates PSD via Welch, detects spectral peaks per band with
10 ``scipy.signal.find_peaks``, fits each peak to a Lorentzian profile to
11 extract (f0, Q, amplitude), and groups nearby peaks into analysis windows.
12
13Stage 2 -- Kalman tracking and subtraction
14 The returned ``InterferometerModel`` (a specialised ``OscillatorBankModel``)
15 tracks each window's phasor in the rotating frame -- the joint Kalman filter
16 of Moisset et al. 2025 (arXiv:2509.20658). After running the filter::
17
18 record = KalmanFilter(model).forward(strain, uk=model.make_controller())
19 clean = model.subtract(strain, record)
20
21 ``make_controller()`` returns a ``ViolinBankController`` that attaches one
22 ``AdaptiveQRKalmanController`` per window, self-adjusting each window's
23 process-noise level (Q) from empirical innovation statistics.
24
25Noise sources
26-------------
27``NOISE_SOURCES = ("violin",)`` -- violin-mode suspension noise only.
28Future additions:
29
30* ``"powergrid"`` -- use ``PowerGridStateModel`` (already implemented in
31 ``modules/apps/powergrids.py``) as a **separate parallel filter** since its
32 PhasorOscillatorModel architecture (lab-frame, time-varying H, scalar voltage)
33 is incompatible with ``OscillatorBankModel._broadcast_preprocess``.
34* ``"omc_pcal"`` -- fixed-frequency PCal calibration lines can be added as
35 additional ``OscillatorModel`` windows inside this bank (Q approx inf, known fc).
36
37Detector-specific subclasses
38-----------------------------
39Pre-loaded per-run violin-mode parameters are available via detector subclasses::
40
41 from kalman.apps.interferometer import InterferometerK1
42
43 # Compute band limits from the built-in catalog and fit from synthetic data:
44 bands = InterferometerK1.violin_bands("O3GK")
45 model = InterferometerK1.fit("K1", strain, fs=4096, bands=bands)
46
47 # Or download O3GK strain directly (requires gwpy):
48 model = InterferometerK1.from_gps(1271267328, 1271281728, run="O3GK")
49 ctrl = model.make_controller()
50 rec = model.filter(strain, uk=ctrl, smooth=True)
51 clean = model.subtract(strain, rec)
52
53Available: InterferometerK1, InterferometerH1, InterferometerL1, InterferometerV1.
54"""

66class InterferometerModel(OscillatorBankModel):
67 """Noise-subtraction model for GW interferometer spectral lines.
68
69 Inherits ``OscillatorBankModel`` (bank of windowed rotating-frame Kalman
70 filters) and adds:
71
72 * A classmethod :meth:`fit` that auto-builds the window bank from raw strain
73 data by fitting Lorentzian profiles to detected PSD peaks.
74 * :meth:`make_controller` returning a ``ViolinBankController`` with one
75 ``AdaptiveQRKalmanController`` per window for closed-loop Q adaptation.
76 * :meth:`subtract` convenience method (thin wrapper over inherited
77 ``reconstruct``).
78
79 Typical usage::
80
81 model = InterferometerModel.fit(
82 "H1", strain, fs=4096,
83 bands=[(170, 186), (402, 432), (718, 770)], # KAGRA violin bands
84 )
85 ctrl = model.make_controller()
86 record = KalmanFilter(model).forward(strain, uk=ctrl)
87 clean = model.subtract(strain, record)
88 print(model.peak_summary())
89
90 Pre-fitted parameters can be passed directly::
91
92 win1 = OscillatorModel("H1", [175.3, 177.1], fc=176.2, dfc=4.0,
93 Q=[1e5, 8e4], fs=4096)
94 model = InterferometerModel("H1", [win1])
95
96 Noise sources
97 -------------
98 ``NOISE_SOURCES = ("violin",)``
99
100 Future: ``"powergrid"`` via ``PowerGridStateModel`` (separate parallel
101 filter — incompatible with this bank's broadcast-preprocess architecture);
102 ``"omc_pcal"`` as additional fixed-frequency windows inside this bank.
103 """
104
105 NAME = "Interferometer Noise Subtraction"
106 DESCRIPTION = ("OscillatorBankModel with Lorentzian peak-fitting and per-window "
107 "AdaptiveQR closed-loop Q adaptation. Violin suspension only.")

112 def __init__(self, sensor_id: str, windows: list[OscillatorModel], *,
113 aqr_window: int = 30,
114 aqr_q_min: float = 0.1,
115 aqr_q_max: float = 10.0,
116 **kwargs):
117 """
118 Parameters
119 ----------
120 sensor_id : str
121 Channel / detector identifier (e.g. ``"H1"``, ``"K1"``).
122 windows : list of OscillatorModel
123 One window per violin-mode group. Build via :meth:`fit` or supply
124 manually for pre-known peak parameters.
125 aqr_window : int
126 Innovation buffer length for each ``AdaptiveQRKalmanController``.
127 aqr_q_min, aqr_q_max : float
128 Clipping bounds for the Q-scaling factor per window.
129 """
130 self.sensor_id = sensor_id
131 super().__init__(*windows, **kwargs)
132
133 self._window_aqr: list[_LeafAdaptiveQR] = [
134 _LeafAdaptiveQR(win, window_len=aqr_window,
135 q_min=aqr_q_min, q_max=aqr_q_max)
136 for win in self.windows
137 ]
138 # Cumulative state offsets for bank-level nu/P slicing in ViolinBankController.
139 self._state_offsets: list[int] = list(
140 np.cumsum([0] + [w.n for w in self.windows[:-1]])
141 )

159 @classmethod
160 def fit(cls, sensor_id: str,
161 strain,
162 fs: float,
163 bands: list[tuple[float, float]],
164 *,
165 Q_init: float = 1e4,
166 merge_gap_hz: float = 2.0,
167 window_margin_hz: float = 1.0,
168 min_prominence=None,
169 measurement_uncertainty: float = 1.0,
170 process_uncertainty: float = 1e-3,
171 nperseg: int = None,
172 **kwargs) -> "InterferometerModel":
173 """Fit Lorentzian peaks in each band and build an ``InterferometerModel``.
174
175 Parameters
176 ----------
177 sensor_id : str
178 strain : array-like, shape (N,)
179 Raw strain time series.
180 fs : float
181 Sample rate [Hz].
182 bands : list of (f_low, f_high) tuples
183 Frequency bands to search for peaks, e.g.
184 ``[(170, 186), (402, 432), (718, 770)]`` for KAGRA violin modes.
185 Q_init : float
186 Fallback Q when Lorentzian fit is unavailable for a peak.
187 merge_gap_hz : float
188 Peaks within this Hz are merged into one FUSED window.
189 window_margin_hz : float
190 Extra padding on each side of the outermost peak in a window.
191 min_prominence : float, optional
192 Minimum PSD prominence for ``find_peaks_in_band``. Auto-scaled
193 per band if ``None``.
194 measurement_uncertainty, process_uncertainty : float
195 Passed through to each ``OscillatorModel``.
196 nperseg : int, optional
197 Welch segment length. Default: ``int(fs)`` (1-second segments).
198
199 Returns
200 -------
201 InterferometerModel
202
203 Workflow
204 --------
205 1. ``welch_psd`` → (freqs, psd)
206 2. For each band: ``find_peaks_in_band`` → candidate peak frequencies
207 3. For each candidate: ``fit_lorentzian_peak`` → (f0, Q, amplitude)
208 4. ``group_peaks_into_windows`` → [(fc, dfc, peaks), …]
209 5. Build one ``OscillatorModel`` per window (Q per mode from fit)
210 6. Construct ``InterferometerModel``
211 """
212 freqs, psd = welch_psd(strain, fs, nperseg=nperseg)
213
214 all_fits: list[dict] = []
215 for f_low, f_high in bands:
216 peak_freqs = find_peaks_in_band(freqs, psd, f_low, f_high,
217 min_prominence=min_prominence)
218 for fp in peak_freqs:
219 half_w = _estimate_half_max_width(freqs, psd, fp, f_low, f_high)
220 try:
221 fit = fit_lorentzian_peak(freqs, psd, fp, half_w)
222 except RuntimeError:
223 fit = {"f0": fp, "gamma": fp / Q_init, "Q": Q_init,
224 "amplitude": float(psd[np.argmin(np.abs(freqs - fp))])}
225 all_fits.append(fit)
226
227 if not all_fits:
228 raise ValueError(
229 "No peaks found in any of the supplied bands. "
230 "Check that `strain` contains the expected spectral lines and "
231 "that the band limits are correct."
232 )
233
234 win_defs = group_peaks_into_windows(all_fits,
235 merge_gap_hz=merge_gap_hz,
236 margin_hz=window_margin_hz)
237
238 windows: list[OscillatorModel] = []
239 for wd in win_defs:
240 peak_freqs_w = [p["f0"] for p in wd["peaks"]]
241 Qs_w = [p["Q"] for p in wd["peaks"]]
242 win = OscillatorModel(
243 sensor_id,
244 peak_freqs_w,
245 fc=wd["fc"],
246 dfc=wd["dfc"],
247 Q=Qs_w if len(Qs_w) > 1 else Qs_w[0],
248 fs=fs,
249 measurement_uncertainty=measurement_uncertainty,
250 process_uncertainty=process_uncertainty,
251 )
252 win._fit_results = wd["peaks"] # store for peak_summary()
253 windows.append(win)
254
255 return cls(sensor_id, windows, **kwargs)

261 def subtract(self, strain, record) -> np.ndarray:
262 """Return ``strain − reconstruct(record)``: the noise-subtracted residual."""
263 return np.asarray(strain, dtype=float).ravel() - self.reconstruct(record)
264
265 def peak_summary(self) -> list[dict]:
266 """Fitted peak parameters (f0, Q, fc, dfc) per window.
267
268 Returns a list of dicts — one per window — each containing:
269 ``fc``, ``dfc``, and a ``peaks`` list with ``f0``, ``Q``, ``amplitude``
270 per mode. Only populated when the model was built via :meth:`fit`.
271 """
272 out = []
273 for win in self.windows:
274 entry: dict = {"fc": win.fc, "dfc": win.dfc}
275 entry["peaks"] = getattr(win, "_fit_results", [
276 {"f0": float(f), "Q": None, "amplitude": None}
277 for f in win.frequencies
278 ])
279 out.append(entry)
280 return out

304def _download_strain(ifo: str, gps_start: float, gps_end: float,
305 channel: str = None, target_fs: float = 4096.0):
306 """Download strain via gwpy and return (strain_array, actual_fs).
307
308 Lazy-imports gwpy so the rest of the module works without it.
309 Uses GWOSC open data (``fetch_open_data``) by default; pass a NDS2
310 *channel* name to use ``TimeSeries.get`` instead.
311 """
312 try:
313 from gwpy.timeseries import TimeSeries # type: ignore[import]
314 except ImportError as exc:
315 raise ImportError(
316 "gwpy is required for GPS-based data download. "
317 "Install it with: pip install gwpy"
318 ) from exc
319
320 if channel is not None:
321 ts = TimeSeries.get(channel, gps_start, gps_end, verbose=True)
322 else:
323 ts = TimeSeries.fetch_open_data(ifo, gps_start, gps_end,
324 sample_rate=int(target_fs))
325 return np.asarray(ts.value, dtype=float), float(ts.sample_rate.value)

332class _DetectorBase(InterferometerModel):
333 """Base for detector-specific ``InterferometerModel`` subclasses.
334
335 Subclasses set ``_IFO`` and ``_CATALOG`` to register known violin-mode
336 parameters per observation run.
337
338 Two construction paths are available:
339
340 * :meth:`from_peaks` — build directly from catalog peak list (no strain
341 needed; K1 O3GK only so far).
342 * :meth:`from_gps` — download strain from GWOSC via gwpy, cache it, and
343 call :meth:`InterferometerModel.fit` with the catalog bands.
344 """
345
346 _IFO: str = ""
347 _CATALOG: dict = {}
348
349 @classmethod
350 def latest_run(cls) -> str:
351 """Return the most recently registered observation run key."""
352 return list(cls._CATALOG.keys())[-1]
353
354 @classmethod
355 def violin_bands(cls, run: str) -> list:
356 """Return ``[(f_low, f_high), …]`` search bands for a given run.
357
358 When the catalog entry contains a ``"bands"`` key (from direct
359 measurement), those values are returned as-is. Otherwise, band
360 centres are computed from the Fletcher inharmonicity series
361 ``f_n = n·f0·√(1 + B·n²)`` and padded by ``band_margin_hz``.
362
363 Parameters
364 ----------
365 run : str
366 Observation run key (e.g. ``"O3GK"``, ``"O4"``).
367
368 Returns
369 -------
370 list of (f_low, f_high) float tuples [Hz]
371 """
372 from ...resonators.oscillators import partial_frequencies as _pf
373 if run not in cls._CATALOG:
374 raise KeyError(
375 f"Run '{run}' not in {cls.__name__}._CATALOG. "
376 f"Available: {list(cls._CATALOG.keys())}"
377 )
378 p = cls._CATALOG[run]
379 if "bands" in p:
380 return [(float(b[0]), float(b[1])) for b in p["bands"]]
381 partials = _pf(p["f0"], p["B"], p["n_harmonics"])
382 m = p.get("band_margin_hz", 10.0)
383 return [(float(f - m), float(f + m)) for f in partials]
384
385 @classmethod
386 def preloaded_peaks(cls, run: str) -> list:
387 """Return the catalog peak list for *run* as ``[mode1, mode2, ...]``.
388
389 Each element is a list of ``(f0_hz, Q)`` tuples measured from real data.
390 Raises ``KeyError`` for unknown runs and ``ValueError`` when no peaks are
391 stored for that run (e.g. K1 O4 placeholder).
392 """
393 if run not in cls._CATALOG:
394 raise KeyError(
395 f"Run '{run}' not in {cls.__name__}._CATALOG. "
396 f"Available: {list(cls._CATALOG.keys())}"
397 )
398 peaks = cls._CATALOG[run].get("peaks")
399 if not peaks:
400 raise ValueError(
401 f"No preloaded peaks for {cls._IFO} {run}. "
402 "Use from_gps() + fit() to detect peaks from strain data."
403 )
404 return peaks
405
406 @classmethod
407 def from_peaks(cls, run: str = None, *,
408 dfc: float = None,
409 measurement_uncertainty: float = 1.0,
410 process_uncertainty: float = 1e-3,
411 **kwargs) -> "_DetectorBase":
412 """Build an ``InterferometerModel`` from catalog peak positions.
413
414 Uses the preloaded ``(f0, Q)`` pairs stored in the catalog, bypassing
415 the Welch PSD fitting stage entirely. Suitable for K1 O3GK where the
416 88 peak positions are known from Moisset et al. 2025 (arXiv:2509.20658).
417
418 Creates one ``OscillatorModel`` window per mode (one per band), with all
419 catalogue peaks of that mode passed as ``frequencies``. All windows share
420 the same ``dfc`` (required by ``OscillatorBankModel``), chosen as the
421 widest band half-width plus 15 % margin — or overridden via ``dfc``.
422
423 Parameters
424 ----------
425 run : str, optional
426 Observation run (default: :meth:`latest_run`).
427 dfc : float, optional
428 Common half-bandwidth [Hz] for all mode windows. Default: auto
429 from the widest catalog band (e.g. ~30 Hz for K1 O3GK).
430 measurement_uncertainty, process_uncertainty : float
431 Passed to each ``OscillatorModel``.
432
433 Returns
434 -------
435 Instance of the calling subclass.
436
437 Example
438 -------
439 ::
440
441 model = InterferometerK1.from_peaks("O3GK")
442 rec = model.filter(strain, uk=model.make_controller())
443 clean = model.subtract(strain, rec)
444 """
445 if run is None:
446 run = cls.latest_run()
447 mode_peak_lists = cls.preloaded_peaks(run)
448 p = cls._CATALOG[run]
449 ifo = cls._IFO
450 fs = float(p.get("fs", 4096.0))
451 bands = [(float(b[0]), float(b[1])) for b in p["bands"]]
452
453 if dfc is None:
454 # One common dfc for all windows (required by OscillatorBankModel).
455 # Use the widest mode half-span + 15% margin so all peaks stay in-band.
456 dfc = max((hi - lo) / 2.0 for lo, hi in bands) * 1.15
457
458 windows: list[OscillatorModel] = []
459 for mode_peaks, (band_lo, band_hi) in zip(mode_peak_lists, bands):
460 fc = (band_lo + band_hi) / 2.0
461 peak_freqs = [float(f0) for f0, _ in mode_peaks]
462 Qs = [float(Q) for _, Q in mode_peaks]
463 win = OscillatorModel(
464 ifo, peak_freqs,
465 fc=fc, dfc=dfc,
466 Q=Qs if len(Qs) > 1 else Qs[0],
467 fs=fs,
468 measurement_uncertainty=measurement_uncertainty,
469 process_uncertainty=process_uncertainty,
470 )
471 windows.append(win)
472
473 return cls(ifo, windows, **kwargs)

475 @classmethod
476 def from_gps(cls, gps_start: float, gps_end: float, *,
477 run: str = None,
478 channel: str = None,
479 fs: float = None,
480 cache: bool = True,
481 cache_dir=None,
482 **kwargs) -> "_DetectorBase":
483 """Download strain from GWOSC and build a fitted interferometer model.
484
485 Requires gwpy (``pip install gwpy``). Downloaded data is cached as a
486 ``.npz`` file in *cache_dir* (default ``~/.cache/kalman_framework/``)
487 so subsequent calls with the same GPS range are instant.
488
489 Parameters
490 ----------
491 gps_start, gps_end : float
492 GPS time range [s].
493 run : str, optional
494 Observation run key (e.g. ``"O3GK"``). Defaults to
495 :meth:`latest_run`.
496 channel : str, optional
497 NDS2 channel name for private data. When set, uses
498 ``gwpy.TimeSeries.get`` instead of ``fetch_open_data``.
499 fs : float, optional
500 Target sample rate [Hz]. Defaults to the catalog entry's ``fs``.
501 cache : bool
502 Cache the downloaded timeseries on disk. Default ``True``.
503 cache_dir : str or Path, optional
504 Cache directory. Default ``~/.cache/kalman_framework/``.
505 **kwargs
506 Forwarded to :meth:`InterferometerModel.fit` (e.g.
507 ``measurement_uncertainty``, ``process_uncertainty``).
508
509 Returns
510 -------
511 Instance of the calling subclass.
512
513 Example
514 -------
515 ::
516
517 from kalman.apps.interferometer import InterferometerK1
518 model = InterferometerK1.from_gps(1271267328, 1271281728, run="O3GK")
519 rec = model.filter(strain, uk=model.make_controller(), smooth=True)
520 clean = model.subtract(strain, rec)
521 """
522 if run is None:
523 run = cls.latest_run()
524 p = cls._CATALOG[run]
525 tfs = fs or p.get("fs", 4096.0)
526 ifo = cls._IFO
527
528 if cache:
529 cache_path = _get_cache_path(cache_dir, ifo, gps_start, gps_end, tfs)
530 if cache_path.exists():
531 data = np.load(str(cache_path))
532 strain = data["strain"]
533 actual_fs = float(data["fs"])
534 else:
535 strain, actual_fs = _download_strain(ifo, gps_start, gps_end,
536 channel, tfs)
537 np.savez(str(cache_path), strain=strain, fs=actual_fs)
538 else:
539 strain, actual_fs = _download_strain(ifo, gps_start, gps_end,
540 channel, tfs)
541
542 bands = cls.violin_bands(run)
543 return cls.fit(ifo, strain, actual_fs, bands, **kwargs)
