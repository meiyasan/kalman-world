4local error tolerance requires, so a stiff-ish or fast-varying ``f`` is resolved
5without the caller shrinking ``dt``. The 5th-order solution is returned; the

29class DormandPrinceIntegrator(KalmanIntegrator):
30 """Adaptive RK45 (Dormand–Prince); ``rtol``/``atol`` control the substeps."""
31
32 NAME = "Dormand-Prince Integrator"
33 DESCRIPTION = "Adaptive RK45 (Dormand–Prince 5(4)) with error-controlled substeps."
34 VERSION = "1.0"
35 KIND = KalmanIntegratorKind.DORMAND_PRINCE
36 # 5th-order tableau (the b5 weights) — used as a *fixed-step* RK5 by field
37 # solvers (adaptive per-member step control is skipped for batched ensembles;
38 # the ODE ``step`` above keeps the full adaptive 5(4) control).
39 tableau = (list(_C), [list(r) for r in _A], list(_B5))

49 def _step(self, f, x, u=None, t=None, dt=1.0):
50 x = np.asarray(x, dtype=float)
51 t0 = 0.0 if t is None else float(t)
52 target = t0 + dt
53 t, h = t0, dt
54 for _ in range(self.max_substeps):
55 if (h > 0 and t >= target) or (h < 0 and t <= target):
56 break
57 h = np.sign(dt) * min(abs(h), abs(target - t))
58 ks = []
59 for i in range(7):
60 xi = x + h * sum(_A[i][j] * ks[j] for j in range(i)) if i else x
61 ks.append(np.asarray(f(xi, u, t + _C[i] * h), dtype=float))
62 K = np.stack(ks)
63 x5 = x + h * np.tensordot(_B5, K, axes=1)
64 x4 = x + h * np.tensordot(_B4, K, axes=1)
65 err = np.linalg.norm((x5 - x4) / (self.atol + self.rtol * np.maximum(np.abs(x), np.abs(x5))))
66 err = err / max(np.sqrt(x.size), 1)
67 if err <= 1.0 or abs(h) <= 1e-14 * abs(dt):
68 x, t = x5, t + h
69 # PI-free step-size update (clamped)
70 h *= min(5.0, max(0.2, 0.9 * (1.0 / max(err, 1e-12)) ** 0.2))
71 return x
