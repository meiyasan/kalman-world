13class KalmanProxyMixin:
14 """
15 Model-side convenience sugar (intent: the model is *not* the algorithm).
16
17 ``filter()`` / ``smooth()`` lazily build and cache a default ``KalmanFilter``
18 (composition ``model + engine``, the engine resolved from ``DEFAULT_ENGINE``);
19 ``sample()`` builds a ``KalmanSampler`` over the model. The model holds no
20 filtering equations or history — the transient filter does.
21 """
22 _filter: Any = None # cached default KalmanFilter from the last filter()/smooth()
23
24 def _make_filter(self):
25 from ..filter import KalmanFilter
26 self._filter_filter = KalmanFilter(self) # engine resolved from self.DEFAULT_ENGINE
27 return self._filter_filter
28
29 def filter(self, z, u=None, **kwargs):
30 # Accept uk= as an alias for u= (mirrors forward()'s parameter name).
31 if u is None:
32 u = kwargs.pop("uk", None)
33
34 flt = self._make_filter()
35 if kwargs.pop("smooth", False):
36 flt.forward(z, u, record=False, **kwargs)
37 return flt.backward(**kwargs)
38
39 kwargs["record"] = True
40 return flt.forward(z, u, **kwargs)
41
42 def smooth(self, **kwargs):
43 if self._filter_filter is None:
44 raise RuntimeError("No prior forward pass; call filter() first.")
45 return self._filter_filter.backward(**kwargs)
46
47 # --- consistency diagnostics (proxy to the cached filter; run filter() first) ---
48 def _require_filter(self):
49 if self._filter_filter is None:
50 raise RuntimeError("No prior filter run; call filter() first.")
51 return self._filter_filter
52 def innovations(self): return self._require_filter().innovations()
53 def gains(self): return self._require_filter().gains()
54 def nis(self): return self._require_filter().nis()
55 def nees(self, truth): return self._require_filter().nees(truth)
56 def diagnostics(self, sample=None, **kwargs):
57 return self._require_filter().diagnostics(sample=sample, **kwargs)
58
59 def sample(self, duration: float, **kwargs) -> KalmanRecord:
60 from ..sampler import KalmanSampler
61 return KalmanSampler(self).measurement(duration, **kwargs)
62
63 def truth(self, duration: float = None):
64 record = self.sample(duration, process_noise=True, measurement_noise=False)
65 return record.xk, record.Pk
66 def ideal(self, duration: float = None):
67 record = self.sample(duration, process_noise=False, measurement_noise=False)
68 return record.xk

70class KalmanShapeMixin:
71 """
72 Mixin providing comprehensive dimension and consistency validation
73 for Kalman models and engines once (n, m, l) are known.
74
75 Expected dimensions:
76 x ∈ ℝⁿ×¹
77 P, Q ∈ ℝⁿ×ⁿ
78 z ∈ ℝᵐ×¹
79 R ∈ ℝᵐ×ᵐ
80 F ∈ ℝⁿ×ⁿ
81 H ∈ ℝᵐ×ⁿ
82 L ∈ ℝⁿ×q (usually q=n for additive noise)
83 M ∈ ℝᵐ×r (usually r=m for additive noise)
84 u ∈ ℝˡ×¹ (if applicable)
85 """
86
87 def _check_matrix(self, M, expected_shape, name: str, strict=False, symmetric=False):
88 """Validate matrix shape and optionally symmetry/PSD."""
89 if M is None:
90 raise ValueError(f"{name} is None (expected shape {expected_shape})")
91 M = np.asarray(M)
92 if M.shape != expected_shape:
93 raise ValueError(f"{name}: got {format_shape(M)}, expected {expected_shape}")
94 if symmetric:
95 if not np.allclose(M, M.T, atol=1e-10):
96 raise ValueError(f"{name} not symmetric.")
97 eigvals = np.linalg.eigvalsh(M)
98 if np.min(eigvals) < -1e-12:
99 raise ValueError(f"{name} not PSD (min eigenvalue {np.min(eigvals):.2e})")
100 return M
101
102 def _check_vector(self, v, expected_length, name: str):
103 """Validate vector shape (column vector form)."""
104 if v is None:
105 raise ValueError(f"{name} is None (expected length {expected_length})")
106 v = as_col(v, name)
107 if v.shape[0] != expected_length:
108 raise ValueError(f"{name}: got length {v.shape[0]}, expected {expected_length}")
109 return v
110
111 # ------------------------------------------------------
112 # Top-level consistency checker
113 # ------------------------------------------------------
114 def _check_shapes(self : KalmanModelMetadata, *, strict=False):
115 """
116 Validate internal model dimensions (x, P, Q, R, F, H, L, M)
117 according to (n, m, l).
118 """
119 n, m, l = self.n, self.m, self.l
120
121 # --- State quantities ---
122 if hasattr(self, "x0"): self._check_vector(self.x0, n, "x0")
123 if hasattr(self, "x"): self._check_vector(self.x, n, "x")
124 if hasattr(self, "P"): self._check_matrix(self.P, (n, n), "P", symmetric=strict)
125 if hasattr(self, "Q"): self._check_matrix(self.Q(), (n, n), "Q", symmetric=strict)
126
127 # --- Measurement quantities ---
128 if m > 0:
129 if hasattr(self, "R"): self._check_matrix(self.R(), (m, m), "R", symmetric=strict)
130 if hasattr(self, "H"): self._check_matrix(self.H(np.zeros((n,1))), (m, n), "H")
131 if hasattr(self, "M") and self.M is not None:
132 M = self.M(np.zeros((n,1)))
133 if M is not None:
134 if M.ndim == 2 and M.shape[0] != m:
135 raise ValueError(f"M: invalid rows {M.shape[0]}, expected {m}")
136
137 # --- Process quantities ---
138 if hasattr(self, "F"): self._check_matrix(self.F(np.zeros((n,1))), (n, n), "F")
139 if hasattr(self, "L") and self.L is not None:
140 L = self.L(np.zeros((n,1)))
141 if L is not None:
142 if L.ndim == 2 and L.shape[0] != n:
143 raise ValueError(f"L: invalid rows {L.shape[0]}, expected {n}")
144
145 # --- Optional control input ---
146 if isinstance(l, int) and l > 0:
147 self._check_vector(self.u0, l, "u0")
148
149 return True

199 def _infer_P0(self, P0=None):
200 """Infer or validate initial covariance P0."""
201 n = getattr(self, "n", None)
202 if n is None:
203 raise ValueError("State dimension 'n' must be defined before inferring P0.")
204
205 # --- Direct argument or fallback ---
206 if P0 is not None:
207 P0 = np.asarray(P0)
208 else:
209 Q = self._infer_Q()
210 Q0 = Q(0) if callable(Q) else np.asarray(Q)
211 P0 = Q0.copy()
212
213 # --- Validation ---
214 if P0.shape != (n, n):
215 raise ValueError(f"P0 must be ({n},{n}); got {format_shape(P0)}")
216
217 return P0
218
219 def _infer_Q(self, Q=None):
220 """Infer or validate process noise covariance Q (matrix or callable)."""
221 n = getattr(self, "n", None)
222 if n is None:
223 raise ValueError("State dimension 'n' must be defined before inferring Q.")
224
225 # --- Priority: argument, attribute, or default ---
226 _Q = Q
227 if _Q is None:
228 # Prefer self.Q if available
229 _Q = self.Q() if callable(getattr(self, "Q", None)) else getattr(self, "Q", None)
230
231 # --- Default to zero covariance if nothing found ---
232 if _Q is None:
233 return np.zeros((n, n))
234
235 # --- Callable covariance (time-varying) ---
236 if callable(_Q):
237 Q0 = _Q(0)
238 if np.shape(Q0) != (n, n):
239 raise ValueError(f"Callable Q(t) must return shape ({n},{n}); got {format_shape(Q0)}")
240 return _Q
241
242 # --- Static covariance matrix ---
243 Qarr = np.asarray(_Q)
244 if Qarr.shape != (n, n):
245 raise ValueError(f"Q must be ({n},{n}); got {format_shape(Qarr)}")
246 return Qarr
247
248 def _infer_R(self, R=None):
249 """Infer or validate measurement noise covariance R (matrix or callable)."""
250 m = getattr(self, "m", None)
251 if m is None:
252 raise ValueError("Measurement dimension 'm' must be defined before inferring R.")
253
254 # --- Priority: argument, attribute, or default ---
255 _R = R
256 if _R is None:
257 _R = self.R() if callable(getattr(self, "R", None)) else getattr(self, "R", None)

259 # --- Default to zero covariance if nothing found ---

300 def initial_state(self, x0=None, P0=None):
301
302 import warnings
303 n = int(self.n)
304
305 if x0 is None:
306 # Preserve an already-set x0; fall back to zeros only on the first call.
307 existing = getattr(self, "x0", None)
308 x0 = existing if existing is not None else np.zeros((n, 1))
309 x0 = KalmanRecord._normalize_in(x0)
310 if x0.shape != (n, 1):
311 warnings.warn(
312 f"Initial state vector dimension expected {n}x1, got {format_shape(x0)}. Resizing."
313 )
314 x0 = np.asarray(x0, dtype=np.float64).reshape(-1, 1)
315 if x0.shape[0] != n:
316 x0 = np.resize(x0, (n, 1))
317 self.x0x0 = x0
318
319 if P0 is None:
320 # Preserve an already-set positive-definite P0; fall back to Q(0) only on first call.
321 existing = getattr(self, "P0", None)
322 P0 = existing if existing is not None else self.Q(0)
323 P0 = KalmanRecord._normalize_in(P0)
324 if P0.shape != (n, n):
325 warnings.warn(
326 f"Initial state covariance dimension expected {n}x{n}, got {format_shape(P0)}. Resizing."
327 )
328 P0 = np.asarray(P0, dtype=np.float64)
329 if P0.ndim == 1:
330 P0 = np.diag(P0)
331 P0 = np.resize(P0, (n, n))
332
333 self.P0 = P0
334
335 # Return the (vector, covariance) so initial_state_vector()/_covariance() — which
336 # unpack this — work, and callers can read back what was set.
337 return self.x0x0, self.P0

340 # Measurement <-> state mapping (via the observation matrix H)
341 # ------------------------------------------------------
342 def exclusive_mapping(self, state_idx, meas_idx) -> bool:
343 """Check if measurement(s) exclusively observe state(s) (via H)."""
344 if isinstance(state_idx, (tuple, list, np.ndarray)) or isinstance(meas_idx, (tuple, list, np.ndarray)):
345 n = max(len(state_idx) if hasattr(state_idx, "__len__") else 1,
346 len(meas_idx) if hasattr(meas_idx, "__len__") else 1)
347 state_idx = KalmanRecord._normalize_index(state_idx, n)
348 meas_idx = KalmanRecord._normalize_index(meas_idx, n)
349 for i in range(n):
350 if not self.exclusive_mapping(state_idx[i], meas_idx[i]):
351 return False
352 return True if n else False
353 else:
354 s = state_idx.value if hasattr(state_idx, 'value') else int(state_idx)
355 assert 0 <= s < self.n
356 m = meas_idx.value if hasattr(meas_idx, 'value') else int(meas_idx)
357 assert 0 <= m < self.m
358 return self.H()[m, s] != 0
359
360 def inclusive_mapping(self, state_idx, meas_idx) -> bool:
361 """Check if measurement(s) depend on state(s) in any way (via H)."""
362 if isinstance(state_idx, (tuple, list, np.ndarray)) or isinstance(meas_idx, (tuple, list, np.ndarray)):
363 n = max(len(state_idx) if hasattr(state_idx, "__len__") else 1,
364 len(meas_idx) if hasattr(meas_idx, "__len__") else 1)
365 state_idx = KalmanRecord._normalize_index(state_idx, n)
366 meas_idx = KalmanRecord._normalize_index(meas_idx, n)
367 for i in range(n):
368 if self.inclusive_mapping(state_idx[i], meas_idx[i]) is False:
369 return False
370 return True if n else False
371 else:
372 s = state_idx.value if hasattr(state_idx, 'value') else int(state_idx)
373 m = meas_idx.value if hasattr(meas_idx, 'value') else int(meas_idx)
374 assert 0 <= s < self.n
375 assert 0 <= m < self.m
376 return self.H()[m, s] != 0

582 def get_record(self : KalmanFilterMetadata, *, flavor=KalmanStateFlavor.POSTERIOR, **kwargs) -> KalmanRecord:
