74 Vector PID::Policy(const Vector &xHat, const Matrix &, double, const Vector *)
75 {
76 Vector e = xHat;
77 e -= fXRef;
78 fIntegral += fDt * e;
79 Vector de = e;
80 de -= fPrevError;
81 de *= 1. / fDt;
82 fPrevError = e;
83
84 Vector u = fKp * e;
85 u += fKi * fIntegral;
86 u += fKd * de;
87 u *= -1.;
88 return u;

151 std::size_t nIters, std::uint64_t seed, const Vector &xRef, const Matrix &Qcost, const Matrix &Rcost)
152 : Controller(model, xRef, Qcost, Rcost), fNRollouts(nRollouts), fHorizon(horizon),
153 fNoiseSigma(noiseSigma), fLambda(lambda), fNIters(nIters), fSeed(seed), fRandom(seed)
154 {

157 fNominal.assign(fHorizon, Vector(NControl(), 0.));

159
160 Vector MPPI::Policy(const Vector &xHat, const Matrix &, double t, const Vector *)
161 {
162 const std::size_t l = NControl(), n = fModel->NState();
163 const double dt = fModel->DeltaT();
164 std::normal_distribution<double> gauss(0., fNoiseSigma);
165
166 // The caller's perturbations, when it gave some, are used once.
167 const std::vector<double> draws = std::move(fDraws);
168 fDraws.clear();
169 std::size_t cursor = 0;
170
171 std::vector<Vector> nominal = fNominal;
172 std::vector<std::vector<Vector>> trial(fNRollouts, std::vector<Vector>(fHorizon, Vector(l, 0.)));
173 std::vector<double> costs(fNRollouts);
174
175 for (std::size_t iter = 0; iter < fNIters; ++iter)
176 {
177 for (std::size_t i = 0; i < fNRollouts; ++i)
178 {
179 Vector xi = xHat;
180 double cost = 0.;
181 for (std::size_t k = 0; k < fHorizon; ++k)
182 {
183 Vector uk = nominal[k];
184 for (std::size_t j = 0; j < l; ++j) uk(j) += draws.empty() ? gauss(fRandom) : draws[cursor++];
185 // Clip the candidate before rolling it out or costing it, so the
186 // averaged nominal stays inside what the actuator can deliver.
187 uk = fModel->ClipControl(uk);
188 trial[i][k] = uk;
189
190 xi = fModel->f(xi, fModel->NControl() > 0 ? uk : Vector(), t + static_cast<double>(k) * dt);
191 Vector e = xi;
192 e -= fXRef;
193 for (std::size_t a = 0; a < n; ++a)
194 for (std::size_t b = 0; b < n; ++b) cost += e(a) * fQCost(a, b) * e(b);
195 for (std::size_t a = 0; a < l; ++a)
196 for (std::size_t b = 0; b < l; ++b) cost += uk(a) * fRCost(a, b) * uk(b);
197 }
198 costs[i] = cost;
199 }
200
201 const double best = *std::min_element(costs.begin(), costs.end());
202 std::vector<double> weights(fNRollouts);
203 double total = 0.;
204 for (std::size_t i = 0; i < fNRollouts; ++i)
205 {
206 weights[i] = std::exp(-(costs[i] - best) / fLambda);
207 total += weights[i];
208 }
209 total += 1e-12;
210
211 for (std::size_t k = 0; k < fHorizon; ++k)
212 {
213 Vector acc(l, 0.);
214 for (std::size_t i = 0; i < fNRollouts; ++i) acc += (weights[i] / total) * trial[i][k];
215 nominal[k] = acc;
216 }
217 }
218
219 const Vector u0 = nominal[0];
220 // Shift the warm-started sequence for the next call's receding horizon.
221 for (std::size_t k = 0; k + 1 < fHorizon; ++k) fNominal[k] = nominal[k + 1];
222 fNominal[fHorizon - 1] = Vector(l, 0.);
223 return u0;
224 }

261 void Boost::Update(const Vector &xHat, const Matrix &P, double t, const Vector *nu)
262 {
263 if (!nu) return;
264 if (fRNominal.Empty()) fRNominal = fModel->R(t);
265
266 const std::vector<std::size_t> keep = Utils::ValidChannels(*nu);
267 if (keep.empty()) return;
268
269 const Matrix H = Utils::CompressRows(fModel->JacobianH(xHat, t), keep);
270 Matrix S = Utils::Similarity(H, P);
271 S += Utils::CompressBoth(fModel->R(t), keep);
272 fNis.push_back(Utils::Mahalanobis(Utils::Compress(*nu, keep), Utils::Inverse(S)));
273
274 const std::size_t first = fNis.size() > fWindow ? fNis.size() - fWindow : 0;
275 double mean = 0.;
276 for (std::size_t i = first; i < fNis.size(); ++i) mean += fNis[i];
277 mean /= static_cast<double>(fNis.size() - first);
278
279 const double scale = mean > fThreshold * static_cast<double>(keep.size()) ? 1. / fBoostFactor : 1.;
280 Matrix R = fRNominal;
281 R *= scale;
282 fModel->SetR(R);
283 fScaleLog.emplace_back(t, scale);
284 }

287 {
288 Controller::Reset();
289 if (!fRNominal.Empty()) fModel->SetR(fRNominal);
290 fNis.clear();
291 fScaleLog.clear();

308 void AdaptiveQR::Update(const Vector &xHat, const Matrix &P, double t, const Vector *nu)
309 {
310 if (!nu) return;
311 if (fQNominal.Empty()) fQNominal = fModel->Q(t);
312 if (!Utils::IsComplete(*nu)) return; // a partial innovation has no full outer product
313
314 fInnovations.push_back(*nu);
315 if (fInnovations.size() < fWindow) return;
316
317 const std::size_t m = nu->Size();
318 Matrix empirical(m, m, 0.);
319 for (std::size_t s = fInnovations.size() - fWindow; s < fInnovations.size(); ++s)
320 for (std::size_t i = 0; i < m; ++i)
321 for (std::size_t j = 0; j < m; ++j) empirical(i, j) += fInnovations[s](i) * fInnovations[s](j);
322 empirical *= 1. / static_cast<double>(fWindow);
323
324 Matrix S = Utils::Similarity(fModel->JacobianH(xHat, t), P);
325 S += fModel->R(t);
326
327 const double lambda = std::min(fQMax, std::max(fQMin, Trace(empirical) / std::max(Trace(S), 1e-12)));
328 Matrix Q = fQNominal;
329 Q *= lambda;
330 fModel->SetQ(Q);
331 fScaleLog.emplace_back(t, lambda);
332 }
333
334 void AdaptiveQR::Reset()
335 {
336 Controller::Reset();
337 if (!fQNominal.Empty()) fModel->SetQ(fQNominal);
338 fInnovations.clear();
339 fScaleLog.clear();
