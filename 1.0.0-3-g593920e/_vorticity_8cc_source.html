33 SpectralField Vorticity::RHSHat(const SpectralField &wh, double) const
34 {
35 const std::size_t size = wh.Size();
36
37 // Stream function, velocity and gradient, all from wh by multiplications.
38 SpectralField psih = wh;
39 for (std::size_t i = 0; i < size; ++i) psih.Data()[i] *= fK2Inv.Data()[i];
40 SpectralField vh = DerivativeX(psih);
41 vh *= -1.;
42 const Field u = Real(FFT::Inverse(DerivativeY(psih)));
43 const Field v = Real(FFT::Inverse(vh));
44 const Field wx = Real(FFT::Inverse(DerivativeX(wh)));
45 const Field wy = Real(FFT::Inverse(DerivativeY(wh)));
46
47 // The products are taken on the grid, then dealiased in Fourier space.
48 Field product(fNy, fNx);
49 for (std::size_t i = 0; i < size; ++i)
50 product.Data()[i] = u.Data()[i] * wx.Data()[i] + v.Data()[i] * wy.Data()[i] + fBeta * v.Data()[i];
51 SpectralField out = FFT::Forward(product);
52
53 const bool forced = fForcing.Size() != 0;
54 for (std::size_t i = 0; i < size; ++i)
55 {
56 cd value = -out.Data()[i] * fDealias.Data()[i] - fNu * fK2.Data()[i] * wh.Data()[i];
57 if (forced) value += fForcing.Data()[i];
58 out.Data()[i] = value;
59 }
60 return out;
61 }
