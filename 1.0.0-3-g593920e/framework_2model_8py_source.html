11def _recording(init):
12 """``init`` wrapped so that the instance remembers the arguments it was
13 built with (``_init_arguments``, the outermost constructor's, defaults
14 included). That is what lets a model given by code be written to a file as
15 a reference to its class: the file carries those arguments, and reading it
16 calls the constructor with them again."""
17 import functools
18 import inspect
19 if getattr(init, "_kalman_recording", False):
20 return init
21 try:
22 signature = inspect.signature(init)
23 except (TypeError, ValueError):
24 return init
25
26 @functools.wraps(init)
27 def wrapper(self, *args, **kwargs):
28 if "_init_arguments" not in self.__dict__:
29 try:
30 bound = signature.bind(self, *args, **kwargs)
31 bound.apply_defaults()
32 recorded, first = {}, None
33 for position, (name, value) in enumerate(bound.arguments.items()):
34 kind = signature.parameters[name].kind
35 if position == 0 or kind is inspect.Parameter.VAR_POSITIONAL:
36 continue
37 if position == 1 and name in ("sensor_id", "sensor", "name"):
38 first = name # the sensor id: the reader gives its own
39 if kind is inspect.Parameter.VAR_KEYWORD:
40 recorded.update(value)
41 else:
42 recorded[name] = value
43 self.__dict__["_init_arguments"] = recorded
44 self.__dict__["_init_first"] = first
45 except TypeError:
46 pass # the call itself will say what is wrong
47 return init(self, *args, **kwargs)
48
49 wrapper._kalman_recording = True
50 return wrapper

53class KalmanModel(KalmanModelMetadata, KalmanModelMixin):
54 """
55 The single, flexible Kalman **model** contract (the *system*, WHAT).
56
57 A model defines ``f/h`` and the noise covariances ``Q/R``; the Jacobians
58 ``F/H/B/L/M`` are optional refinements with a numerical default. Construct it two
59 ways, **same contract**, runnable by **any** engine:
60
61 - **matrices** — pass ``F=,H=,B=,L=,M=`` (constant or callable ``M(t)``); ``f``/``h``
62 then default to ``f(x)=Fx+Bu+Lw`` / ``h(x)=Hx+Mv`` and ``F()/H()/…`` return them;
63 - **continuous** — pass ``A=`` (and, for process noise, ``Qc=`` with optional
64 ``Lc=``) instead of ``F=``/``Q=``; the model discretizes exactly via
65 ``expm``/Van Loan (needs ``fs=``/``sample_rate=``). ``Ac()``/``Qc()``/``Lc()``
66 return the continuous inputs back for inspection (e.g.
67 ``helpers.symbolic_discretize(model.Ac())``); mutually exclusive with
68 ``F=``/``Q=`` respectively;
69 - **functions** — override ``f``/``h`` (nonlinear); ``F/H`` come from
70 :func:`numerical_jacobian`.
71
72 Engines consume only what they need: UKF/PF use ``f/h/Q/R``; EKF additionally reads
73 ``F/H`` (numerical/analytic); KF reads ``F/H`` as constants.
74
75 Engine binding is **declared, not imported**: ``DEFAULT_ENGINE`` /
76 ``ALLOWED_ENGINES`` are :class:`KalmanEngineKind` values, so a model never imports an
77 engine; the :class:`~kalman.framework.filter.KalmanFilter` resolves/validates them.
78 """
79
80 NAME = "Kalman Model"
81 DESCRIPTION = "Generic model for Kalman-family filters; matrices (F/H/…) or override f/h."
82 VERSION = "2.0"
83
84 # Engine binding (declared via KalmanEngineKind so the model never imports an engine).
85 DEFAULT_ENGINE = KalmanEngineKind.EXTENDED # safe universal default (falls back to linear on matrices)
86 ALLOWED_ENGINES = ALL_ENGINES # any engine
87
88 # What f returns (declared, like the engine): the step itself (DISCRETE),
89 # or the derivative dx/dt (CONTINUOUS), which the framework decodes into
90 # the step with the model's integrator, in SOLVER_SUBSTEPS steps per sample.
91 # A field model binds a solver instead and calls it from f.
92 DYNAMICS = KalmanDynamicsKind.DISCRETE
93 # Where the Jacobians come from when the model gives no formula: forward
94 # differences (the default) or central ones. A model that overrides F()/H()
95 # or holds matrices is analytic whatever this says.
96 DIFFERENTIATOR = KalmanDifferentiatorKind.FORWARD
97 #: What the model is made of (KalmanStructureKind), or None to read it from the model: see ``structure``.
98 STRUCTURE = None
99 DEFAULT_INTEGRATOR = KalmanIntegratorKind.RK4
100 ALLOWED_INTEGRATORS = ALL_INTEGRATORS
101 DEFAULT_SOLVER = KalmanSolverKind.SPECTRAL
102 ALLOWED_SOLVERS = ALL_SOLVERS
103 SOLVER_SUBSTEPS = 1

123 self._solver_override = self._validate_kind(

131 # --- Optional explicit matrices (the "linear" construction path) ---
132 def _store(M):
133 if M is None:
134 return None
135 return M if callable(M) else np.asarray(M, dtype=self.dtype)
136
137 # --- Continuous-time construction path: A (+ optional Qc/Lc) --------
138 # Alternative to passing F=/Q= directly: give the continuous generator
139 # ẋ=Ax(+Lw) and let the model discretize exactly (expm for F, Van Loan
140 # for Q). Kept as self._Ac/_Qc/_Lc for introspection — NOT the same
141 # slot as the framework's own _L (discrete process-noise map ∂f/∂w,
142 # set below), which is a distinct thing left untouched here.
143 A, Qc, Lc = kwargs.pop("A", None), kwargs.pop("Qc", None), kwargs.pop("Lc", None)
144 self._Ac = self._Qc = self._Lc = None
145 if A is not None:
146 if kwargs.get("F") is not None:
147 raise ValueError("Pass either F= or A=, not both.")
148 if not self.fsfsfs:
149 raise ValueError("A= needs fs= (or sample_rate=) to decode F = exp(A / fs).")
150 A = np.asarray(A, dtype=self.dtype)
151 dt = 1.0 / self.fsfsfs
152 kwargs["F"] = matrix_exponential(A * dt)
153 self._Ac = A
154 if Qc is not None:
155 if kwargs.get("Q") is not None:
156 raise ValueError("Pass either Q= or Qc=, not both.")
157 Lc = np.eye(A.shape[0], dtype=self.dtype) if Lc is None else np.asarray(Lc, dtype=self.dtype)
158 _, kwargs["Q"] = van_loan_discretize(A, Lc, np.asarray(Qc, dtype=self.dtype), dt)
159 self._Qc, self._Lc = Qc, Lc
160 elif Qc is not None:
161 # A continuous model with no generator matrix: its noise intensity is
162 # decoded in Q(), with the Jacobian of its derivative as generator.
163 if self.DYNAMICS is not KalmanDynamicsKind.CONTINUOUS:
164 raise ValueError("Qc= needs a continuous model: pass A=, or declare DYNAMICS = CONTINUOUS.")
165 if kwargs.get("Q") is not None:
166 raise ValueError("Pass either Q= or Qc=, not both.")
167 if not self.fsfsfs:
168 raise ValueError("Qc= needs fs= (or sample_rate=) to decode the per-step Q.")
169 self._Qc = np.asarray(Qc, dtype=self.dtype)
170 self._Lc = None if Lc is None else np.asarray(Lc, dtype=self.dtype)
171 kwargs["Q"] = self._decode_noise
172 kwargs.setdefault("L", np.eye(self.nnn, dtype=self.dtype)) # the coupling is inside the decoded Q
173 self._decoded_noise = None
174
175 self._F = _store(kwargs.pop("F", None))
176 self._H = _store(kwargs.pop("H", None))
177 self._B = _store(kwargs.pop("B", None))
178 self._L = _store(kwargs.pop("L", None))
179 self._M = _store(kwargs.pop("M", None))
180 self._is_linear = self._F is not None and self._H is not None
181
182 # Infer l from B columns so _infer_l() picks it up
183 if self._B is not None:
184 B_mat = self._B(0) if callable(self._B) else self._B
185 if B_mat is not None:
186 self.ll = int(np.asarray(B_mat).shape[1])
187
188 # --- Infer dimensions ---
189 self.nnn = self._infer_n()
190 self.mmm = self._infer_m()
191 self.ll = self._infer_l()
192
193 # --- Infer covariances ---
194 self._Q_Q = self._infer_Q(kwargs.pop("Q", None))
195 self._R_R = self._infer_R(kwargs.pop("R", None))
196
197 # --- Initial state ---
198 x0 = kwargs.pop("x0", None)
199 if x0 is None: x0 = np.zeros((self.nnn, 1))
200
201 P0 = kwargs.pop("P0", None)
202 self.initial_state(x0, P0)
203
204 self._check_shapes(strict=False)

219 @staticmethod
220 def _validate_kind(value, allowed, label):
221 """Normalise an ``integrator=``/``solver=`` override.
222
223 Accepts a Kind, a registered name, a concrete class, a ready instance,
224 or None. A **custom** scheme — one whose ``KIND`` is not a member of the
225 framework enum — is validated by its *base class* instead of by
226 ``ALLOWED_*``, since a closed enum cannot describe a kind it has never
227 heard of. Names are checked by resolving them (raising on a typo) but
228 returned unresolved, so ``default_solver(**kwargs)`` can still pass
229 constructor arguments.
230 """
231 if value is None:
232 return None
233
234 from .integrator import KalmanIntegrator
235 from .solver import KalmanSolver
236 if label == "integrator":
237 base, enum_cls = KalmanIntegrator, KalmanIntegratorKind
238 from .integrators import resolve_integrator as _resolve
239 else:
240 base, enum_cls = KalmanSolver, KalmanSolverKind
241 from .solvers import resolve_solver as _resolve
242
243 if isinstance(value, str):
244 _resolve(value) # raises ValueError on an unknown name
245 return value # resolved later, with kwargs
246
247 is_impl = isinstance(value, base) or (isinstance(value, type) and issubclass(value, base))
248 kind = getattr(value, "KIND", value) # instance/class → its KIND; else the Kind itself
249 if is_impl and not isinstance(kind, enum_cls):
250 return value # custom scheme: the base class is the contract
251 if allowed is not None and kind not in allowed:
252 raise TypeError(f"{label} {kind} not in ALLOWED_{label.upper()}S "
253 f"({sorted(k.name for k in allowed)}).")
254 return value

280 # ------------------------------------------------------------------
281 # Continuous dynamics. A class with DYNAMICS = CONTINUOUS writes f as the
282 # derivative dx/dt. When the class is created, that f is kept as
283 # `derivative` and f becomes the decoded step below, so engines and
284 # filters only ever see a discrete step, and the Jacobian F = df/dx is the
285 # Jacobian of the integrated step.
286 # ------------------------------------------------------------------
287 def __init_subclass__(cls, **kwargs):
288 super().__init_subclass__(**kwargs)
289 if "__init__" in cls.__dict__:
290 cls.__init____init__ = _recording(cls.__dict__["__init__"])
291 written = cls.__dict__.get("f")
292 if (written is not None and cls.DYNAMICS is KalmanDynamicsKind.CONTINUOUS
293 and written is not _decoded_step):
294 cls.derivativederivative = written
295 cls.fff = _decoded_step

297 def _jacobian(self, fn, x):
298 """``d fn / dx`` at ``x`` by the model's declared differentiator."""
299 kind = self.DIFFERENTIATOR
300 if kind is KalmanDifferentiatorKind.CENTRAL:
301 return numerical_jacobian(fn, x, scheme="central")
302 if kind is KalmanDifferentiatorKind.FORWARD:
303 return numerical_jacobian(fn, x, scheme="forward")
304 if kind is KalmanDifferentiatorKind.ANALYTIC:
305 raise NotImplementedError(
306 f"{type(self).__name__} declares DIFFERENTIATOR = ANALYTIC but gives no formula for this "
307 "Jacobian: override F()/H() (or pass matrices), or declare FORWARD or CENTRAL.")
308 raise NotImplementedError("DIFFERENTIATOR = AUTOMATIC is reserved for libKalmanLearn.")
309
310 def derivative(self, x, u=None, w=None, t=None):
311 """dx/dt of a CONTINUOUS model: the ``f`` its class wrote."""
312 raise NotImplementedError(f"{type(self).__name__} is not a continuous model.")

314 def substeps(self, x=None) -> int:
315 """Integrator steps per sample when decoding a continuous model."""
316 return max(1, int(getattr(self, "SOLVER_SUBSTEPS", 1)))

318 def after_step(self, x, u=None, t=None):
319 """Hook: adjust the state a continuous model lands on after one sample
320 (a clamp, a slot that is not integrated). Returns the state."""
321 return x

323 def decode_step(self, x, u=None, t=None):
324 """One sample of a CONTINUOUS model: the derivative integrated over
325 ``1/fs`` by the model's integrator, then :meth:`after_step`. No noise."""
326 steps = self.substeps(x)
327 dt = ((1.0 / self.fsfsfs) if self.fsfsfs else 1.0) / steps
328 # The state is a column throughout, whatever shape the caller passed: a
329 # flat state plus a column derivative would broadcast to a matrix.
330 rate = lambda xx, uu, tt: np.asarray(self.derivativederivative(xx, u=uu, t=tt), dtype=self.dtype).reshape(-1, 1)
331 xk = self.default_integrator().integrate(rate, np.asarray(x, dtype=self.dtype).reshape(-1, 1), u=u,
332 t=(0.0 if t is None else t), dt=dt, n=steps)
333 return self.after_step(xk, u=u, t=t)

335 # ============================================================
336 # Core equations (matrix path; subclasses override for nonlinear f/h)
337 # ============================================================
338 def f(self, x, u=None, w=None, t=None):
339 """State transition, the discrete step. Uses constant matrices if
340 provided; otherwise a subclass writes it (the step, or with
341 ``DYNAMICS = CONTINUOUS`` the derivative, which is decoded)."""
342 if self._F is None:
343 raise NotImplementedError(
344 "Model must define f() or be constructed with matrices (F=, H=, ...).")
345 x = self.FFF(t=t) @ x
346 if u is not None:
347 B = self.BBB(t=t)
348 x = x + (u if B is None else B @ u)
349 if w is not None:
350 L = self.LLL(t=t)
351 x = x + (w if L is None else L @ w)
352 return x
353
354 def h(self, x, v=None, t=None):
355 """Observation map. Uses constant matrices if provided; else override me."""
356 if self._H is None:
357 raise NotImplementedError(
358 "Model must define h() or be constructed with matrices (F=, H=, ...).")
359 z = self.HHH(t=t) @ x
360 if v is not None:
361 M = self.MMM(t=t)
362 z = z + (v if M is None else M @ v)
363 return z

368 def F(self, x: np.ndarray = None, u: Optional[np.ndarray] = None, t: Optional[float] = None) -> np.ndarray:

382 def B(self, x: Optional[np.ndarray] = None, u: Optional[np.ndarray] = None, t: Optional[float] = None) -> Optional[np.ndarray]:
383 """Control-input map ∂f/∂u (constant matrix if provided, numerical if u given)."""

385 return self._B(t) if callable(self._B) else self._B

387 return None

391 def clip_u(self, u: np.ndarray) -> np.ndarray:
392 """Actuator-limit hook: map a controller's raw command to what the plant can

403 return self._L(t) if callable(self._L) else self._L

405 return None

407 return self._jacobian(lambda ww: self.fff(as_col(x), u=u, w=ww, t=t), as_col(w))

409 def M(self, x: Optional[np.ndarray] = None, v: Optional[np.ndarray] = None,
410 t: Optional[float] = None) -> Optional[np.ndarray]:
411 """Measurement-noise map ∂h/∂v (constant matrix if provided, numerical if v given)."""
412 if self._M is not None:
413 return self._M(t) if callable(self._M) else self._M
414 if v is None or self._H is not None:
415 return None
416 x = self.x0x0x0 if x is None else x
417 return self._jacobian(lambda vv: self.hh(as_col(x), v=vv, t=t), as_col(v))

419 # Unified interface returning all Jacobians
420 def jacobians(self, x: np.ndarray,
421 u: Optional[np.ndarray] = None,
422 v: Optional[np.ndarray] = None,

435 def _decode_noise(self, t=None):
436 """Per-step Q of a continuous model that stated a noise intensity ``Qc``:
437 Van Loan over one sample, with the Jacobian of the derivative at the
438 initial state as generator. Decoded once per (sample rate, initial state)."""
439 x0 = np.zeros((self.nnn, 1)) if getattr(self, "x0", None) is None else np.asarray(self.x0x0x0, dtype=float).reshape(-1, 1)
440 key = (float(self.fsfsfs), x0.tobytes())
441 if self._decoded_noise is None or self._decoded_noise[0] != key:
442 u = np.zeros((self.ll, 1)) if getattr(self, "l", 0) else None
443 A = self._jacobian(lambda xx: self.derivativederivative(xx, u=u, t=t), x0)
444 Lc = np.eye(self.nnn) if self._Lc is None else self._Lc
445 _, Q = van_loan_discretize(np.asarray(A, dtype=float), Lc, self._Qc, 1.0 / self.fsfsfs)
446 self._decoded_noise = (key, Q.astype(self.dtype))
447 return self._decoded_noise[1]

470 def covariances(self, t: Optional[float] = None,
471 x: Optional[np.ndarray] = None,
472 u: Optional[np.ndarray] = None,
473 w: Optional[np.ndarray] = None,
474 v: Optional[np.ndarray] = None):
475 """Return (Q, R) with optional L Q Lᵀ / M R Mᵀ coupling."""
476 Q = self.QQQ(t)
477 L = self.LLL(x, u, w, t)
478 if L is not None:
479 Q = L @ Q @ L.T
480 R = self.RRR(t)
481 M = self.MMM(x, v, t)
482 if M is not None:
483 R = M @ R @ M.T
484 return Q, R

500 return self.STRUCTURE
501 cls = type(self)
502 untouched = all(getattr(cls, k) is getattr(KalmanModel, k) for k in ("f", "h", "F", "H", "Q", "R", "B", "L", "M"))
503 if not untouched or self._F is None or self._H is None or self._M is not None:
