1"""
2A libKalman model as a :class:`~kalman.framework.model.KalmanModel`.
3
4:class:`NativeModel` holds an instance built by the C++ factory and forwards the
5model contract — ``f/h``, ``F/H/B/L``, ``Q/R``, the initial state — to it through
6the C API (:mod:`kalman.framework.native.core`). Everything above the contract is
7the ordinary Python framework: the engines, the filters, the sampler, the record,
8``plot``, the API verbs and the MCP tools all see just another ``KalmanModel``.
9
10That is the one crossing between the two implementations for *models*: the C++
11side owns the physics, the Python side owns everything a notebook does with it.
12A crossing costs one C call per ``f``/``h``/coefficient evaluation, which is what
13the measured numbers in ``docs/15_Cxx-Library.md`` say is affordable in this
14direction (a C call is cheap; it is a *Python* callback from C++ that is not).
15
16:func:`native_model_classes` turns every registered name into a class — ``NAME``,
17``DESCRIPTION``, ``VERSION`` and ``DEFAULT_ENGINE`` filled from the C++ metadata —
18so the catalogue, ``describe()`` and ``make_model()`` treat a plugin model exactly
19like a class in ``apps/``: ``{"base": "constant-velocity", "sensor_id": "cv"}``.
20"""

34class NativeModel(KalmanModel):
35 """A C++ model from the factory, driven by the Python engines.
36
37 ``NativeModel("constant-velocity", "sensor-1")`` constructs the registered
38 C++ model and adopts its dimensions, sample rate, engine binding and initial
39 state. ``fs=``, ``x0=``, ``P0=`` and ``t0=`` override those and are pushed
40 back into the C++ instance, so both sides agree on every step length.
41 """
42
43 NAME = "Native Model"
44 DESCRIPTION = "A libKalman model (C++) exposed through the C API."
45 VERSION = "1.0"

47 NATIVE_NAME: str | None = None # set on the classes native_model_classes() builds
48 DEFAULT_ENGINE = KalmanEngineKind.EXTENDED

51 def __init__(self, name: str | None = None, sensor_id=None, **kwargs):
52 L = core.require()
53 # A model that exists already (read from a file) is wrapped as it is.
54 handle = kwargs.pop("_handle", None)
55 if handle is not None:
56 name = name or core._s(L.kalman_model_name(handle))
57 else:
58 name = name or self.NATIVE_NAME
59 if not name:
60 raise ValueError("NativeModel needs the registered name of a C++ model.")
61 handle = L.kalman_model_create(name.encode())
62 if handle == core.ffi.NULL:
63 raise KeyError(f"NativeModel: {core.last_error()}")
64 self._handle = core.ffi.gc(handle, L.kalman_model_destroy)
65 self.native_name = name
66
67 n, m, l = core.ffi.new("int *"), core.ffi.new("int *"), core.ffi.new("int *")
68 core.check(L.kalman_model_dims(self._handle, n, m, l), "kalman_model_dims")
69 self.nnnn, self.mmmm, self.lll = int(n[0]), int(m[0]), int(l[0])
70
71 # Time base: an explicit fs wins and is pushed down; otherwise the C++ one is adopted.
72 fs = kwargs.pop("fs", None) or kwargs.pop("sample_rate", None)
73 native_fs = float(L.kalman_model_sample_rate(self._handle))
74 if fs is not None:
75 core.check(L.kalman_model_set_sample_rate(self._handle, float(fs)), "set_sample_rate")
76 else:
77 fs = native_fs if native_fs > 0 else None
78 if "t0" in kwargs and kwargs["t0"] is not None:
79 core.check(L.kalman_model_set_t0(self._handle, float(kwargs["t0"])), "set_t0")
80 else:
81 kwargs["t0"] = float(L.kalman_model_t0(self._handle))
82
83 # Initial state: same rule. A C++ P0 that was never set is all zeros,
84 # which the Python side reads as "infer from Q", as it does for any model.
85 x0 = kwargs.pop("x0", None)
86 if x0 is not None:
87 x0 = np.asarray(x0, dtype=float).reshape(-1)
88 core.check(L.kalman_model_set_x0(self._handle, core.cdptr(np.ascontiguousarray(x0))), "set_x0")
89 else:
90 x0 = self._vec(L.kalman_model_x0, "x0")
91 P0 = kwargs.pop("P0", None)
92 if P0 is not None:
93 P0 = np.ascontiguousarray(np.asarray(P0, dtype=float).reshape(self.nnnn, self.nnnn))
94 core.check(L.kalman_model_set_P0(self._handle, core.cdptr(P0)), "set_P0")
95 else:
96 P0 = self._mat(L.kalman_model_P0, (self.nnnn, self.nnnn), "P0")
97 if not np.any(P0):
98 P0 = None
99
100 # Noise and named parameters live in the C++ instance (Q(), R() and
101 # parameters() read them from there), so what is passed here is pushed
102 # down, not kept on the Python side where nothing would read it.
103 for key, setter, size in (("Q", L.kalman_model_set_Q, self.nnnn), ("R", L.kalman_model_set_R, self.mmmm)):
104 value = kwargs.pop(key, None)
105 if value is not None:
106 value = np.ascontiguousarray(np.asarray(value, dtype=float).reshape(size, size))
107 core.check(setter(self._handle, core.cdptr(value)), "set_" + key)
108 # A model that computes its noise from its own parameters does
109 # not take a matrix: say so instead of running with another one.
110 getter = L.kalman_model_Q if key == "Q" else L.kalman_model_R
111 if not np.allclose(self._mat(getter, (size, size), key, 0.0), value, rtol=1e-12, atol=0.0):
112 raise ValueError(
113 f"the C++ model {name!r} computes its own {key} and does not take one; "
114 f"set its parameters instead ({', '.join(sorted(self.parameters())) or 'it declares none'}).")
115 declared = self.parameters()
116 for key in [k for k in kwargs if k in declared]:
117 self.set_parameter(key, kwargs.pop(key))
118
119 self.DEFAULT_ENGINEDEFAULT_ENGINE = ENGINE_BY_INT.get(
120 core.check(L.kalman_model_engine(self._handle), "engine"), KalmanEngineKind.EXTENDED)
121 self._analytic = core.check(L.kalman_model_analytic_jacobian(self._handle), "analytic") == 1
122
123 super().__init__(sensor_id if sensor_id is not None else name,
124 fs=fs, x0=x0.reshape(-1, 1), P0=P0, **kwargs)

126 def native_handle(self):
127 """The C++ model itself, so that a filtering pass runs wholly in
128 Kalman::Filter instead of asking for f, h and the matrices step by step.
129 ``None`` for a subclass that rewrites part of the model in Python."""
130 import os
131 if not USE_NATIVE_TWINS or os.environ.get("KALMAN_NATIVE_TWINS", "1") == "0":
132 return None
133 for name in ("f", "h", "F", "H", "B", "L", "Q", "R"):
134 if name in self.__dict__ or getattr(type(self), name) is not getattr(NativeModel, name):
135 return None
136 return self._handle

139 def _vec(self, fn, what, *args):
140 out = np.empty(self.nnnn if what != "z" else self.mmmm, dtype=np.float64)
141 core.check(fn(self._handle, *args, core.dptr(out)), what)
142 return out

144 def _mat(self, fn, shape, what, *args):
145 out = np.empty(shape, dtype=np.float64)
146 core.check(fn(self._handle, *args, core.dptr(out)), what)
147 return out
148
149 def _mat_optional(self, fn, shape, what, t):
150 out = np.empty(shape, dtype=np.float64)
151 return out if core.check(fn(self._handle, t, core.dptr(out)), what) == 1 else None

153 @staticmethod
154 def _t(t):
155 return 0.0 if t is None else float(t)

157 def _flat(self, v, size, what):
158 a = np.ascontiguousarray(np.asarray(v, dtype=np.float64).reshape(-1))
159 if a.size != size:
160 raise ValueError(f"{self.native_name}: {what} has {a.size} elements, expected {size}.")
161 return a

163 def _u(self, u):
164 if u is None or self.lll == 0:
165 return core.ffi.NULL
166 return core.cdptr(self._flat(u, self.lll, "u"))

168 @staticmethod
169 def _like(out, x):
170 # Column in, column out; flat in, flat out — the engines use both.
171 return out.reshape(-1, 1) if np.ndim(x) == 2 else out

173 # --- the contract: coefficients --------------------------------------------
174 def F(self, x=None, u=None, t=None):
175 L = core.lib
176 if self._analytic:
177 return self._mat(L.kalman_model_F, (self.nnnn, self.nnnn), "F", self._t(t))
178 xf = self._flat(self.x0x0x0 if x is None else x, self.nnnn, "x")
179 return self._mat(L.kalman_model_jacobian, (self.nnnn, self.nnnn), "df/dx",
180 core.cdptr(xf), self._u(u), self._t(t))
181
182 def H(self, x=None, t=None):
183 L = core.lib
184 if self._analytic:
185 return self._mat(L.kalman_model_H, (self.mmmm, self.nnnn), "H", self._t(t))
186 xf = self._flat(self.x0x0x0 if x is None else x, self.nnnn, "x")
187 return self._mat(L.kalman_model_jacobian_h, (self.mmmm, self.nnnn), "dh/dx",
188 core.cdptr(xf), self._t(t))
189
190 def B(self, x=None, u=None, t=None):
191 if self.lll == 0:
192 return None
193 return self._mat_optional(core.lib.kalman_model_B, (self.nnnn, self.lll), "B", self._t(t))
194
195 def L(self, x=None, u=None, w=None, t=None):
196 return self._mat_optional(core.lib.kalman_model_L, (self.nnnn, self.nnnn), "L", self._t(t))
197
198 def M(self, x=None, v=None, t=None):
199 return None # additive measurement noise
200
201 def Q(self, t=None):
202 return self._mat(core.lib.kalman_model_Q, (self.nnnn, self.nnnn), "Q", self._t(t))
203
204 def R(self, t=None):
205 return self._mat(core.lib.kalman_model_R, (self.mmmm, self.mmmm), "R", self._t(t))
206
207 # --- the contract: maps ----------------------------------------------------
208 def f(self, x, u=None, w=None, t=None):
209 xf = self._flat(x, self.nnnn, "x")
210 out = np.empty(self.nnnn, dtype=np.float64)
211 core.check(core.lib.kalman_model_f(self._handle, core.cdptr(xf), self._u(u), self._t(t),
212 core.dptr(out)), "f")
213 if w is not None:
214 Lw = self.LLLL(t=t)
215 wf = self._flat(w, self.nnnn, "w")
216 out = out + (wf if Lw is None else Lw @ wf)
217 return self._like(out, x)
218
219 def h(self, x, v=None, t=None):
220 xf = self._flat(x, self.nnnn, "x")
221 out = np.empty(self.mmmm, dtype=np.float64)
222 core.check(core.lib.kalman_model_h(self._handle, core.cdptr(xf), self._t(t), core.dptr(out)), "h")
223 if v is not None:
224 out = out + self._flat(v, self.mmmm, "v")
225 return self._like(out, x)

228 def run_native(self, zk, uk=None, *, engine="extended", smooth=False, **engine_params) -> dict:
229 """Kalman::Filter over ``zk`` entirely in C++ with a native ``engine`` (name
230 or kind, plus its constructor parameters). Returns the per-step arrays:
231 ``x_prior/P_prior/x_post/P_post/innovations/transitions`` and, with
232 ``smooth=True``, ``x_smooth/P_smooth``."""
233 eng = core.create_engine(engine.name.lower() if hasattr(engine, "name") else str(engine).lower(),
234 **engine_params)
235 return core.run_filter(self._handle, eng, self.nnnn, self.mmmm, self.lll, zk, uk, smooth=smooth)

237 def filter_native(self, zk, uk=None, *, engine="extended", smooth=False, **engine_params):
238 """:meth:`run_native` as a :class:`~kalman.framework.io.record.KalmanRecord` —
239 posterior, or smoothed when ``smooth=True`` — so it plots, records and
240 compares like any Python filter's output."""
241 from ..io.record import KalmanRecord
242 from ..constants import KalmanStateFlavor
243 out = self.run_native(zk, uk, engine=engine, smooth=smooth, **engine_params)
244 N = out["x_post"].shape[0]
245 t0 = float(self.t0t0 or 0.0)
246 time = t0 + np.arange(N, dtype=float) / (self.fsfsfs or 1.0)
247 zk = np.asarray(zk, dtype=float).reshape(N, self.mmmm)
248 if smooth:
249 return KalmanRecord(time=time, xk=out["x_smooth"], Pk=out["P_smooth"], zk=zk,
250 model=self, flavor=KalmanStateFlavor.SMOOTHED)
251 return KalmanRecord(time=time, xk=out["x_post"], Pk=out["P_post"], zk=zk,
252 model=self, flavor=KalmanStateFlavor.POSTERIOR)

387class NativeTwin:
388 """Mixin for an application model whose physics also exists in libKalmanApps.
389
390 The Python class stays the model: its constructor, its attributes, ``plot``
391 and the notebooks are unchanged. What changes is who runs the filtering
392 pass. When the configuration has a C++ counterpart, ``KalmanFilter.forward``
393 hands the whole pass to the twin (``Kalman::Filter`` in C++, one call), and
394 otherwise the Python loop runs as before.
395
396 A model declares its twin with two methods:
397
398 - :meth:`native_name` — the registered C++ name for *this* configuration, or
399 ``None`` when it has none (a Python callback, a feature not ported);
400 - :meth:`native_parameters` — the scalars to push, by the names the C++
401 class declares.
402
403 The sample rate, ``Q`` and ``R`` are pushed from the Python instance on every
404 run, and the initial state by the filter, so nothing is defined twice and a
405 change made to the Python model after construction reaches the twin.
406 """
407
408 #: What must be exactly the declaring class's own for the twin to stand in.
409 #: A continuous model's class keeps what it wrote as ``f`` under
410 #: ``derivative`` (KalmanModel.__init_subclass__), so its ``f`` is the shared
411 #: decoded step: the derivative, the closing step and the sub-stepping are
412 #: what a subclass changes, and are checked too.
413 _TWIN_CONTRACT = ("f", "h", "F", "H", "B", "L", "Q", "R", "derivative", "after_step", "substeps")
414
415 def native_name(self) -> str | None:
416 return None
417
418 def native_parameters(self) -> dict:
419 return {}
420
421 def native_arrays(self) -> dict:
422 """The named arrays to push (the C++ class's ``DeclareArray`` names). The
423 twin may size itself from them."""
424 return {}
425
426 #: Stored matrices the twin computes itself from what is pushed (``("_F",)``
427 #: for a model built with ``A=``): they do not keep the twin from standing in.
428 _TWIN_DECODES = ()
429
430 def _twin_owner(self) -> type:
431 for cls in type(self).__mro__:
432 if "native_name" in cls.__dict__ and cls is not NativeTwin:
433 return cls
434 return NativeTwin

436 def native_handle(self):
437 """The configured C++ twin, or ``None`` when the Python physics must run."""
438 import os
439 if not USE_NATIVE_TWINS or os.environ.get("KALMAN_NATIVE_TWINS", "1") == "0":
440 return None
441 if not core.HAS_CXX_CORE:
442 return None
443 name = self.native_name()
444 if not name or name not in apps_library_names():
445 return None
446 # A subclass or an instance that redefines the contract is a different
447 # model from the one that was ported.
448 owner = self._twin_owner()
449 for attr in self._TWIN_CONTRACT:
450 if attr in self.__dict__ or getattr(type(self), attr, None) is not getattr(owner, attr, None):
451 return None
452 # A control input is fine: the twin's f takes u, and the filter hands
453 # the twin the control sequence (a controller function keeps the Python
454 # model, see KalmanFilter._native_twin).
455 if np.dtype(getattr(self, "dtype", np.float64)) != np.float64:
456 return None
457 if any(getattr(self, attr) is not None for attr in ("_F", "_H") if attr not in self._TWIN_DECODES_TWIN_DECODES):
458 return None
459 if callable(self._Q) or callable(self._R):
460 return None
461 L_ = self._L
462 if L_ is not None and (callable(L_) or not np.array_equal(np.asarray(L_), np.eye(self.n))):
463 return None
464
465 lib = core.lib
466 cached = self.__dict__.get("_native_twin")
467 if cached is None or cached[0] != name:
468 handle = lib.kalman_model_create(name.encode())
469 if handle == core.ffi.NULL:
470 raise KeyError(f"{type(self).__name__}: {core.last_error()}")
471 handle = core.ffi.gc(handle, lib.kalman_model_destroy)
472 self._push_arrays(handle) # a twin may size itself from them
473 n, m, l = core.ffi.new("int *"), core.ffi.new("int *"), core.ffi.new("int *")
474 core.check(lib.kalman_model_dims(handle, n, m, l), "kalman_model_dims")
475 if (int(n[0]), int(m[0]), int(l[0])) != (self.n, self.m, self.l):
476 raise RuntimeError(
477 f"{type(self).__name__}: the C++ twin {name!r} has (n, m, l) = "
478 f"({n[0]}, {m[0]}, {l[0]}), the Python model ({self.n}, {self.m}, {self.l}).")
479 self._native_twin = cached = (name, handle)
480 handle = cached[1]
481
482 if self.fs:
483 core.check(lib.kalman_model_set_sample_rate(handle, float(self.fs)), "set_sample_rate")
484 # The twin linearises with the scheme the Python model declares.
485 scheme = getattr(getattr(self, "DIFFERENTIATOR", None), "name", "FORWARD").lower()
486 if scheme not in ("forward", "central"):
487 return None

528# --- a Python-written model on the C++ engines ----------------------------------
529def callback_handle(model, engine="extended", live_noise=False):
530 """A C++ ``Kalman::Models::Callback`` whose ``f``, ``h`` and Jacobians call
531 ``model``'s own Python methods, so the C++ filter can run a model written in
532 Python. Returns ``(handle, keep)``: ``keep`` holds the callback objects and
533 must stay referenced for as long as the handle is used.
534
535 The Jacobians are the model's ``F(x, u, t)`` and ``H(x, t)``, analytic where
536 the model defines them and differenced by the Python framework otherwise, so
537 both sides linearise alike. ``Q`` and ``R`` are pushed as constants (``Q``
538 with the noise coupling ``L`` applied); the sample rate too. A Python
539 exception inside a callback stops the pass and is re-raised by the caller
540 (see :func:`run_callback_filter`).
541 """
542 lib = core.require()
543 ffi = core.ffi
544 n, m, l = int(model.n), int(model.m), int(model.l)
545 errors: list[BaseException] = []
546
547 def col(ptr, size):
548 return np.frombuffer(ffi.buffer(ptr, size * 8), dtype=np.float64).reshape(size, 1).copy()
549
550 def guard(fn):
551 def wrapped(*args):
552 try:
553 fn(*args)
554 return 0
555 except BaseException as exc: # must not unwind through C++
556 errors.append(exc)
557 return 1
558 return wrapped
559
560 def write(out, values, size):
561 ffi.memmove(out, np.ascontiguousarray(np.asarray(values, dtype=np.float64).reshape(-1)), size * 8)
562
563 def control(u):
564 return None if (u == ffi.NULL or l == 0) else col(u, l)
565
566 @ffi.callback("int(void *, const double *, const double *, double, double *)")
567 @guard
568 def step(user, x, u, t, out):
569 write(out, model.f(col(x, n), u=control(u), t=t), n)
570
571 @ffi.callback("int(void *, const double *, double, double *)")
572 @guard
573 def measure(user, x, t, out):
574 write(out, model.h(col(x, n), t=t), m)
575
576 @ffi.callback("int(void *, const double *, const double *, double, double *)")
577 @guard
578 def step_jacobian(user, x, u, t, out):
579 write(out, model.F(x=col(x, n), u=control(u), t=t), n * n)
580
581 @ffi.callback("int(void *, const double *, double, double *)")
582 @guard
583 def measure_jacobian(user, x, t, out):
584 try:
585 H = model.H(x=col(x, n), t=t)
586 except TypeError:
587 # A model whose h takes no noise argument: the base class cannot
588 # difference it (it passes v=). Only the sigma-point engines accept
589 # such a model, and they use H for the reported innovation alone.
590 from ...helpers import numerical_jacobian
591 H = numerical_jacobian(lambda xx: model.h(xx, t=t), col(x, n))
592 write(out, H, m * n)
593
594 handle = lib.kalman_model_create_callback(type(model).__name__.encode(), n, m, l,
595 step, measure, step_jacobian, measure_jacobian, ffi.NULL)
596 if handle == ffi.NULL:
597 raise RuntimeError(f"callback model: {core.last_error()}")
598 handle = ffi.gc(handle, lib.kalman_model_destroy)
599
600 if model.fs:
601 core.check(lib.kalman_model_set_sample_rate(handle, float(model.fs)), "set_sample_rate")
602
603 keep = [step, measure, step_jacobian, measure_jacobian]
604 if engine in ("unscented", "particle"):
605 keep += _unscented_noise(model, handle, guard, col, control, write)
606 elif constant_noise(model) and not live_noise:
607 Lm = np.asarray(model._L, dtype=np.float64)
608 Q = Lm @ np.asarray(model.Q(), dtype=np.float64).reshape(n, n) @ Lm.T
609 R = np.asarray(model.R(), dtype=np.float64).reshape(m, m)
610 core.check(lib.kalman_model_set_Q(handle, core.cdptr(np.ascontiguousarray(Q))), "set_Q")
611 core.check(lib.kalman_model_set_R(handle, core.cdptr(np.ascontiguousarray(R))), "set_R")
612 else:
613 # The noise as the Python engine works it out at each step: Q at the
614 # state the step starts from when Q takes one, then the coupling L.
615 from ..engine import KalmanEngine
616 takes_state = KalmanEngine._accepts_param(model.Q, "x")
617
618 @ffi.callback("int(void *, const double *, const double *, double, double *)")
619 @guard
620 def process_noise(user, x, u, t, out):
621 xc, uc = col(x, n), control(u)
622 Q = model.Q(t=t, x=xc) if takes_state else model.Q(t=t)
623 Lm = KalmanEngine._safe_L(model, xc, uc, t)
624 if Lm is not None:
625 Q = Lm @ Q @ Lm.T
626 write(out, Q, n * n)
627
628 @ffi.callback("int(void *, double, double *)")
629 @guard
630 def measurement_noise(user, t, out):
631 write(out, model.R(t=t), m * m)
632
633 core.check(lib.kalman_model_callback_set_noise(handle, process_noise, measurement_noise), "set_noise")
634 keep += [process_noise, measurement_noise]
635 return handle, (*keep, errors)
636

638def _unscented_noise(model, handle, guard, col, control, write):
639 """The noise hooks of the unscented engine, as the Python engine applies
640 them: where ``f`` takes ``w`` (or ``h`` takes ``v``) the noise goes through
641 the map and the covariance is handed over as it is; otherwise it is added
642 afterwards, coupled by ``L`` (or ``M``)."""
643 from ..engine import KalmanEngine
644 from ..engines.unscented import UnscentedKalmanEngine as U
645 lib, ffi = core.lib, core.ffi
646 n, m = int(model.n), int(model.m)
647 through_f = KalmanEngine._noise_through_f(model)
648 through_h = KalmanEngine._noise_through_h(model)
649
650 @ffi.callback("int(void *, const double *, const double *, double, double *)")
651 @guard
652 def process_noise(user, x, u, t, out):
653 Q = U._Q(model, t)
654 if not through_f:
655 Lm = U._L(model, col(x, n), control(u), t)
656 Q = Lm @ Q @ Lm.T
657 write(out, Q, n * n)
658
659 @ffi.callback("int(void *, double, double *)")
660 @guard
661 def measurement_noise(user, t, out):
662 R = U._R(model, t)
663 if not through_h:
664 Mm = U._M(model, np.asarray(model.x0, dtype=np.float64).reshape(-1, 1), t)
665 R = Mm @ R @ Mm.T
666 write(out, R, m * m)
667
668 @ffi.callback("int(void *, const double *, const double *, const double *, double, double *)")
669 @guard
670 def noisy_step(user, x, u, w, t, out):
671 write(out, model.f(col(x, n), u=control(u), w=col(w, n), t=t), n)
672
673 @ffi.callback("int(void *, const double *, const double *, double, double *)")
674 @guard
675 def noisy_measure(user, x, v, t, out):
676 write(out, model.h(col(x, n), v=col(v, m), t=t), m)
677
678 core.check(lib.kalman_model_callback_set_noise(handle, process_noise, measurement_noise), "set_noise")
679 core.check(lib.kalman_model_callback_set_noise_inputs(
680 handle, noisy_step if through_f else ffi.NULL, noisy_measure if through_h else ffi.NULL), "set_noise_inputs")
681 return [process_noise, measurement_noise, noisy_step, noisy_measure]

684def run_callback_filter(model, zk, uk=None, *, x0=None, P0=None, smooth=False,
685 engine="extended", control=None, **engine_params) -> dict:
686 """Filter ``zk`` with a C++ engine (``"extended"`` or ``"unscented"``, with
687 its constructor parameters) on a Python-written ``model``. Returns the
688 per-step arrays of :func:`core.run_filter`."""
689 lib = core.require()
690 # A controller may retune the model while it runs (adaptive Q and R), so in
691 # closed loop the noise is asked for at every step, never handed over once.
692 # The engine by name, or a C++ engine already made (the particle engine on numpy's draws).
693 kind = engine if isinstance(engine, str) else {v: k for k, v in core.ENGINE_KINDS.items()}.get(
694 int(core.lib.kalman_engine_kind(engine)), "extended")
695 handle, keep = callback_handle(model, kind, live_noise=control is not None)
696 n = int(model.n)
697 x0 = np.ascontiguousarray(np.asarray(model.x0 if x0 is None else x0, dtype=np.float64).reshape(-1))
698 P0 = np.ascontiguousarray(np.asarray(model.P0 if P0 is None else P0, dtype=np.float64).reshape(n, n))
699 core.check(lib.kalman_model_set_x0(handle, core.cdptr(x0)), "set_x0")
700 core.check(lib.kalman_model_set_P0(handle, core.cdptr(P0)), "set_P0")
701 core.check(lib.kalman_model_set_t0(handle, float(model.t0) if model.t0 is not None else 0.0), "set_t0")
702 try:
703 made = engine if not isinstance(engine, (str, int)) else core.create_engine(engine, **engine_params)
704 return core.run_filter(handle, made, n, int(model.m),
705 int(model.l), zk, uk, smooth=smooth, control=control)
706 except RuntimeError:
707 if keep[-1]:
708 raise keep[-1][0] # the Python exception, not its C++ shadow
709 raise
