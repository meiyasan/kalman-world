47 Vector Neural::Features(const Matrix &nis) const
48 {
49 const Matrix block = NisWindow(nis, fWindow);
50
51 // Row-major flatten: [step0_model0, step0_model1, ..., step1_model0, ...].
52 // Must match numpy's .ravel() in routers/neural.py exactly — get it wrong
53 // and the policy still returns probabilities, just meaningless ones.
54 Vector feat(block.Rows() * block.Cols());
55 std::size_t k = 0;
56 for (std::size_t r = 0; r < block.Rows(); ++r)
57 for (std::size_t j = 0; j < block.Cols(); ++j)
58 feat(k++) = block(r, j);
59
60 if (fMean.Size() == feat.Size())
61 for (std::size_t i = 0; i < feat.Size(); ++i)
62 feat(i) = (feat(i) - fMean(i)) / fScale(i);
63
64 return feat;
65 }

67 Vector Neural::Route(const Matrix &nis, const Vector &c, const Vector &likelihoods) const
68 {
69 if (!IsReady())
70 return Bayesian().Route(nis, c, likelihoods); // documented fallback
71
72 const Vector feat = Features(nis);
73
74 if (fModel->NInputs() && feat.Size() != fModel->NInputs())
75 {
76 std::ostringstream os;
77 os << "Kalman::Routers::Neural: the policy expects " << fModel->NInputs()
78 << " features but the NIS window produced " << feat.Size()
79 << " (window=" << fWindow << ", models=" << nis.Cols()
80 << "). The window must match the one used at training.";
81 throw std::runtime_error(os.str());
82 }
83
84 const Vector mu = (*fModel)(feat);

86 {
87 std::ostringstream os;
88 os << "Kalman::Routers::Neural: the policy returned " << mu.Size()
89 << " probabilities for " << c.Size() << " models.";
90 throw std::runtime_error(os.str());
91 }
