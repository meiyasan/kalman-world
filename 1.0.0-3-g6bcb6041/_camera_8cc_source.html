15 Sensor Make(std::string name, std::size_t n, const std::vector<std::size_t> &position, double focal,

17 {
18 const std::string who = "Sensors::Camera '" + name + "'";
19 Detail::CheckIndices(who, position, n);
20 if (position.size() != 3) throw std::invalid_argument(who + ": a point in space has three coordinates.");
21 if (!(focal > 0.)) throw std::invalid_argument(who + ": the focal length must be positive.");
22 if (at.Size() && at.Size() != 3) throw std::invalid_argument(who + ": where the camera is takes 3 numbers.");
23 if (rotation.Rows() && (rotation.Rows() != 3 || rotation.Cols() != 3))
24 throw std::invalid_argument(who + ": the rotation must be 3 x 3.");
25 const Matrix W = rotation.Rows() ? rotation : Matrix::Identity(3);
26
27 // Its settings, read at every call: [0] the focal length, [1..3] where it is.
28 auto settings = std::make_shared<Sensor::Settings>();
29 settings->emplace_back("focal", focal);
30 for (std::size_t i = 0; i < 3; ++i)
31 settings->emplace_back(i == 0 ? "x" : i == 1 ? "y" : "z", at.Size() ? at(i) : 0.);

33 // The point in the camera's axes, its depth kept above NEAR.
34 const auto seen = [position, settings, W](const Vector &x) {
35 Vector q(3, 0.);
36 for (std::size_t i = 0; i < 3; ++i)
37 for (std::size_t j = 0; j < 3; ++j)
38 q(i) += W(i, j) * (x(position[j]) - (*settings)[1 + j].second);
39 if (q(2) < Camera::NEAR) q(2) = Camera::NEAR;
40 return q;
41 };

43 Sensor::Measure measure = [seen, settings](const Vector &x, double) {
44 const Vector q = seen(x);
45 const double f = (*settings)[0].second;
46 Vector z(2);
47 z(0) = f * q(0) / q(2);
48 z(1) = f * q(1) / q(2);
49 return z;
50 };
51 // d(u, v) / dq = (focal / q_z) [ 1 0 -q_x/q_z ; 0 1 -q_y/q_z ], dq / dp = W
52 Sensor::MeasureJacobian jacobian = [seen, settings, position, W, n](const Vector &x, double) {
53 const Vector q = seen(x);
54 const double s = (*settings)[0].second / q(2);
55 Matrix J(2, n, 0.);
56 for (std::size_t j = 0; j < 3; ++j)
57 {
58 J(0, position[j]) += s * (W(0, j) - q(0) / q(2) * W(2, j));
59 J(1, position[j]) += s * (W(1, j) - q(1) / q(2) * W(2, j));
60 }
61 return J;
62 };
63 Sensor sensor(std::move(name), std::move(measure), Detail::Diagonal(2, Detail::Variance(who, sigma)),
64 std::move(jacobian));
65 sensor.settings = std::move(settings);
