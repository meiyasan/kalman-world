26 std::size_t CheckedCount(const Model &inner, const std::vector<Joint::Parameter> &parameters)
27 {
28 if (parameters.empty())
29 throw std::invalid_argument("Models::Joint: name at least one parameter to estimate.");
30 std::set<std::string> seen;
31 for (const auto &p : parameters)
32 {
33 if (!inner.HasParameter(p.name))
34 inner.Parameter(p.name); // throws, naming the known ones
35 if (!seen.insert(p.name).second)
36 throw std::invalid_argument("Models::Joint: the parameter '" + p.name + "' is given twice.");
37 if (!(p.sigma > 0.))
38 throw std::invalid_argument("Models::Joint: the parameter '" + p.name
39 + "' needs an initial standard deviation > 0.");
40 if (!(p.drift >= 0.))
41 throw std::invalid_argument("Models::Joint: the drift of '" + p.name + "' must be >= 0.");
42 if (!(p.tau >= 0.))
43 throw std::invalid_argument("Models::Joint: the tau of '" + p.name + "' must be >= 0.");
44 }
45 return parameters.size();
46 }

49 Joint::Joint(std::shared_ptr<Model> inner, std::vector<Parameter> parameters)
50 : fInner(std::move(inner)), fEstimated(std::move(parameters))
51 {
52 if (!fInner) throw std::invalid_argument("Models::Joint: the inner model is null.");
53 const std::size_t n = fInner->NState(), p = CheckedCount(*fInner, fEstimated);
54 fNInner = n;
55 SetName(fInner->Name());
56 Allocate(n + p, fInner->NMeasure(), fInner->NControl());
57
58 SetSampleRate(fInner->SampleRate());
59 SetT0(fInner->T0());
60 SetTimeUnit(fInner->TimeUnit());
61
62 // The parameters make the step nonlinear in the state even when the inner
63 // model is linear: the linear engine would read F(t), which is not dJ/dx.
64 const EngineKind engine = fInner->DefaultEngine();
65 SetDefaultEngine(engine == EngineKind::Linear ? EngineKind::Extended : engine);
66 SetAnalyticJacobian(false);
67 SetStructure(StructureKind::Maps);
68
69 AdoptInitialState();
70
71 std::vector<Component> components = fInner->StateComponents();
72 for (const auto &e : fEstimated) components.push_back({e.name, ""});
73 DeclareState(std::move(components));
74 DeclareMeasurement(fInner->MeasurementComponents());
75
76 CheckActs();
77 }

79 void Joint::AdoptInitialState()
80 {
81 // The inner model's initial state, then the parameters as they are now.
82 const std::size_t n = fNInner, p = fEstimated.size();
83 const Vector &s0 = fInner->X0();
84 const Matrix &P0 = fInner->P0();
85 Vector x0(n + p, 0.);
86 Matrix Px(n + p, n + p, 0.);
87 for (std::size_t i = 0; i < n; ++i)
88 {
89 x0(i) = s0.Size() == n ? s0(i) : 0.;
90 for (std::size_t j = 0; j < n; ++j)
91 Px(i, j) = (P0.Rows() == n && P0.Cols() == n) ? P0(i, j) : (i == j ? 1. : 0.);
92 }
93 fMean.assign(p, 0.);
94 for (std::size_t k = 0; k < p; ++k)
95 {
96 x0(n + k) = fInner->Parameter(fEstimated[k].name);
97 fMean[k] = x0(n + k);
98 Px(n + k, n + k) = fEstimated[k].sigma * fEstimated[k].sigma;
99 }
100 SetInitialState(x0);
101 SetInitialCovariance(Px);
102 }

104 void Joint::CheckActs() const
105 {
106 Sync();
107 const Vector s0 = Head(fX0);
108 Vector shifted(s0);
109 for (std::size_t i = 0; i < fNInner; ++i) shifted(i) += std::max(1., std::fabs(s0(i)));
110 const Vector u(fNControl, 0.);
111 const double t = fT0;
112
113 for (std::size_t k = 0; k < fEstimated.size(); ++k)
114 {
115 const std::string &name = fEstimated[k].name;
116 const double value = fInner->Parameter(name);
117 const double step = 1e-3 * std::max(1., std::fabs(value));
118 bool acts = false;
119 for (const Vector *s : std::initializer_list<const Vector *>{&s0, &shifted})
120 {
121 Vector f0, h0, f1, h1;
122 try
123 {
124 f0 = fInner->f(*s, u, t); h0 = fInner->h(*s, t);
125 fInner->SetParameter(name, value + step);
126 f1 = fInner->f(*s, u, t); h1 = fInner->h(*s, t);
127 }
128 catch (...) { fInner->SetParameter(name, value); throw; }
129 fInner->SetParameter(name, value);
130 for (std::size_t i = 0; i < f0.Size() && !acts; ++i) acts = f0(i) != f1(i);
131 for (std::size_t i = 0; i < h0.Size() && !acts; ++i) acts = h0(i) != h1(i);
132 if (acts) break;
133 }
134 if (!acts)
135 throw std::invalid_argument("Models::Joint: changing the parameter '" + name + "' of "
136 + fInner->Name() + " changes neither f nor h. The model may read it "
137 "only when it is built; the filter could not estimate it.");
138 }
139 }

198 Vector Joint::f(const Vector &x, const Vector &u, double t) const
199 {
200 Sync();
201 const Vector s = Head(x);
202 fLastX = x;
203 const auto previous = Apply(x);
204 Vector next;
205 try { next = fInner->f(s, u, t); }
206 catch (...) { Restore(previous); throw; }
207 Restore(previous);
208
209 Vector out(x); // the parameters stay, or return towards their mean
210 for (std::size_t i = 0; i < fNInner; ++i) out(i) = next(i);
211 for (std::size_t k = 0; k < fEstimated.size(); ++k)
212 out(fNInner + k) = fMean[k] + Decay(k) * (x(fNInner + k) - fMean[k]);
213 return out;
214 }

249 std::vector<Sensor> Joint::Sensors() const
250 {
251 Sync();
252 std::vector<Sensor> out = fInner->Sensors();
253 std::size_t first = 0;
254 for (Sensor &sensor : out)
255 {
256 const std::size_t k = sensor.Channels(), row = first;
257 first += k;
258 sensor.G = Matrix();
259 sensor.calibrated = false;
260 sensor.g = [this, row, k](const Vector &x, double t) {
261 const Vector z = this->h(x, t);
262 Vector part(k);
263 for (std::size_t i = 0; i < k; ++i) part(i) = z(row + i);
264 return part;
265 };
266 sensor.jacobian = [this, row, k](const Vector &x, double t) {
267 const Matrix H = this->JacobianH(x, t);
268 Matrix part(k, fNState, 0.);
269 for (std::size_t i = 0; i < k; ++i)
270 for (std::size_t j = 0; j < fNState; ++j) part(i, j) = H(row + i, j);
271 return part;
272 };
273 }
274 return out;

289 Matrix Joint::Jacobian(const Vector &x, const Vector &u, double t) const
290 {
291 Sync();
292 const std::size_t n = fNInner, p = fEstimated.size();
293 const Vector s = Head(x);
294 Vector theta(p);
295 for (std::size_t k = 0; k < p; ++k) theta(k) = x(n + k);
296
297 const auto previous = Apply(x);
298 Matrix Js, Jt;
299 try
300 {
301 Js = fInner->Jacobian(s, u, t);
302 const Vector uu(u);
303 Jt = Differentiators::Jacobian(
304 ParameterScheme(),
305 [this, &s, &uu, t](const Vector &th) {
306 for (std::size_t k = 0; k < th.Size(); ++k) fInner->SetParameter(fEstimated[k].name, th(k));
307 return fInner->f(s, uu, t);
308 },
309 theta, ParameterStep());
310 }
311 catch (...) { Restore(previous); throw; }
312 Restore(previous);
313
314 Matrix J(n + p, n + p, 0.);
315 for (std::size_t i = 0; i < n; ++i)
316 {
317 for (std::size_t j = 0; j < n; ++j) J(i, j) = Js(i, j);
318 for (std::size_t k = 0; k < p; ++k) J(i, n + k) = Jt(i, k);
319 }
320 for (std::size_t k = 0; k < p; ++k) J(n + k, n + k) = Decay(k);
321 return J;
322 }

324 Matrix Joint::JacobianH(const Vector &x, double t) const
325 {
326 Sync();
327 const std::size_t n = fNInner, p = fEstimated.size(), m = fNMeasure;
328 const Vector s = Head(x);
329 Vector theta(p);
330 for (std::size_t k = 0; k < p; ++k) theta(k) = x(n + k);
331
332 const auto previous = Apply(x);
333 Matrix Hs, Ht;
334 try
335 {
336 Hs = fInner->JacobianH(s, t);
337 Ht = Differentiators::Jacobian(
338 ParameterScheme(),
339 [this, &s, t](const Vector &th) {
340 for (std::size_t k = 0; k < th.Size(); ++k) fInner->SetParameter(fEstimated[k].name, th(k));
341 return fInner->h(s, t);
342 },
343 theta, ParameterStep());
344 }
345 catch (...) { Restore(previous); throw; }
346 Restore(previous);
347
348 Matrix H(m, n + p, 0.);
349 for (std::size_t i = 0; i < m; ++i)
350 {
351 for (std::size_t j = 0; j < n; ++j) H(i, j) = Hs(i, j);
352 for (std::size_t k = 0; k < p; ++k) H(i, n + k) = Ht(i, k);
353 }
354 return H;
355 }

360 {
