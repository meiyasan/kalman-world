1"""Instruments: a set of sensors with one name, one rate and one calibration.
2
3``compose(s1, s2, name=...)`` (``s1 | s2``) makes a ``KalmanInstrument``, the
4Python face of ``Kalman::Instrument``; ``model.add_instrument(instrument)`` gives

99 def test_calibration_is_two_parameters_of_the_model(self):
100 model = room().add_instrument(compose(probe(), humidity(), name="station").calibrate(2.0, 1.5))
101 self.assertEqual(model.parameters()["station.gain"], 2.0)
102 self.assertEqual(model.parameters()["station.offset"], 1.5)
103 self.assertIs(model.structure, KalmanStructureKind.MAPS) # gain * G x + offset: no matrix H
104 x = np.array([[20.0], [400.0]])
105 np.testing.assert_array_equal(model.h(x).ravel(), [400.0, 41.5, 801.5])
106 np.testing.assert_array_equal(model.sense(np.hstack([x, x]))[:, 1], [400.0, 41.5, 801.5])
107 model.set_parameter("station.offset", -1.0) # one value for all its sensors
108 np.testing.assert_array_equal(model.h(x).ravel(), [400.0, 39.0, 799.0])
109 sensors = model.sensors
110 self.assertEqual([s.offset for s in sensors], [0.0, -1.0, -1.0])
111 self.assertEqual([s.calibrated for s in sensors], [False, True, True])
112 self.assertEqual(float(sensors[1].measure(x)[0, 0]), 39.0) # read through the C++ sensor
113 station = model.instruments[1]

132class TestCalibratingWhileMeasuring(unittest.TestCase):

136 @staticmethod
137 def thermo(offset=0.0):
138 return compose(probe(), name="thermo").calibrate(1.0, offset)

140 def test_one_thermometer_cannot_tell_its_offset_from_the_temperature(self):
141 blind = room().add_instrument(self.thermo()).joint({"thermo.offset": (2.0, 0.0)})
142 with warnings.catch_warnings():
143 warnings.simplefilter("ignore")
144 report = blind.observability(50)
145 self.assertEqual(report["rank"], 2) # one direction is not determined
146 temperature, co2, offset = report["weakness"].values()
147 self.assertGreater(min(temperature, offset), 0.1) # it mixes these two,
148 self.assertLess(co2, 1e-6) # not the CO2
149 self.assertIn("thermo.offset", report["weakness"])
150
151 def test_a_reference_thermometer_settles_it(self):
152 reference = reads("reference", 0, 0.01)
153 checked = room().add_instrument(self.thermo()).add_sensor(reference)
154 model = checked.joint({"thermo.offset": (2.0, 0.0)})
155 self.assertTrue(model.observability(50)["observable"])
156 self.assertEqual(list(model.states.__members__)[2], "thermo.offset")
157
158 truth = room().add_instrument(self.thermo(self.OFFSETOFFSET)).add_sensor(reference)
159 rng = np.random.default_rng(20261117)
160 x, zs = truth.x0.copy(), []
161 Lq, Lr = np.linalg.cholesky(truth.Q()), np.linalg.cholesky(truth.R())
162 for _ in range(400):
163 x = truth.f(x) + Lq @ rng.standard_normal((2, 1))
164 zs.append((truth.h(x) + Lr @ rng.standard_normal((3, 1))).ravel())
165 with warnings.catch_warnings():
166 warnings.simplefilter("ignore")
167 rec = model.filter(np.array(zs), smooth=False, reset=True)
168 estimate, sigma = rec.xk[-1, 2], np.sqrt(rec.Pk[-1, 2, 2])
169 self.assertLess(abs(estimate - self.OFFSETOFFSET), 3 * sigma)
170 self.assertLess(sigma, 0.05)
171 self.assertEqual(checked.parameters()["thermo.offset"], 0.0) # the inner model is left as found
