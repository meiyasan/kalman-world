33 Oscillator(double omega, double damping, double fs) : Models::Extended("oscillator", 2, 1)
34 {
35 SetSampleRate(fs);
36 SetDifferentiator(DifferentiatorKind::Forward, 1e-6);
37 DeclareParameter("omega", omega);
38 DeclareParameter("damping", damping);
39 ParametersChanged();
40 DeclareState({{"position", "m"}, {"velocity", "m/s"}});
41 DeclareMeasurement({{"position", "m"}});
42
43 Matrix Q(2, 2, 0.); Q(0, 0) = 1e-6; Q(1, 1) = 1e-4;
44 SetQ(Q);
45 Matrix R(1, 1, 0.); R(0, 0) = 1e-2;
46 SetR(R);
47 Matrix H(1, 2, 0.); H(0, 0) = 1.;
48 SetH(H);
49 }

51 Vector f(const Vector &x, const Vector &, double = 0.) const override
52 {
53 const double dt = DeltaT();
54 Vector out(2);
55 out(0) = x(0) + dt * x(1);
56 out(1) = x(1) + dt * (-fOmega * fOmega * x(0) - fDamping * x(1));
57 return out;
58 }
59 };
